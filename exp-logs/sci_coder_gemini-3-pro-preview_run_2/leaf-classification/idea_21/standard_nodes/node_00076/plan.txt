    def _apply_preprocessing(self, X_train_dict, X_eval_dicts):
        """
        Fits PowerTransformer on the training views and transforms eval views.
        This ensures features are Gaussianized for LDA experts.

        Args:
            X_train_dict (dict): Dictionary of training views.
            X_eval_dicts (list of dict): List of dictionaries for evaluation sets (e.g. [X_val, X_test]).

        Returns:
            tuple: (X_train_transformed, [X_eval_transformed_1, ...])
        """
        X_train_transformed = {}
        X_eval_transformed_list = [{} for _ in X_eval_dicts]

        views = ["original", "morphological", "combined"]

        for view in views:
            # Initialize PowerTransformer with Yeo-Johnson
            # This handles skewness and forces a more Gaussian distribution
            pt = PowerTransformer(**PREPROCESSOR_PARAMS)

            # Fit on training data
            # float64 precision is maintained for numerical stability
            pt.fit(X_train_dict[view])

            # Transform Train
            X_train_transformed[view] = pt.transform(X_train_dict[view])

            # Transform Eval sets
            for i, eval_dict in enumerate(X_eval_dicts):
                X_eval_transformed_list[i][view] = pt.transform(eval_dict[view])

        return X_train_transformed, X_eval_transformed_list

    def create_pipeline(preprocessor_type="power", model_type="lda", **kwargs):
        """
        Creates a Pipeline with a preprocessor and a classifier.
        """
        steps = []

        # 1. Preprocessor
        if preprocessor_type == "power":
            steps.append(
                (
                    "preprocessor",
                    PowerTransformer(method="yeo-johnson", standardize=True),
                )

            steps.append(
                (
                    "preprocessor",
                    QuantileTransformer(
                        output_distribution="normal", n_quantiles=700, random_state=42
                    ),
                )
            )
        elif preprocessor_type == "standard":
            steps.append(("preprocessor", StandardScaler()))
        else:
            # No preprocessing
            pass

        # 2. Classifier
        if model_type == "lda":
            # Default LDA params from config, can be overridden by kwargs
            params = LDA_PARAMS.copy()

            steps.append(("classifier", LinearDiscriminantAnalysis(**params)))

            calibrated_model = CalibratedClassifierCV(
                estimator=base_estimator, **CALIBRATION_PARAMS
            )
            steps.append(("classifier", calibrated_model))

        return Pipeline(steps)
