    def _add_last_item_similarity(
        self, df: pd.DataFrame, transactions: pd.DataFrame
    ) -> pd.DataFrame:
        """
        Computes cosine similarity between the candidate item and the user's last purchased item.
        Cite solution_lesson_node_00021
        """
        if self.embedder.embeddings is None:
            df["last_item_sim"] = 0.0
            return df

        # 1. Identify last item per user
        # transactions is sorted by t_dat. We take the last occurrence for each customer.
        # Filter to relevant users first to speed up if transactions is huge
        relevant_users = df["customer_id"].unique()
        subset = transactions[transactions["customer_id"].isin(relevant_users)][
            ["customer_id", "article_id"]
        ]
        last_items = subset.drop_duplicates("customer_id", keep="last")
        last_item_map = dict(zip(last_items["customer_id"], last_items["article_id"]))

        # 2. Prepare Embeddings
        vocab = self.embedder.vocab
        embeddings = self.embedder.embeddings

        # Normalize embeddings
        norms = np.linalg.norm(embeddings, axis=1, keepdims=True)
        norms[norms == 0] = 1e-10
        norm_embeddings = embeddings / norms

        # 3. Compute Similarity
        # Pre-fetch vectors for needed articles
        needed_articles = set(df["article_id"].unique()) | set(last_item_map.values())
        vec_map = {}
        for aid in needed_articles:
            if aid in vocab:
                vec_map[aid] = norm_embeddings[vocab[aid]]

        dim = embeddings.shape[1]
        zero_vec = np.zeros(dim, dtype=np.float32)

        # Compute in chunks
        chunk_size = 100000
        sims = []

        for start in range(0, len(df), chunk_size):
            end = min(start + chunk_size, len(df))
            chunk = df.iloc[start:end]

            cust_ids = chunk["customer_id"].values
            cand_ids = chunk["article_id"].values

            # Get vectors
            v_last = np.array(
                [
                    vec_map.get(last_item_map.get(u), zero_vec)
                    if u in last_item_map
                    else zero_vec
                    for u in cust_ids
                ]
            )
            v_cand = np.array([vec_map.get(c, zero_vec) for c in cand_ids])

            # Dot product
            batch_sims = np.sum(v_last * v_cand, axis=1)
            sims.append(batch_sims)

        df["last_item_sim"] = np.concatenate(sims)
        return df
