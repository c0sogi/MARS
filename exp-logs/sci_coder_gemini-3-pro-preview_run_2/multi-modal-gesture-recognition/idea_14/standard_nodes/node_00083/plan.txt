My strategy is to re-implement the **Boundary-Aware Masked Dual-Stage Cascaded Recurrent-Convolutional Network (BA-MD-CRCN)** which was identified as the high-performing architecture in the lessons, specifically addressing the catastrophic failure of the previous model caused by the removal of explicit boundary supervision.

1.  **Data Loader (`library/data_loader.py`)**: I will modify `GestureDataset` to generate frame-wise **boundary targets** (binary labels indicating label changes). This is required to provide the explicit supervision mandated by **Lesson 82** and **Lesson 73**.
2.  **Loss Function (`library/loss.py`)**: I will update `DeepSupervisionLoss` to include a **Binary Cross Entropy (BCE)** term for the boundary heads. This ensures that the auxiliary transition channels in the cascaded network are trained to recognize gesture boundaries, rather than injecting noise into downstream stages (Cite **solution_lesson_node_00073**).
3.  **Model Architecture (`library/model.py`)**:
    *   I will replace the `GatedMSTCN` with a standard **Dilated TCN** (using `DilatedResidualLayer`) to match the robust "Current Best" configuration.
    *   I will ensure the TCN channel capacity is set to **256** (Cite **solution_lesson_node_00078**) to maintain sufficient representational power for temporal refinement.
    *   The model will output both classification logits and boundary logits at every stage (Encoder, Stage 2, Stage 3), realizing the "Latent Transition Channel" concept (Cite **solution_lesson_node_00075**).
4.  **Training Loop (`library/train_eval.py` & `runfile.py`)**: I will update the training loop to pass the new boundary targets to the loss function and increase the epoch count to 50, taking advantage of the fast training time observed in the previous run.

class DeepSupervisionLoss(nn.Module):
    """
    Aggregates losses from all model stages (Encoder, Refinement, Sharpening).
    Handles auxiliary channels (transition head) by slicing logits.
    """

    def __init__(self, num_classes: int = 21, smoothing_weight: float = 0.15):
        super(DeepSupervisionLoss, self).__init__()
        self.cls_loss_fn = MaskedWeightedCrossEntropy(num_classes=num_classes)
        self.smooth_loss_fn = TMSELoss(threshold=0.15)
        self.smoothing_weight = smoothing_weight
        self.num_classes = num_classes

    def forward(
        self, outputs: list, targets: torch.Tensor, mask: torch.Tensor
    ) -> torch.Tensor:
        """
        Args:
            outputs: List of tensors from the model stages.
                     Stage 1 & 2: (B, NumClasses + 1, T)
                     Stage 3: (B, NumClasses, T)
            targets: (B, T)
            mask: (B, T)
        """
        total_loss = 0.0

        for output in outputs:
            # Slice the first 'num_classes' channels for classification
            # This ignores the auxiliary 'transition' channel in Stage 1 & 2
            cls_logits = output[:, : self.num_classes, :]

            # 1. Classification Loss
            c_loss = self.cls_loss_fn(cls_logits, targets, mask)

            # 2. Smoothing Loss (T-MSE)
            # Apply Softmax to get probabilities as per requirement
            probs = F.softmax(cls_logits, dim=1)
            s_loss = self.smooth_loss_fn(probs, mask)

            # Aggregate
            total_loss += c_loss + (self.smoothing_weight * s_loss)

        return total_loss

class DeepSupervisionLoss(nn.Module):
    """
    Aggregates losses from all model stages.
    Includes explicit boundary supervision (Cite solution_lesson_node_00073, solution_lesson_node_00082).
    """

    def __init__(self, num_classes: int = 21, smoothing_weight: float = 0.15, boundary_weight: float = 1.0):
        super(DeepSupervisionLoss, self).__init__()
        self.cls_loss_fn = MaskedWeightedCrossEntropy(num_classes=num_classes)
        self.smooth_loss_fn = TMSELoss(threshold=0.15)
        self.bnd_loss_fn = nn.BCEWithLogitsLoss(reduction='none')
        self.smoothing_weight = smoothing_weight
        self.boundary_weight = boundary_weight
        self.num_classes = num_classes

    def forward(
        self, outputs: list, targets: torch.Tensor, boundaries: torch.Tensor, mask: torch.Tensor
    ) -> torch.Tensor:
        """
        Args:
            outputs: List of tuples (cls_logits, bnd_logits) from each stage.
            targets: (B, T) Class labels.
            boundaries: (B, T) Boundary labels (0 or 1).
            mask: (B, T)
        """
        total_loss = 0.0

        for cls_logits, bnd_logits in outputs:
            # 1. Classification Loss
            c_loss = self.cls_loss_fn(cls_logits, targets, mask)

            # 2. Boundary Loss (Explicit Supervision)
            # bnd_logits: (B, 1, T) -> squeeze to (B, T)
            b_loss = self.bnd_loss_fn(bnd_logits.squeeze(1), boundaries)
            # Masked mean
            b_loss = (b_loss * mask).sum() / (mask.sum() + 1e-8)

            # 3. Smoothing Loss (T-MSE) on Class Probabilities
            probs = F.softmax(cls_logits, dim=1)
            s_loss = self.smooth_loss_fn(probs, mask)

            # Aggregate
            total_loss += c_loss + (self.boundary_weight * b_loss) + (self.smoothing_weight * s_loss)

        return total_loss

class GatedConvBlock(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, dilation, dropout=0.2):
        super(GatedConvBlock, self).__init__()
        self.padding = (kernel_size - 1) * dilation

        self.filter_conv = nn.Conv1d(
            in_channels, out_channels, kernel_size, dilation=dilation
        )
        self.gate_conv = nn.Conv1d(
            in_channels, out_channels, kernel_size, dilation=dilation
        )
        self.conv_1x1 = nn.Conv1d(out_channels, out_channels, 1)
        self.dropout = nn.Dropout(dropout)

        # Residual connection handling
        if in_channels != out_channels:
            self.res_conv = nn.Conv1d(in_channels, out_channels, 1)
        else:
            self.res_conv = None

    def forward(self, x):
        # x: (B, C, T)

        # Compute filter and gate
        # Pad manually to handle causal or same padding. Here 'same' via slicing
        f = self.filter_conv(
            F.pad(x, (self.padding, 0))
        )  # Causal-like padding logic or just left pad
        # Wait, usually for TCN we pad left.
        # But let's stick to simple padding and slice if needed.
        # If we pad (padding, 0), output length is T + padding - (K-1)*D + 1 - 1 = T. Correct.

        g = self.gate_conv(F.pad(x, (self.padding, 0)))

        # Activation
        z = torch.tanh(f) * torch.sigmoid(g)

        # Projection
        out = self.conv_1x1(z)
        out = self.dropout(out)

        # Residual
        res = x if self.res_conv is None else self.res_conv(x)

        return out + res


class BiLSTMEncoder(nn.Module):
    def __init__(self, input_dim, hidden_dim, num_classes):
        super(BiLSTMEncoder, self).__init__()
        self.lstm = nn.LSTM(
            input_dim, hidden_dim, num_layers=2, batch_first=True, bidirectional=True
        )
        # Bidirectional -> 2 * hidden_dim
        self.cls_head = nn.Linear(hidden_dim * 2, num_classes)
        self.trans_head = nn.Linear(hidden_dim * 2, 1)

    def forward(self, x):
        # x: (B, C, T) -> Permute to (B, T, C)
        x = x.permute(0, 2, 1)

        feat, _ = self.lstm(x)  # (B, T, 2*H)

        cls_logits = self.cls_head(feat)  # (B, T, num_classes)
        trans_logits = self.trans_head(feat)  # (B, T, 1)
        trans_prob = torch.sigmoid(trans_logits)

        # Concatenate: (B, T, num_classes + 1)
        out = torch.cat([cls_logits, trans_prob], dim=2)

        # Permute back to (B, C, T)
        out = out.permute(0, 2, 1)
        return out


class GatedMSTCN(nn.Module):
    def __init__(self, input_dim, num_layers, num_f_maps, output_dim):
        super(GatedMSTCN, self).__init__()
        self.input_conv = nn.Conv1d(input_dim, num_f_maps, 1)

        self.layers = nn.ModuleList(
            [
                GatedConvBlock(
                    num_f_maps,
                    num_f_maps,
                    kernel_size=3,
                    dilation=2**i,
                    dropout=Config.dropout,
                )
                for i in range(num_layers)
            ]
        )

        self.output_conv = nn.Conv1d(num_f_maps, output_dim, 1)

    def forward(self, x, mask):
        # x: (B, C, T)
        out = self.input_conv(x)

        for layer in self.layers:
            out = layer(out)

        out = self.output_conv(out)

        # Apply mask
        if mask is not None:
            out = out * mask.unsqueeze(1)

        return out

class DilatedResidualLayer(nn.Module):
    """
    Standard Dilated Residual Block for TCN.
    """
    def __init__(self, channels, kernel_size, dilation, dropout):
        super(DilatedResidualLayer, self).__init__()
        # Standard padding to maintain temporal dimension with kernel 3
        self.conv1 = nn.Conv1d(channels, channels, kernel_size, dilation=dilation, padding=dilation)
        self.conv2 = nn.Conv1d(channels, channels, 1)
        self.dropout = nn.Dropout(dropout)

    def forward(self, x):
        out = F.relu(self.conv1(x))
        out = self.conv2(out)
        return x + self.dropout(out)


class BiLSTMEncoder(nn.Module):
    def __init__(self, input_dim, hidden_dim, num_classes):
        super(BiLSTMEncoder, self).__init__()
        self.lstm = nn.LSTM(
            input_dim, hidden_dim, num_layers=2, batch_first=True, bidirectional=True
        )
        # Bidirectional -> 2 * hidden_dim
        self.cls_head = nn.Linear(hidden_dim * 2, num_classes)
        self.bnd_head = nn.Linear(hidden_dim * 2, 1)

    def forward(self, x):
        # x: (B, C, T) -> Permute to (B, T, C)
        x = x.permute(0, 2, 1)

        feat, _ = self.lstm(x)  # (B, T, 2*H)

        cls_logits = self.cls_head(feat)  # (B, T, num_classes)
        bnd_logits = self.bnd_head(feat)  # (B, T, 1)

        # Permute back to (B, C, T)
        cls_logits = cls_logits.permute(0, 2, 1)
        bnd_logits = bnd_logits.permute(0, 2, 1)
        
        return cls_logits, bnd_logits


class SingleStageTCN(nn.Module):
    """
    Standard Multi-Stage TCN block (Cite solution_lesson_node_00078 for capacity).
    """
    def __init__(self, input_dim, num_layers, num_f_maps, num_classes):
        super(SingleStageTCN, self).__init__()
        self.input_conv = nn.Conv1d(input_dim, num_f_maps, 1)

        self.layers = nn.ModuleList(
            [
                DilatedResidualLayer(
                    num_f_maps,
                    kernel_size=3,
                    dilation=2**i,
                    dropout=Config.dropout,
                )
                for i in range(num_layers)
            ]
        )

        self.cls_conv = nn.Conv1d(num_f_maps, num_classes, 1)
        self.bnd_conv = nn.Conv1d(num_f_maps, 1, 1)

    def forward(self, x, mask):
        # x: (B, C, T)
        out = self.input_conv(x)

        for layer in self.layers:
            out = layer(out)

        cls_logits = self.cls_conv(out)
        bnd_logits = self.bnd_conv(out)

        # Apply mask
        if mask is not None:
            mask_expanded = mask.unsqueeze(1)
            cls_logits = cls_logits * mask_expanded
            bnd_logits = bnd_logits * mask_expanded

        return cls_logits, bnd_logits


class BAMD_CRCN(nn.Module):
    """
    Boundary-Aware Masked Dual-Stage CRCN (Cite solution_lesson_node_00082).
    """
    def __init__(self):
        super(BAMD_CRCN, self).__init__()

        # Stage 1: Encoder
        self.stage1 = BiLSTMEncoder(
            Config.input_dim, Config.hidden_dim, Config.num_classes
        )

        # Stage 2: Refinement
        # Input: Softmax(Class) + Sigmoid(Boundary) -> NumClasses + 1
        self.stage2 = SingleStageTCN(
            Config.num_classes + 1,
            Config.tcn_layers,
            Config.tcn_f_maps,
            Config.num_classes,
        )

        # Stage 3: Refinement
        self.stage3 = SingleStageTCN(
            Config.num_classes + 1,
            Config.tcn_layers,
            Config.tcn_f_maps,
            Config.num_classes,
        )

    def forward(self, x, mask):
        # x: (B, C, T)
        # mask: (B, T)

        # Stage 1
        cls1, bnd1 = self.stage1(x)
        if mask is not None:
            cls1 = cls1 * mask.unsqueeze(1)
            bnd1 = bnd1 * mask.unsqueeze(1)

        # Prepare input for Stage 2 (Latent Transition Channel - Cite solution_lesson_node_00075)
        probs1 = F.softmax(cls1, dim=1)
        bnd_prob1 = torch.sigmoid(bnd1)
        in2 = torch.cat([probs1, bnd_prob1], dim=1)

        # Stage 2
        cls2, bnd2 = self.stage2(in2, mask)

        # Prepare input for Stage 3
        probs2 = F.softmax(cls2, dim=1)
        bnd_prob2 = torch.sigmoid(bnd2)
        in3 = torch.cat([probs2, bnd_prob2], dim=1)

        # Stage 3
        cls3, bnd3 = self.stage3(in3, mask)

        return [(cls1, bnd1), (cls2, bnd2), (cls3, bnd3)]
