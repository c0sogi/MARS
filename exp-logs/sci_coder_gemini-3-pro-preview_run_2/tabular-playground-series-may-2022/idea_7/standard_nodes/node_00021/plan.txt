class InputInjector(nn.Module):
    """
    Projects raw input features to the current hidden dimension and adds them
    to the main signal stream.
    """

    def __init__(self, raw_dim, hidden_dim):
        super(InputInjector, self).__init__()
        self.projector = nn.Linear(raw_dim, hidden_dim)

    def forward(self, current_features, raw_features):
        return current_features + self.projector(raw_features)


class IdentityInjector(nn.Module):
    """
    Placeholder for stages where no input injection occurs.
    """

    def forward(self, current_features, raw_features):
        return current_features


class IIResFunnelGLU(nn.Module):
    """
    Input-Injected Residual Funnel Gated Network.

    Features:
    - Categorical Embeddings
    - Funnel Architecture (decreasing width)
    - Residual Gated Blocks
    - Input Injection at specific stages
    """

    def __init__(self):
        super(IIResFunnelGLU, self).__init__()

        # Configuration
        self.vocab_size = ModelConfig.VOCAB_SIZE
        self.embed_dim = ModelConfig.EMBED_DIM
        self.seq_len = ModelConfig.SEQUENCE_LENGTH
        self.num_cont = ModelConfig.NUM_CONT_FEATURES
        self.hidden_dims = ModelConfig.HIDDEN_DIMS
        self.dropout_rate = ModelConfig.DROPOUT_RATE
        self.inject_indices = set(ModelConfig.INJECT_AT_STAGES)

        # 1. Input Processing
        # Embedding layer for character tokens
        self.embedding = nn.Embedding(self.vocab_size + 1, self.embed_dim)

        # Calculate dimension of the raw concatenated input
        # (10 chars * 32 dim) + 30 continuous features
        self.raw_input_dim = (self.seq_len * self.embed_dim) + self.num_cont

        # 2. Backbone Stages
        self.downsamples = nn.ModuleList()
        self.injectors = nn.ModuleList()
        self.stages = nn.ModuleList()

        # Track input dimension for the next layer
        prev_dim = self.raw_input_dim

        for i, dim in enumerate(self.hidden_dims):
            # Downsampling / Projection Layer
            # For Stage 0, this projects Raw Input -> Hidden[0]
            # For Stage > 0, this projects Hidden[i-1] -> Hidden[i]
            self.downsamples.append(nn.Linear(prev_dim, dim))

            # Input Injection Mechanism
            if i in self.inject_indices:
                self.injectors.append(InputInjector(self.raw_input_dim, dim))
            else:
                self.injectors.append(IdentityInjector())

            # Stacked Gated Blocks
            # Using 3 blocks per stage for depth
            blocks = nn.Sequential(
                GatedBlock(dim, self.dropout_rate),
                GatedBlock(dim, self.dropout_rate),
                GatedBlock(dim, self.dropout_rate),
            )
            self.stages.append(blocks)

            prev_dim = dim

        # 3. Output Head
        self.head = nn.Linear(self.hidden_dims[-1], 1)

    def forward(self, cont_data, cat_data):
        # Embed Categorical Features
        # cat_data: (B, 10) -> (B, 10, 32)
        embs = self.embedding(cat_data)
        # Flatten embeddings: (B, 320)
        embs_flat = embs.view(embs.size(0), -1)

        # Concatenate with Continuous Features to form Raw Input
        # (B, 320) + (B, 30) -> (B, 350)
        x_raw = torch.cat([embs_flat, cont_data], dim=1)

        # Initial state
        x = x_raw
        prev_stage_out = x_raw

        # Iterate through stages
        for i in range(len(self.hidden_dims)):
            # 1. Downsample / Project
            if i == 0:
                # First stage projects raw input
                x = self.downsamples[i](x_raw)
            else:
                # Subsequent stages project output of previous stage
                x = self.downsamples[i](prev_stage_out)

            # 2. Input Injection
            # Adds projected raw input to current signal if configured for this stage
            x = self.injectors[i](x, x_raw)

            # 3. Process through Gated Blocks
            x = self.stages[i](x)

            # Store output for next stage's projection
            prev_stage_out = x

        # Output Head
        logits = self.head(x)
        return logits
