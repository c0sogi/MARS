The following changes implement a **Dual-Stream Transformer Architecture** (Cite `solution_lesson_node_00032`) and a **Learning Rate Scheduler** (Cite `solution_lesson_node_00033`) to improve performance.

1.  **Architecture**: The `TransformerResFunnel` is modified to process the categorical sequence `f_27` and continuous features in separate streams. The sequence stream uses a Transformer with explicit positional embeddings (Cite `solution_lesson_node_00030`), while continuous features are concatenated directly with the flattened Transformer output (Late Fusion). This avoids "polluting" the attention mechanism with non-sequential data.
2.  **Optimization**: A `StepLR` scheduler is added to the training loop to decay the learning rate, enabling the model to settle into sharper minima.
3.  **Hyperparameters**: `EMBED_DIM` is set to 32, `BATCH_SIZE` to 4096, and `EPOCHS` increased to 35 to match the successful configuration identified in the lessons.

class FeatureTokenizer(nn.Module):
    """
    Tokenizes continuous and categorical features into a unified sequence.
    Continuous features are projected via feature-specific linear layers.
    Categorical features are embedded and summed with positional embeddings.
    """

    def __init__(self, num_cont, cat_seq_len, vocab_size, embed_dim):
        super().__init__()
        self.num_cont = num_cont
        self.cat_seq_len = cat_seq_len
        self.embed_dim = embed_dim

        # Continuous feature projections: 1 -> embed_dim for each feature
        self.cont_projections = nn.ModuleList(
            [nn.Linear(1, embed_dim) for _ in range(num_cont)]
        )

        # Categorical embedding
        # vocab_size + 1 to handle potential padding or 1-based indexing safety
        self.cat_embedding = nn.Embedding(vocab_size + 1, embed_dim)

        # Learnable positional embedding for the categorical sequence
        self.cat_pos_embedding = nn.Parameter(torch.randn(1, cat_seq_len, embed_dim))

    def forward(self, x_cont, x_cat):
        # x_cont: (Batch, num_cont)
        # x_cat: (Batch, cat_seq_len)

        # 1. Process Continuous Features
        cont_tokens = []
        for i in range(self.num_cont):
            # Extract ith feature: (Batch, 1)
            val = x_cont[:, i].unsqueeze(1)
            # Project: (Batch, embed_dim)
            emb = self.cont_projections[i](val)
            cont_tokens.append(emb)

        # Stack to form sequence: (Batch, num_cont, embed_dim)
        cont_emb = torch.stack(cont_tokens, dim=1)

        # 2. Process Categorical Features
        # Embed: (Batch, cat_seq_len, embed_dim)
        cat_emb = self.cat_embedding(x_cat)
        # Add Positional Embeddings (broadcasting over batch)
        cat_emb = cat_emb + self.cat_pos_embedding

        # 3. Concatenate
        # Output: (Batch, num_cont + cat_seq_len, embed_dim)
        x = torch.cat([cont_emb, cat_emb], dim=1)
        return x

class TransformerResFunnel(nn.Module):
    """
    Hybrid architecture:
    1. Tokenize features.
    2. Apply Transformer Encoder for global interaction.
    3. Flatten and compress via Residual Funnel MLP.
    """

    def __init__(self):
        super().__init__()

        # Configuration
        self.num_cont = len(Config.CONT_FEATURES)
        self.cat_seq_len = Config.CAT_SEQ_LEN
        self.total_seq_len = self.num_cont + self.cat_seq_len
        self.embed_dim = Config.EMBED_DIM

        # 1. Tokenizer
        self.tokenizer = FeatureTokenizer(
            num_cont=self.num_cont,
            cat_seq_len=self.cat_seq_len,
            vocab_size=Config.VOCAB_SIZE,
            embed_dim=self.embed_dim,
        )

        # 2. Transformer Encoder
        encoder_layer = nn.TransformerEncoderLayer(
            d_model=self.embed_dim,
            nhead=Config.TRANSFORMER_HEADS,
            dim_feedforward=Config.TRANSFORMER_FF_DIM,
            dropout=Config.TRANSFORMER_DROPOUT,
            batch_first=True,
            activation="gelu",
        )
        self.transformer = nn.TransformerEncoder(
            encoder_layer, num_layers=Config.TRANSFORMER_DEPTH
        )

        # 3. Residual Funnel Backbone
        # Input dimension after flattening
        in_dim = self.total_seq_len * self.embed_dim

        layers = []
        current_dim = in_dim

        for next_dim in Config.FUNNEL_STAGES:

                ResGatedBlock(
                    in_dim=current_dim, out_dim=next_dim, dropout=Config.FUNNEL_DROPOUT
                )
            )
            current_dim = next_dim

        self.backbone = nn.Sequential(*layers)

        # 4. Output Head
        self.head = nn.Linear(current_dim, 1)

    def forward(self, x_cont, x_cat):
        # 1. Tokenize
        x = self.tokenizer(x_cont, x_cat)  # (Batch, Seq, Embed)

        # 2. Transformer
        x = self.transformer(x)  # (Batch, Seq, Embed)

        # 3. Flatten
        x = x.flatten(start_dim=1)  # (Batch, Seq*Embed)

        # 4. Backbone
        x = self.backbone(x)

        # 5. Head (Logits)
        logits = self.head(x)
        return logits

class TransformerResFunnel(nn.Module):
    """
    Hybrid architecture with Dual-Stream Processing.
    Cite solution_lesson_node_00032: Specialized Stream Processing vs Unified Tokenization.
    Stream 1: Character sequence -> Transformer.
    Stream 2: Continuous features -> Dense.
    Fusion: Concatenation -> Residual Funnel.
    """

    def __init__(self):
        super().__init__()

        # Configuration
        self.num_cont = len(Config.CONT_FEATURES)
        self.cat_seq_len = Config.CAT_SEQ_LEN
        self.embed_dim = Config.EMBED_DIM

        # 1. Character Tokenizer (Stream 1)
        self.char_tokenizer = CharTokenizer(
            cat_seq_len=self.cat_seq_len,
            vocab_size=Config.VOCAB_SIZE,
            embed_dim=self.embed_dim,
        )

        # 2. Transformer Encoder (Stream 1)
        encoder_layer = nn.TransformerEncoderLayer(
            d_model=self.embed_dim,
            nhead=Config.TRANSFORMER_HEADS,
            dim_feedforward=Config.TRANSFORMER_FF_DIM,
            dropout=Config.TRANSFORMER_DROPOUT,
            batch_first=True,
            activation="gelu",
        )
        self.transformer = nn.TransformerEncoder(
            encoder_layer, num_layers=Config.TRANSFORMER_DEPTH
        )

        # 3. Residual Funnel Backbone
        # Input dimension = Flattened Transformer Output + Continuous Features
        # Transformer Out: cat_seq_len * embed_dim
        # Cont Features: num_cont
        in_dim = (self.cat_seq_len * self.embed_dim) + self.num_cont

        layers = []
        current_dim = in_dim

        # Build stages
        for next_dim in Config.FUNNEL_STAGES:
            layers.append(
                ResGatedBlock(
                    in_dim=current_dim, out_dim=next_dim, dropout=Config.FUNNEL_DROPOUT
                )
            )
            current_dim = next_dim

        self.backbone = nn.Sequential(*layers)

        # 4. Output Head
        self.head = nn.Linear(current_dim, 1)

    def forward(self, x_cont, x_cat):
        # Stream 1: Categorical Sequence
        x_seq = self.char_tokenizer(x_cat)  # (Batch, Seq, Embed)
        x_seq = self.transformer(x_seq)     # (Batch, Seq, Embed)

        # Stream 2: Continuous Features (Late Fusion)
        # x_cont is (Batch, num_cont)
        
        # Fusion
        x = torch.cat([x_seq, x_cont], dim=1)

        # Backbone
        x = self.backbone(x)

        # Head
        logits = self.head(x)
        return logits
