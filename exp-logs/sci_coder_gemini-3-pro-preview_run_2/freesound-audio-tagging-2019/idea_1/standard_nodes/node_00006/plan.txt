The optimization strategy involves three key changes:
1.  **Model Upgrade**: Replacing the custom MobileNetV2 with a pre-trained version from `torchvision`. This leverages ImageNet weights for better feature extraction. We adapt the first layer for 1-channel audio and explicitly retain **Global Max Pooling** to handle zero-padded variable-length audio without signal dilution (Cite solution_lesson_node_00002).
2.  **Regularization**: Implementing **Mixup** augmentation in the training loop. This technique mixes inputs and targets, encouraging smoother decision boundaries and improving generalization.
3.  **Training Configuration**: Increasing `MAX_EPOCHS` to 50. This decouples the training duration from arbitrary limits, allowing the Early Stopping mechanism (patience=7) to determine the optimal stopping point (Cite solution_lesson_node_00005).

def _make_divisible(v, divisor, min_value=None):
    """
    Ensures that all layers have a channel number that is divisible by 8.
    This is important for optimizing performance on hardware accelerators.
    """
    if min_value is None:
        min_value = divisor
    new_v = max(min_value, int(v + divisor / 2) // divisor * divisor)
    # Make sure that round down does not go down by more than 10%.
    if new_v < 0.9 * v:
        new_v += divisor
    return new_v


class ConvBNReLU(nn.Sequential):
    """
    Standard block: Conv2d -> BatchNorm2d -> ReLU6
    """

    def __init__(self, in_planes, out_planes, kernel_size=3, stride=1, groups=1):
        padding = (kernel_size - 1) // 2
        super(ConvBNReLU, self).__init__(
            nn.Conv2d(
                in_planes,
                out_planes,
                kernel_size,
                stride,
                padding,
                groups=groups,
                bias=False,
            ),
            nn.BatchNorm2d(out_planes),
            nn.ReLU6(inplace=True),
        )


class InvertedResidual(nn.Module):
    """
    Inverted Residual Block (MobileNetV2 Block).
    Structure: Pointwise (Expansion) -> Depthwise -> Pointwise (Projection)
    """

    def __init__(self, inp, oup, stride, expand_ratio):
        super(InvertedResidual, self).__init__()
        self.stride = stride
        assert stride in [1, 2]

        hidden_dim = int(round(inp * expand_ratio))
        self.use_res_connect = self.stride == 1 and inp == oup

        layers = []
        if expand_ratio != 1:
            # Pointwise Convolution (Expansion)
            layers.append(ConvBNReLU(inp, hidden_dim, kernel_size=1))

        layers.extend(
            [
                # Depthwise Convolution
                ConvBNReLU(hidden_dim, hidden_dim, stride=stride, groups=hidden_dim),
                # Pointwise Convolution (Projection) - No non-linearity
                nn.Conv2d(hidden_dim, oup, 1, 1, 0, bias=False),
                nn.BatchNorm2d(oup),
            ]
        )
        self.conv = nn.Sequential(*layers)

    def forward(self, x):
        if self.use_res_connect:
            return x + self.conv(x)
        else:
            return self.conv(x)


class AudioMobileNet(nn.Module):
    """
    Lightweight 2D CNN based on MobileNetV2 architecture.
    Adapted for Audio Spectrogram inputs.

    def __init__(self, width_mult=1.0):

        input_channel = 32

        num_classes = Config.NUM_CLASSES

        # MobileNetV2 Configuration:
        # t: expansion factor, c: output channels, n: number of blocks, s: stride
        inverted_residual_setting = [
            [1, 16, 1, 1],
            [6, 24, 2, 2],
            [6, 32, 3, 2],
            [6, 64, 4, 2],
            [6, 96, 3, 1],
            [6, 160, 3, 2],
            [6, 320, 1, 1],
        ]

        # Build first layer
        input_channel = _make_divisible(input_channel * width_mult, 8)
        self.last_channel = (
            _make_divisible(last_channel * width_mult, 8)
            if width_mult > 1.0
            else last_channel
        )

        # Initial Feature Extraction
        # Input channels defined in Config (1 for Log-Mel Spectrogram)
        self.features = [ConvBNReLU(Config.IN_CHANNELS, input_channel, stride=2)]

        # Build Inverted Residual Blocks
        for t, c, n, s in inverted_residual_setting:
            output_channel = _make_divisible(c * width_mult, 8)
            for i in range(n):

                self.features.append(
                    InvertedResidual(
                        input_channel, output_channel, stride, expand_ratio=t

                )
                input_channel = output_channel

        self.features.append(
            ConvBNReLU(input_channel, self.last_channel, kernel_size=1)

        self.features = nn.Sequential(*self.features)

        # Classifier Head
        self.classifier = nn.Sequential(
            nn.Dropout(0.2),
            nn.Linear(self.last_channel, num_classes),
        )

        # Weight Initialization
        self._initialize_weights()

    def _initialize_weights(self):
        for m in self.modules():
            if isinstance(m, nn.Conv2d):
                nn.init.kaiming_normal_(m.weight, mode="fan_out")
                if m.bias is not None:
                    nn.init.zeros_(m.bias)
            elif isinstance(m, nn.BatchNorm2d):
                nn.init.ones_(m.weight)
                nn.init.zeros_(m.bias)
            elif isinstance(m, nn.Linear):
                nn.init.normal_(m.weight, 0, 0.01)
                nn.init.zeros_(m.bias)

        x = nn.functional.adaptive_max_pool2d(x, (1, 1))

        x = torch.flatten(x, 1)

        x = self.classifier(x)

def train_one_epoch(model, loader, criterion, optimizer, device):
    """
    Trains the model for one epoch using Mixup augmentation.

    Args:
        model (nn.Module): The neural network model.
        loader (DataLoader): DataLoader for training data.
        criterion (nn.Module): Loss function.
        optimizer (Optimizer): Optimizer.
        device (torch.device): Device to run training on.

    Returns:
        float: Average training loss for the epoch.
    """
    model.train()
    losses = AverageMeter()
    
    # Mixup configuration
    use_mixup = True
    alpha = 0.2

    for batch_idx, (images, targets, _) in enumerate(loader):
        images = images.to(device)
        targets = targets.to(device)

        optimizer.zero_grad()

        if use_mixup:
            # Generate mixup lambda
            if alpha > 0:
                lam = np.random.beta(alpha, alpha)
            else:
                lam = 1

            # Mix images
            batch_size = images.size(0)
            index = torch.randperm(batch_size).to(device)
            mixed_images = lam * images + (1 - lam) * images[index, :]

            # Mix targets (BCEWithLogitsLoss supports soft labels)
            mixed_targets = lam * targets + (1 - lam) * targets[index]

            # Forward pass
            outputs = model(mixed_images)
            loss = criterion(outputs, mixed_targets)
        else:
            outputs = model(images)
            loss = criterion(outputs, targets)

        loss.backward()
        optimizer.step()

        losses.update(loss.item(), images.size(0))

    return losses.avg
