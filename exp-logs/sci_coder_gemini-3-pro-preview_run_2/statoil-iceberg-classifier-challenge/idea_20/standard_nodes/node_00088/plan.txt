class QuadrantReadout(nn.Module):
    """
    Quadrant-Based Spatial Readout.
    Reduces spatial dimensions to a 2x2 grid using Adaptive Max Pooling,
    then flattens. Preserves spatial locality (quadrants) while reducing parameters.
    """

    def __init__(self, output_size=(2, 2)):
        super(QuadrantReadout, self).__init__()
        self.adaptive_pool = nn.AdaptiveMaxPool2d(output_size)

    def forward(self, x):
        # x: (B, C, H, W) -> (B, C, 2, 2)
        x = self.adaptive_pool(x)
        # Flatten: (B, C * 2 * 2)
        x = x.view(x.size(0), -1)
        return x
