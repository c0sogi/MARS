class RandomRadarTransform:
    """
    Applies random 90-degree rotations and flips using native PyTorch operations.
    Cite solution_lesson_node_00011 (Native Tensor Ops)
    Cite solution_lesson_node_00014 (Rotational Invariance)
    """

    def __call__(self, x):
        # x is Tensor (C, H, W)

        # Random rotation (0, 90, 180, 270 degrees)
        k = np.random.randint(0, 4)
        if k > 0:
            x = torch.rot90(x, k, [1, 2])

        # Random horizontal flip
        if np.random.random() > 0.5:
            x = torch.flip(x, [2])

        # Random vertical flip
        if np.random.random() > 0.5:
            x = torch.flip(x, [1])

        return x
