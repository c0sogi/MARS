class RandomFlip:
    """
    Applies random horizontal and vertical flips to the image tensor.
    Cite solution_lesson_node_00009
    """

    def __call__(self, x):
        # x is (C, H, W)
        if random.random() < 0.5:
            x = torch.flip(x, [2])  # Horizontal flip
        if random.random() < 0.5:
            x = torch.flip(x, [1])  # Vertical flip
        return x
