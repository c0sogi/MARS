def main():
    # Set seeds for reproducibility
    seed_everything(42)

    print("========================================")
    print("      DN-WBN Ensemble Pipeline          ")
    print("========================================")

    # ==========================================
    # 2. DATA LOADING
    # ==========================================
    print("\n[1/5] Loading Data...")
    # load_data returns loaders corresponding to metadata/train.csv and metadata/val.csv
    train_loader, val_loader, test_loader = load_data(load_cached_data=True)

    # ==========================================
    # 3. TRAINING (Seed Ensemble)
    # ==========================================
    # Cite Lesson 00143: The Ensemble Discrepancy Principle
    # We train 5 models with different seeds on the same training split to maximize
    # performance on the fixed validation hold-out.
    print("\n[2/5] Training Seed Ensemble (5 Models)...")
    
    models = []
    val_probs_list = []
    
    # Extract validation targets and features for analysis once
    val_targets = []
    val_angles = []
    val_img_means = []
    val_img_stds = []
    
    for images, angles, labels in val_loader:
        val_targets.extend(labels.numpy())
        val_angles.extend(angles.numpy())
        
        imgs_np = images.numpy()
        val_img_means.extend(np.mean(imgs_np, axis=(1, 2, 3)))
        val_img_stds.extend(np.std(imgs_np, axis=(1, 2, 3)))
        
    val_targets = np.array(val_targets)
    val_angles = np.array(val_angles)
    val_img_means = np.array(val_img_means)
    val_img_stds = np.array(val_img_stds)

    NUM_MODELS = 5
    from library.train import get_predictions
    from sklearn.metrics import log_loss

    for i in range(NUM_MODELS):
        print(f"\n--- Training Model {i+1}/{NUM_MODELS} (Seed {42+i}) ---")
        seed_everything(42 + i)
        
        # Train model (saves to dn_wbn_fold_{i}.pth)
        model = run_fold(i, train_loader, val_loader, DEVICE)
        models.append(model)
        
        # Get validation predictions
        probs = get_predictions(model, val_loader, DEVICE)
        val_probs_list.append(probs)
        
        # Individual score
        score = log_loss(val_targets, probs)
        print(f"Model {i+1} Val LogLoss: {score:.6f}")

    # ==========================================
    # 4. VALIDATION EVALUATION (ENSEMBLE)
    # ==========================================
    print("\n[3/5] Evaluating Ensemble on Hold-out Validation Set...")
    
    # Average predictions
    avg_val_probs = np.mean(val_probs_list, axis=0)
    
    # Calculate final metric
    val_metric = log_loss(val_targets, avg_val_probs)

    # REQUIRED OUTPUT FORMAT
    print(f"Final Validation Metric: {val_metric:.15f}")

    # ==========================================
    # 5. FAILURE ANALYSIS
    # ==========================================
    print("\n[4/5] Performing Failure Analysis on Ensemble...")
    
    # Calculate Error Magnitude (Absolute Difference)
    errors = np.abs(val_targets - avg_val_probs)

    # Create DataFrame for correlation analysis
    df_analysis = pd.DataFrame(
        {
            "error_magnitude": errors,
            "incidence_angle": val_angles,
            "image_mean_intensity": val_img_means,
            "image_std_intensity": val_img_stds,
        }
    )

    # Drop any NaNs

    # Calculate correlations

    print("Correlation between Error Magnitude and Input Features:")

    # ==========================================
    # 6. SUBMISSION GENERATION
