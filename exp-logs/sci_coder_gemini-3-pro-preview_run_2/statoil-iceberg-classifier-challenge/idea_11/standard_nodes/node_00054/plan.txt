def get_raw_data_and_stats():
    """
    Returns the raw processed data and the normalization statistics derived from train.csv.
    Used for custom splitting strategies.
    """
    # 1. Load Raw Data
    raw_data = process_and_cache_data(load_cached_data=True)
    all_ids = raw_data["ids"]
    all_images = raw_data["images"]
    all_angles = raw_data["angles"]
    
    # ID Lookup
    id_to_idx = {id_: i for i, id_ in enumerate(all_ids)}

    # 2. Impute Angles (using Train set mean as per standard procedure)
    train_meta = pd.read_csv(config.TRAIN_META_CSV)
    train_indices = [id_to_idx[i] for i in train_meta["id"].values]
    
    train_angles = all_angles[train_indices]
    angle_mean = np.nanmean(train_angles)
    all_angles = np.nan_to_num(all_angles, nan=angle_mean)
    raw_data["angles"] = all_angles # Update raw data

    # 3. Calculate Stats (from Train set)
    train_imgs = all_images[train_indices]
    train_b1 = train_imgs[:, :, :, 0]
    train_b2 = train_imgs[:, :, :, 1]
    train_b3 = (train_b1 + train_b2) / 2.0

    stats = {
        "min": np.array([np.min(train_b1), np.min(train_b2), np.min(train_b3)], dtype=np.float32),
        "max": np.array([np.max(train_b1), np.max(train_b2), np.max(train_b3)], dtype=np.float32),
    }

    return raw_data, stats, id_to_idx
