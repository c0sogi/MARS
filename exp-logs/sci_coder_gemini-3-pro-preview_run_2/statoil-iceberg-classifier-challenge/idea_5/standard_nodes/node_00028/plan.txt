def apply_tta(model, inputs, angles):
    """
    Applies Test-Time Augmentation using the D4 symmetry group (8 transforms).
    Cite solution_lesson_node_00014: Rotational Invariance.
    """
    probs_sum = torch.zeros(inputs.size(0), 1, device=inputs.device)
    
    # List of transform functions
    # Input shape: (B, C, H, W) -> dims 2,3 are H,W
    transforms = [
        lambda x: x,                                            # Identity
        lambda x: torch.rot90(x, 1, [2, 3]),                    # 90 deg
        lambda x: torch.rot90(x, 2, [2, 3]),                    # 180 deg
        lambda x: torch.rot90(x, 3, [2, 3]),                    # 270 deg
        lambda x: torch.flip(x, [3]),                           # Flip H
        lambda x: torch.rot90(torch.flip(x, [3]), 1, [2, 3]),   # Flip H + 90
        lambda x: torch.rot90(torch.flip(x, [3]), 2, [2, 3]),   # Flip H + 180
        lambda x: torch.rot90(torch.flip(x, [3]), 3, [2, 3]),   # Flip H + 270
    ]
    
    for t in transforms:
        aug_inputs = t(inputs)
        logits = model(aug_inputs, angles)
        probs_sum += torch.sigmoid(logits)
        
    return probs_sum / len(transforms)
