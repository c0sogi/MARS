Based on the provided lessons, the previous solution's underperformance can be attributed to the use of a Multi-Task Learning (MTL) approach (Idea 26) which disconnected the depth information from the decoder, rather than the superior **Explicit Feature Injection** strategy (Idea 8) identified in the lessons. Additionally, the complex semi-supervised pipeline consumed training time without yielding gains over a well-tuned supervised baseline (Cite solution_lesson_node_00045).

To optimize performance, I will revert to the "Current Best" architecture described in the lessons:
1.  **Architecture**: Modify `ResNet34WideLinkNet` to explicitly inject depth information at the bottleneck via concatenation. I will use a non-linear MLP to project the scalar depth into a 32-dimensional embedding before concatenation, as simple linear projections are less effective (Cite solution_lesson_node_00029, solution_lesson_node_00041).
2.  **Training Schedule**: Increase `EPOCHS` from 15 to 50. Short schedules with cosine annealing can cause underfitting due to rapid learning rate decay (Cite solution_lesson_node_00035).
3.  **Pipeline**: Simplify to a single-stage supervised training loop on the full training set. Ensembling provided no gain over the single-model baseline in previous experiments (Cite solution_lesson_node_00067), and pseudo-labeling is risky without a maximized supervised baseline (Cite solution_lesson_node_00049).
4.  **Loss**: Remove the auxiliary depth regression loss (MTL) and focus solely on the sum of BCE and Lovasz-Hinge loss (Cite solution_lesson_node_00036).

class MultiTaskLoss(nn.Module):
    """
    Multi-task loss combining Segmentation Loss (Lovasz + BCE) and Depth Regression Loss (MSE).
    """

    def __init__(self):
        super().__init__()
        self.bce = nn.BCEWithLogitsLoss()
        self.lovasz = LovaszHingeLoss()
        self.mse = nn.MSELoss()
        self.depth_weight = Config.DEPTH_LOSS_WEIGHT

    def forward(self, preds, targets):
        """
        Args:
            preds (dict): Dictionary containing:
                - 'mask': Tensor of shape (B, 1, H, W) containing logits.
                - 'depth': Tensor of shape (B, 1) or (B,) containing predicted depths.
            targets (dict): Dictionary containing:
                - 'mask': Tensor of shape (B, 1, H, W) containing binary ground truth.
                - 'depth': Tensor of shape (B, 1) or (B,) containing true depths.

        Returns:
            torch.Tensor: Weighted total loss.
            dict: Dictionary of loss components for logging.
        """
        # Unpack
        mask_logits = preds["mask"]
        pred_depth = preds["depth"]

        gt_mask = targets["mask"].float()
        gt_depth = targets["depth"].float()

        # --- Segmentation Loss ---
        # BCE expects (B, 1, H, W) or (B, H, W) matching input.
        # Lovasz expects (B, H, W) usually.

        # Squeeze channel dim for Lovasz if present (B, 1, H, W) -> (B, H, W)
        if mask_logits.dim() == 4 and mask_logits.shape[1] == 1:
            mask_logits_flat = mask_logits.squeeze(1)
        else:
            mask_logits_flat = mask_logits

        if gt_mask.dim() == 4 and gt_mask.shape[1] == 1:
            gt_mask_flat = gt_mask.squeeze(1)
        else:
            gt_mask_flat = gt_mask

        bce_loss = self.bce(mask_logits, gt_mask)
        lovasz_loss = self.lovasz(mask_logits_flat, gt_mask_flat)
        seg_loss = bce_loss + lovasz_loss

        # --- Depth Loss ---
        # Ensure shapes match (B,)
        depth_loss = self.mse(pred_depth.view(-1), gt_depth.view(-1))

        # --- Total Loss ---
        total_loss = seg_loss + (self.depth_weight * depth_loss)

        return total_loss, {
            "loss": total_loss.item(),
            "bce": bce_loss.item(),
            "lovasz": lovasz_loss.item(),
            "seg_loss": seg_loss.item(),
            "depth_loss": depth_loss.item(),
        }

def train_one_epoch(model, dataloader, optimizer, device, epoch):
    """
    Trains the model for one epoch.
    """
    model.train()
    criterion = MultiTaskLoss()

    total_loss_meter = 0.0
    bce_loss_meter = 0.0
    lovasz_loss_meter = 0.0
    depth_loss_meter = 0.0
    count = 0

    for batch in dataloader:
        images = batch["image"].to(device)
        masks = batch["mask"].to(device)
        depths = batch["depth"].to(device)

        optimizer.zero_grad()

        preds = model(images)
        targets = {"mask": masks, "depth": depths}

        loss, loss_dict = criterion(preds, targets)

        loss.backward()
        optimizer.step()

        batch_size = images.size(0)
        total_loss_meter += loss.item() * batch_size
        bce_loss_meter += loss_dict["bce"] * batch_size
        lovasz_loss_meter += loss_dict["lovasz"] * batch_size
        depth_loss_meter += loss_dict["depth_loss"] * batch_size
        count += batch_size

    metrics = {
        "loss": total_loss_meter / count,
        "bce": bce_loss_meter / count,
        "lovasz": lovasz_loss_meter / count,
        "depth": depth_loss_meter / count,
    }

    print(
        f"Epoch {epoch} Train Loss: {metrics['loss']:.10f} (BCE: {metrics['bce']:.6f}, Lovasz: {metrics['lovasz']:.6f}, Depth: {metrics['depth']:.6f})"
    )
    return metrics["loss"]
