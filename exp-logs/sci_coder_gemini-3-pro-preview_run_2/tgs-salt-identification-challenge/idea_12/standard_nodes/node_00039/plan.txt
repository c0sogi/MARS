class ResFiLM(nn.Module):
    """
    Residual Feature-wise Linear Modulation.
    Projects a scalar depth value into scale (gamma) and shift (beta) vectors.
    Initialized to Identity (gamma=0, beta=0).
    """

    def __init__(self, in_channels, depth_dim=1):
        super(ResFiLM, self).__init__()
        self.mlp = nn.Sequential(
            nn.Linear(depth_dim, in_channels),
            nn.ReLU(inplace=True),
            nn.Linear(in_channels, in_channels * 2),
        )
        # Initialize final layer to 0 for identity fallback
        nn.init.constant_(self.mlp[2].weight, 0)
        nn.init.constant_(self.mlp[2].bias, 0)

    def forward(self, x, depth):
        # depth: (B, 1)
        params = self.mlp(depth)  # (B, 2*C)
        gamma, beta = params.chunk(2, dim=1)

        # Reshape for broadcasting: (B, C, 1, 1)
        gamma = gamma.unsqueeze(2).unsqueeze(3)
        beta = beta.unsqueeze(2).unsqueeze(3)

        # F' = (1 + gamma) * F + beta
        return (1 + gamma) * x + beta

class DepthAwareLinkNet34(nn.Module):
    """
    LinkNet34 with explicit depth injection via concatenation at the bottleneck.
    Cite solution_lesson_node_00037: Additive concatenation outperforms multiplicative modulation (FiLM).
    Cite solution_lesson_node_00009: Concatenation is more efficient than heavy convolutional fusion.
    """

    def __init__(self):
        super(DepthAwareLinkNet34, self).__init__()

        # Load Pretrained ResNet34
        base = models.resnet34(pretrained=True)

        # Input Adaptation: Modify first conv for 1-channel input
        w = base.conv1.weight
        base.conv1 = nn.Conv2d(1, 64, kernel_size=7, stride=2, padding=3, bias=False)
        base.conv1.weight.data = w.sum(dim=1, keepdim=True)

        # Encoder Layers
        self.in_block = nn.Sequential(base.conv1, base.bn1, base.relu)
        self.maxpool = base.maxpool
        self.enc1 = base.layer1  # 64 channels
        self.enc2 = base.layer2  # 128 channels
        self.enc3 = base.layer3  # 256 channels
        self.enc4 = base.layer4  # 512 channels

        # Depth Projector
        self.depth_projector = DepthProjector(1, 32)

        # Decoder Layers (Wide-LinkNet)
        # Bottleneck Input: 512 (Encoder) + 32 (Depth) = 544
        self.dec4 = WideDecoderBlock(544, 256)
        self.dec3 = WideDecoderBlock(256, 128)
        self.dec2 = WideDecoderBlock(128, 64)
        self.dec1 = WideDecoderBlock(64, 64)

        # Final Upsample
        self.final_dec = nn.Sequential(
            nn.ConvTranspose2d(
                64, 32, kernel_size=3, stride=2, padding=1, output_padding=1, bias=False
            ),
            nn.BatchNorm2d(32),
            nn.ReLU(inplace=True),
            nn.Conv2d(32, 32, kernel_size=3, padding=1, bias=False),
            nn.BatchNorm2d(32),
            nn.ReLU(inplace=True),
        )

        self.final_conv = nn.Conv2d(32, 1, kernel_size=1)

    def forward(self, x, depth):
        # Encoder
        x0 = self.in_block(x)
        x1 = self.maxpool(x0)
        e1 = self.enc1(x1)
        e2 = self.enc2(e1)
        e3 = self.enc3(e2)
        e4 = self.enc4(e3)  # (B, 512, 4, 4)

        # Depth Injection
        d_emb = self.depth_projector(depth)  # (B, 32)
        # Expand depth embedding to spatial dimensions
        d_emb = (
            d_emb.unsqueeze(2).unsqueeze(3).expand(-1, -1, e4.size(2), e4.size(3))
        )  # (B, 32, 4, 4)

        # Concatenate: (B, 544, 4, 4)
        b = torch.cat([e4, d_emb], dim=1)

        # Decoder
        d4 = self.dec4(b)
        d4 = d4 + e3

        d3 = self.dec3(d4)
        d3 = d3 + e2

        d2 = self.dec2(d3)
        d2 = d2 + e1

        d1 = self.dec1(d2)
        d1 = d1 + x0

        out = self.final_dec(d1)
        logits = self.final_conv(out)

        return logits
