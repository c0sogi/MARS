The strategy involves reverting the complex semi-supervised pipeline to a robust, single-stage supervised learning approach using 5-Fold Cross-Validation, as recommended by the lessons derived from the "Current Best" solution.

**Key Optimizations:**
1.  **Architecture:** Revert to the **ResNet34 + Wide-LinkNet + Depth Injection** architecture. This explicitly models the depth dependency (Cite solution_lesson_node_00024, solution_lesson_node_00037).
2.  **Training Schedule:** Increase training epochs from 15 to **50** to ensure convergence, addressing the underfitting observed in the previous run (Cite solution_lesson_node_00035).
3.  **Validation Strategy:** Implement **5-Fold Cross-Validation** and use the aggregated Out-Of-Fold (OOF) predictions to evaluate performance, avoiding single-split bias (Cite solution_lesson_node_00079).
4.  **Threshold Optimization:** Implement dynamic binarization threshold search during validation to ensure model checkpoints represent the best possible separation capability, rather than just calibration at 0.5 (Cite solution_lesson_node_00033).
5.  **Ensembling:** Generate the final submission by averaging predictions from all 5 fold models (Cite solution_lesson_node_00067).
6.  **Data Augmentation:** Maintain `ElasticTransform` with `p=0.2` to handle non-rigid deformations without shifting the domain too heavily (Cite solution_lesson_node_00055, solution_lesson_node_00073).

class SaltNet(nn.Module):
    """
    ResNet34-based Wide-LinkNet for Salt Segmentation.
    Supports two modes:
    1. 'teacher': Injects depth information into the bottleneck.
    2. 'student': Multi-task learning with an auxiliary depth regression head.
    """

    def __init__(self, mode="student"):
        super(SaltNet, self).__init__()
        self.mode = mode
        assert mode in ["teacher", "student"], "Mode must be 'teacher' or 'student'"

        # Load ResNet34 backbone
        # Using default weights (ImageNet)
        base = models.resnet34(weights="DEFAULT")

        # Modify first convolution for 1-channel input (Grayscale)
        # We sum the weights of the original RGB filters to preserve structural kernels
        original_conv1 = base.conv1
        self.conv1 = nn.Conv2d(1, 64, kernel_size=7, stride=2, padding=3, bias=False)
        with torch.no_grad():
            self.conv1.weight.data = original_conv1.weight.data.sum(dim=1, keepdim=True)

        self.bn1 = base.bn1
        self.relu = base.relu
        self.maxpool = base.maxpool

        # Encoder Layers
        self.layer1 = base.layer1  # 64 channels
        self.layer2 = base.layer2  # 128 channels
        self.layer3 = base.layer3  # 256 channels
        self.layer4 = base.layer4  # 512 channels

        # Depth Injection Module (Teacher Only)
        self.depth_emb_dim = 64
        if self.mode == "teacher":
            self.depth_mlp = nn.Sequential(
                nn.Linear(1, 32),
                nn.ReLU(inplace=True),
                nn.Linear(32, self.depth_emb_dim),
                nn.ReLU(inplace=True),
            )

        # Auxiliary Regression Head (Student Only)
        if self.mode == "student":
            self.aux_head = nn.Sequential(
                nn.AdaptiveAvgPool2d((1, 1)), nn.Flatten(), nn.Linear(512, 1)
            )

        # Decoder Construction
        # Input to dec4 depends on mode (concatenation in teacher mode)
        in_ch_dec4 = 512 + self.depth_emb_dim if self.mode == "teacher" else 512

        self.dec4 = DecoderBlock(in_ch_dec4, 256)
        self.dec3 = DecoderBlock(256, 128)
        self.dec2 = DecoderBlock(128, 64)
        self.dec1 = DecoderBlock(64, 64)

        # Final upsampling block to restore original resolution (H/2 -> H)
        # Input: 64ch (from dec1 + x0 skip), Output: 1ch (Logits)
        self.final = nn.Sequential(
            nn.ConvTranspose2d(64, 32, kernel_size=4, stride=2, padding=1, bias=False),
            nn.BatchNorm2d(32),
            nn.ReLU(inplace=True),
            nn.Conv2d(32, 1, kernel_size=3, padding=1),
        )

    def forward(self, x, depth=None):
        """
        Forward pass.
        Args:
            x: Input image (B, 1, H, W)
            depth: Depth scalar (B, 1), required for Teacher mode.
        """
        # ---------------------
        # Encoder
        # ---------------------
        x0 = self.conv1(x)  # 64, H/2, W/2
        x0 = self.bn1(x0)
        x0 = self.relu(x0)

        x_pool = self.maxpool(x0)  # 64, H/4, W/4

        x1 = self.layer1(x_pool)  # 64, H/4, W/4
        x2 = self.layer2(x1)  # 128, H/8, W/8
        x3 = self.layer3(x2)  # 256, H/16, W/16
        x4 = self.layer4(x3)  # 512, H/32, W/32

        # ---------------------
        # Bottleneck / Mode Specifics
        # ---------------------
        aux_out = None
        bottleneck = x4

        if self.mode == "student":
            # Predict depth from texture
            aux_out = self.aux_head(x4)
        elif self.mode == "teacher":
            if depth is None:
                raise ValueError("Depth input is required for Teacher mode")

            # Project and inject depth
            d = self.depth_mlp(depth)  # (B, 64)
            d = d.unsqueeze(2).unsqueeze(3)  # (B, 64, 1, 1)
            # Expand to match spatial dimensions of x4
            d = d.expand(-1, -1, x4.size(2), x4.size(3))

            # Concatenate depth features with encoder features
            bottleneck = torch.cat([x4, d], dim=1)  # (B, 576, H/32, W/32)

        # ---------------------
        # Decoder (Additive Skips)
        # ---------------------
        d4 = self.dec4(bottleneck)
        d4 = d4 + x3  # Skip from Layer3

        d3 = self.dec3(d4)
        d3 = d3 + x2  # Skip from Layer2

        d2 = self.dec2(d3)
        d2 = d2 + x1  # Skip from Layer1

        d1 = self.dec1(d2)
        d1 = d1 + x0  # Skip from Conv1 (before pooling)

        # Final Upsample to original resolution
        out = self.final(d1)

        if self.mode == "student":
            return out, aux_out
        else:
            return out

    def train_teacher_epoch(model, loader, optimizer, device, loss_fn, scheduler=None):
        """
        Stage 1: Trains the Specialist Teacher with Depth Injection.
        """
        model.train()
        running_loss = 0.0

            images = images.to(device)

            # Teacher Forward: Requires image and depth
            outputs = model(images, depths)

            loss = loss_fn(outputs, masks)

        running_loss = 0.0

        # but here we simply assume we iterate based on the labeled loader length
        # and cycle the unlabeled one.
        unlabeled_iter = iter(unlabeled_loader)

        for labeled_batch in labeled_loader:
            # Fetch unlabeled batch
            try:
                unlabeled_batch = next(unlabeled_iter)
            except StopIteration:
                unlabeled_iter = iter(unlabeled_loader)
                unlabeled_batch = next(unlabeled_iter)

            # --- Prepare Data ---
            # Labeled Data
            l_images, l_masks, l_depths, _ = labeled_batch
            l_images = l_images.to(device)
            l_masks = l_masks.to(device)
            l_depths = l_depths.to(device)

            # Unlabeled Data (Pseudo-labels)
            # Note: u_masks here are the Soft Pseudo-Labels generated in Stage 2
            u_images, u_masks, _, _ = unlabeled_batch
            u_images = u_images.to(device)
            u_masks = u_masks.to(device)

            optimizer.zero_grad()

            # --- Forward & Loss ---

            # 1. Supervised Step (Labeled)
            # Student returns (mask_logits, depth_pred)
            l_pred_mask, l_pred_depth = model(l_images)
            loss_sup = loss_fn(
                l_pred_mask, l_pred_depth, l_masks, target_depth=l_depths
            )

            # 2. Distillation Step (Unlabeled)
            u_pred_mask, u_pred_depth = model(u_images)
            # Loss vs Soft Targets (BCE only), ignore depth regression for unlabeled
            loss_unsup = loss_fn(u_pred_mask, u_pred_depth, u_masks, target_depth=None)

            # Combine losses
            loss = loss_sup + loss_unsup

            loss.backward()
            optimizer.step()

            running_loss += loss.item()

        if scheduler:
            scheduler.step()

        return running_loss / len(labeled_loader)
