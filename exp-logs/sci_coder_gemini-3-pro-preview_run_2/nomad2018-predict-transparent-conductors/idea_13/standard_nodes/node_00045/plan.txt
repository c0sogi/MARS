class CGCNNBlock(nn.Module):
    """
    Standard CGCNN Interaction Block.
    Structure: Conv -> BatchNorm -> Dropout -> + Residual -> Softplus
    """

    def __init__(self, hidden_dim, dropout=0.0):
        super().__init__()
        self.conv = CGCNNConv(hidden_dim, hidden_dim)
        self.bn = nn.BatchNorm1d(hidden_dim)
        self.act = nn.Softplus()
        self.dropout = nn.Dropout(dropout)

    def forward(self, x, edge_index, edge_attr):
        residual = x

        # Convolution
        x = self.conv(x, edge_index, edge_attr)

        # Batch Normalization
        x = self.bn(x)

        # Regularization
        x = self.dropout(x)

        # Residual connection and Activation
        return self.act(residual + x)
