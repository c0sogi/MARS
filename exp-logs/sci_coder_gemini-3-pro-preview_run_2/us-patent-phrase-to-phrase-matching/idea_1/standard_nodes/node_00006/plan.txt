import torch
from torch.utils.data import Dataset
from library.config import Config


class PhraseDataset(Dataset):
    """
    PyTorch Dataset for the Phrase Similarity Task.
    Handles tokenization of text pairs (Context+Anchor, Target).
    """

    def __init__(self, df, tokenizer, is_test=False):
        """
        Args:
            df (pd.DataFrame): Dataframe containing 'text_a', 'text_b', and optionally 'score'.
            tokenizer (transformers.PreTrainedTokenizer): Tokenizer to process text.
            is_test (bool): If True, does not look for 'score' column.
        """
        self.df = df
        self.tokenizer = tokenizer
        self.is_test = is_test

        # Extract columns as numpy arrays for faster access
        self.text_a = df["text_a"].values
        self.text_b = df["text_b"].values

        if not self.is_test:
            self.labels = df["score"].values

    def __len__(self):
        return len(self.df)

    def __getitem__(self, idx):
        text_a = str(self.text_a[idx])
        text_b = str(self.text_b[idx])

        # Tokenize the pair
        # text_a = Context + Anchor
        # text_b = Target
        inputs = self.tokenizer(
            text_a,
            text_b,
            add_special_tokens=True,
            max_length=Config.MAX_LENGTH,
            padding="max_length",
            truncation=True,
            return_tensors="pt",
        )

        # The tokenizer returns tensors with a batch dimension of 1 (e.g., [1, seq_len])
        # We squeeze this dimension to get [seq_len] for the DataLoader to batch correctly.
        item = {
            "input_ids": inputs["input_ids"].squeeze(0),
            "attention_mask": inputs["attention_mask"].squeeze(0),
        }

        if not self.is_test:
            # Return label as a float tensor
            item["labels"] = torch.tensor(self.labels[idx], dtype=torch.float)

        return item
