import torch
from torch.utils.data import Dataset
from library.config import Config


class PhraseDataset(Dataset):
    def __init__(self, df, tokenizer, is_test=False):
        self.df = df
        self.tokenizer = tokenizer
        self.is_test = is_test
        
        # Ensure inputs are strings
        self.text_a = df["text_a"].astype(str).values
        self.text_b = df["text_b"].astype(str).values
        
        if not self.is_test:
            self.targets = df["score"].values

    def __len__(self):
        return len(self.df)

    def __getitem__(self, idx):
        text_a = self.text_a[idx]
        text_b = self.text_b[idx]

        # Tokenize the pair (Context+Anchor, Target)
        inputs = self.tokenizer(
            text_a,
            text_b,
            add_special_tokens=True,
            max_length=Config.MAX_LENGTH,
            padding="max_length",
            truncation=True,
            return_tensors="pt"
        )

        # Flatten tensors to remove the batch dimension added by return_tensors="pt"
        item = {
            "input_ids": inputs["input_ids"].flatten(),
            "attention_mask": inputs["attention_mask"].flatten()
        }

        if not self.is_test:
            # Regression target
            item["labels"] = torch.tensor(self.targets[idx], dtype=torch.float)

        return item
