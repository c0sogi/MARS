Optimization Strategy:
1.  **Architecture Change**: Replace the Bi-LSTM model with a **Windowed MLP**. Lessons indicate that simpler models with short context windows outperform deep sequence models on sparse trajectory data like this (Cite solution_lesson_node_00010).
2.  **Feature Engineering**: Add absolute baseline coordinates (`feat_WlsLat`, `feat_WlsLon`, `feat_WlsAlt`) to the input features. The previous solution failed partly because it excluded absolute state, preventing the model from learning location-dependent error biases (Cite solution_lesson_node_00011). We create copies prefixed with `feat_` to allow scaling them for the model while keeping original columns for reconstruction.
3.  **Hyperparameters**:
    *   Reduce `WINDOW_SIZE` from 30 to 5 to prevent overfitting to specific trajectory histories (Cite solution_lesson_node_00010).
    *   Increase `BATCH_SIZE` to 1024 to improve gradient estimation and training speed (Cite solution_lesson_node_00006).
    *   Adjust model hidden dimensions for the MLP structure.

class Attention(nn.Module):
    """
    Simple Attention Mechanism to aggregate LSTM hidden states.
    Computes a weighted sum of the sequence outputs.
    """

    def __init__(self, hidden_dim):
        super(Attention, self).__init__()
        self.attention = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim // 2),
            nn.Tanh(),
            nn.Linear(hidden_dim // 2, 1),
        )

    def forward(self, x):
        # x shape: (batch_size, seq_len, hidden_dim)
        # weights shape: (batch_size, seq_len, 1)
        weights = self.attention(x)
        weights = torch.softmax(weights, dim=1)

        # Context vector shape: (batch_size, hidden_dim)
        context = torch.sum(x * weights, dim=1)
        return context


class BiLSTMRegressor(nn.Module):
    """
    Sequence-Aware Residual Regressor using a Bidirectional LSTM backbone.
    Predicts the (DeltaEast, DeltaNorth) correction for the center timestamp of the window.
    """

    def __init__(
        self,
        input_dim=len(Config.INPUT_FEATURES),
        hidden_dim=Config.HIDDEN_DIM,
        num_layers=Config.NUM_LAYERS,
        dropout=Config.DROPOUT,
        output_dim=len(Config.TARGET_COLUMNS),
    ):
        super(BiLSTMRegressor, self).__init__()

        # Input projection to hidden dimension
        self.input_proj = nn.Linear(input_dim, hidden_dim)
        self.relu = nn.ReLU()

        # Bidirectional LSTM Backbone
        self.lstm = nn.LSTM(
            input_size=hidden_dim,
            hidden_size=hidden_dim,
            num_layers=num_layers,
            batch_first=True,
            bidirectional=True,
            dropout=dropout if num_layers > 1 else 0,
        )

        # Attention Layer (Input dim is hidden_dim * 2 because of bidirectionality)
        self.attention = Attention(hidden_dim * 2)

        # Output Regression Head
        self.fc = nn.Sequential(
            nn.Linear(hidden_dim * 2, hidden_dim),
            nn.ReLU(),
            nn.Dropout(dropout),
            nn.Linear(hidden_dim, output_dim),
        )

    def forward(self, x):
        # x shape: (batch_size, seq_len, input_dim)

        # Project features
        x = self.input_proj(x)
        x = self.relu(x)

        # Pass through LSTM
        # lstm_out shape: (batch_size, seq_len, 2 * hidden_dim)
        lstm_out, _ = self.lstm(x)

        # Aggregate sequence using attention
        # context shape: (batch_size, 2 * hidden_dim)
        context = self.attention(lstm_out)

        # Predict residuals
        out = self.fc(context)
        return out

class WindowedMLP(nn.Module):
    """
    Simple MLP that takes a flattened window of features as input.
    Cite solution_lesson_node_00010: Simpler Models with Short Contexts Outperform Deep Sequences
    """

    def __init__(
        self,
        input_dim=len(Config.INPUT_FEATURES),
        window_size=Config.WINDOW_SIZE,
        hidden_dim=Config.HIDDEN_DIM,
        output_dim=len(Config.TARGET_COLUMNS),
        dropout=Config.DROPOUT,
        **kwargs
    ):
        super(WindowedMLP, self).__init__()
        
        self.flatten_dim = input_dim * window_size
        
        self.net = nn.Sequential(
            nn.Linear(self.flatten_dim, hidden_dim),
            nn.BatchNorm1d(hidden_dim),
            nn.ReLU(),
            nn.Dropout(dropout),
            
            nn.Linear(hidden_dim, hidden_dim // 2),
            nn.BatchNorm1d(hidden_dim // 2),
            nn.ReLU(),
            nn.Dropout(dropout),
            
            nn.Linear(hidden_dim // 2, hidden_dim // 4),
            nn.BatchNorm1d(hidden_dim // 4),
            nn.ReLU(),
            
            nn.Linear(hidden_dim // 4, output_dim)
        )

    def forward(self, x):
        # x shape: (batch_size, window_size, input_dim)
        # Flatten the window dimension
        x = x.view(x.size(0), -1)
        return self.net(x)
