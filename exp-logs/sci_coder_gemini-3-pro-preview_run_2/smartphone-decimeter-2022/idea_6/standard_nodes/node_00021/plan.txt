def create_windows(df, window_size):
    """
    Creates sliding windows for the dataset.
    Returns a numpy array of shape (N, window_size, features).
    """
    # Features to window
    feature_cols = [
        "RelE",
        "RelN",
        "RelU",
        "VelE",
        "VelN",
        "VelU",
        "MeanCn0",
        "MeanUncertainty",
        "SatelliteCount",
    ]

    # Ensure data is sorted by trip and time
    df = df.sort_values(["tripId", "UnixTimeMillis"]).reset_index(drop=True)

    # We need to pad per trip to maintain output size = input size
    # The center of the window corresponds to the prediction target
    pad = window_size // 2

    windows_list = []

    for trip_id, group in df.groupby("tripId"):
        data = group[feature_cols].values

        # Pad the beginning and end of the trip sequence
        # Replicate the first and last elements
        pad_start = np.tile(data[0], (pad, 1))
        pad_end = np.tile(data[-1], (pad, 1))
        padded_data = np.vstack([pad_start, data, pad_end])

        # Create sliding windows
        # Shape: (num_samples, window_size, num_features)
        # Using stride_tricks for efficiency
        stride0, stride1 = padded_data.strides
        num_windows = len(data)

        windows = np.lib.stride_tricks.as_strided(
            padded_data,
            shape=(num_windows, window_size, len(feature_cols)),
            strides=(stride0, stride0, stride1),
        )

        windows_list.append(windows)

    return np.vstack(windows_list)

        df["WlsPositionZEcefMeters"].values,

    df["WlsLon"] = wls_lon
    df["WlsAlt"] = wls_alt

    # We construct "Relative" features.
    # However, the windowing function logic above just slices existing columns.
    # To make the window "Relative to Center", we need to do it AFTER windowing or
    # use a clever trick.
    # Trick: The network needs translation invariance.
    # Instead of absolute ECEF or LLA, we can use the difference between
    # current epoch WLS and previous epoch WLS (Velocity) as inputs,
    # plus maybe a local shape relative to the window center.

    # Implementing "Relative to Window Center" efficiently:
    # It is computationally expensive to re-center every window in a loop.
    # Approximation:
    # 1. Calculate global ENU relative to the FIRST point of the trip.
    # 2. Inside the window, subtract the center's ENU from neighbors.
    # This preserves the metric shape perfectly locally.

    # Calculate ENU relative to trip start
    df["TripStartLat"] = df.groupby("tripId")["WlsLat"].transform("first")
    df["TripStartLon"] = df.groupby("tripId")["WlsLon"].transform("first")
    df["TripStartAlt"] = df.groupby("tripId")["WlsAlt"].transform("first")

    # Vectorized ECEF to ENU relative to trip start
    # Note: ecef_to_enu takes scalar ref lat/lon usually, but we can pass arrays if implementation supports it.
    # The provided utils ecef_to_enu uses numpy, so passing arrays for ref should work if dimensions match.

    e, n, u = ecef_to_enu(
        df["WlsPositionXEcefMeters"].values,
        df["WlsPositionYEcefMeters"].values,
        df["WlsPositionZEcefMeters"].values,
        df["TripStartLat"].values,
        df["TripStartLon"].values,
        df["TripStartAlt"].values,
    )

    df["GlobalE"] = e
    df["GlobalN"] = n
    df["GlobalU"] = u

    # Fill NA for first element with 0
    df["VelE"] = df.groupby("tripId")["GlobalE"].diff().fillna(0)

    df["MeanUncertainty"] = df["MeanUncertainty"].fillna(
        100
    )  # High uncertainty default
    df["SatelliteCount"] = df["SatelliteCount"].fillna(0)

    return df

    """
    Generates relative metric features (ENU) and velocities.
    Cite solution_lesson_node_00020: Minimize Coordinate Transformations.
    We use simple scaling of Lat/Lon instead of ECEF->ENU to avoid instability.
    """
    # Ensure WLS columns are filled
    cols = ["WlsPositionXEcefMeters", "WlsPositionYEcefMeters", "WlsPositionZEcefMeters"]
    df[cols] = df[cols].fillna(0)

    # Convert WLS ECEF to LLA
    wls_lat, wls_lon, wls_alt = ecef_to_lla(
        df["WlsPositionXEcefMeters"].values,
        df["WlsPositionYEcefMeters"].values,
        df["WlsPositionZEcefMeters"].values,
    )

    df["WlsLat"] = wls_lat
    df["WlsLon"] = wls_lon
    df["WlsAlt"] = wls_alt

    # Calculate Velocities (First order difference) in meters approx
    deg_to_m = 111320.0
    
    # Diffs
    dLat = df.groupby("tripId")["WlsLat"].diff().fillna(0)
    dLon = df.groupby("tripId")["WlsLon"].diff().fillna(0)
    dAlt = df.groupby("tripId")["WlsAlt"].diff().fillna(0)
    
    # Scale velocities to meters
    df["VelLat"] = dLat * deg_to_m
    df["VelLon"] = dLon * deg_to_m * np.cos(np.radians(df["WlsLat"]))
    df["VelAlt"] = dAlt

    # Fill missing signal metrics
    df["MeanCn0"] = df["MeanCn0"].fillna(0)
    df["MeanUncertainty"] = df["MeanUncertainty"].fillna(100)
    df["SatelliteCount"] = df["SatelliteCount"].fillna(0)

    return df


def re_center_windows(windows):
    """
    Adjusts the position features to be relative to the center of the window
    and scales them to meters.
    """
    # windows shape: (N, W, F)
    # Indices: 0:Lat, 1:Lon, 2:Alt
    
    center_idx = windows.shape[1] // 2
    center_pos = windows[:, center_idx : center_idx + 1, 0:3]

    # Subtract center from all positions in window
    windows[:, :, 0:3] -= center_pos
    
    # Scale to meters
    deg_to_m = 111320.0
    
    # RelLat (North)
    windows[:, :, 0] *= deg_to_m
    
    # RelLon (East) - use center lat for cosine scaling
    center_lats_rad = np.radians(center_pos[:, :, 0])
    cos_lat = np.cos(center_lats_rad)
    windows[:, :, 1] *= (deg_to_m * cos_lat)
    
    # RelAlt is already in meters (diff of meters)
    
    return windows
