def read_raw_dicom_bytes(path: str) -> Optional[np.ndarray]:
    """
    Fallback reader for DICOM files when OpenCV fails.
    Assumes uncompressed pixel data located at the end of the file.
    Dynamically checks for common square resolutions (uint16).
    """
    try:
        with open(path, "rb") as f:
            data = f.read()

        file_size = len(data)

        # List of common square resolutions to check
        # We check from largest to smallest, though it shouldn't strictly matter
        # as the file size constraints are non-overlapping.
        resolutions = [512, 384, 320, 256, 240, 224, 192, 128]

        for res in resolutions:
            num_pixels = res * res
            expected_bytes = num_pixels * 2  # uint16 = 2 bytes per pixel

            # Header size = Total File Size - Pixel Data Size
            header_size = file_size - expected_bytes

            # Heuristic:
            # 1. Header must be positive and at least 128 bytes (DICOM preamble)
            # 2. Header shouldn't be excessively large for a standard uncompressed file.
            #    We allow up to 50KB to account for private tags/icons.
            if 128 <= header_size < 50000:
                pixel_data = data[-expected_bytes:]
                img = np.frombuffer(pixel_data, dtype=np.uint16).reshape(res, res)
                return img

        return None
    except Exception:
        return None
