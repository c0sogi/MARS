def select_roi_indices(subject_id, paths_dict):
    """
    Implements Hierarchical-Gated ROI Selection.

    Stage 1: FLAIR Integral -> Defines Safe Zone (15-85%)
    Stage 2: T1wCE Max Intensity -> Defines Target Slice within Safe Zone

    Returns:
        int: The selected anchor slice index (relative to the sorted file list).
    """
    # Construct full paths
    path_flair = os.path.join(Config.INPUT_DIR, paths_dict["path_FLAIR"])
    path_t1wce = os.path.join(Config.INPUT_DIR, paths_dict["path_T1wCE"])

    # Get sorted file lists
    files_flair = get_sorted_image_files(path_flair)
    files_t1wce = get_sorted_image_files(path_t1wce)

    num_slices = len(files_flair)
    if num_slices == 0:
        return 0

    # --- Stage 1: Anatomical Gating (FLAIR) ---
    # Calculate sum of intensity for all FLAIR slices to find the brain volume
    flair_sums = []
    for f in files_flair:
        img = read_dicom_robust(os.path.join(path_flair, f))
        flair_sums.append(np.sum(img))

    # Define Safe Zone
    start_idx = int(num_slices * Config.ROI_SAFE_START_PCT)
    end_idx = int(num_slices * Config.ROI_SAFE_END_PCT)

    # Ensure valid range
    if start_idx >= end_idx:
        start_idx = 0
        end_idx = num_slices

    # --- Stage 2: Pathological Targeting (T1wCE) ---
    # We look for the brightest spot (tumor core) within the safe zone
    # If T1wCE is missing or empty, fallback to FLAIR sum peak

    best_idx = -1
    max_intensity = -1.0

    # Check if T1wCE is available and has enough slices
    # We assume spatial correspondence: slice N in FLAIR ~ slice N in T1wCE
    if len(files_t1wce) > 0:
        # Limit search to the intersection of available files and safe zone
        search_limit = min(len(files_t1wce), end_idx)

        for i in range(start_idx, search_limit):
            f_name = files_t1wce[i]
            img = read_dicom_robust(os.path.join(path_t1wce, f_name))

            # Use raw max intensity (no smoothing) to find sharp hyperintensities
            current_max = np.max(img)

            if current_max > max_intensity:
                max_intensity = current_max
                best_idx = i

    # --- Fallback Strategy ---
    if best_idx == -1:
        # Fallback to the slice with the most brain tissue (FLAIR Sum Peak)
        # restricted to the safe zone
        safe_sums = flair_sums[start_idx:end_idx]
        if safe_sums:
            local_argmax = np.argmax(safe_sums)
            best_idx = start_idx + local_argmax
        else:
            best_idx = num_slices // 2

    return int(best_idx)

def select_roi_indices(subject_id, paths_dict):
    """
    Implements Integral-Statistic ROI Selection.
    Cite solution_lesson_node_00038: Prefer integral statistics (Sum) over Extremal (Max).
    Cite solution_lesson_node_00053: Derive selection from a single dominant reference modality.
    Cite solution_lesson_node_00026: Minimize hard constraints (no arbitrary safe zones).

    Returns:
        int: The selected anchor slice index (relative to the sorted file list).

    dir_path = ""
