def calculate_safe_zone(flair_full_path):
    """
    Stage 1: Anatomical Gating using FLAIR.
    Returns (start_idx, end_idx) containing central 80% of signal energy.
    """
    files = get_sorted_image_files(flair_full_path)
    if not files:
        return 0, 0

    slice_sums = []

    for f in files:
        path = os.path.join(flair_full_path, f)
        img = read_dicom_raw(path)
        slice_sums.append(np.sum(img))

    slice_sums = np.array(slice_sums)
    total_energy = np.sum(slice_sums)

    if total_energy == 0:
        return 0, len(files) - 1

    cumsum = np.cumsum(slice_sums)

    # Find indices where cumulative energy crosses 10% and 90%
    start_idx = np.argmax(cumsum >= 0.10 * total_energy)
    end_idx = np.argmax(cumsum >= 0.90 * total_energy)

    # Ensure valid range
    if start_idx >= end_idx:
        end_idx = start_idx

    return start_idx, end_idx


def get_anchor_index(t1wce_full_path, start_idx, end_idx):
    """
    Stage 2: Pathological Targeting using T1wCE within Safe Zone.
    Returns index of slice with max intensity.
    """
    files = get_sorted_image_files(t1wce_full_path)
    if not files:
        return 0

    # Clamp indices to valid range for T1wCE
    safe_start = max(0, min(start_idx, len(files) - 1))
    safe_end = max(0, min(end_idx, len(files) - 1))

    if safe_start == safe_end:
        search_indices = [safe_start]
    else:
        search_indices = range(safe_start, safe_end + 1)

    max_intensity = -1.0
    best_idx = safe_start

    for idx in search_indices:
        path = os.path.join(t1wce_full_path, files[idx])
        img = read_dicom_raw(path)
        # Use raw max to find tumor core
        curr_max = np.max(img)
        if curr_max > max_intensity:
            max_intensity = curr_max
            best_idx = idx

    return best_idx
