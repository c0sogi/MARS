class StripPooling(nn.Module):
    """
    Strip Pooling Module (SPM).
    Captures long-range dependencies using horizontal and vertical strip pooling.
    This is critical for identifying long, linear structures like contrails that
    may span across the entire image or be disconnected.
    """

    def __init__(self, in_channels, norm_layer=nn.BatchNorm2d):
        super(StripPooling, self).__init__()
        self.pool1 = nn.AdaptiveAvgPool2d((1, None))  # 1xW pooling
        self.pool2 = nn.AdaptiveAvgPool2d((None, 1))  # Hx1 pooling

        inter_channels = int(in_channels / 2)

        # 1x1 convolutions to reduce channels before expansion
        self.conv1 = nn.Conv2d(in_channels, inter_channels, 1, bias=False)
        self.conv1_bn = norm_layer(inter_channels)

        self.conv2 = nn.Conv2d(in_channels, inter_channels, 1, bias=False)
        self.conv2_bn = norm_layer(inter_channels)

        # 1x1 convolution to fuse features
        self.conv3 = nn.Conv2d(inter_channels, in_channels, 1, bias=False)
        self.conv3_bn = norm_layer(in_channels)

        self.relu = nn.ReLU(inplace=True)

    def forward(self, x):
        _, _, h, w = x.size()

        # Horizontal path
        x1 = self.pool1(x)
        x1 = self.conv1(x1)
        x1 = self.conv1_bn(x1)
        x1 = self.relu(x1)
        x1 = F.interpolate(x1, (h, w), mode="bilinear", align_corners=False)

        # Vertical path
        x2 = self.pool2(x)
        x2 = self.conv2(x2)
        x2 = self.conv2_bn(x2)
        x2 = self.relu(x2)
        x2 = F.interpolate(x2, (h, w), mode="bilinear", align_corners=False)

        # Fusion
        out = self.conv3(x1 + x2)
        out = self.conv3_bn(out)
        out = torch.sigmoid(out)

        # Attention-like scaling of input features
        return x * out

class ASPP(nn.Module):
    """
    Atrous Spatial Pyramid Pooling (ASPP).
    Captures multi-scale context using dilated convolutions.
    Preferred over Strip Pooling for non-axis-aligned features.
    """

    def __init__(self, in_channels, out_channels=256, rates=[6, 12, 18]):
        super(ASPP, self).__init__()
        modules = []
        # 1x1 conv
        modules.append(
            nn.Sequential(
                nn.Conv2d(in_channels, out_channels, 1, bias=False),
                nn.BatchNorm2d(out_channels),
                nn.ReLU(inplace=True),
            )
        )
        # Dilated convolutions
        for rate in rates:
            modules.append(
                nn.Sequential(
                    nn.Conv2d(
                        in_channels,
                        out_channels,
                        3,
                        padding=rate,
                        dilation=rate,
                        bias=False,
                    ),
                    nn.BatchNorm2d(out_channels),
                    nn.ReLU(inplace=True),
                )
            )
        # Global Average Pooling
        modules.append(
            nn.Sequential(
                nn.AdaptiveAvgPool2d(1),
                nn.Conv2d(in_channels, out_channels, 1, bias=False),
                nn.BatchNorm2d(out_channels),
                nn.ReLU(inplace=True),
            )
        )

        self.convs = nn.ModuleList(modules)

        # Project after concatenation
        self.project = nn.Sequential(
            nn.Conv2d(len(modules) * out_channels, in_channels, 1, bias=False),
            nn.BatchNorm2d(in_channels),
            nn.ReLU(inplace=True),
            nn.Dropout(0.5),
        )

    def forward(self, x):
        res = []
        for conv in self.convs:
            out = conv(x)
            # Handle Global Pooling branch upsampling
            if out.size(2) != x.size(2):
                out = F.interpolate(
                    out, size=x.shape[2:], mode="bilinear", align_corners=False
                )
            res.append(out)
        res = torch.cat(res, dim=1)
        return self.project(res)

    """

        # ===========================

        self.encoder = models.resnet18(weights=weights)

        # ResNet18 conv1: 7x7, stride 2, padding 3
        if in_channels != 3:

                in_channels,

                kernel_size=original_conv1.kernel_size,
                stride=original_conv1.stride,
                padding=original_conv1.padding,
                bias=original_conv1.bias,
            )

            # Initialize new weights
            if pretrained:

        self.dec1 = DecoderBlock(in_channels=64, skip_channels=64, out_channels=64)

        d1 = self.dec1(d2, x0)  # 64->128
