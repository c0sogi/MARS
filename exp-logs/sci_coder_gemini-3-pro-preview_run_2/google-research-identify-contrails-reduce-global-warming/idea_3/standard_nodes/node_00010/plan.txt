The auxiliary classifier, being less sensitive than the pixel-wise segmenter, discards valid predictions, capping the maximum possible score. 1.  **Remove the Multi-Task Architecture and Gated Inference**: Revert to a standard ResNet18 U-Net focused solely on segmentation. This eliminates the "single point of failure" introduced by the classifier (Cite solution_lesson_node_00009).
2.  **Restore Training Duration**: Increase `EPOCHS` back to 20 to allow sufficient convergence for the sparse segmentation task (Cite solution_lesson_node_00007).
3.  **Simplify Loss Function**: Use a standard combination of Dice Loss and BCE Loss without the auxiliary classification component.

This strategy aims to recover the performance of the "Current Best Solution" (~0.545 Dice) by removing the detrimental architectural changes while maintaining the robust ResNet18 backbone and 6-channel input (Cite solution_lesson_node_00002).

class MultiTaskLoss(nn.Module):
    """
    Joint Loss for Multi-Task Learning:
    1. Segmentation: Dice Loss + BCE Loss
    2. Classification: BCE Loss
    Total Loss = L_seg + lambda * L_cls
    """

    def __init__(self):
        super(MultiTaskLoss, self).__init__()

        # Sub-losses
        self.dice_loss = DiceLoss()
        self.bce_seg_loss = nn.BCEWithLogitsLoss()
        self.bce_cls_loss = nn.BCEWithLogitsLoss()

        # Weight for the auxiliary classification task
        self.cls_weight = Config.CLS_WEIGHT

    def forward(self, seg_logits, cls_logits, mask_targets, cls_targets):
        """
        Args:
            seg_logits (torch.Tensor): (B, 1, H, W) Segmentation head output.
            cls_logits (torch.Tensor): (B, 1) Classification head output.
            mask_targets (torch.Tensor): (B, 1, H, W) Ground truth masks.
            cls_targets (torch.Tensor): (B,) or (B, 1) Ground truth class labels.

        Returns:
            dict: Dictionary containing 'loss' (total loss for backprop) and components.
        """
        # Ensure targets are float for BCE/Dice calculations
        if not mask_targets.is_floating_point():
            mask_targets = mask_targets.float()
        if not cls_targets.is_floating_point():
            cls_targets = cls_targets.float()

        # ---------------------------------------------------
        # 1. Segmentation Loss (Dice + BCE)
        # ---------------------------------------------------
        loss_dice = self.dice_loss(seg_logits, mask_targets)
        loss_bce_seg = self.bce_seg_loss(seg_logits, mask_targets)

        loss_seg = loss_dice + loss_bce_seg

        # ---------------------------------------------------
        # 2. Classification Loss (BCE)
        # ---------------------------------------------------
        # Ensure cls_targets matches logits shape (B, 1)
        if cls_targets.ndim == 1:
            cls_targets = cls_targets.view(-1, 1)

        loss_cls = self.bce_cls_loss(cls_logits, cls_targets)

        # ---------------------------------------------------
        # 3. Total Loss
        # ---------------------------------------------------
        total_loss = loss_seg + (self.cls_weight * loss_cls)

        return {
            "loss": total_loss,
            "seg_loss": loss_seg,
            "cls_loss": loss_cls,
            "dice_loss": loss_dice,
            "bce_seg_loss": loss_bce_seg,
        }

def train_one_epoch(model, loader, optimizer, criterion, device, scheduler=None):
    """
    Performs one epoch of training.
    """
    model.train()
    running_loss = 0.0
    running_seg_loss = 0.0
    running_cls_loss = 0.0

    dataset_size = 0

    for images, masks, labels, _ in loader:
        batch_size = images.size(0)

        images = images.to(device)
        masks = masks.to(device)
        labels = labels.to(device)

        optimizer.zero_grad()

        # Forward pass
        seg_logits, cls_logits = model(images)

        # Compute loss
        loss_dict = criterion(seg_logits, cls_logits, masks, labels)
        loss = loss_dict["loss"]

        # Backward pass
        loss.backward()
        optimizer.step()

        if scheduler:
            scheduler.step()

        # Accumulate metrics
        running_loss += loss.item() * batch_size
        running_seg_loss += loss_dict["seg_loss"].item() * batch_size
        running_cls_loss += loss_dict["cls_loss"].item() * batch_size
        dataset_size += batch_size

    epoch_loss = running_loss / dataset_size
    epoch_seg_loss = running_seg_loss / dataset_size
    epoch_cls_loss = running_cls_loss / dataset_size

    return epoch_loss, epoch_seg_loss, epoch_cls_loss


def validate(model, loader, criterion, device):
    """
    Validates the model on the validation set.
    Computes Global Dice Coefficient using Gated Inference.
    """
    model.eval()
    running_loss = 0.0

    # Global Dice Accumulators
    intersection_sum = 0.0
    union_sum = 0.0

    # Classification Accuracy Accumulator
    correct_cls = 0
    total_cls = 0

    dataset_size = 0

    with torch.no_grad():
        for images, masks, labels, _ in loader:
            batch_size = images.size(0)

            images = images.to(device)
            masks = masks.to(device)
            labels = labels.to(device)

            # Forward pass
            seg_logits, cls_logits = model(images)

            # Loss calculation
            loss_dict = criterion(seg_logits, cls_logits, masks, labels)
            running_loss += loss_dict["loss"].item() * batch_size
            dataset_size += batch_size

            # --------------------------------------------------
            # Metrics Calculation
            # --------------------------------------------------

            # 1. Classification Accuracy
            cls_probs = torch.sigmoid(cls_logits)
            cls_preds = (cls_probs > Config.CLS_THRESHOLD).float()
            # Ensure labels are (B, 1)
            if labels.ndim == 1:
                labels = labels.view(-1, 1)
            correct_cls += (cls_preds == labels).sum().item()
            total_cls += batch_size

            # 2. Global Dice with Gated Inference
            seg_probs = torch.sigmoid(seg_logits)

            # Pixel-level thresholding
            pred_masks = (seg_probs > Config.PIXEL_THRESHOLD).float()

            # Apply Classification Gate
            # If image-level prob < threshold, zero out the entire mask for that image
            # cls_probs is (B, 1), pred_masks is (B, 1, H, W)
            gate_mask = (
                (cls_probs > Config.CLS_THRESHOLD).float().view(batch_size, 1, 1, 1)
            )
            pred_masks = pred_masks * gate_mask

            # Flatten for global intersection/union
            pred_flat = pred_masks.view(-1)
            true_flat = masks.view(-1)

            intersection_sum += (pred_flat * true_flat).sum().item()
            union_sum += pred_flat.sum().item() + true_flat.sum().item()

    epoch_loss = running_loss / dataset_size
    cls_accuracy = correct_cls / total_cls if total_cls > 0 else 0.0

    # Global Dice Formula: 2 * |X n Y| / (|X| + |Y|)
    smooth = 1e-6
    global_dice = (2.0 * intersection_sum + smooth) / (union_sum + smooth)

    return epoch_loss, global_dice, cls_accuracy
