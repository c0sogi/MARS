def mixup_criterion(criterion, pred, y_a, y_b, lam):
    """
    Calculates Mixup loss.
    Handles Multi-Sample Dropout output: (Batch, Num_Drops, Num_Classes).
    """
    # pred shape: (Batch, Num_Drops, Num_Classes)
    # y_a, y_b shape: (Batch, Num_Classes)

    loss = 0
    num_drops = pred.size(1)

    # Iterate over each dropout head output
    for i in range(num_drops):
        # Calculate loss for this head against both targets
        loss_a = criterion(pred[:, i, :], y_a)
        loss_b = criterion(pred[:, i, :], y_b)
        loss += lam * loss_a + (1 - lam) * loss_b

    # Average loss across all dropout heads
    return loss / num_drops
