def train_one_epoch(model, loader, optimizer, criterion, device, epoch):
    """
    Trains the model for one epoch using AMP and Gradient Accumulation.
    """
    model.train()
    running_loss = 0.0
    dataset_size = 0

    # Configuration with defaults for safety
    smoothing = getattr(Config, "LABEL_SMOOTHING", 0.05)
    use_amp = getattr(Config, "USE_AMP", False)
    grad_acc_steps = getattr(Config, "GRAD_ACC_STEPS", 1)

    # Initialize Scaler for AMP
    scaler = torch.amp.GradScaler('cuda', enabled=use_amp)

    optimizer.zero_grad()

    for step, (images, targets) in enumerate(loader):
        images = images.to(device)
        targets = targets.to(device)
        batch_size = images.size(0)

        # Apply label smoothing to targets manually
        with torch.no_grad():
            smoothed_targets = targets * (1.0 - smoothing) + 0.5 * smoothing

        # Forward pass with Mixed Precision
        with torch.amp.autocast('cuda', enabled=use_amp):
            outputs = model(images)
            loss = criterion(outputs, smoothed_targets)
            # Normalize loss for gradient accumulation
            loss = loss / grad_acc_steps

        # Backward pass with Scaler
        scaler.scale(loss).backward()

        # Optimizer Step (only every grad_acc_steps)
        if (step + 1) % grad_acc_steps == 0:
            scaler.step(optimizer)
            scaler.update()
            optimizer.zero_grad()

        # Update metrics
        # We multiply by grad_acc_steps to recover the actual loss value for reporting
        running_loss += loss.item() * grad_acc_steps * batch_size
        dataset_size += batch_size

    # Handle any remaining gradients at the end of the epoch
    if len(loader) % grad_acc_steps != 0:
        scaler.step(optimizer)
        scaler.update()
        optimizer.zero_grad()

    return running_loss / dataset_size
