To optimize the performance of the text normalization model, I will implement a **Neural Backoff Strategy** as suggested by the lesson "The Effectiveness of Hierarchical N-Gram Backoff". The lesson highlights that while the HFBB baseline is excellent for memorizable tokens (the "head" of the distribution), it fails on generative semiotic classes like `MONEY`, `DATE`, and `CARDINAL` (the "tail").

My strategy is to:
1.  **Augment the HFBB Model**: Instead of backing off to a simple "Identity" mapping when a token is unseen in the N-gram tables, I will introduce a character-level Sequence-to-Sequence (Seq2Seq) model.
2.  **Train a Targeted Neural Model**: I will train a lightweight LSTM-based Seq2Seq model specifically on "hard" examples (tokens where `before != after`) and a subsample of "easy" examples (identity mappings). This ensures the neural model focuses on learning the generative rules (e.g., converting digit strings to words) without being overwhelmed by the massive amount of trivial data.
3.  **Integrate and Optimize**: The neural model will be integrated into the `HFBBModel` class. During inference, it will only be invoked for tokens that fail the Trigram and Unigram lookups, ensuring the solution remains computationally efficient and fits within the runtime limits.

class CharTokenizer:
    def __init__(self):
        self.char2idx = {}
        self.idx2char = {}
        self.special_tokens = ["<PAD>", "<SOS>", "<EOS>", "<UNK>"]
        
    def fit(self, texts):
        unique_chars = set("".join(texts))
        all_tokens = self.special_tokens + sorted(list(unique_chars))
        self.char2idx = {char: idx for idx, char in enumerate(all_tokens)}
        self.idx2char = {idx: char for idx, char in enumerate(all_tokens)}
        self.pad_idx = self.char2idx["<PAD>"]
        self.sos_idx = self.char2idx["<SOS>"]
        self.eos_idx = self.char2idx["<EOS>"]
        self.unk_idx = self.char2idx["<UNK>"]
        
    def encode(self, text, add_special_tokens=True):
        indices = [self.char2idx.get(c, self.unk_idx) for c in text]
        if add_special_tokens:
            return [self.sos_idx] + indices + [self.eos_idx]
        return indices
    
    def decode(self, indices):
        chars = []
        for idx in indices:
            if idx == self.eos_idx:
                break
            if idx == self.sos_idx or idx == self.pad_idx:
                continue
            chars.append(self.idx2char.get(idx, ""))
        return "".join(chars)
    
    def __len__(self):
        return len(self.char2idx)
    
    def save(self, path):
        torch.save({
            'char2idx': self.char2idx,
            'idx2char': self.idx2char
        }, path)
        
    def load(self, path):
        checkpoint = torch.load(path)
        self.char2idx = checkpoint['char2idx']
        self.idx2char = checkpoint['idx2char']
        self.pad_idx = self.char2idx["<PAD>"]
        self.sos_idx = self.char2idx["<SOS>"]
        self.eos_idx = self.char2idx["<EOS>"]
        self.unk_idx = self.char2idx["<UNK>"]


class Seq2SeqDataset(Dataset):
    def __init__(self, df, tokenizer, max_len=100):
        self.df = df
        self.tokenizer = tokenizer
        self.max_len = max_len
        
    def __len__(self):
        return len(self.df)
    
    def __getitem__(self, idx):
        row = self.df.iloc[idx]
        src = self.tokenizer.encode(str(row['curr']), add_special_tokens=False)
        tgt = self.tokenizer.encode(str(row['after']), add_special_tokens=True)
        return torch.tensor(src, dtype=torch.long), torch.tensor(tgt, dtype=torch.long)

def collate_fn(batch):
    src_batch, tgt_batch = zip(*batch)
    src_lens = [len(x) for x in src_batch]
    src_padded = pad_sequence(src_batch, batch_first=True, padding_value=0)
    tgt_padded = pad_sequence(tgt_batch, batch_first=True, padding_value=0)
    return src_padded, tgt_padded, torch.tensor(src_lens)


class Seq2Seq(nn.Module):
    def __init__(self, vocab_size, embed_dim, hidden_dim, pad_idx):
        super().__init__()
        self.embedding = nn.Embedding(vocab_size, embed_dim, padding_idx=pad_idx)
        self.encoder = nn.LSTM(embed_dim, hidden_dim, batch_first=True, bidirectional=True)
        # Decoder input dim is embed_dim, hidden is hidden_dim*2 (bidirectional)
        self.decoder = nn.LSTM(embed_dim, hidden_dim * 2, batch_first=True)
        self.fc = nn.Linear(hidden_dim * 2, vocab_size)
        self.dropout = nn.Dropout(0.2)
        
    def forward(self, src, tgt, src_lens):
        # src: [batch, src_len]
        # tgt: [batch, tgt_len]
        
        # Encoder
        embedded = self.dropout(self.embedding(src))
        packed = nn.utils.rnn.pack_padded_sequence(embedded, src_lens.cpu(), batch_first=True, enforce_sorted=False)
        _, (hidden, cell) = self.encoder(packed)
        
        # Reshape hidden/cell to match decoder (layers*dir, batch, hidden) -> (layers, batch, hidden*2)
        hidden = torch.cat([hidden[-2], hidden[-1]], dim=1).unsqueeze(0)
        cell = torch.cat([cell[-2], cell[-1]], dim=1).unsqueeze(0)
        
        # Decoder
        tgt_embedded = self.dropout(self.embedding(tgt[:, :-1])) # Exclude EOS for input
        output, _ = self.decoder(tgt_embedded, (hidden, cell))
        prediction = self.fc(output)
        return prediction


class HFBBModel:
    """
    Hierarchical Frequency-Based Backoff Model with Neural Fallback.
    Cite solution_lesson_node_00002: Implements a generative model for the tail of the distribution.

    Layers:
    1. Trigram Context: (prev, curr, next) -> after
    2. Unigram Context: (curr) -> after
    3. Neural Seq2Seq: (curr) -> after
    4. Identity: curr -> curr (Final fallback)
    """

    def __init__(self):
        self.trigram_map = None
        self.unigram_map = None
        self.tokenizer = CharTokenizer()
        self.neural_model = None
        self.device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
        self.logger = setup_logger("HFBBModel")

        self.trigram_path = f"{MODEL_CACHE_PATH}.trigram.parquet"
        self.unigram_path = f"{MODEL_CACHE_PATH}.unigram.parquet"
        self.neural_path = NEURAL_MODEL_PATH
        self.tokenizer_path = f"{NEURAL_MODEL_PATH}.tokenizer"

    def fit(self, df):
        """
        Trains the HFBB tables and the Neural Seq2Seq model.
        """
        # 1. HFBB Training
        self.logger.info("Fitting Trigram Layer...")
        tri_counts = (
            df.groupby(["prev", "curr", "next", "after"])
            .size()
            .reset_index(name="count")
        )
        tri_counts.sort_values(by="count", ascending=False, inplace=True)
        self.trigram_map = tri_counts.drop_duplicates(subset=["prev", "curr", "next"])[
            ["prev", "curr", "next", "after"]
        ]
        self.logger.info(f"Trigram Layer fitted. Unique keys: {len(self.trigram_map)}")

        self.logger.info("Fitting Unigram Layer...")
        uni_counts = df.groupby(["curr", "after"]).size().reset_index(name="count")
        uni_counts.sort_values(by="count", ascending=False, inplace=True)
        self.unigram_map = uni_counts.drop_duplicates(subset=["curr"])[
            ["curr", "after"]
        ]
        self.logger.info(f"Unigram Layer fitted. Unique keys: {len(self.unigram_map)}")

        # 2. Neural Model Training
        self.logger.info("Preparing data for Neural Model...")
        # Filter for changed tokens (hard cases) + sample of unchanged (identity)
        mask_changed = df['curr'] != df['after']
        df_changed = df[mask_changed]
        df_unchanged = df[~mask_changed].sample(frac=0.1, random_state=42)
        
        df_neural = pd.concat([df_changed, df_unchanged])
        self.logger.info(f"Neural training set size: {len(df_neural)}")
        
        # Fit Tokenizer
        all_text = pd.concat([df_neural['curr'], df_neural['after']]).astype(str).tolist()
        self.tokenizer.fit(all_text)
        
        # Dataset & Loader
        dataset = Seq2SeqDataset(df_neural, self.tokenizer)
        dataloader = DataLoader(dataset, batch_size=1024, shuffle=True, collate_fn=collate_fn, num_workers=4)
        
        # Init Model
        self.neural_model = Seq2Seq(
            vocab_size=len(self.tokenizer),
            embed_dim=64,
            hidden_dim=256,
            pad_idx=self.tokenizer.pad_idx
        ).to(self.device)
        
        optimizer = optim.Adam(self.neural_model.parameters(), lr=0.001)
        criterion = nn.CrossEntropyLoss(ignore_index=self.tokenizer.pad_idx)
        
        # Training Loop
        self.neural_model.train()
        epochs = 5
        self.logger.info(f"Training Neural Model for {epochs} epochs...")
        
        for epoch in range(epochs):
            total_loss = 0
            for src, tgt, src_lens in dataloader:
                src, tgt = src.to(self.device), tgt.to(self.device)
                
                optimizer.zero_grad()
                output = self.neural_model(src, tgt, src_lens)
                
                output = output.reshape(-1, output.shape[-1])
                target = tgt[:, 1:].reshape(-1)
                
                loss = criterion(output, target)
                loss.backward()
                optimizer.step()
                
                total_loss += loss.item()
            
            self.logger.info(f"Epoch {epoch+1}/{epochs}, Loss: {total_loss/len(dataloader):.4f}")

    def _predict_neural(self, tokens):
        """
        Helper to run inference on a list of tokens.
        """
        self.neural_model.eval()
        results = []
        batch_size = 2048
        
        with torch.no_grad():
            for i in range(0, len(tokens), batch_size):
                batch_tokens = tokens[i:i+batch_size]
                src_batch = [self.tokenizer.encode(str(t), add_special_tokens=False) for t in batch_tokens]
                src_lens = torch.tensor([len(x) for x in src_batch])
                src_padded = pad_sequence([torch.tensor(x) for x in src_batch], batch_first=True, padding_value=self.tokenizer.pad_idx).to(self.device)
                
                # Greedy Decoding
                embedded = self.neural_model.embedding(src_padded)
                packed = nn.utils.rnn.pack_padded_sequence(embedded, src_lens.cpu(), batch_first=True, enforce_sorted=False)
                _, (hidden, cell) = self.neural_model.encoder(packed)
                
                hidden = torch.cat([hidden[-2], hidden[-1]], dim=1).unsqueeze(0)
                cell = torch.cat([cell[-2], cell[-1]], dim=1).unsqueeze(0)
                
                batch_curr_size = src_padded.size(0)
                decoder_input = torch.tensor([self.tokenizer.sos_idx] * batch_curr_size, device=self.device).unsqueeze(1)
                
                batch_preds = [[] for _ in range(batch_curr_size)]
                finished = [False] * batch_curr_size
                
                max_len = 100
                for _ in range(max_len):
                    if all(finished): break
                    
                    tgt_embedded = self.neural_model.embedding(decoder_input)
                    output, (hidden, cell) = self.neural_model.decoder(tgt_embedded, (hidden, cell))
                    logits = self.neural_model.fc(output.squeeze(1))
                    top1 = logits.argmax(1)
                    
                    decoder_input = top1.unsqueeze(1)
                    
                    for j, token_idx in enumerate(top1.cpu().numpy()):
                        if not finished[j]:
                            if token_idx == self.tokenizer.eos_idx:
                                finished[j] = True
                            else:
                                batch_preds[j].append(token_idx)
                                
                decoded_strs = [self.tokenizer.decode(idxs) for idxs in batch_preds]
                results.extend(decoded_strs)
                
        return results

    def predict(self, df):
        """
        Predicts normalized text using the backoff strategy.
        Returns a Series of predictions.
        """
        self.logger.info("Starting prediction...")

        # Ensure input columns exist
        req_cols = ["prev", "curr", "next"]
        if not all(c in df.columns for c in req_cols):
            raise ValueError(f"Input dataframe must contain columns: {req_cols}")

        # 1. Trigram Lookup
        self.logger.info("Applying Trigram Layer...")
        pred_df = (
            df[req_cols]
            .merge(self.trigram_map, on=["prev", "curr", "next"], how="left")
            .rename(columns={"after": "pred_tri"})
        )

        # 2. Unigram Lookup
        self.logger.info("Applying Unigram Layer...")
        pred_df = pred_df.merge(self.unigram_map, on=["curr"], how="left").rename(
            columns={"after": "pred_uni"}
        )

        # 3. Backoff Logic
        combined_preds = pred_df["pred_tri"].fillna(pred_df["pred_uni"])
        
        missing_mask = combined_preds.isna()
        missing_count = missing_mask.sum()
        
        if missing_count > 0 and self.neural_model is not None:
            self.logger.info(f"Applying Neural Layer to {missing_count} tokens...")
            missing_tokens = df.loc[missing_mask, "curr"].astype(str).tolist()
            neural_preds = self._predict_neural(missing_tokens)
            
            pred_df.loc[missing_mask, "pred_neural"] = neural_preds
            combined_preds = combined_preds.fillna(pred_df["pred_neural"])
        else:
            self.logger.info("Skipping Neural Layer (no missing tokens or model not loaded).")

        # 4. Final Identity Fallback
        self.logger.info("Applying Final Identity Fallback...")
        predictions = combined_preds.fillna(pred_df["curr"])

        return predictions

    def save(self):
        """
        Saves the lookup tables and neural model.
        """
        self.logger.info(f"Saving model to {WORKING_DIR}...")
        if self.trigram_map is not None:
            self.trigram_map.to_parquet(self.trigram_path, index=False)
        if self.unigram_map is not None:
            self.unigram_map.to_parquet(self.unigram_path, index=False)
        
        if self.neural_model is not None:
            torch.save(self.neural_model.state_dict(), self.neural_path)
            self.tokenizer.save(self.tokenizer_path)
            
        self.logger.info("Model saved.")

    def load(self):
        """
        Loads the lookup tables and neural model.
        Returns True if successful, False otherwise.
        """
        hfbb_exists = os.path.exists(self.trigram_path) and os.path.exists(self.unigram_path)
        
        if hfbb_exists:
            self.logger.info("Loading HFBB tables from cache...")
            self.trigram_map = pd.read_parquet(self.trigram_path)
            self.unigram_map = pd.read_parquet(self.unigram_path)
            
            if os.path.exists(self.neural_path) and os.path.exists(self.tokenizer_path):
                self.logger.info("Loading Neural Model from cache...")
                self.tokenizer.load(self.tokenizer_path)
                self.neural_model = Seq2Seq(
                    vocab_size=len(self.tokenizer),
                    embed_dim=64,
                    hidden_dim=256,
                    pad_idx=self.tokenizer.pad_idx
                ).to(self.device)
                self.neural_model.load_state_dict(torch.load(self.neural_path, map_location=self.device))
            else:
                self.logger.warning("Neural model not found in cache. Proceeding without it.")
                
            self.logger.info("Model loaded successfully.")
            return True
        return False
