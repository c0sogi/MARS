def get_optimizer_params(model, base_lr, head_lr, weight_decay, llrd_decay):
    """
    Constructs parameter groups for Layer-wise Learning Rate Decay (LLRD).
    """
    optimizer_parameters = []
    
    # 1. Head Parameters (High LR)
    head_params_map = list(map(id, model.fc.parameters()))
    optimizer_parameters.append({
        "params": model.fc.parameters(),
        "lr": head_lr,
        "weight_decay": weight_decay
    })
    
    # 2. Backbone Parameters (Decaying LR)
    # DeBERTa-v3-base has 12 layers. 
    # We group params by layer index.
    n_layers = 12
    
    layers = {}
    for name, p in model.backbone.named_parameters():
        if not p.requires_grad:
            continue
            
        # Determine layer index
        if "embeddings" in name:
            idx = 0
        elif "encoder.layer." in name:
            match = re.search(r"encoder\.layer\.(\d+)", name)
            if match:
                idx = int(match.group(1)) + 1
            else:
                idx = 0
        else:
             # Default to 0 (embeddings level) for others
             idx = 0
             
        if idx not in layers:
            layers[idx] = []
        layers[idx].append(p)
        
    for idx, params in layers.items():
        # Distance from top (12)
        # If idx=12 (top layer), dist=0 -> lr = base_lr
        # If idx=0 (embeddings), dist=12 -> lr = base_lr * decay^12
        dist = n_layers - idx
        dist = max(0, dist)
        
        lr = base_lr * (llrd_decay ** dist)
        
        optimizer_parameters.append({
            "params": params,
            "lr": lr,
            "weight_decay": weight_decay
        })
        
    return optimizer_parameters
