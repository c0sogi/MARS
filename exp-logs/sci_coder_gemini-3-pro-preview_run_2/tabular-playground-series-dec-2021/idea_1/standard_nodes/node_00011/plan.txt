def engineer_features(df):
    """
    Applies domain-specific feature engineering to the dataframe.
    """
    df = df.copy()

    # Euclidean distance to hydrology
    if (
        "Horizontal_Distance_To_Hydrology" in df.columns
        and "Vertical_Distance_To_Hydrology" in df.columns
    ):
        df["Euclidean_Distance_To_Hydrology"] = np.sqrt(
            df["Horizontal_Distance_To_Hydrology"] ** 2
            + df["Vertical_Distance_To_Hydrology"] ** 2
        )

    # Elevation at hydrology (Elevation - Vertical Distance)
    if "Elevation" in df.columns and "Vertical_Distance_To_Hydrology" in df.columns:
        df["Elevation_Hydrology"] = (
            df["Elevation"] - df["Vertical_Distance_To_Hydrology"]
        )

    # Aspect Cyclic Features
    if "Aspect" in df.columns:
        df["Aspect_Cos"] = np.cos(np.radians(df["Aspect"]))
        df["Aspect_Sin"] = np.sin(np.radians(df["Aspect"]))

    # Mean Distance to key points
    dist_cols = [
        "Horizontal_Distance_To_Hydrology",
        "Horizontal_Distance_To_Roadways",
        "Horizontal_Distance_To_Fire_Points",
    ]
    existing_dist_cols = [c for c in dist_cols if c in df.columns]
    if existing_dist_cols:
        df["Mean_Distance_To_Points"] = df[existing_dist_cols].mean(axis=1)

    return df
