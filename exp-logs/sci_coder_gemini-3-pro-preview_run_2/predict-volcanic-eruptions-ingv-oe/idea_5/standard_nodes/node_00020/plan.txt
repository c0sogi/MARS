The following modifications implement a strategy based on the lessons learned to optimize performance:

1.  **Single-View Spectrogram & ResNet18 (Cite Lesson 00019)**: Reverting from the multi-view stacked spectrograms (20 channels) and ResNet34 backbone to a single domain-tuned spectrogram view (10 channels) and a lighter **ResNet18** backbone. The previous attempt with ResNet34 and stacked views suffered from instability and poor generalization due to the small dataset size relative to the model complexity.
2.  **SpecAugment (Cite Lesson 00007, 00008)**: Introducing **FrequencyMasking** and **TimeMasking** during training. This regularization is critical for preventing overfitting on the small dataset (~3000 samples) and allowing the model to learn robust features. Conservative masking parameters are used to avoid destroying essential signal transients.
3.  **Extended Training (Cite Lesson 00007)**: Increasing the training duration from 10 to **40 epochs** to allow the model to fully converge, which is necessary when using strong data augmentation like SpecAugment.

def generate_dual_spectrograms(signal_data):
    """
    Generates two Log-Mel Spectrograms (temporal-focus and spectral-focus)
    per sensor and stacks them.

    Args:
        signal_data (pd.DataFrame or np.ndarray): The cleaned sensor data.
                                                  Shape (60001, 10).

    Returns:
        torch.Tensor: Stacked spectrograms with shape (20, 128, Time).
    """
    if isinstance(signal_data, pd.DataFrame):
        signal_data = signal_data.values

    # Convert to Tensor and transpose to (Channels, Time) -> (10, 60001)
    waveform = torch.tensor(signal_data, dtype=torch.float32).T

    # Define Transforms
    # View A: High Temporal Resolution (Short Window)
    spec_transform_short = T.MelSpectrogram(
        sample_rate=Config.SAMPLE_RATE,
        n_fft=Config.N_FFT_SHORT,
        win_length=Config.WIN_LENGTH_SHORT,
        hop_length=Config.HOP_LENGTH,
        n_mels=Config.N_MELS,
        center=True,
        pad_mode="reflect",
        power=2.0,
        norm="slaney",
        mel_scale="slaney",
    )

    # View B: High Frequency Resolution (Long Window)
    # Captures harmonic patterns and continuous rumbling
    spec_transform_long = T.MelSpectrogram(
        sample_rate=Config.SAMPLE_RATE,
        n_fft=Config.N_FFT_LONG,
        win_length=Config.WIN_LENGTH_LONG,
        hop_length=Config.HOP_LENGTH,
        n_mels=Config.N_MELS,
        center=True,
        pad_mode="reflect",
        power=2.0,
        norm="slaney",
        mel_scale="slaney",
    )

    # Amplitude to DB
    db_transform = T.AmplitudeToDB(top_db=Config.TOP_DB)

    # Generate Spectrograms
    # Input: (10, Time) -> Output: (10, n_mels, Time)
    spec_short = spec_transform_short(waveform)
    spec_short = db_transform(spec_short)

    spec_long = spec_transform_long(waveform)
    spec_long = db_transform(spec_long)

    # Stack along channel dimension
    # Result: (20, n_mels, Time)
    # Channels 0-9: Short view, Channels 10-19: Long view
    dual_spec = torch.cat([spec_short, spec_long], dim=0)

    return dual_spec

class HybridResNet34(nn.Module):
    """
    A Hybrid Neural Network that fuses a Multi-Resolution Spectrogram Encoder (ResNet34)
    with a Statistical Feature Encoder (MLP).
    """

    def __init__(self, num_stats_features=140):
        """
        Args:
            num_stats_features (int): Number of input statistical features.
                                      Default is 140 (14 features * 10 sensors).
        """
        super(HybridResNet34, self).__init__()

        # ---------------------------------------------------------------------
        # 1. Multi-Resolution Spectrogram Encoder (ResNet34 Backbone)
        # ---------------------------------------------------------------------
        # Load pretrained weights
        # Using the modern 'weights' parameter for torchvision
        self.backbone = models.resnet34(weights=models.ResNet34_Weights.IMAGENET1K_V1)

        # Modify the first convolutional layer to accept 20 channels
        # Original: Conv2d(3, 64, kernel_size=7, stride=2, padding=3, bias=False)
        original_conv1 = self.backbone.conv1

        # Create new layer with 20 input channels
        new_conv1 = nn.Conv2d(
            in_channels=Config.IN_CHANNELS,
            out_channels=original_conv1.out_channels,
            kernel_size=original_conv1.kernel_size,
            stride=original_conv1.stride,
            padding=original_conv1.padding,
            bias=original_conv1.bias is not None,
        )

        # Initialize weights: Average original RGB weights and replicate 20 times
        # This preserves the spatial filters learned from ImageNet while adapting to 20 channels
        with torch.no_grad():
            # original_conv1.weight shape: [64, 3, 7, 7]
            w_orig = original_conv1.weight
            # Average over channel dimension -> [64, 1, 7, 7]
            w_mean = torch.mean(w_orig, dim=1, keepdim=True)
            # Replicate for new input channels -> [64, 20, 7, 7]
            w_new = w_mean.repeat(1, Config.IN_CHANNELS, 1, 1)
            new_conv1.weight.copy_(w_new)

        self.backbone.conv1 = new_conv1

        # Remove the original Fully Connected (FC) layer
        # We want the output of the Global Average Pooling (GAP) layer
        # ResNet34 GAP output dimension is 512
        self.backbone_out_dim = self.backbone.fc.in_features
        self.backbone.fc = nn.Identity()
