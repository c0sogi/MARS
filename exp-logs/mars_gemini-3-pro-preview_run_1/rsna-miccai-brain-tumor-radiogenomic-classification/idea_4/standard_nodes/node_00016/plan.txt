2.  **Remove Ensemble Inference:** I will simplify the evaluation and prediction loops in `library/engine.py` and `runfile.py` to handle standard 4D tensors `(Batch, Channel, Height, Width)` instead of the 5D ensemble tensors.
3.  **Ensure Reproducibility:** I will add a `seed_worker` function and attach a `torch.Generator` to the DataLoaders in `library/data.py`. This ensures that data loading is deterministic across runs, preventing the stochastic variance that previously made performance evaluation unreliable.

def evaluate(model, dataloader, criterion, device):
    """
    Evaluates the model on the validation set using 2.5D ensemble logic.
    Input images shape: (B, N_SLICES, C, H, W)
    """
    model.eval()
    running_loss = 0.0
    dataset_size = 0

    all_targets = []
    all_preds = []

    # Use BCELoss for validation tracking since we compute probabilities manually
    val_criterion = nn.BCELoss()

    with torch.no_grad():
        for images, targets, _ in dataloader:
            images = images.to(device)
            targets = targets.to(device).unsqueeze(1)  # Shape: (B, 1)

            B, N, C, H, W = images.shape

            # Flatten slices into the batch dimension: (B*N, C, H, W)
            flat_images = images.view(B * N, C, H, W)

            # Forward pass
            flat_logits = model(flat_images)  # (B*N, 1)
            flat_probs = torch.sigmoid(flat_logits)

            # Reshape back to (B, N) and average probabilities across slices
            # This implements the ensemble voting mechanism
            batch_probs = flat_probs.view(B, N).mean(dim=1, keepdim=True)  # (B, 1)

            # Calculate loss on the aggregated predictions
            loss = val_criterion(batch_probs, targets)

            running_loss += loss.item() * B
            dataset_size += B

            all_targets.extend(targets.cpu().numpy())
            all_preds.extend(batch_probs.cpu().numpy())

    total_loss = running_loss / dataset_size

    # Calculate ROC AUC
    # Handle potential edge cases where a batch might have only one class
    try:
        auc_score = roc_auc_score(all_targets, all_preds)
    except ValueError:
        auc_score = 0.5

    return total_loss, auc_score
