import cv2
import numpy as np

def read_dicom(path):
    """
    Reads a DICOM file robustly, falling back to manual binary parsing if pydicom is missing.
    
    Args:
        path (str): Path to the DICOM file.
        
    Returns:
        tuple: (img, (height, width))
            - img (np.ndarray or None): The image array (grayscale).
            - (height, width) (tuple or (None, None)): The dimensions of the image.
    """
    # 1. Try pydicom (Standard)
    try:
        import pydicom
        ds = pydicom.dcmread(path)
        img = ds.pixel_array
        # Ensure grayscale
        if img.ndim == 3:
             # Assume RGB if 3 channels, convert to Gray
             if img.shape[2] == 3:
                 img = cv2.cvtColor(img, cv2.COLOR_RGB2GRAY)
        return img, img.shape[:2]
    except (ImportError, Exception):
        pass

    # 2. Try OpenCV Standard Read (Sometimes works if plugins are present)
    try:
        img = cv2.imread(path, cv2.IMREAD_UNCHANGED)
        if img is not None:
            if img.ndim == 3:
                img = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
            return img, img.shape[:2]
    except Exception:
        pass

    # 3. Manual Binary Parsing (Fallback for encapsulated JPEG/JP2)
    # Cite debug_lesson_1: Implement Low-Level Binary Parsing Fallbacks
    try:
        with open(path, 'rb') as f:
            data = f.read()
        
        # Common headers for embedded images in DICOM
        # JPEG start markers (FF D8)
        # JP2 signature (00 00 00 0C 6A 50 ...)
        # J2K codestream (FF 4F FF 51)
        
        headers = [
            b'\xff\xd8\xff\xe0', # JPEG
            b'\xff\xd8\xff\xe1',
            b'\xff\xd8\xff\xdb',
            b'\x00\x00\x00\x0c\x6a\x50\x20\x20\x0d\x0a\x87\x0a', # JP2
            b'\xff\x4f\xff\x51'  # J2K
        ]
        
        start_offset = -1
        for h in headers:
            idx = data.find(h)
            if idx != -1:
                start_offset = idx
                break
        
        if start_offset != -1:
            img_bytes = np.frombuffer(data[start_offset:], dtype=np.uint8)
            img = cv2.imdecode(img_bytes, cv2.IMREAD_UNCHANGED)
            if img is not None:
                if img.ndim == 3:
                    img = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
                return img, img.shape[:2]
                
    except Exception:
        pass

    return None, (None, None)
