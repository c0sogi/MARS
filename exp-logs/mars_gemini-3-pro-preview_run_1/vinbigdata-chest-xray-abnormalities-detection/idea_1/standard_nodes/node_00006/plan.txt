import os
import numpy as np
import cv2

def read_dicom(path):
    """
    Reads a DICOM file and returns the pixel array and shape.
    Falls back to binary search for image headers if pydicom is missing.
    
    Args:
        path (str): Path to the DICOM file.
        
    Returns:
        tuple: (image_array, (height, width))
               image_array is np.ndarray (grayscale) or None
               (height, width) are ints or None
    """
    if not os.path.exists(path):
        return None, (None, None)

    # 1. Try pydicom (Standard)
    try:
        import pydicom
        ds = pydicom.dcmread(path)
        img = ds.pixel_array
        # Ensure 2D if possible, or let downstream handle it
        if img.ndim == 3 and img.shape[2] == 3:
            img = cv2.cvtColor(img, cv2.COLOR_RGB2GRAY)
        return img, img.shape[:2]
    except (ImportError, Exception):
        pass

    # 2. Manual Binary Parsing (Fallback)
    # Many DICOMs in ML datasets are encapsulated JPEG or JPEG2000
    try:
        with open(path, 'rb') as f:
            data = f.read()
        
        img = None
        
        # Look for JPEG Start of Image (FF D8)
        # We search for the first occurrence.
        jpeg_start = data.find(b'\xff\xd8')
        if jpeg_start != -1:
            img_buf = np.frombuffer(data[jpeg_start:], np.uint8)
            img = cv2.imdecode(img_buf, cv2.IMREAD_GRAYSCALE)

        # Look for JPEG 2000 Codestream (FF 4F FF 51)
        if img is None:
            j2k_start = data.find(b'\xff\x4f\xff\x51')
            if j2k_start != -1:
                img_buf = np.frombuffer(data[j2k_start:], np.uint8)
                img = cv2.imdecode(img_buf, cv2.IMREAD_GRAYSCALE)

        if img is not None:
            return img, img.shape[:2]
            
    except Exception:
        pass

    # 3. OpenCV Direct Read (Last Resort)
    try:
        img = cv2.imread(path, cv2.IMREAD_GRAYSCALE)
        if img is not None:
            return img, img.shape[:2]
    except Exception:
        pass

    return None, (None, None)
