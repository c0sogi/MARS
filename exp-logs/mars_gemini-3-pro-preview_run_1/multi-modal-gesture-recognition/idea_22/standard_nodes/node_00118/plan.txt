class MultiScaleStem(nn.Module):
    """
    Decoupled Multi-Scale Stem processing modality-specific data via parallel temporal branches.
    Captures dynamics at different resolutions (short, medium, long).
    """

    def __init__(self, in_channels, out_channels_per_branch, kernel_sizes):
        super(MultiScaleStem, self).__init__()
        self.branches = nn.ModuleList()

        for k in kernel_sizes:
            # Padding is k // 2 to maintain temporal resolution (same padding)
            self.branches.append(
                nn.Conv1d(
                    in_channels, out_channels_per_branch, kernel_size=k, padding=k // 2
                )
            )

        self.activation = nn.ReLU()
        self.dropout = nn.Dropout(Config.DROPOUT)

    def forward(self, x):
        # x: (Batch, Time, Channels) -> (Batch, Channels, Time) for Conv1d
        x = x.transpose(1, 2)

        branch_outputs = []
        for branch in self.branches:
            out = branch(x)
            branch_outputs.append(out)

        # Concatenate along channel dimension: (Batch, Channels * num_branches, Time)
        x_cat = torch.cat(branch_outputs, dim=1)

        # Transpose back: (Batch, Time, Channels_new)
        x_cat = x_cat.transpose(1, 2)

        return self.dropout(self.activation(x_cat))

class SingleScaleStem(nn.Module):
    """
    Single-Scale Wide Stem.
    Prioritizes channel capacity over multi-scale complexity (Cite solution_lesson_node_00117).
    """

    def __init__(self, in_channels, out_channels, kernel_size):
        super(SingleScaleStem, self).__init__()
        self.conv = nn.Conv1d(
            in_channels, out_channels, kernel_size=kernel_size, padding=kernel_size // 2
        )
        self.activation = nn.ReLU()
        self.dropout = nn.Dropout(Config.DROPOUT)

    def forward(self, x):
        # x: (Batch, Time, Channels) -> (Batch, Channels, Time)
        x = x.transpose(1, 2)
        x = self.conv(x)
        x = x.transpose(1, 2)
        return self.dropout(self.activation(x))
