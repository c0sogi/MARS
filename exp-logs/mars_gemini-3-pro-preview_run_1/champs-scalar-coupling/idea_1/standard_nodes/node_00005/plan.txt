    """
    Generates node-level features for each atom in the structures.
    Features: Electronegativity, Bond Count, Min/Mean Distance to neighbors.
    Cite solution_lesson_node_00003
    """
    # 1. Electronegativity
    en_map = {"H": 2.20, "C": 2.55, "N": 3.04, "O": 3.44, "F": 3.98}
    structures = structures.copy()
    structures["en"] = structures["atom"].map(en_map)

    # 2. Geometric/Topological Features via Self-Join
    # Select coordinates
    coords = structures[["molecule_name", "atom_index", "x", "y", "z"]]

    # Merge to get all pairs within molecule
    pairs = pd.merge(coords, coords, on="molecule_name", suffixes=("", "_neigh"))

    # Remove self-loops
    pairs = pairs[pairs["atom_index"] != pairs["atom_index_neigh"]]

    # Calculate distances
    pairs["dist"] = np.sqrt(
        (pairs["x"] - pairs["x_neigh"]) ** 2
        + (pairs["y"] - pairs["y_neigh"]) ** 2

    """
    Generates node-level features for each atom in the structures.
    Features: Electronegativity, Bond Count, Min/Mean Distance to neighbors.
    Cite solution_lesson_node_00003

    structures = structures.copy()
    structures["en"] = structures["atom"].map(en_map)

    # 2. Geometric/Topological Features via Self-Join
    # Select coordinates AND atom type to identify neighbor types
    coords = structures[["molecule_name", "atom_index", "x", "y", "z", "atom"]]

    # Merge to get all pairs within molecule (Cite solution_lesson_node_00004)
    pairs = pd.merge(coords, coords, on="molecule_name", suffixes=("", "_neigh"))

    # Remove self-loops
    pairs = pairs[pairs["atom_index"] != pairs["atom_index_neigh"]]

    # Calculate distances
    pairs["dist"] = np.sqrt(
        (pairs["x"] - pairs["x_neigh"]) ** 2
        + (pairs["y"] - pairs["y_neigh"]) ** 2
        + (pairs["z"] - pairs["z_neigh"]) ** 2
    )

    # Define bonds (simple distance threshold)
    BOND_THRESHOLD = 1.6  # Angstroms

    # Vectorized calculation of bond flags
    pairs["is_bond"] = pairs["dist"] < BOND_THRESHOLD

    # Create flags for specific neighbor types (Bag of Neighbors)
    # This allows us to count "How many C neighbors?", "How many H neighbors?" etc.
    for at in ATOM_TYPES:

    # Aggregate
    # Use dictionary unpacking for cleaner syntax and avoid slow lambdas
    aggs = {
        "n_bonds": ("is_bond", "sum"),
        "min_dist_neigh": ("dist", "min"),
        "mean_dist_neigh": ("dist", "mean"),

    # Add aggregations for specific neighbor counts
    for at in ATOM_TYPES:
        aggs[f"n_{at}"] = (f"is_bond_{at}", "sum")

        on=["molecule_name", "atom_index"],
        how="left",
    )

    # Fill NaNs for isolated atoms
    atom_features.fillna(0, inplace=True)

    return atom_features
