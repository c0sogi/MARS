import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import Dataset, DataLoader
import pandas as pd
import numpy as np
import random
from library.config import Config
from library.utils import setup_logger

logger = setup_logger("NeuralModel")

# --- Constants ---
SOS_TOKEN = "<sos>"
EOS_TOKEN = "<eos>"
PAD_TOKEN = "<pad>"
UNK_TOKEN = "<unk>"

class CharTokenizer:
    """
    Character-level tokenizer for Seq2Seq model.
    """
    def __init__(self):
        self.char2idx = {}
        self.idx2char = {}
        self.vocab_size = 0
        self.specials = [PAD_TOKEN, SOS_TOKEN, EOS_TOKEN, UNK_TOKEN]

    def fit(self, texts):
        """Builds vocabulary from a list of strings."""
        unique_chars = set()
        for text in texts:
            unique_chars.update(str(text))
        
        # Sort for determinism
        sorted_chars = sorted(list(unique_chars))
        
        # Add specials first
        self.idx2char = {i: char for i, char in enumerate(self.specials + sorted_chars)}
        self.char2idx = {char: i for i, char in self.idx2char.items()}
        self.vocab_size = len(self.idx2char)
        
    def encode(self, text, add_special_tokens=True):
        """Converts string to list of indices."""
        text = str(text)
        indices = [self.char2idx.get(c, self.char2idx[UNK_TOKEN]) for c in text]
        if add_special_tokens:
            indices = [self.char2idx[SOS_TOKEN]] + indices + [self.char2idx[EOS_TOKEN]]
        return indices

    def decode(self, indices):
        """Converts list of indices to string."""
        chars = []
        for idx in indices:
            char = self.idx2char.get(idx, UNK_TOKEN)
            if char == EOS_TOKEN:
                break
            if char not in [SOS_TOKEN, PAD_TOKEN]:
                chars.append(char)
        return "".join(chars)
    
    def __len__(self):
        return self.vocab_size

class TextDataset(Dataset):
    def __init__(self, data, tokenizer):
        self.data = data
        self.tokenizer = tokenizer

    def __len__(self):
        return len(self.data)

    def __getitem__(self, idx):
        src, trg = self.data[idx]
        return (
            torch.tensor(self.tokenizer.encode(src), dtype=torch.long),
            torch.tensor(self.tokenizer.encode(trg), dtype=torch.long)
        )

def collate_fn(batch):
    src_batch, trg_batch = zip(*batch)
    src_pad = torch.nn.utils.rnn.pad_sequence(src_batch, padding_value=0, batch_first=True)
    trg_pad = torch.nn.utils.rnn.pad_sequence(trg_batch, padding_value=0, batch_first=True)
    return src_pad, trg_pad

class Seq2SeqModel(nn.Module):
    def __init__(self, vocab_size, embed_dim, hidden_dim, device):
        super(Seq2SeqModel, self).__init__()
        self.device = device
        self.vocab_size = vocab_size
        
        self.embedding = nn.Embedding(vocab_size, embed_dim, padding_idx=0)
        self.encoder = nn.GRU(embed_dim, hidden_dim, batch_first=True)
        self.decoder = nn.GRU(embed_dim, hidden_dim, batch_first=True)
        self.fc = nn.Linear(hidden_dim, vocab_size)
        
    def forward(self, src, trg, teacher_forcing_ratio=0.5):
        # src: [batch, src_len]
        # trg: [batch, trg_len]
        
        batch_size = src.shape[0]
        trg_len = trg.shape[1]
        
        # Encoder
        embedded_src = self.embedding(src)
        _, hidden = self.encoder(embedded_src)
        
        # Decoder
        outputs = torch.zeros(batch_size, trg_len, self.vocab_size).to(self.device)
        decoder_input = trg[:, 0].unsqueeze(1) # <sos>
        
        for t in range(1, trg_len):
            embedded_trg = self.embedding(decoder_input)
            output, hidden = self.decoder(embedded_trg, hidden)
            prediction = self.fc(output)
            outputs[:, t, :] = prediction.squeeze(1)
            
            if random.random() < teacher_forcing_ratio:
                decoder_input = trg[:, t].unsqueeze(1)
            else:
                top1 = prediction.argmax(2)
                decoder_input = top1
                
        return outputs

    def predict_step(self, src, max_len=100, sos_idx=1, eos_idx=2):
        # src: [1, src_len]
        self.eval()
        with torch.no_grad():
            embedded_src = self.embedding(src)
            _, hidden = self.encoder(embedded_src)
            
            decoder_input = torch.tensor([[sos_idx]], device=self.device)
            result_indices = []
            
            for _ in range(max_len):
                embedded_trg = self.embedding(decoder_input)
                output, hidden = self.decoder(embedded_trg, hidden)
                prediction = self.fc(output)
                top1 = prediction.argmax(2)
                
                idx = top1.item()
                if idx == eos_idx:
                    break
                result_indices.append(idx)
                decoder_input = top1
                
        return result_indices

def train_neural_module(df, device_name="cuda", epochs=3, batch_size=256, limit_samples=50000):
    """
    Trains a Seq2Seq model on the provided dataframe.
    """
    logger.info("Initializing Neural Module training...")
    
    # Filter for interesting cases to speed up training (e.g. where input != output)
    # or just sample to keep it within time limits.
    # The task is normalization, so we care most about things that change.
    
    mask_change = df[Config.INPUT_COL] != df[Config.TARGET_COL]
    df_changed = df[mask_change]
    
    # If we don't have enough changed samples, mix in some unchanged
    if len(df_changed) < limit_samples:
        df_sample = pd.concat([
            df_changed, 
            df[~mask_change].sample(n=min(len(df[~mask_change]), limit_samples - len(df_changed)))
        ])
    else:
        df_sample = df_changed.sample(n=limit_samples, random_state=42)
        
    logger.info(f"Training on {len(df_sample)} samples.")
    
    inputs = df_sample[Config.INPUT_COL].astype(str).tolist()
    targets = df_sample[Config.TARGET_COL].astype(str).tolist()
    
    # Tokenizer
    tokenizer = CharTokenizer()
    tokenizer.fit(inputs + targets)
    
    # Dataset
    data_pairs = list(zip(inputs, targets))
    dataset = TextDataset(data_pairs, tokenizer)
    dataloader = DataLoader(dataset, batch_size=batch_size, shuffle=True, collate_fn=collate_fn)
    
    # Model
    device = torch.device(device_name)
    model = Seq2SeqModel(len(tokenizer), embed_dim=64, hidden_dim=256, device=device).to(device)
    
    optimizer = optim.Adam(model.parameters(), lr=0.001)
    criterion = nn.CrossEntropyLoss(ignore_index=0) # Pad index
    
    model.train()
    for epoch in range(epochs):
        total_loss = 0
        for src, trg in dataloader:
            src, trg = src.to(device), trg.to(device)
            
            optimizer.zero_grad()
            output = model(src, trg)
            
            # output: [batch, trg_len, vocab] -> [batch * trg_len, vocab]
            # trg: [batch, trg_len] -> [batch * trg_len]
            # Exclude SOS from target comparison (it's input to decoder)
            # Actually standard way: output[:, 1:] vs trg[:, 1:]
            
            output_dim = output.shape[-1]
            output = output[:, 1:].reshape(-1, output_dim)
            trg = trg[:, 1:].reshape(-1)
            
            loss = criterion(output, trg)
            loss.backward()
            optimizer.step()
            
            total_loss += loss.item()
            
        logger.info(f"Epoch {epoch+1}/{epochs}, Loss: {total_loss/len(dataloader):.4f}")
        
    return model, tokenizer
