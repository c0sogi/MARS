import torch

import torch.optim as optim
from torch.utils.data import Dataset, DataLoader
import random
import numpy as np

class CharTokenizer:
    """
    Character-level tokenizer for Seq2Seq models.
    Handles mapping between characters and integer indices.
    """
    def __init__(self):
        self.char2idx = {}
        self.idx2char = {}
        self.pad_token = "<PAD>"
        self.sos_token = "<SOS>"
        self.eos_token = "<EOS>"
        self.unk_token = "<UNK>"
        self.specials = [self.pad_token, self.sos_token, self.eos_token, self.unk_token]
        
    def fit(self, text_list):
        """
        Builds the vocabulary from a list of strings.
        """
        unique_chars = set("".join(str(t) for t in text_list))
        # Add specials first
        for s in self.specials:
            self.char2idx[s] = len(self.char2idx)
            self.idx2char[self.char2idx[s]] = s
            
        for c in sorted(list(unique_chars)):
            if c not in self.char2idx:
                self.char2idx[c] = len(self.char2idx)
                self.idx2char[self.char2idx[c]] = c
                
    def encode(self, text, add_special_tokens=True):
        """
        Converts a string to a list of indices.
        """
        indices = [self.char2idx.get(c, self.char2idx[self.unk_token]) for c in str(text)]
        if add_special_tokens:
            indices = [self.char2idx[self.sos_token]] + indices + [self.char2idx[self.eos_token]]
        return indices
    
    def decode(self, indices):
        """
        Converts a list of indices back to a string.
        """
        chars = []
        for idx in indices:
            char = self.idx2char.get(idx, self.unk_token)
            if char == self.eos_token:
                break
            if char not in self.specials:
                chars.append(char)
        return "".join(chars)
        
    def __len__(self):
        return len(self.char2idx)

class Seq2SeqModel(nn.Module):
    """
    Standard Encoder-Decoder GRU model.
    """
    def __init__(self, vocab_size, embed_dim, hidden_dim, device):
        super(Seq2SeqModel, self).__init__()
        self.vocab_size = vocab_size
        self.embed_dim = embed_dim
        self.hidden_dim = hidden_dim
        self.device = device
        
        self.embedding = nn.Embedding(vocab_size, embed_dim)
        self.encoder = nn.GRU(embed_dim, hidden_dim, batch_first=True)
        self.decoder = nn.GRU(embed_dim, hidden_dim, batch_first=True)
        self.fc = nn.Linear(hidden_dim, vocab_size)
        
    def forward(self, src, trg, teacher_forcing_ratio=0.5):
        # src: [batch, src_len]
        # trg: [batch, trg_len]
        batch_size = src.shape[0]
        trg_len = trg.shape[1]
        
        embedded_src = self.embedding(src)
        _, hidden = self.encoder(embedded_src)
        
        decoder_input = trg[:, 0].unsqueeze(1) # SOS
        outputs = torch.zeros(batch_size, trg_len, self.vocab_size).to(self.device)
        
        for t in range(1, trg_len):
            embedded_trg = self.embedding(decoder_input)
            output, hidden = self.decoder(embedded_trg, hidden)
            prediction = self.fc(output)
            outputs[:, t, :] = prediction.squeeze(1)
            
            top1 = prediction.argmax(2)
            
            # Teacher forcing
            if random.random() < teacher_forcing_ratio:
                decoder_input = trg[:, t].unsqueeze(1)
            else:
                decoder_input = top1
                
        return outputs

class TextDataset(Dataset):
    def __init__(self, data, tokenizer):
        self.data = data
        self.tokenizer = tokenizer
        
    def __len__(self):
        return len(self.data)
        
    def __getitem__(self, idx):
        src_txt, trg_txt = self.data[idx]
        src = self.tokenizer.encode(src_txt)
        trg = self.tokenizer.encode(trg_txt)
        return torch.tensor(src), torch.tensor(trg)

def collate_fn(batch):
    """
    Pads sequences in the batch to the maximum length.
    """
    src_list, trg_list = zip(*batch)
    src_lens = [len(s) for s in src_list]
    trg_lens = [len(t) for t in trg_list]
    
    max_src = max(src_lens)
    max_trg = max(trg_lens)
    
    # Pad with 0 (assuming 0 is PAD, need to verify in tokenizer)
    # In CharTokenizer, PAD is the first special token added, so index 0.
    
    src_padded = torch.zeros(len(batch), max_src, dtype=torch.long)
    trg_padded = torch.zeros(len(batch), max_trg, dtype=torch.long)
    
    for i, (s, t) in enumerate(zip(src_list, trg_list)):
        src_padded[i, :len(s)] = s
        trg_padded[i, :len(t)] = t
        
    return src_padded, trg_padded

def train_neural_module(df, device_name="cuda", epochs=5, batch_size=256):

    Trains the Seq2Seq model on the provided dataframe.

    # Filter: train on tokens where input != target (normalization happened)

    mask_change = df[Config.INPUT_COL] != df[Config.TARGET_COL]

    # Sample some identity for stability
    n_identity = min(len(df[~mask_change]), len(df_change) // 2)
    if n_identity > 0:
        df_identity = df[~mask_change].sample(n=n_identity, random_state=42)
        train_data = pd.concat([df_change, df_identity])
    else:
        train_data = df_change

        print("Warning: No training data for neural model found. Returning None.")

    src_texts = train_data[Config.INPUT_COL].astype(str).tolist()

    model.train()
    print(f"Training Neural Model on {len(train_data)} samples...")
    for epoch in range(epochs):
        total_loss = 0
        for src, trg in dataloader:
            src, trg = src.to(device_name), trg.to(device_name)
            
            optimizer.zero_grad()
            output = model(src, trg)
            
            # output: [batch, trg_len, vocab]
            # trg: [batch, trg_len]

            output_dim = output.shape[-1]
            output = output[:, 1:].reshape(-1, output_dim)
            trg = trg[:, 1:].reshape(-1)
            
            loss = criterion(output, trg)
            loss.backward()
            optimizer.step()
            
            total_loss += loss.item()
        print(f"Epoch {epoch+1}, Loss: {total_loss/len(dataloader):.4f}")
        
    return model, tokenizer
