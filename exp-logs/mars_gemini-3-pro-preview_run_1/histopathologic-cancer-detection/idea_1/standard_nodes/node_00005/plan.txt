def predict_and_submit_tta(model, loader, device, output_path):
    """
    Generates predictions for the test set using Test Time Augmentation (TTA)
    and saves them to a CSV file.
    """
    model.eval()
    ids_list = []
    preds_list = []

    print("Generating predictions with TTA (4 views)...")
    with torch.no_grad():
        for images, ids in loader:
            images = images.to(device)
            
            # 1. Original
            out1 = model(images)
            p1 = torch.sigmoid(out1)
            
            # 2. Horizontal Flip
            out2 = model(torch.flip(images, [3]))
            p2 = torch.sigmoid(out2)
            
            # 3. Vertical Flip
            out3 = model(torch.flip(images, [2]))
            p3 = torch.sigmoid(out3)
            
            # 4. Rotate 90
            out4 = model(torch.rot90(images, 1, [2, 3]))
            p4 = torch.sigmoid(out4)
            
            # Average probabilities
            probs = (p1 + p2 + p3 + p4) / 4.0
            probs = probs.cpu().numpy().flatten()

            ids_list.extend(ids)
            preds_list.extend(probs)

    # Ensure output directory exists
    os.makedirs(os.path.dirname(output_path), exist_ok=True)

    df = pd.DataFrame({"id": ids_list, "label": preds_list})
    df.to_csv(output_path, index=False)
    print(f"Submission saved to {output_path}")
