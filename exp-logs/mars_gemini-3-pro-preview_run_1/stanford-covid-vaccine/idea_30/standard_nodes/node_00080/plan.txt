My strategy is to revert the "Zoneout" regularization and "Global Context Initialization" (if present) which were identified as causes for performance regression in **Lesson 00076** and **Lesson 00059**. I will restore the "Current Best" architecture: a **Scalar-Aggregated Wide-Stream Residual BiGRU** with standard **Inter-layer Dropout**.

Specifically:
1.  **Architecture**: Replace the custom `ZoneoutBiGRUBlock` (which uses slow Python loops over `GRUCell`) with a standard `BiGRUBlock` using `nn.GRU`. This leverages cuDNN for speed and applies standard dropout, which outperformed Zoneout (0.620 vs 0.741 MCRMSE, **Cite solution_lesson_node_00076**).
2.  **Hyperparameters**:
    -   Set `HIDDEN_DIM = 384`. **Lesson 00022** highlights the criticality of width, but **Lesson 00076** shows 384 was sufficient for the best result, while 512 with Zoneout failed. 384 is a safer, proven capacity.
    -   Set `DROPOUT = 0.1`.
    -   Set `EPOCHS = 20`. The faster `nn.GRU` implementation allows for more epochs within the time limit.
3.  **Aggregation**: Retain `ScalarMixture` as it outperformed channel-wise and dense aggregation (**Cite solution_lesson_node_00073**, **solution_lesson_node_00055**).

class ZoneoutBiGRUBlock(nn.Module):
    """
    A Bidirectional GRU block with Zoneout regularization.
    Unrolls the sequence to apply stochastic state updates during training.
    """

    def __init__(
        self,
        input_size: int,
        hidden_size: int,
        zoneout_prob: float = Config.ZONEOUT_PROB,
    ):
        super().__init__()
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.zoneout_prob = zoneout_prob

        # Forward and Backward Cells
        self.fwd_cell = nn.GRUCell(input_size, hidden_size)
        self.bwd_cell = nn.GRUCell(input_size, hidden_size)

    def forward(
        self, x: torch.Tensor, h_0: Optional[torch.Tensor] = None
    ) -> torch.Tensor:
        """
        Args:
            x (torch.Tensor): Input sequence of shape (Batch, Seq_Len, Input_Size).
            h_0 (torch.Tensor, optional): Initial hidden state (Batch, Hidden_Size * 2).

        Returns:
            torch.Tensor: Output sequence of shape (Batch, Seq_Len, Hidden_Size * 2).
        """
        batch_size, seq_len, _ = x.size()

        # Initialize hidden states
        if h_0 is None:
            h_fwd = torch.zeros(
                batch_size, self.hidden_size, device=x.device, dtype=x.dtype
            )
            h_bwd = torch.zeros(
                batch_size, self.hidden_size, device=x.device, dtype=x.dtype
            )
        else:
            # Assuming h_0 is concatenated [h_fwd, h_bwd]
            h_fwd = h_0[:, : self.hidden_size]
            h_bwd = h_0[:, self.hidden_size :]

        outputs_fwd = []
        outputs_bwd = []

        # Pre-generate masks for efficiency if training
        # We need masks for every step: (Seq_Len, Batch, Hidden_Size)
        use_zoneout = self.training and self.zoneout_prob > 0.0

        # Forward Pass
        for t in range(seq_len):
            x_t = x[:, t, :]
            h_next = self.fwd_cell(x_t, h_fwd)

            if use_zoneout:
                # Zoneout: h_t = d_t * h_next + (1 - d_t) * h_prev
                # d_t ~ Bernoulli(1 - prob) (1 means keep new, 0 means keep old)
                # We sample a mask where 1 = update, 0 = keep old
                mask = torch.bernoulli(torch.full_like(h_next, 1.0 - self.zoneout_prob))
                h_fwd = mask * h_next + (1 - mask) * h_fwd
            elif self.zoneout_prob > 0.0:
                # Evaluation: Expectation
                # h_t = (1 - p) * h_next + p * h_prev
                h_fwd = (1 - self.zoneout_prob) * h_next + self.zoneout_prob * h_fwd
            else:
                # Standard GRU update
                h_fwd = h_next

            outputs_fwd.append(h_fwd)

        # Backward Pass
        for t in range(seq_len - 1, -1, -1):
            x_t = x[:, t, :]
            h_next = self.bwd_cell(x_t, h_bwd)

            if use_zoneout:
                mask = torch.bernoulli(torch.full_like(h_next, 1.0 - self.zoneout_prob))
                h_bwd = mask * h_next + (1 - mask) * h_bwd
            elif self.zoneout_prob > 0.0:
                h_bwd = (1 - self.zoneout_prob) * h_next + self.zoneout_prob * h_bwd
            else:
                h_bwd = h_next

            outputs_bwd.insert(0, h_bwd)

        # Stack and Concatenate
        # outputs_fwd: List of (B, H) -> (B, L, H)
        out_fwd = torch.stack(outputs_fwd, dim=1)
        out_bwd = torch.stack(outputs_bwd, dim=1)

        # (B, L, 2*H)
        output = torch.cat([out_fwd, out_bwd], dim=2)

        return output

class BiGRUBlock(nn.Module):
    """
    Standard Bidirectional GRU Block with Dropout.
    Uses cuDNN implementation for speed.
    """

    def __init__(
        self,
        input_size: int,
        hidden_size: int,
        dropout: float = Config.DROPOUT,
    ):
        super().__init__()
        self.gru = nn.GRU(
            input_size=input_size,
            hidden_size=hidden_size,
            batch_first=True,
            bidirectional=True,
        )
        self.dropout = nn.Dropout(dropout)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        # x: (Batch, Seq, Input_Size)
        # out: (Batch, Seq, Hidden_Size * 2)
        out, _ = self.gru(x)
        return self.dropout(out)
