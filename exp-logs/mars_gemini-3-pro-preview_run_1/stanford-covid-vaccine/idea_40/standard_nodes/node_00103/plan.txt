class InputNormedWideResBiGRU(nn.Module):
    """
    Input-Normed Wide-Stream Residual BiGRU.

    Architecture Strategy:
    1. Proportional Embeddings (Seq, Loop, Dist).
    2. Independent Layer Normalization for each embedding channel to stabilize input distribution.
    3. BiGRU Stem projecting to a wide hidden dimension (512).
    4. 6x Wide-Stream Residual BiGRU Blocks (Pre-LN, Dropout) maintaining full width.
    5. Scalar Mixture Aggregation to combine multi-level features.
    6. Output Head for 3 target channels.
    """

    def __init__(self):
        super().__init__()

        # --- 1. Embeddings ---
        # Sequence: 4 tokens (A, G, C, U)
        self.seq_embed = nn.Embedding(4, Config.EMBED_DIM_SEQ)

        # Loop Type: 7 tokens (S, M, I, B, H, E, X)
        self.loop_embed = nn.Embedding(7, Config.EMBED_DIM_LOOP)

        # Distance: Fixed Sinusoidal Encoding
        self.dist_embed = SinusoidalDistanceEncoding(Config.EMBED_DIM_DIST)

        # --- 2. Independent Feature Normalization ---
        # Normalize each modality independently before fusion
        self.norm_seq = nn.LayerNorm(Config.EMBED_DIM_SEQ)
        self.norm_loop = nn.LayerNorm(Config.EMBED_DIM_LOOP)
        self.norm_dist = nn.LayerNorm(Config.EMBED_DIM_DIST)

        # Calculate fused input dimension
        stem_input_dim = (
            Config.EMBED_DIM_SEQ + Config.EMBED_DIM_LOOP + Config.EMBED_DIM_DIST
        )

        # --- 3. Recurrent Stem ---
        # Projects concatenated inputs to the residual stream width (512)
        # Bidirectional GRU: hidden_size = HIDDEN_DIM // 2
        self.stem = nn.GRU(
            input_size=stem_input_dim,
            hidden_size=Config.HIDDEN_DIM // 2,
            batch_first=True,
            bidirectional=True,
        )
