class PreNormBiGRU(nn.Module):
    """
    Pre-LayerNorm Bidirectional GRU component.
    Replaces the previous LayerNormBiLSTM.
    """

    def __init__(self, input_dim, hidden_dim):
        super(PreNormBiGRU, self).__init__()
        self.ln = nn.LayerNorm(input_dim)
        self.gru = nn.GRU(
            input_size=input_dim,
            hidden_size=hidden_dim,
            batch_first=True,
            bidirectional=True,
        )

    def forward(self, x):
        x_norm = self.ln(x)
        out, _ = self.gru(x_norm)
        return out
