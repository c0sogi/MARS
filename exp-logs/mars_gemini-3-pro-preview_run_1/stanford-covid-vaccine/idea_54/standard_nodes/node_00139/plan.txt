class ParallelBlock(nn.Module):
    """
    Local-Global Parallel Block.
    Processes input through two parallel branches:
    1. Global: BiGRU to capture long-range dependencies.
    2. Local: Depthwise Conv1D to capture local k-mer motifs.
    Outputs are summed, followed by Dropout and Residual Connection.
    """

    def __init__(self, hidden_dim, dropout=0.2):
        super().__init__()

        # Global Branch: BiGRU
        # Hidden size is halved per direction to maintain total width
        self.gru = nn.GRU(
            input_size=hidden_dim,
            hidden_size=hidden_dim // 2,
            batch_first=True,
            bidirectional=True,
        )

        # Local Branch: Depthwise Conv1d
        # Kernel size 3, padding 1 preserves length
        # Groups=hidden_dim makes it depthwise (channel-independent)
        self.conv = nn.Sequential(
            nn.Conv1d(
                in_channels=hidden_dim,
                out_channels=hidden_dim,
                kernel_size=3,
                padding=1,
                groups=hidden_dim,
            ),
            nn.GELU(),
        )

        self.dropout = nn.Dropout(dropout)
        self.norm = nn.LayerNorm(hidden_dim)

    def forward(self, x):
        # x: (B, L, C)

        # Global Branch
        gru_out, _ = self.gru(x)  # (B, L, C)

        # Local Branch
        # Conv1d expects (B, C, L)
        conv_in = x.permute(0, 2, 1)
        conv_out = self.conv(conv_in)
        conv_out = conv_out.permute(0, 2, 1)  # (B, L, C)

        # Fusion: Sum branches
        fused = gru_out + conv_out

        # Dropout
        fused = self.dropout(fused)

        # Residual + Norm
        out = self.norm(x + fused)
        return out

class ResBiGRUBlock(nn.Module):
    """
    Stabilized Residual BiGRU Block with Pre-LayerNorm.
    Cite solution_lesson_node_00135: Superiority of Pre-LayerNorm Residual Blocks.
    Cite solution_lesson_node_00138: Efficacy of Pure Recurrent Backbones.
    """

    def __init__(self, hidden_dim, dropout=0.1):
        super().__init__()
        self.norm = nn.LayerNorm(hidden_dim)
        self.gru = nn.GRU(
            input_size=hidden_dim,
            hidden_size=hidden_dim // 2,
            batch_first=True,
            bidirectional=True,
        )
        self.dropout = nn.Dropout(dropout)

    def forward(self, x):
        # Pre-LayerNorm Residual connection
        res = x
        out = self.norm(x)
        out, _ = self.gru(out)
        out = self.dropout(out)
        return res + out
