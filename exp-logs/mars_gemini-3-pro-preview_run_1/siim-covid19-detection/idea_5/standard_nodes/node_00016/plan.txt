class AttentionBlock(nn.Module):
    """
    Attention Gate for U-Net skip connections.
    Uses a coarser gating signal (g) to filter the features from the skip connection (x).
    """

    def __init__(self, F_g, F_l, F_int):
        super().__init__()
        self.W_g = nn.Sequential(
            nn.Conv2d(F_g, F_int, kernel_size=1, stride=1, padding=0, bias=True),
            nn.BatchNorm2d(F_int),
        )

        self.W_x = nn.Sequential(
            nn.Conv2d(F_l, F_int, kernel_size=1, stride=1, padding=0, bias=True),
            nn.BatchNorm2d(F_int),
        )

        self.psi = nn.Sequential(
            nn.Conv2d(F_int, 1, kernel_size=1, stride=1, padding=0, bias=True),
            nn.BatchNorm2d(1),
            nn.Sigmoid(),
        )

        self.relu = nn.ReLU(inplace=True)

    def forward(self, g, x):
        # g: gating signal (from coarser layer)
        # x: skip connection (from finer layer)

        g1 = self.W_g(g)
        x1 = self.W_x(x)

        # Downsample x1 to match g1 spatial dimensions for addition
        if g1.shape[2:] != x1.shape[2:]:
            x1 = F.interpolate(
                x1, size=g1.shape[2:], mode="bilinear", align_corners=False
            )

        # Combine and activate
        psi = self.relu(g1 + x1)
        psi = self.psi(psi)

        # Upsample attention map back to x size
        if psi.shape[2:] != x.shape[2:]:
            psi = F.interpolate(
                psi, size=x.shape[2:], mode="bilinear", align_corners=False
            )

        return x * psi


class UpBlock(nn.Module):
    """
    Decoder Block with Attention Gate.
    Flow: Attention(Skip, Input) -> Upsample(Input) -> Concat -> ConvBlock
    """

    def __init__(self, in_c, skip_c, out_c):
        super().__init__()
        self.up = nn.Upsample(scale_factor=2, mode="bilinear", align_corners=False)
        # Internal channels for attention usually half of skip channels or similar
        self.attn = AttentionBlock(F_g=in_c, F_l=skip_c, F_int=skip_c // 2)
        self.conv = ConvBlock(in_c + skip_c, out_c)

    def forward(self, x, skip):
        # x: input from previous decoder layer (coarser)
        # skip: skip connection from encoder (finer)

        # 1. Apply Attention Gate to skip connection
        # We use 'x' (before upsampling) as the gating signal
        skip_att = self.attn(g=x, x=skip)

        # 2. Upsample input
        x_up = self.up(x)

        # 3. Handle potential padding issues (if odd dimensions)
        if x_up.shape[2:] != skip_att.shape[2:]:
            x_up = F.interpolate(
                x_up, size=skip_att.shape[2:], mode="bilinear", align_corners=False
            )

        # 4. Concatenate
        x_cat = torch.cat([x_up, skip_att], dim=1)

        # 5. Convolve
        return self.conv(x_cat)
