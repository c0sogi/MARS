import sys
import traceback
import torch
from torch.utils.data import DataLoader
from library.config import TrainingConfig, DataConfig, ProjectConfig
from library.data_utils import (
    load_dataset_raw, 
    build_vocabularies, 
    TaggerDataset, 
    Seq2SeqDataset, 
    collate_fn_tagger, 
    collate_fn_seq2seq
)
from library.models import MultiGranularityTagger, Seq2SeqNormalizer
from library.engine import Trainer

def run_tagger_task(config_overrides):
    """
    Isolated task for training the Tagger model.
    """
    try:
        # Apply configuration overrides
        for k, v in config_overrides.items():
            if hasattr(TrainingConfig, k):
                setattr(TrainingConfig, k, v)
        
        print(f"\n[Subprocess] Starting Tagger Task (Epochs: {TrainingConfig.TAGGER_EPOCHS})")
        
        # Load Vocabularies
        vocab_words, vocab_chars, vocab_classes = build_vocabularies(load_cached_data=True)
        
        # Load Data
        df_train = load_dataset_raw("train")
        df_val = load_dataset_raw("val")
        
        train_ds = TaggerDataset(df_train, vocab_words, vocab_chars, vocab_classes)
        val_ds = TaggerDataset(df_val, vocab_words, vocab_chars, vocab_classes)
        
        train_loader = DataLoader(
            train_ds, 
            batch_size=TrainingConfig.TAGGER_BATCH_SIZE, 
            shuffle=True, 
            collate_fn=collate_fn_tagger, 
            num_workers=DataConfig.NUM_WORKERS,
            pin_memory=DataConfig.PIN_MEMORY
        )
        val_loader = DataLoader(
            val_ds, 
            batch_size=TrainingConfig.TAGGER_BATCH_SIZE, 
            shuffle=False, 
            collate_fn=collate_fn_tagger, 
            num_workers=DataConfig.NUM_WORKERS,
            pin_memory=DataConfig.PIN_MEMORY
        )
        
        # Initialize Model
        model = MultiGranularityTagger(len(vocab_words), len(vocab_chars), len(vocab_classes))
        
        # Train
        trainer = Trainer()
        trainer.train_tagger(model, train_loader, val_loader, epochs=TrainingConfig.TAGGER_EPOCHS)
        
        print("[Subprocess] Tagger Task Completed Successfully.")

    except Exception:
        # Cite debug_lesson_7: Propagate exceptions from child process
        # Cite debug_lesson_8: Force flush stderr to ensure logs appear
        traceback.print_exc()
        sys.stderr.flush()
        sys.exit(1)
