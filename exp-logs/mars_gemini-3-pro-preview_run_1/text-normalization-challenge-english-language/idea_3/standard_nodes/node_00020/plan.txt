def run_tagger_wrapper(config_overrides, load_cached_data):
    """Wrapper to run tagger training in a separate process to ensure memory release."""
    for k, v in config_overrides.items():
        setattr(Config, k, v)
    train_tagger(load_cached_data=load_cached_data)
