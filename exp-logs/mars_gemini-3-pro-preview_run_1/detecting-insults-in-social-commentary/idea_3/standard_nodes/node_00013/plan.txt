class StructuralFeatureExtractor:
    """
    Extracts structural features using Character and Word N-grams and SVD.
    This captures morphological patterns and explicit lexical triggers (Cite solution_lesson_node_00012).
    """

    def __init__(
        self,
        char_ngram_range=None,
        word_ngram_range=None,
        max_features=None,
        n_components=None,
        seed=None,
    ):
        char_ngram_range = (
            char_ngram_range
            if char_ngram_range is not None
            else Config.CHAR_NGRAM_RANGE
        )
        word_ngram_range = (
            word_ngram_range
            if word_ngram_range is not None
            else Config.WORD_NGRAM_RANGE
        )
        max_features = (
            max_features if max_features is not None else Config.TFIDF_MAX_FEATURES
        )
        n_components = (
            n_components if n_components is not None else Config.SVD_COMPONENTS
        )
        seed = seed if seed is not None else Config.SEED

        self.char_vectorizer = TfidfVectorizer(
            analyzer="char",
            ngram_range=char_ngram_range,
            max_features=max_features,
            min_df=2,
            use_idf=True,
            sublinear_tf=True,
        )
        self.word_vectorizer = TfidfVectorizer(
            analyzer="word",
            ngram_range=word_ngram_range,
            max_features=max_features,
            min_df=2,
            use_idf=True,
            sublinear_tf=True,
        )
        self.svd = TruncatedSVD(
            n_components=n_components, random_state=seed, algorithm="randomized"
        )

    def fit(self, texts):
        """
        Fits the vectorizers and SVD on the provided texts.
        """
        # Fit TF-IDF
        char_matrix = self.char_vectorizer.fit_transform(texts)
        word_matrix = self.word_vectorizer.fit_transform(texts)

        # Combine features (Cite solution_lesson_node_00012)
        combined_matrix = hstack([char_matrix, word_matrix])

        # Fit SVD on the combined matrix
        self.svd.fit(combined_matrix)
        return self

    def transform(self, texts):
        """
        Transforms texts into dense SVD vectors.
        """
        char_matrix = self.char_vectorizer.transform(texts)
        word_matrix = self.word_vectorizer.transform(texts)

        combined_matrix = hstack([char_matrix, word_matrix])
        dense_features = self.svd.transform(combined_matrix)
        return dense_features
