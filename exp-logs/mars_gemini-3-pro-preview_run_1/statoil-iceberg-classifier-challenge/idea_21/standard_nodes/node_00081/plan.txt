My strategy to optimize the solution involves two key changes derived from the provided lessons:

1.  **Remove ImageNet Normalization**: As detailed in **Lesson 00042**, applying ImageNet normalization (mean subtraction and variance scaling) to intensity-based Synthetic Aperture Radar (SAR) data can suppress critical signal information by centering the background at zero. I will modify `library/dataset.py` to use simple Min-Max scaling (effectively passing the [0, 1] data through) by setting the normalization parameters to mean=0 and std=1.

2.  **Revert to Supervised SWA Ensemble with Global Epoch Selection**: Following **Lesson 00053**, I will discard the semi-supervised (Teacher-Student) pipeline which proved detrimental. Instead, I will implement the robust supervised strategy from **Lesson 00049**:
    *   **Phase 1 (Calibration):** Perform 5-Fold Cross-Validation on the training set to determine the global `optimal_epoch` that minimizes the average validation loss. This decouples stopping criteria from a single noisy validation split.
    *   **Phase 2 (Production Training):** Train an ensemble of 5 independent models on the **full** training set. Each model will train for the calibrated `optimal_epoch` count using the standard schedule, followed immediately by `SWA_EPOCHS` (12) of Stochastic Weight Averaging to find a flatter, more generalizable minimum (**Lesson 00055**).

This approach leverages the full dataset for training (maximizing effective data size per **Lesson 00020**) while ensuring rigorous convergence calibration.

def run_calibration_cv(train_data_dict, device, n_splits=5):
    """
    Performs 5-Fold Cross-Validation on the training set to determine the
    Global Optimal Epoch.
    Cite solution_lesson_node_00049: Global Epoch Selection.
    """
    logger.info(f"--- Starting Phase 1: Global Epoch Calibration ({n_splits}-Fold CV) ---")
    
    # Prepare data for KFold
    indices = np.arange(len(train_data_dict["images"]))
    kf = KFold(n_splits=n_splits, shuffle=True, random_state=config.SEED)
    
    # Store validation loss history: shape (n_splits, max_epochs)
    val_loss_history = []
    
    for fold, (train_idx, val_idx) in enumerate(kf.split(indices)):
        logger.info(f"Calibrating Fold {fold+1}/{n_splits}")
        utils.set_seed(config.SEED + fold)
        
        # Create subsets
        train_sub = dataset.IcebergDataset(
            images=train_data_dict["images"][train_idx],
            angles=train_data_dict["angles"][train_idx],
            labels=train_data_dict["labels"][train_idx],
            ids=train_data_dict["ids"][train_idx],
            transform=dataset.get_transforms(phase="train")
        )
        val_sub = dataset.IcebergDataset(
            images=train_data_dict["images"][val_idx],
            angles=train_data_dict["angles"][val_idx],
            labels=train_data_dict["labels"][val_idx],
            ids=train_data_dict["ids"][val_idx],
            transform=dataset.get_transforms(phase="val")
        )
        
        train_loader = DataLoader(train_sub, batch_size=config.BATCH_SIZE, shuffle=True, num_workers=config.NUM_WORKERS)
        val_loader = DataLoader(val_sub, batch_size=config.BATCH_SIZE, shuffle=False, num_workers=config.NUM_WORKERS)
        
        # Initialize Model
        model = network.IcebergResNet().to(device)
        optimizer, scheduler = engine.get_optimizer_scheduler(model)
        
        fold_losses = []
        
        # Train
        # We use a reasonably high epoch count to ensure we see the minimum
        # But respect config.MAX_EPOCHS
        for epoch in range(1, config.MAX_EPOCHS + 1):
            engine.train_one_epoch(model, train_loader, optimizer, device, epoch)
            val_loss, _, _ = engine.evaluate(model, val_loader, device)
            scheduler.step(val_loss)
            fold_losses.append(val_loss)
            
        val_loss_history.append(fold_losses)
        
    # Calculate Average Loss Curve
    avg_losses = np.mean(val_loss_history, axis=0)
    
    # Apply smoothing to find robust minimum (Cite solution_lesson_node_00063)
    # Simple moving average window 3
    smoothed_losses = np.convolve(avg_losses, np.ones(3)/3, mode='valid')
    
    # Find optimal epoch (add 1 because 0-indexed and convolution shift)
    # Convolution reduces size by 2. Index i corresponds to original index i+1
    best_idx = np.argmin(smoothed_losses)
    optimal_epoch = best_idx + 1 + 1 # +1 for 0-index, +1 for window shift center approx
    
    logger.info(f"Calibration Complete. Optimal Epoch: {optimal_epoch} (Min Smoothed Loss: {smoothed_losses[best_idx]:.4f})")
    
    return optimal_epoch


def train_final_ensemble(train_data_dict, optimal_epoch, device, n_models=5):
