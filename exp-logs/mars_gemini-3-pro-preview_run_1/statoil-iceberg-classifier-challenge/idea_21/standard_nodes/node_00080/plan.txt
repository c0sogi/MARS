My strategy to improve the model performance is to revert the semi-supervised learning approach (which degraded performance) and implement a robust **Global Epoch Selection** strategy using 5-Fold Cross-Validation, combined with **Stochastic Weight Averaging (SWA)** and a reduced **Batch Size**.

1.  **Remove Semi-Supervised Learning (Cite solution_lesson_node_00053)**: I will remove the Teacher-Student pipeline (Stage 2 and 3) and the pseudo-labeling logic, as previous attempts showed this degraded performance compared to a well-calibrated supervised baseline.
2.  **Optimize Batch Size (Cite solution_lesson_node_00038)**: I will reduce the `BATCH_SIZE` from 64 to 32. On small datasets, smaller batches increase the number of gradient updates per epoch, improving convergence and generalization.
3.  **Implement Global Epoch Selection (Cite solution_lesson_node_00049, solution_lesson_node_00040)**: Instead of relying on a single validation split for early stopping, I will run a 5-Fold Cross-Validation on the training set to determine the `optimal_epoch` where the average validation loss is minimized. This provides a statistically robust stopping point.
4.  **SWA Ensemble with Decoupled Calibration (Cite solution_lesson_node_00055, solution_lesson_node_00063)**: I will train the final ensemble of 5 models on the full training set. Each model will train for the calibrated `optimal_epoch` using the `ReduceLROnPlateau` scheduler (to match the calibration dynamics), and then transition to a constant-LR SWA phase for `SWA_EPOCHS`. This ensures SWA is initialized only after the model has converged to a basin.

def run_calibration(train_data_dict, n_folds=5):
    """
    Performs K-Fold Cross-Validation to determine the optimal number of epochs.
    Returns the epoch with the minimum average validation loss.
    """
    logger.info(f"--- Starting Calibration (Global Epoch Selection) with {n_folds}-Fold CV ---")
    
    kf = KFold(n_splits=n_folds, shuffle=True, random_state=config.SEED)
    
    # Store validation losses: shape (n_folds, MAX_EPOCHS)
    # Initialize with high values
    val_loss_history = np.full((n_folds, config.MAX_EPOCHS), np.inf)
    
    # Prepare data arrays
    images = train_data_dict["images"]
    angles = train_data_dict["angles"]
    labels = train_data_dict["labels"]
    ids = train_data_dict["ids"]
    
    for fold_idx, (train_idx, val_idx) in enumerate(kf.split(images)):
        logger.info(f"Calibrating Fold {fold_idx + 1}/{n_folds}")
        
        # Create Datasets for this fold
        fold_train_ds = dataset.IcebergDataset(
            images[train_idx], angles[train_idx], labels[train_idx], ids[train_idx],
            transform=dataset.get_transforms(phase="train")
        )
        fold_val_ds = dataset.IcebergDataset(
            images[val_idx], angles[val_idx], labels[val_idx], ids[val_idx],
            transform=dataset.get_transforms(phase="val")
        )
        
        fold_train_loader = DataLoader(
            fold_train_ds, batch_size=config.BATCH_SIZE, shuffle=True, 
            num_workers=config.NUM_WORKERS, pin_memory=True
        )
        fold_val_loader = DataLoader(
            fold_val_ds, batch_size=config.BATCH_SIZE, shuffle=False, 
            num_workers=config.NUM_WORKERS, pin_memory=True
        )
        
        # Initialize Model
        utils.set_seed(config.SEED + fold_idx)
        model = network.IcebergResNet().to(DEVICE)
        optimizer, scheduler = engine.get_optimizer_scheduler(model)
        
        for epoch in range(config.MAX_EPOCHS):
            # Train
            _ = engine.train_one_epoch(model, fold_train_loader, optimizer, DEVICE, epoch + 1)
            
            # Validate
            val_loss, _, _ = engine.evaluate(model, fold_val_loader, DEVICE)
            
            # Store loss
            val_loss_history[fold_idx, epoch] = val_loss
            
            # Scheduler Step
            scheduler.step(val_loss)
            
    # Calculate average loss per epoch
    avg_val_loss = np.mean(val_loss_history, axis=0)
    
    # Find optimal epoch (1-based index)
    optimal_epoch = np.argmin(avg_val_loss) + 1
    min_loss = np.min(avg_val_loss)
    
    logger.info(f"Calibration Complete. Optimal Epoch: {optimal_epoch} (Avg Loss: {min_loss:.6f})")
    return optimal_epoch


def run_final_training(train_data_dict, val_loader, optimal_epoch, n_models=5):
    """
    Trains an ensemble of models on the full training set for optimal_epoch + SWA_EPOCHS.
    Uses val_loader only for scheduler stepping (soft leakage) to maintain trajectory.
    """
    logger.info(f"--- Starting Final Ensemble Training (Epochs: {optimal_epoch} + {config.SWA_EPOCHS} SWA) ---")
    
    model_paths = []
    ckpt_dir = os.path.join(config.WORK_DIR, "checkpoints")
    os.makedirs(ckpt_dir, exist_ok=True)
    
    # Full Training Dataset
    full_train_ds = dataset.IcebergDataset(
        train_data_dict["images"], train_data_dict["angles"], 
        train_data_dict["labels"], train_data_dict["ids"],
        transform=dataset.get_transforms(phase="train")
    )
    full_train_loader = DataLoader(
        full_train_ds, batch_size=config.BATCH_SIZE, shuffle=True,
        num_workers=config.NUM_WORKERS, pin_memory=True
    )
    
    for i in range(n_models):
        logger.info(f"Training Final Model {i+1}/{n_models}")
        utils.set_seed(config.SEED + 100 + i)
        
        model = network.IcebergResNet().to(DEVICE)
        optimizer, scheduler = engine.get_optimizer_scheduler(model)
        swa_model = engine.get_swa_model(model)
        
        # Phase 1: Convergence to Optimal Epoch
        for epoch in range(1, optimal_epoch + 1):
            engine.train_one_epoch(model, full_train_loader, optimizer, DEVICE, epoch)
            
            # Step scheduler using hold-out validation set to mimic calibration dynamics
            # Note: We do NOT use this for early stopping.
            val_loss, _, _ = engine.evaluate(model, val_loader, DEVICE)
            scheduler.step(val_loss)
            
        # Phase 2: SWA
        logger.info("Starting SWA Phase...")
        # Reset LR for SWA
        for param_group in optimizer.param_groups:
            param_group['lr'] = config.SWA_LR
            
        for epoch in range(optimal_epoch + 1, optimal_epoch + config.SWA_EPOCHS + 1):
            engine.train_one_epoch(model, full_train_loader, optimizer, DEVICE, epoch)
            swa_model.update_parameters(model)
            
        # Finalize SWA
        engine.update_swa_bn(full_train_loader, swa_model, DEVICE)
        
        # Save
        save_path = os.path.join(ckpt_dir, f"ensemble_model_{i}.pth")
        utils.save_checkpoint(
            {"model_state_dict": swa_model.module.state_dict()},
            is_best=False, checkpoint_dir=ckpt_dir, filename=f"ensemble_model_{i}.pth"
        )
        model_paths.append(save_path)
        
    return model_paths
