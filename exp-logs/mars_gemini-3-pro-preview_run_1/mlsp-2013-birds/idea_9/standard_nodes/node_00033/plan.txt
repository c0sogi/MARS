class RandomTimeFlip(torch.nn.Module):
    def __init__(self, p=0.5):
        super().__init__()
        self.p = p

    def forward(self, img):
        if torch.rand(1) < self.p:
            return torch.flip(img, [2])  # (C, H, W), flip W (time)
        return img


class RandomMasking(torch.nn.Module):
    def __init__(self, p=0.5, num_masks=2, max_mask_pct=0.15):
        super().__init__()
        self.p = p
        self.num_masks = num_masks
        self.max_mask_pct = max_mask_pct

    def forward(self, img):
        if torch.rand(1) > self.p:
            return img
        
        # img: (C, H, W)
        _, H, W = img.shape
        mask_img = img.clone()
        
        for _ in range(self.num_masks):
            # Time mask
            mask_w = int(torch.rand(1) * self.max_mask_pct * W)
            if mask_w > 0:
                start_w = int(torch.rand(1) * (W - mask_w))
                mask_img[:, :, start_w : start_w + mask_w] = 0
                
            # Freq mask
            mask_h = int(torch.rand(1) * self.max_mask_pct * H)
            if mask_h > 0:
                start_h = int(torch.rand(1) * (H - mask_h))
                mask_img[:, start_h : start_h + mask_h, :] = 0
                
        return mask_img
