def evaluate(model, dataloader, device):
    """
    Evaluates the model on the validation set.

    Args:

        device (str): The device to evaluate on.

    Returns:
        float: The average validation loss.

def evaluate(model, dataloader, device, tta=False):
    """
    Evaluates the model on the validation set.
    Supports Test Time Augmentation (TTA) by averaging predictions of
    original and horizontally flipped images (Cite solution_lesson_node_00014).

    Args:
        model (nn.Module): The model to evaluate.
        dataloader (DataLoader): The validation data loader.
        device (str): The device to evaluate on.
        tta (bool): Whether to use Test Time Augmentation.

    Returns:
        float: The average validation loss.
    """
    model.eval()
    running_loss = 0.0
    dataset_size = 0

    criterion = nn.CrossEntropyLoss()
    nll_criterion = nn.NLLLoss()

    with torch.no_grad():
        for images, labels in dataloader:
            images = images.to(device)
            labels = labels.to(device)

            if tta:
                # Original
                logits = model(images)
                probs = torch.softmax(logits, dim=1)

                # Horizontal Flip
                images_flip = torch.flip(images, dims=[3])
                logits_flip = model(images_flip)
                probs_flip = torch.softmax(logits_flip, dim=1)

                # Average
                avg_probs = 0.5 * (probs + probs_flip)

                # Compute NLL Loss (log(probs))
                log_probs = torch.log(avg_probs + 1e-7)
                loss = nll_criterion(log_probs, labels)
            else:
                outputs = model(images)
                loss = criterion(outputs, labels)

            batch_size = images.size(0)
            running_loss += loss.item() * batch_size
            dataset_size += batch_size

    epoch_loss = running_loss / dataset_size
    return epoch_loss
