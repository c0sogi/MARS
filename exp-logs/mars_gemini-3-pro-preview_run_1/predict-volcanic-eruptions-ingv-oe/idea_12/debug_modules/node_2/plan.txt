    def _compute_robust_mfcc(self, x):
        """
        Extracts MFCCs and aggregates them using robust statistics (Median, Q05, Q95).
        """
        try:
            # Convert to tensor for torchaudio
            waveform = torch.tensor(x, dtype=torch.float32)

            # Define MFCC transform
            # Librosa defaults: n_mels=128, center=True. We mimic this for consistency.
            mfcc_transform = torchaudio.transforms.MFCC(
                sample_rate=self.fs,
                n_mfcc=self.n_mfcc,
                melkwargs={
                    "n_fft": self.n_fft_mfcc,
                    "hop_length": self.hop_length_mfcc,
                    "n_mels": 128,
                    "center": True,
                },
            )

            # Compute MFCCs
            mfccs = mfcc_transform(waveform)  # Shape: (n_mfcc, time)
            mfccs = mfccs.numpy()

            features = {}
            for i in range(mfccs.shape[0]):
                coeff = mfccs[i, :]
                features[f"mfcc_{i}_median"] = np.median(coeff)
                features[f"mfcc_{i}_q05"] = np.quantile(coeff, 0.05)
                features[f"mfcc_{i}_q95"] = np.quantile(coeff, 0.95)
                features[f"mfcc_{i}_std"] = np.std(coeff)

            return features
