class VisionTransformer:
    """
    Handles the conversion of raw sensor data into vision-ready 10-channel Log-Mel Spectrograms.
    Implements per-sample standardization and caching.
    """

    def __init__(self, cache_dir=CACHE_DIR):
        self.cache_manager = CacheManager(cache_dir=cache_dir)
        seed_everything(SEED)

        # Initialize MelSpectrogram transform
        self.mel_transform = torchaudio.transforms.MelSpectrogram(
            sample_rate=SAMPLING_RATE,
            n_fft=SPEC_N_FFT,
            hop_length=SPEC_HOP_LENGTH,
            n_mels=SPEC_N_MELS,
            f_min=SPEC_FMIN,
            f_max=SPEC_FMAX,
            center=True,
            power=2.0,
        )

    def _compute_spectrogram(self, df):
        """
        Computes 10-channel Log-Mel Spectrogram from the sensor dataframe.

        Args:
            df: DataFrame of shape (60001, 10) containing sensor readings.

        Returns:
            np.ndarray: Shape (10, n_mels, time_steps)
        """
        specs = []

        for col in SENSOR_COLS:
            # Ensure float32
            y = df[col].values.astype(np.float32)
            
            # Convert to Tensor: (1, time)
            waveform = torch.from_numpy(y).unsqueeze(0)

            # Compute Mel Spectrogram: (1, n_mels, time)
            S = self.mel_transform(waveform).squeeze(0).numpy()

            # Convert to log scale (dB).
            # librosa.power_to_db(S, ref=max) -> 10 * log10(S / max)
            # = 10 * log10(S) - 10 * log10(max)
            
            max_val = np.max(S)
            if max_val <= 1e-8:
                ref_db = -80.0
            else:
                ref_db = 10.0 * np.log10(max_val)

            # 10 * log10(S) with epsilon
            S_db = 10.0 * np.log10(np.maximum(S, 1e-10))
            
            # Normalize relative to peak
            S_db = S_db - ref_db
            
            # Clip at -80dB (standard dynamic range)
            S_db = np.maximum(S_db, -80.0)

            specs.append(S_db)

        # Stack into (10, n_mels, time)
        img = np.stack(specs, axis=0)
        return img
