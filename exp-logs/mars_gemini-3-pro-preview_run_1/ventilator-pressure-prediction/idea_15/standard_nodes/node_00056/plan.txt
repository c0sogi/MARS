To improve the model's performance and achieve the target metric, I will implement a "Context-Injected Residual Block with Full-Capacity Bi-LSTM". This strategy addresses the specific failures of the previous ReZero and Bottlenecked-LSTM attempts by reverting to proven architectural principles derived from the lessons.

**Implementation Strategy:**
1.  **Restore Standard Additive Residuals:** I will remove the `ReZero` mechanism, which was shown to throttle learning in shallow/medium depth networks (Cite Lesson 00054). Instead, I will use standard additive residual connections (`x = x + f(x)`) with Dropout applied correctly on the branch (Cite Lesson 00024).
2.  **Maximize Bi-LSTM State Capacity:** I will modify the `CompositeBlock` to use a Bidirectional LSTM with `hidden_size=hidden_dim` per direction (outputting `2 * hidden_dim`), followed by a linear projection back to `hidden_dim`. This fixes the bottleneck issue in the previous solution where the hidden dimension was halved, severely limiting the model's ability to capture temporal dynamics (Cite Lesson 00055).
3.  **Optimize Training Schedule:** I will increase the training duration to 30 epochs and set the batch size to 256. This ensures the model has a sufficient "Update Budget" to converge, correcting the underfitting observed when using larger batch sizes or shorter schedules (Cite Lesson 00019).
4.  **Preserve Deep Supervision & Context Injection:** I will retain the deep supervision and context injection mechanisms, which were critical for the best historical performance (Cite Lesson 00039).

class ReZero(nn.Module):
    """
    Learnable Residual Scaling (ReZero).
    Implements y = x + alpha * f(x), where alpha is a learnable scalar.
    This stabilizes deep networks without the signal magnitude normalization of LayerNorm,
    which is crucial for regression tasks where output magnitude matters.
    """

    def __init__(self, init_value=0.1):
        super().__init__()
        self.alpha = nn.Parameter(torch.tensor(init_value))

    def forward(self, function_output, residual_stream):
        return residual_stream + self.alpha * function_output

class CompositeBlock(nn.Module):
    """
    ReZero-Stabilized Composite Block with Deep Context Injection.

    Flow:
    1. Context Injection: Concat(Stream, Static Features)
    2. Temporal Mixing: Bi-LSTM
    3. Residual 1: Stream + alpha1 * Dropout(LSTM_Out)
    4. Channel Mixing: Pointwise FFN
    5. Residual 2: Stream + alpha2 * Dropout(FFN_Out)
    """

    def __init__(
        self, hidden_dim, static_dim, expansion_factor=2, dropout=0.1, rezero_init=0.1
    ):
        super().__init__()

        # Temporal Mixing: Bi-LSTM
        # Input size increases due to context injection
        self.lstm = nn.LSTM(
            input_size=hidden_dim + static_dim,
            hidden_size=hidden_dim // 2,  # Bidirectional sums to hidden_dim
            bidirectional=True,
            batch_first=True,
        )
        self.dropout1 = nn.Dropout(dropout)
        self.rezero1 = ReZero(rezero_init)

        # Channel Mixing: Pointwise FFN
        self.ffn = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim * expansion_factor),
            nn.GELU(),
            nn.Linear(hidden_dim * expansion_factor, hidden_dim),
        )
        self.dropout2 = nn.Dropout(dropout)
        self.rezero2 = ReZero(rezero_init)

    def forward(self, stream, static_features):
        # stream: (Batch, Seq, Hidden)
        # static_features: (Batch, Seq, StaticDim)

        # 1. Deep Context Injection
        # Re-introduce physics constraints at every layer
        combined = torch.cat([stream, static_features], dim=-1)

        # 2. Temporal Mixing
        lstm_out, _ = self.lstm(combined)

        stream = self.rezero1(self.dropout1(lstm_out), stream)

        # 4. Channel Mixing
        ffn_out = self.ffn(stream)

        # 5. ReZero Residual 2
        stream = self.rezero2(self.dropout2(ffn_out), stream)

        return stream

class CompositeBlock(nn.Module):
    """
    Context-Injected Residual Block with Full-Capacity Bi-LSTM.
    
    Flow:
    1. Context Injection: Concat(Stream, Static Features)
    2. Temporal Mixing: Bi-LSTM (Full Capacity) -> Projection
    3. Residual 1: Stream + Dropout(LSTM_Out)
    4. Channel Mixing: Pointwise FFN
    5. Residual 2: Stream + Dropout(FFN_Out)
    """

    def __init__(
        self, hidden_dim, static_dim, expansion_factor=2, dropout=0.1
    ):
        super().__init__()

        # Temporal Mixing: Bi-LSTM
        # Use full hidden_dim per direction (Cite Lesson 00055)
        self.lstm = nn.LSTM(
            input_size=hidden_dim + static_dim,
            hidden_size=hidden_dim, 
            bidirectional=True,
            batch_first=True,
        )
        # Project 2*hidden_dim back to hidden_dim to match residual stream
        self.proj = nn.Linear(hidden_dim * 2, hidden_dim)
        self.dropout1 = nn.Dropout(dropout)

        # Channel Mixing: Pointwise FFN
        self.ffn = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim * expansion_factor),
            nn.GELU(),
            nn.Linear(hidden_dim * expansion_factor, hidden_dim),
        )
        self.dropout2 = nn.Dropout(dropout)

    def forward(self, stream, static_features):
        # stream: (Batch, Seq, Hidden)
        # static_features: (Batch, Seq, StaticDim)

        # 1. Deep Context Injection
        combined = torch.cat([stream, static_features], dim=-1)

        # 2. Temporal Mixing
        lstm_out, _ = self.lstm(combined)
        lstm_out = self.proj(lstm_out)

        # 3. Residual 1 (Additive)
        # Apply dropout to branch only (Cite Lesson 00024)
        stream = stream + self.dropout1(lstm_out)

        # 4. Channel Mixing
        ffn_out = self.ffn(stream)

        # 5. Residual 2 (Additive)
        stream = stream + self.dropout2(ffn_out)

        return stream
