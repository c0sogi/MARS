        def add_features(df):
            # Lags
            df["u_in_lag1"] = df.groupby("breath_id")["u_in"].shift(1).fillna(0)
            df["u_in_lag2"] = df.groupby("breath_id")["u_in"].shift(2).fillna(0)
            
            # Diffs
            df["u_in_diff1"] = df["u_in"] - df["u_in_lag1"]
            df["u_in_diff2"] = df["u_in_diff1"] - df.groupby("breath_id")["u_in_diff1"].shift(1).fillna(0)
            
            # Cumulative Volume (Area)
            df["area"] = df.groupby("breath_id")["u_in"].cumsum()
            return df
