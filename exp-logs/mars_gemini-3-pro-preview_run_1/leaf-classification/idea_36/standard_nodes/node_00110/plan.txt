    def fit(self, X, y):
        """
        Fits the model and analytically estimates the optimal shrinkage alpha.
        """
        # Ensure float64
        X = X.astype(FLOAT_PRECISION)

        # Encode labels
        self.le_ = LabelEncoder()
        y_encoded = self.le_.fit_transform(y)
        self.classes_ = self.le_.classes_

        n_samples, n_features = X.shape
        n_classes = len(self.classes_)

        print("Computing statistics and estimating OAS shrinkage...")

        # Compute empirical stats
        means, priors, cov_emp = self._compute_stats(X, y_encoded)

        # OAS Shrinkage Estimation (Chen et al., 2010)
        # We use the pooled covariance matrix (cov_emp)
        # Effective samples n = N - K (degrees of freedom for pooled cov)
        n = n_samples - n_classes
        p = n_features

        trace_S = np.trace(cov_emp)
        trace_S2 = np.sum(cov_emp ** 2)

        # Formula:
        # alpha = [ (1 - 2/p) * Tr(S^2) + Tr^2(S) ] / [ (n + 1 - 2/p) * (Tr(S^2) - Tr^2(S)/p) ]
        numerator = (1 - 2.0 / p) * trace_S2 + (trace_S ** 2)
        denominator = (n + 1 - 2.0 / p) * (trace_S2 - (trace_S ** 2) / p)

        alpha = numerator / denominator

        # Clip alpha to [0, 1]
        self.best_alpha_ = min(max(alpha, 0.0), 1.0)

        print(f"Analytical OAS Alpha: {self.best_alpha_:.6f}")

        # Compute final shrunk covariance
        cov_final = self._get_shrunk_covariance(cov_emp, self.best_alpha_)

        # Solve for weights and bias
        self.W_, self.b_ = self._solve_parameters(means, priors, cov_final)

        return self
