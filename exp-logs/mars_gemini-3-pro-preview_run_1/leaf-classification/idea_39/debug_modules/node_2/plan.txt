def extract_geometric_features(image_path):
    """
    Extracts geometric features from a binary leaf image.
    Images are black leaves on white backgrounds.
    """
    # Read image
    if not os.path.exists(image_path):
        return np.zeros(len(GEOMETRIC_FEATURES), dtype=np.float64)

    img = cv2.imread(image_path, cv2.IMREAD_GRAYSCALE)
    if img is None:
        return np.zeros(len(GEOMETRIC_FEATURES), dtype=np.float64)

    # Invert: Leaf (black=0) becomes 1, Background (white=255) becomes 0
    _, binary = cv2.threshold(img, 128, 1, cv2.THRESH_BINARY_INV)

    # Label regions
    label_img = measure.label(binary)
    regions = measure.regionprops(label_img)

    if not regions:
        return np.zeros(len(GEOMETRIC_FEATURES), dtype=np.float64)

    # Assume the largest connected component is the leaf
    leaf = max(regions, key=lambda r: r.area)

    # Extract raw properties
    area = float(leaf.area)
    perimeter = float(leaf.perimeter)
    minr, minc, maxr, maxc = leaf.bbox
    bbox_width = float(maxc - minc)
    bbox_height = float(maxr - minr)
    major_axis = float(leaf.major_axis_length)
    minor_axis = float(leaf.minor_axis_length)
    convex_area = float(leaf.convex_area)
    equiv_diameter = float(leaf.equivalent_diameter)
    eccentricity = float(leaf.eccentricity)
    solidity = float(leaf.solidity)
    extent = float(leaf.extent)

    # Derived properties
    aspect_ratio = bbox_width / bbox_height if bbox_height > 0 else 0.0

    # Roundness: 4 * pi * Area / Perimeter^2
    roundness = (4 * np.pi * area) / (perimeter**2) if perimeter > 0 else 0.0

    # Compactness: Perimeter^2 / Area (Inverse of roundness roughly)
    compactness = (perimeter**2) / area if area > 0 else 0.0

    features = np.array(
        [
            area,
            perimeter,
            bbox_width,
            bbox_height,
            major_axis,
            minor_axis,
            convex_area,
            equiv_diameter,
            aspect_ratio,
            solidity,
            extent,
            roundness,
            compactness,
            eccentricity,
        ],
        dtype=np.float64,
    )

    return features

def run_pipeline():
    """
    Orchestrates the training and inference pipeline.
    """
    np.random.seed(SEED)

    print("Loading Data...")
    X_train, y_train, ids_train, feat_cols = get_dataset("train")
    X_val, y_val, ids_val, _ = get_dataset("val")
    X_test, _, ids_test, _ = get_dataset("test")

    print(f"Feature count: {len(feat_cols)}")
    print(
        f"Train shape: {X_train.shape}, Val shape: {X_val.shape}, Test shape: {X_test.shape}"
    )

    # Encode Labels
    le = LabelEncoder()
    y_train_enc = le.fit_transform(y_train)
    y_val_enc = le.transform(y_val)
    classes = le.classes_
    n_classes = len(classes)

    # Preprocessing Pipeline (High Precision)
    print("Preprocessing (Yeo-Johnson + StandardScaler)...")
    # 1. Yeo-Johnson
    pt = PowerTransformer(method="yeo-johnson", standardize=False)
    X_train_pt = pt.fit_transform(X_train)
    X_val_pt = pt.transform(X_val)
    X_test_pt = pt.transform(X_test)

    # 2. Standard Scaling
    ss = StandardScaler()
    X_train_scaled = ss.fit_transform(X_train_pt)
    X_val_scaled = ss.transform(X_val_pt)
    X_test_scaled = ss.transform(X_test_pt)

    # OAS Model Training
    print("Training OAS Linear Discriminant...")

    # Compute Class Means and Priors
    means = np.zeros((n_classes, X_train_scaled.shape[1]), dtype=np.float64)
    priors = np.zeros(n_classes, dtype=np.float64)

    for k in range(n_classes):
        X_k = X_train_scaled[y_train_enc == k]
        means[k] = np.mean(X_k, axis=0)
        priors[k] = len(X_k) / len(X_train_scaled)

    # Compute Residuals (Center data by class mean)
    residuals = np.zeros_like(X_train_scaled)
    for k in range(n_classes):
        mask = y_train_enc == k
        residuals[mask] = X_train_scaled[mask] - means[k]

    # Fit OAS to Residuals
    oas = OAS(assume_centered=True)
    oas.fit(residuals)
    precision = oas.precision_  # Inverse Covariance Matrix

    # Calculate Weights (W) and Bias (b) for Linear Discriminant
    # W = means @ precision (Shape: n_classes x n_features)
    W = np.dot(means, precision)

    # b = -0.5 * diag(means @ W.T) + log(priors)
    quad_term = 0.5 * np.sum(means * W, axis=1)
    b = -quad_term + np.log(priors)

    # Validation Inference
    print("Validating...")
    # Z = X @ W.T + b
    Z_val = np.dot(X_val_scaled, W.T) + b
    probs_val = softmax(Z_val, axis=1)

    # Clip for metric stability
    probs_val_clipped = np.clip(probs_val, 1e-15, 1 - 1e-15)
    probs_val_clipped /= probs_val_clipped.sum(axis=1, keepdims=True)

    score = log_loss(y_val_enc, probs_val_clipped)
    print(f"Validation Multi-class Log Loss: {score:.15f}")

    # Test Inference
    print("Generating Submission...")
    Z_test = np.dot(X_test_scaled, W.T) + b
    probs_test = softmax(Z_test, axis=1)

    df_sub = pd.DataFrame(probs_test, columns=classes)
    df_sub.insert(0, "id", ids_test)

    os.makedirs(SUBMISSION_DIR, exist_ok=True)
    df_sub.to_csv(SUBMISSION_FILE, index=False)
    print(f"Submission saved to {SUBMISSION_FILE}")


def main():
    run_pipeline()
