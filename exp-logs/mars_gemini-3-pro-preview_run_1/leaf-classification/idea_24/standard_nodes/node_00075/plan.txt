        Predicts class probabilities using the Linear Formulation of LDA.
        Cite solution_lesson_node_00055: Use Linear Formulation to avoid precision loss.
        Cite solution_lesson_node_00057: Maintain float64 precision.

            X (np.ndarray): Input features.

        Returns:
            np.ndarray: Predicted probabilities.

        # Get analytic weights (Linear Formulation: w.T * x + b)
        coef, intercept = self.get_analytic_weights()

        # Compute Logits
        # X: (n_samples, n_features)
        # coef: (n_classes, n_features)
        # intercept: (n_classes,)
        # logits: (n_samples, n_classes)
        logits = X @ coef.T + intercept

        # Compute Softmax with high precision
        # Shift logits for numerical stability (exp(x - max))
        logits_max = np.max(logits, axis=1, keepdims=True)
        exp_logits = np.exp(logits - logits_max)

        # Add epsilon to sum to avoid division by zero (unlikely but safe)
        sum_exp = np.sum(exp_logits, axis=1, keepdims=True)
        probs = exp_logits / sum_exp

        # Clip probabilities to avoid log(0) and ensure compliance with metric
        probs = np.clip(probs, config.PROB_CLIP_MIN, config.PROB_CLIP_MAX)

        return probs
