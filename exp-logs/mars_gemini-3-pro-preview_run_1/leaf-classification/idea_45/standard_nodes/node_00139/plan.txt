**Strategy:**
1.  **Restore Absolute Size Signal (Cite Lesson 118):** The previous solution degraded performance by replacing absolute size features with scale-invariant Hu Moments. I will remove `Hu Moments` and reinstate `Equivalent Diameter`. As noted in the EDA and Lesson 118, image dimensions are a significant discriminator for species, and enforcing invariance discards this signal.
2.  **Maintain Macro-Geometric Triad (Cite Lesson 120):** I will retain `Solidity`, `Aspect Ratio`, and `Extent`. This triad provides a complete description of gross shape (convexity, elongation, rectangularity) which is critical for linear separation.
3.  **Prioritize Geometric Ratios (Cite Lesson 119):** I will keep `Roundness` and `Eccentricity` while removing raw `Perimeter` and `Axis Lengths`. Ratios are more robust to pixelation noise than raw boundary integrals, while `Equivalent Diameter` handles the scale information more cleanly than raw `Area`.
4.  **Simplify Feature Space:** By removing redundant (Area, Axes) and noisy (Hu Moments) features, I aim to reduce the dimensionality slightly, allowing the OAS estimator to focus on the most discriminative signals without "smearing" the precision matrix.

def extract_features_from_image(image_path):
    """
    Extracts a comprehensive suite of geometric and moment-based features from a binary leaf image.

    Features include:
    - Absolute Moments: Area, Major/Minor Axis Lengths.
    - Relative Moments: Eccentricity.
    - Invariant Moments: Hu Moments 1-7 (Log-Modulus transformed).
    - Boundary Integrals: Perimeter, Convex Perimeter, Solidity, Roundness.
    - Stable Extrema: Aspect Ratio, Extent.

    Args:
        image_path (str): Full path to the image file.

    Returns:
        dict: A dictionary mapping feature names (from config) to float64 values.
    """
    # Initialize all features to 0.0
    features = {col: 0.0 for col in config.IMAGE_FEATURE_COLS}

    if not os.path.exists(image_path):
        return features

    # Load image as grayscale
    img = cv2.imread(image_path, cv2.IMREAD_GRAYSCALE)
    if img is None:
        return features

    # Binarize: The dataset description states "binary black leaves against white backgrounds".
    # We invert this so the leaf is the foreground (255) and background is black (0).
    # Using thresholding is safer than simple inversion to handle potential compression artifacts.
    _, thresh = cv2.threshold(img, 127, 255, cv2.THRESH_BINARY_INV)

    # Find contours
    contours, _ = cv2.findContours(thresh, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

    if not contours:
        return features

    # Assume the largest contour by area is the leaf
    cnt = max(contours, key=cv2.contourArea)

    # ---------------------------------------------------------
    # 0. Base Moments & Area
    # ---------------------------------------------------------
    M = cv2.moments(cnt)
    area = M["m00"]

    # Filter extremely small noise
    if area < 1e-6:
        return features

    features[config.FEAT_IMG_AREA] = float(area)

    # ---------------------------------------------------------
    # 1. Absolute Axis Lengths & Eccentricity (derived from 2nd Moments)
    # ---------------------------------------------------------
    # Normalized central moments
    mu20 = M["mu20"] / area
    mu02 = M["mu02"] / area
    mu11 = M["mu11"] / area

    # Calculate eigenvalues of the covariance matrix
    # common_term = sqrt((mu20 - mu02)^2 + 4 * mu11^2)
    common_term = np.sqrt((mu20 - mu02) ** 2 + 4 * mu11**2)
    lambda1 = (mu20 + mu02 + common_term) / 2
    lambda2 = (mu20 + mu02 - common_term) / 2

    # Axis lengths (approximation for an equivalent ellipse)
    # Major Axis = 4 * sqrt(lambda_max)
    # Minor Axis = 4 * sqrt(lambda_min)
    major_axis = 4 * np.sqrt(lambda1) if lambda1 > 0 else 0.0
    minor_axis = 4 * np.sqrt(lambda2) if lambda2 > 0 else 0.0

    features[config.FEAT_IMG_MAJOR_AXIS] = float(major_axis)
    features[config.FEAT_IMG_MINOR_AXIS] = float(minor_axis)

    # Eccentricity: sqrt(1 - (minor/major)^2)
    if major_axis > 1e-9:
        eccentricity = np.sqrt(max(0.0, 1.0 - (minor_axis / major_axis) ** 2))
    else:
        eccentricity = 0.0
    features[config.FEAT_IMG_ECCENTRICITY] = float(eccentricity)

    # ---------------------------------------------------------
    # 2. Invariant Moments (Hu Moments)
    # ---------------------------------------------------------
    hu = cv2.HuMoments(M).flatten()
    for i in range(7):
        val = hu[i]
        # Log-Modulus Transform: sign(x) * log(|x|)
        # This compresses dynamic range while preserving sign.
        abs_val = np.abs(val)
        if abs_val > 1e-40:
            log_val = np.sign(val) * np.log(abs_val)
        else:
            log_val = 0.0
        features[f"img_hu_moment_{i+1}"] = float(log_val)

    # ---------------------------------------------------------
    # 3. Boundary Integrals
    # ---------------------------------------------------------
    perimeter = cv2.arcLength(cnt, True)
    features[config.FEAT_IMG_PERIMETER] = float(perimeter)

    hull = cv2.convexHull(cnt)
    convex_perimeter = cv2.arcLength(hull, True)
    features[config.FEAT_IMG_CONVEX_PERIMETER] = float(convex_perimeter)

    convex_area = cv2.contourArea(hull)
    if convex_area > 1e-9:
        solidity = area / convex_area
    else:
        solidity = 0.0
    features[config.FEAT_IMG_SOLIDITY] = float(solidity)

    if perimeter > 1e-9:
        # Roundness: 4 * pi * Area / Perimeter^2
        roundness = (4 * np.pi * area) / (perimeter**2)
    else:
        roundness = 0.0
    features[config.FEAT_IMG_ROUNDNESS] = float(roundness)

    # ---------------------------------------------------------
    # 4. Stable Extrema (Axis-Aligned)
    # ---------------------------------------------------------
    x, y, w, h = cv2.boundingRect(cnt)
    if h > 0:
        aspect_ratio = float(w) / h
    else:
        aspect_ratio = 0.0
    features[config.FEAT_IMG_ASPECT_RATIO] = aspect_ratio

    rect_area = w * h
    if rect_area > 0:
        extent = area / rect_area
    else:
        extent = 0.0
    features[config.FEAT_IMG_EXTENT] = extent

    return features

    Extracts a compact suite of high-signal geometric features from a binary leaf image.

    Features include:
    - Absolute Size: Equivalent Diameter (Cite Lesson 118).
    - Geometric Ratios: Eccentricity, Roundness (Cite Lesson 119).
    - Macro-Geometric Triad: Solidity, Aspect Ratio, Extent (Cite Lesson 120).

    Args:
        image_path (str): Full path to the image file.

    Returns:
        dict: A dictionary mapping feature names (from config) to float64 values.
    """

    if not os.path.exists(image_path):
        return features

    if not contours:

    cnt = max(contours, key=cv2.contourArea)

    # ---------------------------------------------------------
    # 0. Base Moments & Absolute Size
    # ---------------------------------------------------------
    M = cv2.moments(cnt)
    area = M["m00"]

    # Filter extremely small noise
    if area < 1e-6:
        return features

    # Equivalent Diameter: sqrt(4 * Area / pi)
    # Captures absolute size signal linearly, robust to rotation.
    features[config.FEAT_IMG_EQUIVALENT_DIAMETER] = float(np.sqrt(4 * area / np.pi))

    # ---------------------------------------------------------
    # 1. Eccentricity (derived from 2nd Moments)
    # ---------------------------------------------------------

    lambda1 = (mu20 + mu02 + common_term) / 2
    lambda2 = (mu20 + mu02 - common_term) / 2

    if major_axis > 1e-9:

    else:
        eccentricity = 0.0
    features[config.FEAT_IMG_ECCENTRICITY] = float(eccentricity)

    # 2. Boundary Integrals (Roundness, Solidity)
    # ---------------------------------------------------------
    perimeter = cv2.arcLength(cnt, True)

        # Roundness: 4 * pi * Area / Perimeter^2

    else:

    features[config.FEAT_IMG_SOLIDITY] = float(solidity)

    # ---------------------------------------------------------
    x, y, w, h = cv2.boundingRect(cnt)
    
    if h > 0:
        aspect_ratio = float(w) / h
    else:
        aspect_ratio = 0.0
    features[config.FEAT_IMG_ASPECT_RATIO] = aspect_ratio

    if rect_area > 0:
        extent = area / rect_area
    else:
        extent = 0.0
    features[config.FEAT_IMG_EXTENT] = extent

    return features
