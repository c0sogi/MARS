    def predict_proba(self, X):
        """
        Predicts class probabilities using the linear formulation.
        Uses PyTorch for high-precision inference if available.
        Cite {solution_lesson_node_00218}: High-Precision Inference.

        Args:
            X (array-like): Test features (n_samples, n_features).

        Returns:
            array-like: Probabilities (n_samples, n_classes).
        """
        import torch

        # Ensure model is fitted
        if self.W_ is None or self.b_ is None:
            raise RuntimeError("Model not fitted")

        # Detect device
        device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

        # Convert to tensors (float64 is crucial)
        # We use torch.tensor to copy data and ensure float64 precision
        X_t = torch.tensor(X, dtype=torch.float64, device=device)
        W_t = torch.tensor(self.W_, dtype=torch.float64, device=device)
        b_t = torch.tensor(self.b_, dtype=torch.float64, device=device)

        # Linear Scoring: Z = X @ W.T + b
        logits = torch.matmul(X_t, W_t.T) + b_t

        # Probability: Softmax
        probs = torch.nn.functional.softmax(logits, dim=1)

        return probs.cpu().numpy()
