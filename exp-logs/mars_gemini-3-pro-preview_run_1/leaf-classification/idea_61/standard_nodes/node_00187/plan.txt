    def _extract_single_image(self, image_path):
        """
        Process a single image to extract geometric features.

        Args:
            image_path (str): Full path to the image file.

        Returns:
            np.ndarray: A 1D array of 7 float64 features.
        """
        # Default zero vector
        features = np.zeros(len(GEOMETRIC_FEATURES), dtype=FLOAT_PRECISION)

        if not os.path.exists(image_path):
            return features

        # Load image in grayscale
        img = cv2.imread(image_path, cv2.IMREAD_GRAYSCALE)
        if img is None:
            return features

        # Polarity Correction: Leaves are binary black on white.
        # Invert so leaf is white (foreground) and background is black.
        _, binary = cv2.threshold(img, 127, 255, cv2.THRESH_BINARY_INV)

        # Find Contours
        # CHAIN_APPROX_NONE to keep all boundary points for high fidelity
        contours, _ = cv2.findContours(binary, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_NONE)

        if not contours:
            return features

        # Assume the largest contour is the leaf
        cnt = max(contours, key=cv2.contourArea)

        # 1. Absolute Scale: Area
        area = cv2.contourArea(cnt)

        # 2. Internal Topology: Mean Thickness
        # Euclidean Distance Transform on the binary mask
        # dist_transform gives distance to nearest zero pixel (background)
        dist_transform = cv2.distanceTransform(binary, cv2.DIST_L2, 5)
        # We only care about the thickness of the leaf itself (non-zero regions)
        # Using the binary mask to filter
        leaf_pixels = dist_transform[binary > 0]
        mean_thickness = np.mean(leaf_pixels) if leaf_pixels.size > 0 else 0.0

        # 3. Elongation: Eccentricity
        # Needs at least 5 points to fit an ellipse
        eccentricity = 0.0
        if len(cnt) >= 5:
            try:
                (x, y), (MA, ma), angle = cv2.fitEllipse(cnt)
                # MA and ma are lengths of axes (Major and minor, but order not guaranteed)
                a = max(MA, ma) / 2.0
                b = min(MA, ma) / 2.0
                if a > 0:
                    eccentricity = np.sqrt(1 - (b**2 / a**2))
            except:
                eccentricity = 0.0

        # Derived Geometric Properties
        perimeter = cv2.arcLength(cnt, True)

        hull = cv2.convexHull(cnt)
        convex_area = cv2.contourArea(hull)

        x, y, w, h = cv2.boundingRect(cnt)
        rect_area = w * h

        # 4. Roughness: Solidity
        solidity = area / convex_area if convex_area > 0 else 0.0

        # 5. Rectangularity: Extent
        extent = area / rect_area if rect_area > 0 else 0.0

        # 6. Orientation: Aspect Ratio
        aspect_ratio = w / float(h) if h > 0 else 0.0

        # 7. Compactness: Roundness
        # 4 * pi * Area / Perimeter^2
        roundness = (4 * np.pi * area) / (perimeter**2) if perimeter > 0 else 0.0

        # Assemble vector strictly in order defined in config.GEOMETRIC_FEATURES
        # ["Area", "Mean_Thickness", "Eccentricity", "Solidity", "Extent", "Aspect_Ratio", "Roundness"]
        features[0] = area
        features[1] = mean_thickness
        features[2] = eccentricity
        features[3] = solidity
        features[4] = extent
        features[5] = aspect_ratio
        features[6] = roundness

        return features
