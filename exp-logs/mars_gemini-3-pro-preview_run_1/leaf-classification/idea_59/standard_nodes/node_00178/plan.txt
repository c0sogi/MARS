def _process_single_image(image_path):
    """
    Extracts the 9 robust-integral geometric features from a single image.

    Features:
    - Area, Perimeter, Major_Axis_Length, Minor_Axis_Length
    - Mean_Thickness (Internal Topology)
    - Solidity, Extent, Eccentricity (Invariant Shape)
    - Roundness (Non-Linear Ratio)

    Returns a dictionary with keys matching config.GEOMETRIC_FEATURES.
    """
    # Initialize default result (zeros)
    result = {k: 0.0 for k in config.GEOMETRIC_FEATURES}

    # Check existence
    if not os.path.exists(image_path):
        return result

    # Load image in grayscale
    img = cv2.imread(image_path, cv2.IMREAD_GRAYSCALE)
    if img is None:
        return result

    # Polarity Correction: Leaf becomes white (255), background black (0)
    # The dataset description says "binary black leaves against white backgrounds"
    # So we invert it using THRESH_BINARY_INV.
    _, binary = cv2.threshold(img, 127, 255, cv2.THRESH_BINARY_INV)

    # Lossless Contours
    contours, _ = cv2.findContours(binary, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_NONE)

    if not contours:
        return result

    # Assume the largest contour corresponds to the leaf
    cnt = max(contours, key=cv2.contourArea)

    # 1. Area (Integral)
    area = cv2.contourArea(cnt)
    if area <= 0:
        return result
    result["Area"] = float(area)

    # 2. Perimeter
    perimeter = cv2.arcLength(cnt, True)
    result["Perimeter"] = float(perimeter)

    # 3. Roundness (4 * pi * Area / P^2)
    if perimeter > 0:
        result["Roundness"] = (4 * np.pi * area) / (perimeter**2)

    # 4. Extent (Area / Bounding Rect Area)
    x, y, w, h = cv2.boundingRect(cnt)
    rect_area = w * h
    if rect_area > 0:
        result["Extent"] = area / rect_area

    # 5. Solidity (Area / Convex Hull Area)
    hull = cv2.convexHull(cnt)
    hull_area = cv2.contourArea(hull)
    if hull_area > 0:
        result["Solidity"] = area / hull_area

    # 6, 7, 8. Ellipse Properties (Major, Minor, Eccentricity)
    # fitEllipse requires at least 5 points
    if len(cnt) >= 5:
        try:
            (cx, cy), (MA, ma), angle = cv2.fitEllipse(cnt)
            # OpenCV returns (MA, ma) as axes lengths (diameters)
            # Usually the order is (minor, major) or depends on angle,
            # but let's sort them to be sure.
            axes = sorted([MA, ma])
            minor_axis = axes[0]
            major_axis = axes[1]

            result["Minor_Axis_Length"] = float(minor_axis)
            result["Major_Axis_Length"] = float(major_axis)

            if major_axis > 0:
                # Eccentricity = sqrt(1 - (b/a)^2) where a is semi-major, b is semi-minor
                # Using axes lengths: sqrt(1 - (minor/major)^2)
                ratio_sq = (minor_axis / major_axis) ** 2
                result["Eccentricity"] = np.sqrt(1 - ratio_sq)
        except Exception:
            # Fallback if fitEllipse fails numerically
            pass

    # 9. Mean Thickness (Euclidean Distance Transform)
    # Compute distance from every binary pixel to the nearest zero pixel
    # The mask is 'binary' (255 for leaf).
    dist_transform = cv2.distanceTransform(binary, cv2.DIST_L2, 5)

    # We want the mean thickness of the "flesh", i.e., where the leaf exists
    # Mask is binary > 0
    leaf_pixels = dist_transform[binary > 0]
    if leaf_pixels.size > 0:
        result["Mean_Thickness"] = np.mean(leaf_pixels)

    return result
