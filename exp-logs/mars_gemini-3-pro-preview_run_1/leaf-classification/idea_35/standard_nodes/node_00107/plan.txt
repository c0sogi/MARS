The strategy is to optimize the Linear Discriminant Analysis (LDA) model by replacing the manual Cholesky-based solver with the robust SVD-based precision matrix provided by the `OAS` estimator. This addresses the performance degradation identified in Lesson `solution_lesson_node_00062`, which notes that manual linear solvers (like `cho_solve` or `solve`) can introduce numerical instability or require arbitrary jitter in ill-conditioned regimes, whereas the library-provided `precision_` attribute utilizes a stable pseudo-inverse. I will rename the classifier to `PrecisionOASDiscriminant` to reflect this change and update the execution script accordingly, while maintaining the critical `float64` precision and alphanumeric feature ordering.

class CholeskyOASDiscriminant(BaseEstimator, ClassifierMixin):
    """
    A Linear Discriminant Classifier that uses the OAS estimator for covariance
    and solves for weights using exact Cholesky decomposition.

    This architecture avoids the numerical instability of explicit matrix inversion
    (precision matrix) and the spectral truncation of SVD, providing microscopic
    signal preservation required for high-precision tasks.

    Attributes:
        classes_ (np.ndarray): Unique class labels.
        W_ (np.ndarray): Weight matrix of shape (n_features, n_classes).
        b_ (np.ndarray): Bias vector of shape (n_classes,).
        covariance_ (np.ndarray): Estimated covariance matrix.
    """

    def __init__(self):
        self.classes_ = None
        self.W_ = None
        self.b_ = None
        self.covariance_ = None
        self.le_ = None

    def fit(self, X, y):
        """
        Fits the model using OAS covariance estimation and Cholesky solving.

        Args:
            X (array-like): Training features. Must be float64.
            y (array-like): Training labels.

        Returns:
            self
        """
        # Enforce float64 for precision
        X = np.array(X, dtype=np.float64)

        # Encode labels
        self.le_ = LabelEncoder()
        y_enc = self.le_.fit_transform(y)
        self.classes_ = self.le_.classes_

        n_samples, n_features = X.shape
        n_classes = len(self.classes_)

        # 1. Compute Empirical Means and Priors
        # We use arithmetic mean as per Lesson 00066
        means = np.zeros((n_classes, n_features), dtype=np.float64)
        priors = np.zeros(n_classes, dtype=np.float64)

        for k in range(n_classes):
            mask = y_enc == k
            X_k = X[mask]
            means[k] = np.mean(X_k, axis=0)
            priors[k] = len(X_k) / n_samples

        # 2. Compute Centered Residuals
        # R = X - mu_y
        residuals = X - means[y_enc]

        # 3. Estimate Covariance with OAS
        # assume_centered=True is critical as we manually centered residuals
        oas = OAS(**OAS_PARAMS)
        oas.fit(residuals)
        self.covariance_ = oas.covariance_

        # 4. Solve for Weights using Cholesky Decomposition
        # We want to solve Sigma * W = means.T for W.
        # This corresponds to W_k = Sigma^-1 * mu_k
        # Factorize Sigma = L * L.T
        try:
            c, lower = linalg.cho_factor(self.covariance_, lower=True)
        except linalg.LinAlgError:
            # Fallback for extremely rare non-PD cases (though OAS usually prevents this)
            # Add minute jitter
            self.covariance_.flat[:: n_features + 1] += 1e-12
            c, lower = linalg.cho_factor(self.covariance_, lower=True)

        # Solve system
        self.W_ = linalg.cho_solve((c, lower), means.T)

        # 5. Compute Bias Terms
        # b_k = -0.5 * (mu_k . W_k) + log(pi_k)
        # The dot product mu_k . W_k is the diagonal of (means @ W)
        # We compute it efficiently using element-wise multiplication and summation
        # means.T is (n_features, n_classes), W_ is (n_features, n_classes)
        dot_products = np.sum(means.T * self.W_, axis=0)
        self.b_ = -0.5 * dot_products + np.log(priors)

        return self

class PrecisionOASDiscriminant(BaseEstimator, ClassifierMixin):
    """
    A Linear Discriminant Classifier that uses the OAS estimator for covariance
    and utilizes the robust SVD-based precision matrix for weight calculation.

    This architecture prefers the library-provided precision_ attribute over
    manual solvers (like Cholesky) to handle potential ill-conditioning without
    introducing arbitrary jitter (Cite solution_lesson_node_00062).

    Attributes:
        classes_ (np.ndarray): Unique class labels.
        W_ (np.ndarray): Weight matrix of shape (n_features, n_classes).
        b_ (np.ndarray): Bias vector of shape (n_classes,).
        precision_ (np.ndarray): Estimated precision matrix (inverse covariance).
    """

    def __init__(self):
        self.classes_ = None
        self.W_ = None
        self.b_ = None
        self.precision_ = None
        self.le_ = None

    def fit(self, X, y):

        Fits the model using OAS covariance estimation and precision matrix multiplication.

        Args:
            X (array-like): Training features. Must be float64.
            y (array-like): Training labels.

        Returns:
            self
        """

        X = np.array(X, dtype=np.float64)

        # Encode labels
        self.le_ = LabelEncoder()
        y_enc = self.le_.fit_transform(y)
        self.classes_ = self.le_.classes_

        means = np.zeros((n_classes, n_features), dtype=np.float64)

            X_k = X[mask]

            priors[k] = len(X_k) / n_samples

        # R = X - mu_y
        residuals = X - means[y_enc]

        # 3. Estimate Covariance with OAS
        # assume_centered=True is critical as we manually centered residuals (Cite solution_lesson_node_00061)
        oas = OAS(**OAS_PARAMS)
        oas.fit(residuals)
        self.precision_ = oas.precision_

        # 4. Calculate Weights using Precision Matrix
        # W = Sigma^-1 * mu^T = Precision * Means^T
        # Using the SVD-based precision matrix from OAS is more robust than manual solvers (Cite solution_lesson_node_00062)
        self.W_ = self.precision_ @ means.T

        # 5. Compute Bias Terms
        # b_k = -0.5 * (mu_k . W_k) + log(pi_k)
        # The dot product mu_k . W_k is the diagonal of (means @ W)
        dot_products = np.sum(means.T * self.W_, axis=0)
        self.b_ = -0.5 * dot_products + np.log(priors)

        return self
