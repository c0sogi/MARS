def predict_tta(model, inputs):
    """
    Predicts using Test-Time Augmentation (TTA).
    Cite solution_lesson_node_00006: TTA reduces variance and improves RMSE.
    """
    # 1. Original
    out1 = model(inputs)

    # 2. Horizontal Flip
    inputs_h = torch.flip(inputs, [3])
    out_h = model(inputs_h)
    out2 = torch.flip(out_h, [3])

    # 3. Vertical Flip
    inputs_v = torch.flip(inputs, [2])
    out_v = model(inputs_v)
    out3 = torch.flip(out_v, [2])

    # 4. Both Flip (Rot 180)
    inputs_hv = torch.flip(inputs, [2, 3])
    out_hv = model(inputs_hv)
    out4 = torch.flip(out_hv, [2, 3])

    # Average
    return (out1 + out2 + out3 + out4) / 4.0
