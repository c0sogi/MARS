class ShallowUNet(nn.Module):
    """
    Shallow U-Net Architecture (3 Levels).
    Encoder: 32 -> 64 -> 128
    Decoder: 128 -> 64 -> 32
    """

    def __init__(
        self, n_channels=IN_CHANNELS, n_classes=OUT_CHANNELS, base_filters=BASE_FILTERS
    ):
        super(ShallowUNet, self).__init__()
        self.n_channels = n_channels
        self.n_classes = n_classes

        # Encoder
        self.inc = DoubleConv(n_channels, base_filters)  # Level 1
        self.down1 = nn.Sequential(
            nn.MaxPool2d(2), DoubleConv(base_filters, base_filters * 2)  # Level 2
        )
        self.down2 = nn.Sequential(
            nn.MaxPool2d(2),
            DoubleConv(base_filters * 2, base_filters * 4),  # Level 3 (Bottleneck)
        )

        # Decoder
        self.up1 = Up(base_filters * 4, base_filters * 2)  # Level 2
        self.up2 = Up(base_filters * 2, base_filters)  # Level 1

        # Output Head
        self.outc = nn.Conv2d(base_filters, n_classes, kernel_size=1)

    def forward(self, x):
        x1 = self.inc(x)
        x2 = self.down1(x1)
        x3 = self.down2(x2)

        x = self.up1(x3, x2)
        x = self.up2(x, x1)

        # Output is pixel intensity 0-1
        return torch.sigmoid(self.outc(x))

class UNet(nn.Module):
    """
    Standard U-Net Architecture (4 Levels).
    Cite Lesson 00024: Deeper architecture allows aggregating global context.
    Encoder: 32 -> 64 -> 128 -> 256 -> 512
    Decoder: 512 -> 256 -> 128 -> 64 -> 32
    """

    def __init__(
        self, n_channels=IN_CHANNELS, n_classes=OUT_CHANNELS, base_filters=BASE_FILTERS
    ):
        super(UNet, self).__init__()
        self.n_channels = n_channels
        self.n_classes = n_classes

        # Encoder
        self.inc = DoubleConv(n_channels, base_filters)
        self.down1 = nn.Sequential(
            nn.MaxPool2d(2), DoubleConv(base_filters, base_filters * 2)
        )
        self.down2 = nn.Sequential(
            nn.MaxPool2d(2), DoubleConv(base_filters * 2, base_filters * 4)
        )
        self.down3 = nn.Sequential(
            nn.MaxPool2d(2), DoubleConv(base_filters * 4, base_filters * 8)
        )
        self.down4 = nn.Sequential(
            nn.MaxPool2d(2), DoubleConv(base_filters * 8, base_filters * 16)
        )

        # Decoder
        self.up1 = Up(base_filters * 16, base_filters * 8)
        self.up2 = Up(base_filters * 8, base_filters * 4)
        self.up3 = Up(base_filters * 4, base_filters * 2)
        self.up4 = Up(base_filters * 2, base_filters)

        # Output Head
        self.outc = nn.Conv2d(base_filters, n_classes, kernel_size=1)

        x1 = self.inc(x)
        x2 = self.down1(x1)
        x3 = self.down2(x2)
        x4 = self.down3(x3)
        x5 = self.down4(x4)

        x = self.up1(x5, x4)
        x = self.up2(x, x3)
        x = self.up3(x, x2)
        x = self.up4(x, x1)
