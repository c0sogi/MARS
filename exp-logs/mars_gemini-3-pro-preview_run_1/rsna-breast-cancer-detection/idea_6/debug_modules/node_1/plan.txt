def load_image(path):
    """
    Robust image loading function.
    Attempts to load using cv2, dicomsdl, pydicom, and fallbacks.
    Raises FileNotFoundError if file is missing or cannot be loaded.
    """
    full_path = os.path.join(INPUT_DIR, path)

    if not os.path.exists(full_path):
        raise FileNotFoundError(f"Image file not found: {full_path}")

    img = None

    # Method 1: OpenCV (Standard)
    try:
        img = cv2.imread(full_path, cv2.IMREAD_UNCHANGED)
    except Exception:
        pass

    # Method 2: dicomsdl (Fast DICOM library)
    if img is None and HAS_DICOMSDL:
        try:
            dcm = dicomsdl.open(full_path)
            img = dcm.pixelData()
        except Exception:
            pass

    # Method 3: Pydicom (Standard DICOM library)
    if img is None and HAS_PYDICOM:
        try:
            dcm = pydicom.dcmread(full_path)
            img = dcm.pixel_array
        except Exception:
            # Fallback: Try to decode raw bytes with cv2 (e.g. for JPEG 2000)
            try:
                if hasattr(dcm, "PixelData"):
                    img = cv2.imdecode(np.frombuffer(dcm.PixelData, np.uint8), cv2.IMREAD_UNCHANGED)
            except Exception:
                pass

    if img is None:
        raise FileNotFoundError(f"Failed to load image data from: {full_path}")

    # Ensure image is 2D (Grayscale)
    if len(img.shape) == 3:
        # If loaded as BGR/RGB, convert to Grayscale

        # If loaded as RGBA, convert
        elif img.shape[2] == 4:
