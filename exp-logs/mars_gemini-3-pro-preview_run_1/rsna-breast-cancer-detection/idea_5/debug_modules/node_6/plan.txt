def cleanup_gpu_resources():
    """
    Aggressively cleans up GPU memory by deleting exception hooks and forcing GC.
    Cite debug_lesson_6: Distinguish Static Model Memory from Dynamic Activation Memory.
    We must free pre-existing memory occupation from previous runs before allocating the new model.
    """
    print("Attempting to release GPU memory...")

    # Clear exception hooks which might hold references to previous stack frames
    # This is crucial in persistent environments where the last exception keeps
    # the entire stack (and thus local variables/tensors) alive.
    if hasattr(sys, "last_traceback"):
        del sys.last_traceback
    if hasattr(sys, "last_value"):
        del sys.last_value
    if hasattr(sys, "last_type"):
        del sys.last_type

    # Force Garbage Collection to release Python objects
    gc.collect()

    # Clear PyTorch CUDA Cache
    if torch.cuda.is_available():
        torch.cuda.empty_cache()
        torch.cuda.ipc_collect()

        # Print Status to verify clearance
        try:
            free, total = torch.cuda.mem_get_info()
            print(
                f"GPU Memory Status: {free/1024**3:.2f}GB Free / {total/1024**3:.2f}GB Total"
            )
