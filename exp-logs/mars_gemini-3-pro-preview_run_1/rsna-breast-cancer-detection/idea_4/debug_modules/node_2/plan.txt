class SimpleDicomReader:
    """
    A minimal DICOM reader to extract pixel data when pydicom is unavailable.
    Focuses on finding the PixelData tag (7FE0, 0010) and extracting the bytes.
    Handles Encapsulated (JPEG/JPEG2000) data by concatenating fragments.
    """

    def __init__(self, path):
        self.path = path
        with open(path, "rb") as f:
            self.data = f.read()

    def get_cv2_image(self):
        data = self.get_pixel_data()
        if not data:
            return None
        # Convert bytes to numpy array for cv2
        buf = np.frombuffer(data, dtype=np.uint8)
        # Load as Unchanged to preserve bit-depth (often 16-bit for mammograms)
        img = cv2.imdecode(buf, cv2.IMREAD_UNCHANGED)
        return img

    def get_pixel_data(self):
        # Search for Pixel Data Tag (7FE0, 0010)
        # Little Endian byte sequence: E0 7F 10 00
        marker = b"\xe0\x7f\x10\x00"
        offset = self.data.find(marker)

        if offset == -1:
            # Fallback: Try to find JPEG/JPEG2000 headers directly
            return self._try_find_image_header()

        # Found the tag. Parse VR and Length.
        # Check VR (Value Representation) at offset + 4
        vr = self.data[offset + 4 : offset + 6]

        if vr in [b"OB", b"OW"]:
            # Explicit VR: VR (2 bytes) + Reserved (2 bytes) + Length (4 bytes)
            length_bytes = self.data[offset + 8 : offset + 12]
            length = struct.unpack("<I", length_bytes)[0]
            value_start = offset + 12
        else:
            # Implicit VR or Explicit with short length.
            # Assuming Implicit VR for simplicity if not OB/OW: Length (4 bytes)
            length_bytes = self.data[offset + 4 : offset + 8]
            length = struct.unpack("<I", length_bytes)[0]
            value_start = offset + 8

        # Check for Undefined Length (0xFFFFFFFF) -> Encapsulated Format
        if length == 0xFFFFFFFF:
            return self._read_encapsulated(value_start)
        else:
            # Native format (uncompressed)
            return self.data[value_start : value_start + length]

    def _read_encapsulated(self, start_offset):
        # Encapsulated data consists of Fragments.
        # Structure: Item Tag (FFFE E000) + Length (4 bytes) + Data
        current = start_offset
        fragments = []

        while current < len(self.data):
            # Read Item Tag
            tag = self.data[current : current + 4]

            if tag == b"\xfe\xff\xdd\xe0":  # Sequence Delimiter
                break

            if tag != b"\xfe\xff\x00\xe0":  # Item Tag
                # If we lost the track, search for next Item Tag
                next_item = self.data.find(b"\xfe\xff\x00\xe0", current)
                if next_item == -1:
                    break
                current = next_item
                continue

            # Read Length
            length = struct.unpack("<I", self.data[current + 4 : current + 8])[0]
            data_start = current + 8

            if length > 0:
                fragments.append(self.data[data_start : data_start + length])

            current = data_start + length

        if not fragments:
            return self._try_find_image_header()

        # If multiple fragments, the first is usually the Basic Offset Table (often empty).
        # We concatenate everything and let cv2 find the start of the stream.
        combined = b"".join(fragments)
        return self._strip_to_image_start(combined)

    def _try_find_image_header(self):
        # Fallback: Search for JPEG/J2K markers globally
        # J2K SOC: FF 4F FF 51
        j2k_start = self.data.find(b"\xff\x4f\xff\x51")
        if j2k_start != -1:
            return self.data[j2k_start:]

        # JPEG SOI: FF D8
        jpeg_start = self.data.find(b"\xff\xd8")
        if jpeg_start != -1:
            return self.data[jpeg_start:]

        return b""

    def _strip_to_image_start(self, blob):
        # Find start of image in the blob to avoid garbage at the beginning
        j2k = blob.find(b"\xff\x4f\xff\x51")
        if j2k != -1:
            return blob[j2k:]

        jpeg = blob.find(b"\xff\xd8")
        if jpeg != -1:
            return blob[jpeg:]

        return blob
