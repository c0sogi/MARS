from library.train_eval import train_one_epoch, evaluate, inference


def cleanup_resources():
    """
    Forces a cleanup of GPU memory and system tracebacks.
    """
    print("Cleaning up GPU resources...")
    # Clear system tracebacks that might hold references to old frames/tensors
    if hasattr(sys, "last_traceback"):
        del sys.last_traceback
    if hasattr(sys, "last_value"):
        del sys.last_value
    if hasattr(sys, "last_type"):
        del sys.last_type

    # Force garbage collection
    gc.collect()

    # Clear CUDA cache
    if torch.cuda.is_available():
        torch.cuda.empty_cache()

    print("Cleanup complete.")
