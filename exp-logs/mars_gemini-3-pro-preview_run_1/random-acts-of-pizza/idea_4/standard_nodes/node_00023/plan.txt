def engineer_text_content_features(df):
    """
    Generates explicit counts for persuasion-related keywords.
    Cite Lesson 00013: Explicit text meta-features enhance tree models.
    """
    if config.TEXT_COL in df.columns:
        # Lowercase for matching
        texts = df[config.TEXT_COL].fillna("").astype(str).str.lower()

        # Financial stress (Need)
        financial_keywords = [
            "money",
            "broke",
            "job",
            "rent",
            "bill",
            "paycheck",
            "cash",
            "funds",
        ]
        df["text_financial_count"] = texts.apply(
            lambda x: sum(x.count(w) for w in financial_keywords)
        )

        # Gratitude (Politeness)
        gratitude_keywords = ["thank", "appreciate", "grateful"]
        df["text_gratitude_count"] = texts.apply(
            lambda x: sum(x.count(w) for w in gratitude_keywords)
        )

        # Reciprocity (Social Exchange)
        reciprocity_keywords = ["repay", "return", "forward"]
        df["text_reciprocity_count"] = texts.apply(
            lambda x: sum(x.count(w) for w in reciprocity_keywords)
        )

    return df
