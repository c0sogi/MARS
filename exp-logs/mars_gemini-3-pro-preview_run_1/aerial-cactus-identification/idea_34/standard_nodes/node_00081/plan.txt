The issue is that the meta-learner is trained on the validation set features (`X_val`) and then evaluated on the same set to report the final metric, causing data leakage and invalid scores. To fix this, I will implement Nested Cross-Validation within `train_meta_learner` to generate unbiased Out-Of-Fold (OOF) predictions for scoring, while still retraining the final model on the full dataset for inference. This directly applies the strategy described in `debug_lesson_6`.

    def train_meta_learner(self, X_val, y_val):
        """
        Trains a Logistic Regression meta-learner on the geometric features.
        Uses Nested Cross-Validation to generate unbiased OOF predictions for scoring.
        """
        print("Training Meta-Learner (Logistic Regression) with Internal CV...")

        # Internal CV for valid metric
        skf = StratifiedKFold(n_splits=5, shuffle=True, random_state=Config.SEED)
        oof_preds = np.zeros(len(y_val))

        # Generate OOF predictions to evaluate the meta-learner without leakage
        for train_idx, valid_idx in skf.split(X_val, y_val):
            X_train_fold, X_valid_fold = X_val[train_idx], X_val[valid_idx]
            y_train_fold, y_valid_fold = y_val[train_idx], y_val[valid_idx]

            model = LogisticRegression(
                solver="liblinear", random_state=Config.SEED, C=1.0
            )
            model.fit(X_train_fold, y_train_fold)
            oof_preds[valid_idx] = model.predict_proba(X_valid_fold)[:, 1]

        cv_auc = roc_auc_score(y_val, oof_preds)
        print(f"Meta-Learner CV AUC: {cv_auc}")

        # Final fit on all data for inference
        print("Retraining Meta-Learner on full validation set for inference...")
        clf = LogisticRegression(solver="liblinear", random_state=Config.SEED, C=1.0)
        clf.fit(X_val, y_val)

        # Save Model
        joblib.dump(clf, Config.META_LEARNER_PATH)
        print(f"Meta-Learner saved to {Config.META_LEARNER_PATH}")

        return clf, oof_preds
