class DepthProjectionCNN(nn.Module):
    """
    A lightweight fully convolutional neural network for 3D to 2D ink detection.

    Architecture:
    1. Depth Projection: 1x1 Conv to compress Z-slices (65) -> Feature map (16).
    2. Spatial Denoising: Stack of 3x3 Convs to aggregate spatial context.
    3. Classification: 1x1 Conv to output pixel-wise logits.
    """

    def __init__(self, in_channels=Z_DIM):
        super(DepthProjectionCNN, self).__init__()

        # Stage 1: Learnable Surface Projection
        # Compresses depth information into 2D feature maps
        self.projection = nn.Sequential(
            nn.Conv2d(in_channels, 16, kernel_size=1, bias=False),
            nn.BatchNorm2d(16),
            nn.ReLU(inplace=True),
        )

        # Stage 2: Spatial Denoising
        # Aggregates context to form coherent strokes
        self.spatial = nn.Sequential(
            nn.Conv2d(16, 32, kernel_size=3, padding=1, bias=False),
            nn.BatchNorm2d(32),
            nn.ReLU(inplace=True),
            nn.Conv2d(32, 32, kernel_size=3, padding=1, bias=False),
            nn.BatchNorm2d(32),
            nn.ReLU(inplace=True),
        )

        # Stage 3: Classification Head
        # Projects features to a single channel logit map
        self.classifier = nn.Conv2d(32, 1, kernel_size=1)

    def forward(self, x):
        # Input shape: (Batch, Z_DIM, H, W)
        x = self.projection(x)
        x = self.spatial(x)
        logits = self.classifier(x)
        return logits

class DilatedBlock(nn.Module):
    """
    Residual block with dilated convolutions to expand receptive field.
    """

    def __init__(self, channels, dilation=1):
        super(DilatedBlock, self).__init__()
        self.conv1 = nn.Conv2d(
            channels,
            channels,
            kernel_size=3,
            padding=dilation,
            dilation=dilation,
            bias=False,
        )
        self.bn1 = nn.BatchNorm2d(channels)
        self.conv2 = nn.Conv2d(
            channels,
            channels,
            kernel_size=3,
            padding=dilation,
            dilation=dilation,
            bias=False,
        )
        self.bn2 = nn.BatchNorm2d(channels)
        self.relu = nn.ReLU(inplace=True)

    def forward(self, x):
        residual = x
        out = self.relu(self.bn1(self.conv1(x)))
        out = self.bn2(self.conv2(out))
        out += residual
        return self.relu(out)
