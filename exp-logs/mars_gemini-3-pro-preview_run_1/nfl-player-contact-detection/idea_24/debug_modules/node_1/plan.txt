class CatBoostExpert(ModelWrapper):
    """
    CatBoost Expert utilizing Symmetric/Oblivious trees for robust generalization.
    """

    def __init__(self):
        super().__init__()
        self.params = Config.CAT_PARAMS.copy()
        # Ensure iterations is set from Config
        self.params["iterations"] = Config.NUM_BOOST_ROUND
        self.params["early_stopping_rounds"] = Config.EARLY_STOPPING_ROUNDS

    def fit(self, X_train, y_train, X_val=None, y_val=None):
        self.model = CatBoostClassifier(**self.params)

        eval_set = None
        if X_val is not None and y_val is not None:
            eval_set = (X_val, y_val)

        self.model.fit(
            X_train,
            y_train,
            eval_set=eval_set,
            verbose=Config.VERBOSE_EVAL,
            use_best_model=True,
        )

    def predict(self, X):
        if self.model is None:
            raise ValueError("Model has not been trained yet.")
        # CatBoostClassifier predict_proba returns [prob_0, prob_1]
        return self.model.predict_proba(X)[:, 1]
