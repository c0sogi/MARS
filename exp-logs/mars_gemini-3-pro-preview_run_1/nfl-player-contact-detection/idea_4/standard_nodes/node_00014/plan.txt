    def _compute_spatial_density(self, df_track):
        """
        Computes the number of other players within 1.5 yards for each player/step.
        Also computes distance to the nearest neighbor.
        Uses cKDTree for efficiency.
        """
        print("Computing spatial density and nearest neighbor features...")

        # Initialize result arrays
        density_map = np.zeros(len(df_track), dtype=np.int8)
        nearest_dist_map = np.full(len(df_track), 100.0, dtype=np.float32) # Default large distance

        # Filter columns to minimal needed
        coords = df_track[["x_position", "y_position"]].values
        
        # Group by game_play and step
        df_track["_frame_id"] = (
            df_track["game_play"].astype(str) + "_" + df_track["step"].astype(str)
        )
        groups = df_track.groupby("_frame_id")

        for _, group_indices in tqdm(
            groups.indices.items(), desc="Spatial Context", miniters=1000
        ):
            # Get coordinates for this frame
            frame_coords = coords[group_indices]

            if len(frame_coords) > 1:
                # Build Tree
                tree = cKDTree(frame_coords)

                # 1. Density: count neighbors within 1.5 yards
                neighbors = tree.query_ball_point(frame_coords, r=1.5)
                counts = [len(n) - 1 for n in neighbors]
                density_map[group_indices] = counts

                # 2. Nearest Neighbor Distance
                if getattr(self.config, "USE_NEAREST_DIST", False):
                    # k=2 because the closest is the point itself (dist=0)
                    # query returns (distances, indices)
                    dists, _ = tree.query(frame_coords, k=2)
                    # Column 1 is the nearest neighbor distance
                    # If k > n points, scipy fills with inf, but we checked len > 1
                    if dists.shape[1] >= 2:
                        nearest_dist_map[group_indices] = dists[:, 1]
            else:
                density_map[group_indices] = 0
                # nearest_dist_map remains default (100.0)

        df_track["spatial_density"] = density_map
        if getattr(self.config, "USE_NEAREST_DIST", False):
            df_track["dist_nearest"] = nearest_dist_map
        else:
            df_track["dist_nearest"] = 0.0
            
        df_track = df_track.drop(columns=["_frame_id"])
        return df_track
