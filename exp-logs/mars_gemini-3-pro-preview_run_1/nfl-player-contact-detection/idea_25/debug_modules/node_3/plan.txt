        Fits all three models in the ensemble.

        Args:
            X (pd.DataFrame or np.ndarray): Training features.
            y (pd.Series or np.ndarray): Training labels (can be soft/continuous).
            eval_set (list of tuple): List containing [(X_val, y_val)] for early stopping.

        Returns:
            self: Returns the instance itself.
        """
        print("Fitting Tri-Ensemble...")

        # 1. Dynamic scale_pos_weight calculation for XGBoost
        # Calculate ratio of negative to positive samples (works for soft labels too)
        n_pos = np.sum(y)
        n_neg = len(y) - n_pos
        scale_pos_weight = n_neg / n_pos if n_pos > 0 else 1.0

        # Update XGBoost parameter
        self.xgb.set_params(scale_pos_weight=scale_pos_weight)
        print(f"  [XGBoost] Dynamic scale_pos_weight set to: {scale_pos_weight:.4f}")

        # 2. Fit LightGBM
        print("  [LightGBM] Training...")
        callbacks_lgb = [
            lgb.early_stopping(
                stopping_rounds=Config.EARLY_STOPPING_ROUNDS, verbose=False
            ),
            lgb.log_evaluation(period=0),  # Suppress log output
        ]
        self.lgbm.fit(
            X,
            y,
            eval_set=eval_set,
            eval_metric="average_precision",
            callbacks=callbacks_lgb,
        )
        if eval_set:
            # Print validation score manually since we suppressed logs
            # Access best_score_ safely
            try:
                valid_key = list(self.lgbm.best_score_.keys())[0]
                metric_key = list(self.lgbm.best_score_[valid_key].keys())[0]
                score = self.lgbm.best_score_[valid_key][metric_key]
                print(f"  [LightGBM] Best Validation {metric_key}: {score}")
            except (IndexError, AttributeError):
                print("  [LightGBM] Validation score not available.")

        # 3. Fit XGBoost
        print("  [XGBoost] Training...")
        self.xgb.fit(X, y, eval_set=eval_set, verbose=False)

        if eval_set:

            results = self.xgb.evals_result()

                final_loss = results["validation_0"][metric_name][-1]

        Returns the unweighted average of the models.

        Args:

        Returns:
            np.ndarray: Shape (n_samples, 2), probability of class 0 and 1.
        """
        if not self.is_fitted:
            raise RuntimeError("Model is not fitted yet.")

        # Get probabilities from each model
        # Regressors with binary objective return 1D array of p(y=1)
        p_lgbm_1 = self.lgbm.predict(X)
        p_xgb_1 = self.xgb.predict(X)

        p_lgbm = np.vstack([1.0 - p_lgbm_1, p_lgbm_1]).T

        avg_proba = (p_lgbm + p_xgb) / 2.0
        return avg_proba
