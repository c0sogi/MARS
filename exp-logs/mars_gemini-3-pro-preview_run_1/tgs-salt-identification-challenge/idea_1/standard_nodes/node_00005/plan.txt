                inputs = inputs.to(self.device)
                masks = masks.to(self.device)
                batch_size = inputs.size(0)

                # Forward pass (Original)
                logits = self.model(inputs)
                loss = self.criterion(logits, masks)
                running_loss += loss.item() * batch_size

                # TTA for IoU Calculation (Cite solution_lesson_node_00004)
                probs = torch.sigmoid(logits)
                
                # Flip inputs horizontally
                inputs_flipped = torch.flip(inputs, dims=[3])
                logits_flipped = self.model(inputs_flipped)
                probs_flipped = torch.sigmoid(logits_flipped)
                probs_flipped_back = torch.flip(probs_flipped, dims=[3])
                
                # Average predictions
                avg_probs = (probs + probs_flipped_back) / 2.0

                # calculate_iou_batch returns the mean IoU for the batch
                batch_iou = calculate_iou_batch(avg_probs, masks, threshold=0.5)

                # Accumulate weighted IoU

        val_loss = running_loss / len(val_loader.dataset)
