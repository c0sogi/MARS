class CompoundLoss(nn.Module):
    def __init__(self):
        super(CompoundLoss, self).__init__()
        self.bce_seg = nn.BCEWithLogitsLoss()
        self.dice = DiceLoss()
        self.lovasz = LovaszHingeLoss()
        self.bce_cls = nn.BCEWithLogitsLoss()

    def forward(self, outputs, targets):
        """
        outputs: Dictionary containing:
            - 'logits': Segmentation logits (B, 1, H, W)
            - 'cls': Classification logits (B, 1)
        targets: Ground truth masks (B, 1, H, W) or (B, H, W)
        """
        seg_logits = outputs["logits"]
        cls_logits = outputs["cls"]

        # Ensure targets are (B, 1, H, W) float for BCE/Dice
        if targets.dim() == 3:
            targets = targets.unsqueeze(1)
        targets = targets.float()

        # 1. Segmentation Losses
        # BCE
        l_bce = self.bce_seg(seg_logits, targets)

        # Dice (expects logits, applies sigmoid internally)
        l_dice = self.dice(seg_logits, targets)

        # Lovasz (expects logits, handles flattening internally)
        l_lovasz = self.lovasz(seg_logits, targets)

        # Combined Segmentation Loss
        l_seg = l_bce + l_dice + 0.1 * l_lovasz

        # 2. Classification Loss
        # Derive targets: 1 if salt exists in mask, 0 otherwise
        # targets is (B, 1, H, W) -> flatten to (B, -1) -> sum -> check > 0
        batch_size = targets.size(0)
        has_salt = targets.view(batch_size, -1).sum(1) > 0
        cls_targets = has_salt.float().view(batch_size, 1)

        l_cls = self.bce_cls(cls_logits, cls_targets)

        # 3. Total Loss
        total_loss = l_seg + 0.05 * l_cls

        return total_loss

class GlobalContextBlock(nn.Module):
    def __init__(self, in_channels, ratio=16):
        super(GlobalContextBlock, self).__init__()
        self.conv_mask = nn.Conv2d(in_channels, 1, kernel_size=1)
        self.softmax = nn.Softmax(dim=1)

        self.channel_add_conv = nn.Sequential(
            nn.Conv2d(in_channels, in_channels // ratio, kernel_size=1),
            nn.LayerNorm([in_channels // ratio, 1, 1]),
            nn.ReLU(inplace=True),
            nn.Conv2d(in_channels // ratio, in_channels, kernel_size=1),
        )

    def forward(self, x):
        b, c, h, w = x.size()
        # Context Modeling
        input_x = x
        input_x = input_x.view(b, c, h * w)
        context_mask = self.conv_mask(x)
        context_mask = context_mask.view(b, 1, h * w)
        context_mask = self.softmax(context_mask)
        context_mask = context_mask.permute(0, 2, 1)  # (B, HW, 1)

        context = torch.matmul(input_x, context_mask)  # (B, C, 1)
        context = context.view(b, c, 1, 1)

        # Transform
        context_transform = self.channel_add_conv(context)

        # Fusion
        return x + context_transform
