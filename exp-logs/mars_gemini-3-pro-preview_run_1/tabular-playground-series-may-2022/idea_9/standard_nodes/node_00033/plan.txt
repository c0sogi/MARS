class DiGUT(nn.Module):
    """
    Discriminative Granular Unified Transformer (DiGUT).

    Architecture:
    1. Granular Input:
       - Numerical features -> Linear Projection + Feature Identity Embedding
       - Sequence features -> Entity Embedding + Positional Embedding
       - [CLS] Token prepended
    2. Transformer Encoder: Stack of layers with GEGLU.
    3. Dual Heads:
       - Target Head: MLP on [CLS] for state prediction.
       - Discriminator Head: Linear on feature tokens for corruption detection.
    """

    def __init__(
        self,
        num_numerical_features: int,
        vocab_size: int,
        sequence_length: int,
        config: Config,
    ):
        super().__init__()
        self.config = config
        dim = config.HIDDEN_DIM

        # ---------------------------------------------------------------------
        # 1. Granular Embeddings
        # ---------------------------------------------------------------------

        # Numerical: Project scalar to vector + Add Feature Identity
        # We use a shared linear projection for all numerical features to map R^1 -> R^d
        self.num_proj = nn.Linear(1, dim)
        # Learnable embedding for each numerical feature index (to distinguish f_00 from f_01 etc.)
        self.num_id_emb = nn.Parameter(torch.zeros(1, num_numerical_features, dim))
        nn.init.normal_(self.num_id_emb, std=0.02)

        # Sequence: Entity Embedding + Positional Embedding
        self.seq_emb = nn.Embedding(vocab_size, dim)
        # Learnable positional embedding for the sequence part
        self.seq_pos_emb = nn.Parameter(torch.zeros(1, sequence_length, dim))
        nn.init.normal_(self.seq_pos_emb, std=0.02)

        # [CLS] Token
        self.cls_token = nn.Parameter(torch.zeros(1, 1, dim))
        nn.init.normal_(self.cls_token, std=0.02)

        # Dropout applied after embeddings
        self.emb_dropout = nn.Dropout(config.DROPOUT)

        # ---------------------------------------------------------------------
        # 2. Transformer Encoder
        # ---------------------------------------------------------------------
        # Explicit instantiation using ModuleList
        self.layers = nn.ModuleList(
            [
                DiGUTLayer(
                    hidden_dim=dim,
                    num_heads=config.NUM_HEADS,
                    forward_dim=config.FORWARD_DIM,
                    dropout=config.DROPOUT,
                    attention_dropout=config.ATTENTION_DROPOUT,
                )
                for _ in range(config.NUM_LAYERS)
            ]
        )

        self.final_norm = nn.LayerNorm(dim)

        # ---------------------------------------------------------------------
        # 3. Heads
        # ---------------------------------------------------------------------

        # Target Head (High-Capacity MLP)
        self.target_head = nn.Sequential(
            nn.Linear(dim, dim),
            nn.GELU(),
            nn.Dropout(config.DROPOUT),
            nn.Linear(dim, 1),
        )

        # Discriminator Head (Token-wise Binary Classification)
        # Predicts probability of token being corrupted (1) or original (0)
        self.discriminator_head = nn.Linear(dim, 1)

        self._init_weights()

    def _init_weights(self):
        # Standard Transformer initialization
        for p in self.parameters():
            if p.dim() > 1:
                nn.init.xavier_uniform_(p)

    def forward(self, x_num, x_seq):
        """
        Args:
            x_num: (Batch, Num_Feats) - Continuous numerical features
            x_seq: (Batch, Seq_Len) - Integer indices for sequence features
        Returns:
            target_logits: (Batch, 1)
            discriminator_logits: (Batch, Total_Tokens - 1, 1)
                                  Note: Discriminator doesn't predict for [CLS]
        """
        B = x_num.shape[0]

        # --- Embedding Numerical ---
        # (B, N) -> (B, N, 1) -> (B, N, H)
        num_emb = self.num_proj(x_num.unsqueeze(-1))
        # Add feature identity
        num_emb = num_emb + self.num_id_emb

        # --- Embedding Sequence ---
        # (B, S) -> (B, S, H)
        seq_emb = self.seq_emb(x_seq)
        # Add positional info
        seq_emb = seq_emb + self.seq_pos_emb

        # --- Unified Sequence Construction ---
        # Expand CLS token to batch size: (1, 1, H) -> (B, 1, H)
        cls_tokens = self.cls_token.expand(B, -1, -1)

        # Concatenate: [CLS, Num_Feats, Seq_Feats]
        x = torch.cat((cls_tokens, num_emb, seq_emb), dim=1)

        x = self.emb_dropout(x)

        # --- Transformer Encoder ---
        for layer in self.layers:
            x = layer(x)

        x = self.final_norm(x)

        # --- Output Heads ---

        # 1. Target Prediction (from [CLS] token at index 0)
        cls_out = x[:, 0, :]
        target_logits = self.target_head(cls_out)

        # 2. Discriminator Prediction (from all feature tokens at indices 1:)
        # We want to detect corruption in the input features
        feature_out = x[:, 1:, :]
        discriminator_logits = self.discriminator_head(feature_out)

        return target_logits, discriminator_logits

class DeGUT(nn.Module):
    """
    Denoising Granular Unified Transformer (DeGUT).

    Architecture:
    1. Granular Input:
       - Numerical features -> Linear Projection + Feature Identity Embedding
       - Sequence features -> Entity Embedding + Positional Embedding
       - [CLS] Token prepended
       - [MASK] Token for denoising task
    2. Transformer Encoder: Stack of layers with GEGLU.
    3. Dual Heads:
       - Target Head: MLP on [CLS] for state prediction.
       - Reconstruction Heads: Predict original values of masked tokens.
    """

    def __init__(
        self,
        num_numerical_features: int,
        vocab_size: int,
        sequence_length: int,
        config: Config,
    ):
        super().__init__()
        self.config = config
        dim = config.HIDDEN_DIM

        # ---------------------------------------------------------------------
        # 1. Granular Embeddings
        # ---------------------------------------------------------------------

        # Numerical: Project scalar to vector + Add Feature Identity
        self.num_proj = nn.Linear(1, dim)
        self.num_id_emb = nn.Parameter(torch.zeros(1, num_numerical_features, dim))
        nn.init.normal_(self.num_id_emb, std=0.02)

        # Sequence: Entity Embedding + Positional Embedding
        self.seq_emb = nn.Embedding(vocab_size, dim)
        self.seq_pos_emb = nn.Parameter(torch.zeros(1, sequence_length, dim))
        nn.init.normal_(self.seq_pos_emb, std=0.02)

        # [CLS] Token
        self.cls_token = nn.Parameter(torch.zeros(1, 1, dim))
        nn.init.normal_(self.cls_token, std=0.02)

        # [MASK] Token (Shared for both modalities)
        self.mask_emb = nn.Parameter(torch.zeros(1, 1, dim))
        nn.init.normal_(self.mask_emb, std=0.02)

        # Dropout applied after embeddings
        self.emb_dropout = nn.Dropout(config.DROPOUT)

        # ---------------------------------------------------------------------
        # 2. Transformer Encoder
        # ---------------------------------------------------------------------
        self.layers = nn.ModuleList(
            [
                DiGUTLayer(
                    hidden_dim=dim,
                    num_heads=config.NUM_HEADS,
                    forward_dim=config.FORWARD_DIM,
                    dropout=config.DROPOUT,
                    attention_dropout=config.ATTENTION_DROPOUT,
                )
                for _ in range(config.NUM_LAYERS)
            ]
        )

        self.final_norm = nn.LayerNorm(dim)

        # ---------------------------------------------------------------------
        # 3. Heads
        # ---------------------------------------------------------------------

        # Target Head
        self.target_head = nn.Sequential(
            nn.Linear(dim, dim),
            nn.GELU(),
            nn.Dropout(config.DROPOUT),
            nn.Linear(dim, 1),
        )

        # Reconstruction Heads (Cite solution_lesson_node_00032)
        # Numerical: Regress the scalar value
        self.num_recon_head = nn.Linear(dim, 1)
        # Sequence: Predict the character class
        self.seq_recon_head = nn.Linear(dim, vocab_size)

        self._init_weights()

    def _init_weights(self):
        for p in self.parameters():
            if p.dim() > 1:
                nn.init.xavier_uniform_(p)

    def forward(self, x_num, x_seq, mask_num=None, mask_seq=None):
        """
        Args:
            x_num: (Batch, Num_Feats)
            x_seq: (Batch, Seq_Len)
            mask_num: (Batch, Num_Feats) Boolean mask (True = Masked)
            mask_seq: (Batch, Seq_Len) Boolean mask (True = Masked)
        """
        B = x_num.shape[0]

        # --- Embedding Numerical ---
        num_emb = self.num_proj(x_num.unsqueeze(-1))
        
        # Apply Masking to Numerical Embeddings
        if mask_num is not None:
            mask_emb_expanded = self.mask_emb.expand(B, num_emb.size(1), -1)
            num_emb = torch.where(mask_num.unsqueeze(-1), mask_emb_expanded, num_emb)

        num_emb = num_emb + self.num_id_emb

        # --- Embedding Sequence ---
        seq_emb = self.seq_emb(x_seq)

        # Apply Masking to Sequence Embeddings
        if mask_seq is not None:
            mask_emb_expanded = self.mask_emb.expand(B, seq_emb.size(1), -1)
            seq_emb = torch.where(mask_seq.unsqueeze(-1), mask_emb_expanded, seq_emb)

        seq_emb = seq_emb + self.seq_pos_emb

        # --- Unified Sequence Construction ---
        cls_tokens = self.cls_token.expand(B, -1, -1)
        x = torch.cat((cls_tokens, num_emb, seq_emb), dim=1)

        x = self.emb_dropout(x)

        # --- Transformer Encoder ---
        for layer in self.layers:
            x = layer(x)

        x = self.final_norm(x)

        # --- Output Heads ---

        # 1. Target Prediction
        cls_out = x[:, 0, :]
        target_logits = self.target_head(cls_out)

        # 2. Reconstruction Prediction
        # Extract tokens corresponding to features
        num_tokens = x[:, 1 : 1 + num_emb.size(1), :]
        seq_tokens = x[:, 1 + num_emb.size(1) :, :]

        num_preds = self.num_recon_head(num_tokens).squeeze(-1)
        seq_preds = self.seq_recon_head(seq_tokens)

        return target_logits, num_preds, seq_preds

def apply_swap_noise(x_num, x_seq, swap_prob):
    """
    Applies swap noise to a batch of data on the GPU.
    Replaces tokens with values from the same feature in a random row within the batch.

    Args:
        x_num: (Batch, Num_Features) FloatTensor
        x_seq: (Batch, Seq_Len) LongTensor
        swap_prob: Probability of swapping a token

    Returns:
        x_num_corr: Corrupted numerical features
        x_seq_corr: Corrupted sequence features
        mask_combined: (Batch, Total_Tokens) Binary mask (1=corrupted, 0=original)
    """
    B = x_num.size(0)
    device = x_num.device

    # Generate random permutation of the batch
    perm = torch.randperm(B, device=device)

    # --- Numerical Corruption ---
    # Create mask: 1 if we should swap, 0 otherwise
    mask_num = (torch.rand(x_num.shape, device=device) < swap_prob).float()
    # Apply swap: mask * permuted + (1-mask) * original
    x_num_corr = mask_num * x_num[perm] + (1 - mask_num) * x_num

    # --- Sequence Corruption ---
    # Create boolean mask
    mask_seq = torch.rand(x_seq.shape, device=device) < swap_prob
    # Apply swap using torch.where for LongTensor
    x_seq_corr = torch.where(mask_seq, x_seq[perm], x_seq)

    # --- Combine Masks ---
    # Concatenate numerical and sequence masks to match discriminator output
    mask_combined = torch.cat([mask_num, mask_seq.float()], dim=1)

    return x_num_corr, x_seq_corr, mask_combined


def train_one_epoch(model, dataloader, optimizer, scheduler, device, config):
    model.train()

    total_loss = 0.0
    total_target_loss = 0.0
    total_disc_loss = 0.0

    # Loss functions
    # Main target loss with label smoothing
    criterion_target = nn.BCEWithLogitsLoss()
    # Auxiliary discriminator loss
    criterion_disc = nn.BCEWithLogitsLoss()

    for batch_idx, (x_num, x_seq, target, _) in enumerate(dataloader):
        # Move to GPU
        x_num = x_num.to(device)
        x_seq = x_seq.to(device)
        target = target.to(device).unsqueeze(1)  # (B) -> (B, 1)

        # Apply GPU-based Swap Noise
        # Note: We ignore the mask returned by the dataset and generate fresh ones on GPU
        x_num_corr, x_seq_corr, mask_combined = apply_swap_noise(
            x_num, x_seq, config.SWAP_PROB
        )

        optimizer.zero_grad()

        # Forward pass with corrupted data
        target_logits, disc_logits = model(x_num_corr, x_seq_corr)

        # Calculate Losses
        loss_target = criterion_target(target_logits, target)

        # Discriminator loss: Predict mask (1=corrupted)
        # disc_logits: (B, N+S, 1) -> squeeze to (B, N+S)
        loss_disc = criterion_disc(disc_logits.squeeze(-1), mask_combined)

        # Composite Loss
        loss = loss_target + config.AUX_WEIGHT * loss_disc

        # Backward
        loss.backward()

        # Gradient Clipping (optional but good practice for Transformers)
        torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)

        optimizer.step()
        if scheduler is not None:
            scheduler.step()

        # Logging
        total_loss += loss.item()
        total_target_loss += loss_target.item()
        total_disc_loss += loss_disc.item()

    avg_loss = total_loss / len(dataloader)
    return avg_loss
