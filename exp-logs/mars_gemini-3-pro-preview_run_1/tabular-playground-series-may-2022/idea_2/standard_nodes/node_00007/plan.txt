def add_bag_of_characters(series, vocab):
    """
    Calculates the count of each character in the vocabulary for each string.
    Returns a numpy array of shape (N, len(vocab)).
    """
    # vocab maps char -> int (1-based)
    # We want a column for each char in vocab
    # Sort keys by value to ensure consistent order
    sorted_chars = sorted(vocab.keys(), key=lambda k: vocab[k])
    char_to_col = {c: i for i, c in enumerate(sorted_chars)}

    n_samples = len(series)
    n_features = len(sorted_chars)
    counts = np.zeros((n_samples, n_features), dtype=np.float32)

    for i, s in enumerate(series):
        for char in s:
            if char in char_to_col:
                counts[i, char_to_col[char]] += 1
    return counts
