class ResidualDepthInjector(nn.Module):
    """
    Injects depth information into the feature map via a residual connection.
    Initialized to perform identity mapping (zero residual) to preserve pretrained features.
    """

    def __init__(self, in_channels, embedding_dim=32):
        super().__init__()

        # MLP to project scalar depth to embedding
        self.depth_mlp = nn.Sequential(
            nn.Linear(1, embedding_dim),
            nn.ReLU(inplace=True),
            nn.Linear(embedding_dim, embedding_dim),
        )

        # Fusion block: Concatenates features + depth embedding -> transforms
        self.fusion = nn.Sequential(
            nn.Conv2d(
                in_channels + embedding_dim, in_channels, kernel_size=1, bias=False
            ),
            nn.BatchNorm2d(in_channels),
            nn.ReLU(inplace=True),
        )

        # Projection to calculate the 'delta' (residual)
        # Initialized to zero so the module starts as an identity function
        self.project = nn.Conv2d(in_channels, in_channels, kernel_size=1, bias=True)
        nn.init.constant_(self.project.weight, 0)
        nn.init.constant_(self.project.bias, 0)

    def forward(self, x, z):
        """
        Args:
            x: Feature map (B, C, H, W)
            z: Normalized depth scalar (B, 1)
        """
        B, C, H, W = x.shape

        # 1. Embed Depth
        d = self.depth_mlp(z)  # (B, emb_dim)

        # 2. Spatially Expand
        d = d.unsqueeze(2).unsqueeze(3)  # (B, emb_dim, 1, 1)
        d = d.expand(-1, -1, H, W)  # (B, emb_dim, H, W)

        # 3. Concatenate
        cat = torch.cat([x, d], dim=1)

        # 4. Compute Delta
        feat = self.fusion(cat)
        delta = self.project(feat)

        # 5. Add Residual
        return x + delta

def validate(model, loader, criterion, device):
    """
    Evaluates the model on the validation set using adaptive thresholding.
    Cite solution_lesson_node_00033: Align checkpointing with evaluation via adaptive thresholding.
    """
    model.eval()
    running_loss = 0.0
    dataset_size = 0
    
    all_preds = []
    all_masks = []

    with torch.no_grad():
        for images, masks, depths, _ in loader:
            batch_size = images.size(0)

            images = images.to(device)
            masks = masks.to(device)
            depths = depths.to(device)

            # Forward pass
            outputs = model(images, depths)

            # Loss
            loss = criterion(outputs, masks)
            running_loss += loss.item() * batch_size
            dataset_size += batch_size

            # Store predictions for threshold search
            preds = torch.sigmoid(outputs)
            all_preds.append(preds.cpu())
            all_masks.append(masks.cpu())

    epoch_loss = running_loss / dataset_size
    
    # Concatenate all batches
    all_preds = torch.cat(all_preds)
    all_masks = torch.cat(all_masks)
    
    # Search for optimal threshold
    best_score = 0.0
    thresholds = np.linspace(0.3, 0.7, 9)  # 0.3, 0.35, ..., 0.7
    
    for t in thresholds:
        score = do_kaggle_metric(all_preds, all_masks, threshold=t)
        if score > best_score:
            best_score = score

    return epoch_loss, best_score
