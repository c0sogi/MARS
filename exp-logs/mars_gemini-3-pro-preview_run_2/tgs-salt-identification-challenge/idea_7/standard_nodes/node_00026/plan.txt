class SaltNetLoss(nn.Module):
    """
    Composite Multi-Task Loss for Salt Segmentation.

    L_total = (L_bce + L_lovasz) + lambda * L_mse
    """

    def __init__(self):
        super(SaltNetLoss, self).__init__()
        self.bce = nn.BCEWithLogitsLoss()
        self.lovasz = LovaszHingeLoss()
        self.mse = nn.MSELoss()
        self.depth_weight = config.DEPTH_LOSS_WEIGHT

    def forward(self, seg_logits, depth_preds, seg_targets, depth_targets):
        """
        Args:
            seg_logits: (Tensor) Segmentation logits [B, 1, H, W]
            depth_preds: (Tensor) Depth predictions [B, 1]
            seg_targets: (Tensor) Ground truth masks [B, 1, H, W] or [B, H, W]
            depth_targets: (Tensor) Ground truth depths [B, 1] or [B]

        Returns:
            loss: (Tensor) Scalar loss value
            metrics: (Dict) Dictionary containing individual loss components for logging
        """
        # --- Segmentation Loss ---
        # Ensure targets are float for BCE
        if seg_targets.dtype != torch.float32:
            seg_targets = seg_targets.float()

        # Ensure shape alignment for BCE
        if seg_targets.ndim == 3:
            seg_targets = seg_targets.unsqueeze(1)

        loss_bce = self.bce(seg_logits, seg_targets)
        loss_lovasz = self.lovasz(seg_logits, seg_targets)

        total_seg_loss = loss_bce + loss_lovasz

        # --- Regression Loss ---
        # Ensure shape alignment for MSE
        if depth_targets.ndim == 1:
            depth_targets = depth_targets.unsqueeze(1)

        # Ensure types match
        if depth_targets.dtype != depth_preds.dtype:
            depth_targets = depth_targets.to(depth_preds.dtype)

        loss_depth = self.mse(depth_preds, depth_targets)

        # --- Total Loss ---
        total_loss = total_seg_loss + (self.depth_weight * loss_depth)

        metrics = {
            "loss_bce": loss_bce.item(),
            "loss_lovasz": loss_lovasz.item(),
            "loss_seg_total": total_seg_loss.item(),
            "loss_depth": loss_depth.item(),
            "loss_total": total_loss.item(),
        }

        return total_loss, metrics
