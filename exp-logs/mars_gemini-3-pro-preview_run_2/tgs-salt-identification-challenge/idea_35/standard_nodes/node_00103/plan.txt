class BCELovaszLoss(nn.Module):
    """
    Compound loss: BCE + Lovasz.
    Stabilizes training for segmentation tasks by combining
    pixel-wise smoothing (BCE) with IoU optimization (Lovasz).
    """

    def __init__(self):
        super().__init__()
        self.bce = nn.BCEWithLogitsLoss()
        self.lovasz = LovaszHingeLoss()

    def forward(self, logits, labels):
        return self.bce(logits, labels) + self.lovasz(logits, labels)
