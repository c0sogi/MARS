class FlatCNN(nn.Module):
    """
    A Plain Multi-Layer Convolutional Network (Flat-CNN).
    Maintains input resolution throughout the network.
    """

    def __init__(
        self,
        num_layers=Config.CNN_NUM_LAYERS,
        hidden_channels=Config.CNN_HIDDEN_CHANNELS,
    ):
        super(FlatCNN, self).__init__()

        layers = []
        # Input Layer: 1 channel (grayscale) -> hidden_channels
        layers.append(
            nn.Conv2d(1, hidden_channels, kernel_size=Config.CNN_KERNEL_SIZE, padding=1)
        )
        layers.append(nn.ReLU(inplace=True))

        # Hidden Layers
        # Total layers = num_layers. We already added 1, and will add 1 output layer.
        # So we add num_layers - 2 intermediate layers.
        for _ in range(num_layers - 2):
            layers.append(
                nn.Conv2d(
                    hidden_channels,
                    hidden_channels,
                    kernel_size=Config.CNN_KERNEL_SIZE,
                    padding=1,
                )
            )
            layers.append(nn.ReLU(inplace=True))

        # Output Layer: hidden_channels -> 1 channel
        layers.append(
            nn.Conv2d(hidden_channels, 1, kernel_size=Config.CNN_KERNEL_SIZE, padding=1)
        )
        layers.append(nn.Sigmoid())  # Output range [0, 1]

        self.net = nn.Sequential(*layers)

    def forward(self, x):
        return self.net(x)

class UNet(nn.Module):
    """
    U-Net Architecture for Image Denoising.
    Provides a larger receptive field to capture context.
    """

    def __init__(self):
        super(UNet, self).__init__()

        # Encoder
        self.enc1 = self.conv_block(1, 32)
        self.pool1 = nn.MaxPool2d(2)
        self.enc2 = self.conv_block(32, 64)
        self.pool2 = nn.MaxPool2d(2)
        self.enc3 = self.conv_block(64, 128)
        self.pool3 = nn.MaxPool2d(2)

        # Bottleneck
        self.bottleneck = self.conv_block(128, 256)

        # Decoder
        self.up3 = nn.ConvTranspose2d(256, 128, kernel_size=2, stride=2)
        self.dec3 = self.conv_block(256, 128)  # 128 from up + 128 from skip

        self.up2 = nn.ConvTranspose2d(128, 64, kernel_size=2, stride=2)
        self.dec2 = self.conv_block(128, 64)  # 64 from up + 64 from skip

        self.up1 = nn.ConvTranspose2d(64, 32, kernel_size=2, stride=2)
        self.dec1 = self.conv_block(64, 32)  # 32 from up + 32 from skip

        self.final = nn.Conv2d(32, 1, kernel_size=1)
        self.sigmoid = nn.Sigmoid()

    def conv_block(self, in_ch, out_ch):
        return nn.Sequential(
            nn.Conv2d(in_ch, out_ch, 3, padding=1),
            nn.ReLU(inplace=True),
            nn.Conv2d(out_ch, out_ch, 3, padding=1),
            nn.ReLU(inplace=True),
        )

    def forward(self, x):
        # Pad input to be divisible by 8 (2^3 pooling layers)
        h, w = x.shape[2], x.shape[3]
        pad_h = (8 - h % 8) % 8
        pad_w = (8 - w % 8) % 8

        if pad_h > 0 or pad_w > 0:
            x = torch.nn.functional.pad(x, (0, pad_w, 0, pad_h), mode="reflect")

        e1 = self.enc1(x)
        p1 = self.pool1(e1)

        e2 = self.enc2(p1)
        p2 = self.pool2(e2)

        e3 = self.enc3(p2)
        p3 = self.pool3(e3)

        b = self.bottleneck(p3)

        d3 = self.up3(b)
        # Skip connection concatenation
        d3 = torch.cat([d3, e3], dim=1)
        d3 = self.dec3(d3)

        d2 = self.up2(d3)
        d2 = torch.cat([d2, e2], dim=1)
        d2 = self.dec2(d2)

        d1 = self.up1(d2)
        d1 = torch.cat([d1, e1], dim=1)
        d1 = self.dec1(d1)

        out = self.final(d1)
        out = self.sigmoid(out)

        # Crop back to original size
        if pad_h > 0 or pad_w > 0:
            out = out[:, :, :h, :w]

        return out
