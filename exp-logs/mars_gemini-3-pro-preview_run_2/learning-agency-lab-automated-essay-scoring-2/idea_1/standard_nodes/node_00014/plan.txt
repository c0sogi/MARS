import torch
from torch.utils.data import Dataset


class EssayDataset(Dataset):
    """
    Custom Dataset for Essay Scoring using Transformers.
    """

    def __init__(self, texts, scores, tokenizer, max_len):
        """
        Args:
            texts (list): List of essay texts.
            scores (list or np.array): List of scores (targets). Can be None for inference.
            tokenizer (transformers.PreTrainedTokenizer): Tokenizer instance.
            max_len (int): Maximum sequence length.
        """
        self.texts = texts
        self.scores = scores
        self.tokenizer = tokenizer
        self.max_len = max_len

    def __len__(self):
        return len(self.texts)

    def __getitem__(self, idx):
        text = str(self.texts[idx])

        # Tokenize
        encoding = self.tokenizer.encode_plus(
            text,
            add_special_tokens=True,
            max_length=self.max_len,
            padding="max_length",
            truncation=True,
            return_attention_mask=True,
            return_tensors="pt",
        )

        # Flatten to remove batch dimension added by return_tensors='pt'
        ids = encoding["input_ids"].flatten()
        mask = encoding["attention_mask"].flatten()

        sample = {
            "input_ids": ids,
            "attention_mask": mask,
        }

        if self.scores is not None:
            sample["labels"] = torch.tensor(self.scores[idx], dtype=torch.float)

        return sample
