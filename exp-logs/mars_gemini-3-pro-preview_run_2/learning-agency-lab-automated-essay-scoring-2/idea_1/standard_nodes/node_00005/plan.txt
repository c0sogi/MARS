def extract_features(df: pd.DataFrame, split: str):
    """
    Extracts TF-IDF features from the dataframe.
    Combines Word n-grams and Character n-grams.

    Logic:
    - If split is 'train': Fits new vectorizers (word & char), saves them, and transforms.
    - If split is 'val' or 'test': Loads saved vectorizers and transforms.

    Args:
        df (pd.DataFrame): The dataframe containing the text column.
        split (str): The data split ('train', 'val', 'test').

    Returns:
        scipy.sparse.csr_matrix: The combined sparse feature matrix.
    """
    # Ensure text column exists
    if Config.TEXT_COL not in df.columns:
        raise KeyError(f"Column '{Config.TEXT_COL}' not found in dataframe.")

    # Fill NaNs with empty string to prevent vectorizer errors and ensure string type
    texts = df[Config.TEXT_COL].fillna("").astype(str)

    if split == "train":
        print("Fitting Word Vectorizer...")
        word_vec = TextVectorizer(params=Config.TFIDF_PARAMS)
        X_word = word_vec.fit_transform(texts)
        print(f"Saving Word Vectorizer to {Config.VECTORIZER_PATH}")
        word_vec.save(Config.VECTORIZER_PATH)

        print("Fitting Char Vectorizer...")
        char_vec = TextVectorizer(params=Config.TFIDF_CHAR_PARAMS)
        X_char = char_vec.fit_transform(texts)
        print(f"Saving Char Vectorizer to {Config.CHAR_VECTORIZER_PATH}")
        char_vec.save(Config.CHAR_VECTORIZER_PATH)

    else:
        # Load existing vectorizers
        if not os.path.exists(Config.VECTORIZER_PATH) or not os.path.exists(Config.CHAR_VECTORIZER_PATH):
            raise FileNotFoundError(
                "Vectorizers not found. Please run with split='train' first."
            )

        word_vec = TextVectorizer.load(Config.VECTORIZER_PATH)
        X_word = word_vec.transform(texts)

        char_vec = TextVectorizer.load(Config.CHAR_VECTORIZER_PATH)
        X_char = char_vec.transform(texts)

    # Combine features
    print(f"Concatenating features: Word {X_word.shape} + Char {X_char.shape}")
    features = hstack([X_word, X_char]).tocsr()

    return features
