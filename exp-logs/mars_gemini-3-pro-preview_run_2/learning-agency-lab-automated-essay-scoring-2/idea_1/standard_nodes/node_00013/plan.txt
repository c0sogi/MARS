import torch
from torch.utils.data import Dataset


class EssayDataset(Dataset):
    """
    PyTorch Dataset for Essay Scoring.
    Tokenizes text on-the-fly using the provided tokenizer.
    """

    def __init__(self, texts, scores, tokenizer, max_length):
        """
        Args:
            texts (list): List of essay texts.
            scores (list or np.array): List of scores (targets). Can be None for inference.
            tokenizer: HuggingFace tokenizer instance.
            max_length (int): Maximum sequence length.
        """
        self.texts = texts
        self.scores = scores
        self.tokenizer = tokenizer
        self.max_length = max_length

    def __len__(self):
        return len(self.texts)

    def __getitem__(self, idx):
        text = str(self.texts[idx])

        # Tokenize
        encoding = self.tokenizer.encode_plus(
            text,
            add_special_tokens=True,
            max_length=self.max_length,
            padding="max_length",
            truncation=True,
            return_attention_mask=True,
            return_tensors="pt",
        )

        # Flatten to remove batch dimension added by return_tensors="pt" (1, SeqLen) -> (SeqLen)
        input_ids = encoding["input_ids"].flatten()
        attention_mask = encoding["attention_mask"].flatten()

        item = {
            "input_ids": input_ids,
            "attention_mask": attention_mask,
        }

        # Handle targets if present
        if self.scores is not None:
            # SmoothL1Loss expects float targets
            item["labels"] = torch.tensor(self.scores[idx], dtype=torch.float)

        return item
