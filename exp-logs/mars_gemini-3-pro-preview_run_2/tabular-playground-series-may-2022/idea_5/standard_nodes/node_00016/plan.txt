The strategy involves:
1.  **Replacing the Architecture**: Switching from the `ResGLUNet` (Constant-Width Residual Network, 6 blocks of 512) to a `GatedMLP` with a funnel structure (layers: 512 $\to$ 256 $\to$ 128). Lesson 15 explicitly states this architecture achieved the highest validation AUC of 0.9953, outperforming the residual network's 0.9944.
2.  **Increasing Training Duration**: Extending the training epochs from 10 to 50. While **Lesson 14** noted that the Residual network converges faster, the Funnel MLP reaches a higher peak performance given sufficient time. **Lesson 3** also advises training past loss stabilization to maximize AUC.

class ResidualGatedBlock(nn.Module):
    """
    A residual block with Gated Linear Unit (GLU) activation.
    Structure: Input -> Linear(d->2d) -> GLU(2d->d) -> BatchNorm -> Dropout -> + Input
    """

    def __init__(self, dim, dropout_rate):
        super(ResidualGatedBlock, self).__init__()
        self.linear = nn.Linear(dim, dim * 2)
        self.glu = nn.GLU(dim=1)
        self.bn = nn.BatchNorm1d(dim)
        self.dropout = nn.Dropout(dropout_rate)

    def forward(self, x):
        residual = x
        out = self.linear(x)
        out = self.glu(out)
        out = self.bn(out)
        out = self.dropout(out)
        return out + residual


class ResGLUNet(nn.Module):
    """
    Deep Residual Gated Network for tabular data.
    Combines learned embeddings for categorical tokens and normalized continuous features.
    """

    def __init__(self):
        super(ResGLUNet, self).__init__()

        # Hyperparameters
        vocab_size = Config.VOCAB_SIZE
        embed_dim = Config.EMBED_DIM
        num_cont = Config.NUM_CONT_FEATURES
        seq_len = Config.TOKEN_SEQ_LEN
        hidden_dim = Config.HIDDEN_DIM
        num_blocks = Config.NUM_BLOCKS
        dropout_rate = Config.DROPOUT

        # 1. Input Processing
        # Shared embedding for the decomposed characters
        self.embedding = nn.Embedding(vocab_size, embed_dim)

        # Calculate input dimension for projection
        # Continuous features + Flattened embeddings (10 tokens * 32 dim)
        input_proj_dim = num_cont + (seq_len * embed_dim)

        self.input_proj = nn.Linear(input_proj_dim, hidden_dim)

        # 2. Residual Gated Backbone
        self.blocks = nn.ModuleList(
            [ResidualGatedBlock(hidden_dim, dropout_rate) for _ in range(num_blocks)]
        )

        # 3. Output Head
        self.final_norm = nn.LayerNorm(hidden_dim)
        self.head = nn.Linear(hidden_dim, 1)
        self.sigmoid = nn.Sigmoid()

    def forward(self, x_cat, x_cont):
        """
        Args:
            x_cat: (Batch, 10) LongTensor - Integer encoded characters
            x_cont: (Batch, 30) FloatTensor - Normalized continuous features
        """
        # Embed categorical tokens: (B, 10) -> (B, 10, 32)
        emb = self.embedding(x_cat)
        # Flatten embeddings: (B, 320)
        emb_flat = emb.view(emb.size(0), -1)

        # Concatenate with continuous features: (B, 350)
        x = torch.cat([emb_flat, x_cont], dim=1)

        # Project to hidden dimension
        x = self.input_proj(x)

        # Pass through residual blocks
        for block in self.blocks:
            x = block(x)

        # Final processing
        x = self.final_norm(x)
        logits = self.head(x)
        probs = self.sigmoid(logits)

        return probs

class GatedBlock(nn.Module):
    """
    A block with Gated Linear Unit (GLU) activation.
    Structure: Input -> Linear(d_in->2*d_out) -> GLU(2*d_out->d_out) -> BatchNorm -> Dropout
    """

    def __init__(self, in_dim, out_dim, dropout_rate):
        super(GatedBlock, self).__init__()
        self.linear = nn.Linear(in_dim, out_dim * 2)
        self.glu = nn.GLU(dim=1)
        self.bn = nn.BatchNorm1d(out_dim)
        self.dropout = nn.Dropout(dropout_rate)

    def forward(self, x):
        out = self.linear(x)
        out = self.glu(out)
        out = self.bn(out)
        out = self.dropout(out)
        return out


class GatedMLP(nn.Module):
    """
    Funnel-Shaped Gated MLP for tabular data.
    Combines learned embeddings for categorical tokens and normalized continuous features.
    Cite solution_lesson_node_00015: Funnel-Shaped Gated MLPs vs. Constant-Width Residual Networks.
    """

    def __init__(self):
        super(GatedMLP, self).__init__()

        # Hyperparameters
        vocab_size = Config.VOCAB_SIZE
        embed_dim = Config.EMBED_DIM
        num_cont = Config.NUM_CONT_FEATURES
        seq_len = Config.TOKEN_SEQ_LEN
        hidden_layers = Config.HIDDEN_LAYERS
        dropout_rate = Config.DROPOUT

        # 1. Input Processing
        self.embedding = nn.Embedding(vocab_size, embed_dim)
        
        # Calculate input dimension
        # Continuous features + Flattened embeddings (10 tokens * 32 dim)
        input_dim = num_cont + (seq_len * embed_dim)

        # 2. Gated Backbone (Funnel)
        layers = []
        current_dim = input_dim
        
        for hidden_dim in hidden_layers:
            layers.append(GatedBlock(current_dim, hidden_dim, dropout_rate))
            current_dim = hidden_dim
            
        self.backbone = nn.Sequential(*layers)

        # 3. Output Head
        self.head = nn.Linear(current_dim, 1)
        self.sigmoid = nn.Sigmoid()

    def forward(self, x_cat, x_cont):
        """
        Args:
            x_cat: (Batch, 10) LongTensor - Integer encoded characters
            x_cont: (Batch, 30) FloatTensor - Normalized continuous features
        """
        # Embed categorical tokens: (B, 10) -> (B, 10, 32)
        emb = self.embedding(x_cat)
        # Flatten embeddings: (B, 320)
        emb_flat = emb.view(emb.size(0), -1)

        # Concatenate with continuous features
        x = torch.cat([emb_flat, x_cont], dim=1)

        # Pass through backbone
        x = self.backbone(x)

        # Final prediction
        logits = self.head(x)
        probs = self.sigmoid(logits)

        return probs
