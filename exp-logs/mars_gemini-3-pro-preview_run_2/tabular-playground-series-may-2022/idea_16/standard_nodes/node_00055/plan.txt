    Formula: x_out = x + Dropout(BN(GLU(Linear(x))))
    Strictly uses Batch Normalization.
    """

        super().__init__()

        # Linear projects to 2 * out_dim for GLU
        self.linear = nn.Linear(in_dim, out_dim * 2)
