To improve the model's performance, I will transition from a Siamese Network with Contrastive Loss to a classification-based approach using **ArcFace** and an **EfficientNet-B3** backbone. This strategy addresses the high cardinality of the dataset and the fine-grained nature of the task.

**Implementation Strategy:**
1.  **Architecture Upgrade**: Replace ResNet18 with **EfficientNet-B3** (Cite `solution_lesson_node_00013`) and implement **Generalized Mean (GeM) Pooling** (Cite `solution_lesson_node_00020`) to better capture specific fluke details compared to global average pooling.
2.  **Loss Function**: Implement **ArcFace (Additive Angular Margin Loss)** (Cite `solution_lesson_node_00008`), which is empirically superior for identification tasks with thousands of classes.
3.  **Data Curation**: Explicitly exclude the `new_whale` class from the training set (Cite `solution_lesson_node_00005`) to avoid collapsing the embedding space with diverse "unknown" images.
4.  **Validation Protocol**: Construct the validation gallery using only known identities (Cite `solution_lesson_node_00019`) and use a dedicated, deterministic data loader for the gallery (Cite `solution_lesson_node_00024`) to ensure stable metric calculation.
5.  **Hyperparameters**: Set input resolution to 256x256 and batch size to 32 (Cite `solution_lesson_node_00022`) to balance capacity and training stability.

class SiameseWhaleDataset(Dataset):
    """
    Dataset for training a Siamese Network.
    Yields pairs of images (anchor, other) and a binary label (same/different).
    """

    def __init__(
        self,
        df,
        load_cached_data=True,
        transform=None,
        cache_name="train_images_cache.npy",
    ):
        """
        Args:
            df (pd.DataFrame): Metadata dataframe.
            load_cached_data (bool): Whether to use cached image data.
            transform (callable, optional): Transform to be applied on a sample.
            cache_name (str): Filename for the image cache.
        """
        self.df = df.reset_index(drop=True)
        self.transform = transform or get_transforms("train")

        # Group images by Id for fast sampling
        self.id_to_images = self.df.groupby("Id")["Image"].apply(list).to_dict()
        self.ids = list(self.id_to_images.keys())

        # Pre-load images into RAM
        self.image_cache = cache_images(
            self.df, cache_name, load_cached_data=load_cached_data
        )

        # Filter dataframe to only include images that were successfully loaded
        available_images = set(self.image_cache.keys())
        self.df = self.df[self.df["Image"].isin(available_images)].reset_index(
            drop=True
        )

    def __len__(self):
        return len(self.df)

    def __getitem__(self, index):
        # Anchor
        row = self.df.iloc[index]
        anchor_id = row["Id"]
        anchor_filename = row["Image"]

        # Determine if positive or negative pair (50/50)
        # We use random.random() < 0.5.
        # Note: We use numpy random for consistency if seeded externally
        is_positive = np.random.random() < 0.5

        if is_positive:
            label = 1.0
            # Get all images for this whale ID
            candidates = self.id_to_images[anchor_id]

            # Filter candidates to those available in cache
            candidates = [c for c in candidates if c in self.image_cache]

            if len(candidates) > 1:
                # Pick a different image if possible, or same (rarely)
                # To ensure robustness, we can pick any from the list
                second_filename = np.random.choice(candidates)
            else:
                # Singleton: Use the same image.
                # The transform will create a synthetic view.
                second_filename = anchor_filename
        else:
            label = 0.0
            # Pick a different class
            while True:
                neg_id = np.random.choice(self.ids)
                if neg_id != anchor_id:
                    break

            # Pick a random image from that class
            neg_candidates = self.id_to_images[neg_id]
            # Filter
            neg_candidates = [c for c in neg_candidates if c in self.image_cache]

            if not neg_candidates:
                # Fallback if the chosen negative class has no valid images (rare edge case)
                # Just use anchor as negative (label 0) - creates noise but prevents crash
                second_filename = anchor_filename
                label = 1.0  # Revert to positive if we can't find negative
            else:
                second_filename = np.random.choice(neg_candidates)

        # Retrieve images from cache
        img1_array = self.image_cache[anchor_filename]
        img2_array = self.image_cache[second_filename]

        # Apply transforms
        # Transforms expect PIL Image or Tensor. ToPILImage handles ndarray (H,W,C)
        if self.transform:
            img1 = self.transform(img1_array)
            img2 = self.transform(img2_array)

        return (img1, img2), torch.tensor(label, dtype=torch.float32)

class EmbeddingNet(nn.Module):
    """
    A neural network that maps images to an embedding space.
    Uses ResNet-18 as the backbone.
    """

    def __init__(self):
        super(EmbeddingNet, self).__init__()

        # Determine weights based on configuration
        if Config.PRETRAINED:
            weights = models.ResNet18_Weights.DEFAULT
        else:
            weights = None

        # Load the backbone
        self.backbone = models.resnet18(weights=weights)

        # Replace the final fully connected layer
        # ResNet18's fc layer has 512 input features
        num_ftrs = self.backbone.fc.in_features
        self.backbone.fc = nn.Linear(num_ftrs, Config.EMBEDDING_DIM)

    def forward(self, x):
        """
        Forward pass of the embedding network.

        Args:
            x (torch.Tensor): Input images of shape (B, C, H, W).

        Returns:
            torch.Tensor: L2-normalized embeddings of shape (B, EMBEDDING_DIM).
        """
        x = self.backbone(x)
        # L2-normalize the output vectors
        x = F.normalize(x, p=2, dim=1)
        return x


class SiameseNet(nn.Module):
    """
    Siamese Network wrapper that processes pairs of images using a shared EmbeddingNet.
    """

    def __init__(self, embedding_net):
        super(SiameseNet, self).__init__()
        self.embedding_net = embedding_net

    def forward(self, x1, x2):
        """
        Forward pass for image pairs.

        Args:
            x1 (torch.Tensor): First batch of images.
            x2 (torch.Tensor): Second batch of images.

        Returns:
            tuple: (output1, output2) embeddings.
        """
        output1 = self.embedding_net(x1)
        output2 = self.embedding_net(x2)
        return output1, output2

import math
import torch
import torch.nn as nn
import torch.nn.functional as F
import timm

class GeM(nn.Module):
    """
    Generalized Mean Pooling.
    Cite solution_lesson_node_00020
    """
    def __init__(self, p=3, eps=1e-6):
        super(GeM, self).__init__()
        self.p = nn.Parameter(torch.ones(1) * p)
        self.eps = eps

    def forward(self, x):
        return self.gem(x, p=self.p, eps=self.eps)

    def gem(self, x, p=3, eps=1e-6):
        return F.avg_pool2d(x.clamp(min=eps).pow(p), (x.size(-2), x.size(-1))).pow(1.0 / p)


class ArcMarginProduct(nn.Module):
    """
    ArcFace Head.
    Cite solution_lesson_node_00008
    """
    def __init__(self, in_features, out_features, s=30.0, m=0.50, easy_margin=False):
        super(ArcMarginProduct, self).__init__()
        self.in_features = in_features
        self.out_features = out_features
        self.s = s
        self.m = m
        self.weight = nn.Parameter(torch.FloatTensor(out_features, in_features))
        nn.init.xavier_uniform_(self.weight)

        self.easy_margin = easy_margin
        self.cos_m = math.cos(m)
        self.sin_m = math.sin(m)
        self.th = math.cos(math.pi - m)
        self.mm = math.sin(math.pi - m) * m

    def forward(self, input, label):
        # --------------------------- cos(theta) & phi(theta) ---------------------------
        cosine = F.linear(F.normalize(input), F.normalize(self.weight))
        sine = torch.sqrt(1.0 - torch.pow(cosine, 2))
        phi = cosine * self.cos_m - sine * self.sin_m
        if self.easy_margin:
            phi = torch.where(cosine > 0, phi, cosine)
        else:
            phi = torch.where(cosine > self.th, phi, cosine - self.mm)
        # --------------------------- convert label to one-hot ---------------------------
        one_hot = torch.zeros_like(cosine)
        one_hot.scatter_(1, label.view(-1, 1).long(), 1)
        output = (one_hot * phi) + ((1.0 - one_hot) * cosine)
        output *= self.s
        return output

    """
    Backbone (EfficientNet) + GeM + ArcFace Head.
    Cite solution_lesson_node_00013, solution_lesson_node_00020
    """
    def __init__(self, num_classes):

        # num_classes=0 removes the head, global_pool='' keeps spatial dims
        self.backbone = timm.create_model(
            Config.BACKBONE, pretrained=Config.PRETRAINED, num_classes=0, global_pool=""
        )
        
        # Get feature dimension
        dummy_input = torch.randn(1, 3, 256, 256)
        features = self.backbone(dummy_input)
        in_features = features.shape[1]
        
        # Pooling
        self.pooling = GeM()
        
        # Neck (BN -> FC -> BN)
        self.bn1 = nn.BatchNorm1d(in_features)
        self.fc = nn.Linear(in_features, Config.EMBEDDING_DIM)
        self.bn2 = nn.BatchNorm1d(Config.EMBEDDING_DIM)
        
        # ArcFace Head
        self.arcface = ArcMarginProduct(
            Config.EMBEDDING_DIM, 
            num_classes, 
            s=Config.ARC_S, 
            m=Config.ARC_M
        )

    def forward(self, x, labels=None):
        # Feature Extraction
        x = self.backbone(x)
        x = self.pooling(x)
        x = x.view(x.size(0), -1)
        
        # Neck
        x = self.bn1(x)
        x = self.fc(x)
        x = self.bn2(x)
        
        # If labels are provided, return ArcFace logits (Training)
        if labels is not None:
            return self.arcface(x, labels)

def train_one_epoch(model, dataloader, criterion, optimizer, device):
    """
    Trains the Siamese Network for one epoch.

    Args:
        model (nn.Module): The SiameseNet model.
        dataloader (DataLoader): DataLoader for SiameseWhaleDataset.
        criterion (nn.Module): Loss function (ContrastiveLoss).
        optimizer (Optimizer): Optimizer.
        device (torch.device): Device to run training on.

    Returns:
        float: Average training loss.
    """
    model.train()
    running_loss = 0.0
    total_samples = 0

    for batch_idx, ((img1, img2), labels) in enumerate(dataloader):
        img1 = img1.to(device)
        img2 = img2.to(device)
        labels = labels.to(device)

        optimizer.zero_grad()

        # Forward pass
        # SiameseNet returns (output1, output2)
        out1, out2 = model(img1, img2)

        # Compute loss
        loss = criterion(out1, out2, labels)

        # Backward and optimize
        loss.backward()
        optimizer.step()

        # Aggregate loss
        # loss.item() is the average loss for the batch (default reduction='mean')
        batch_size = labels.size(0)
        running_loss += loss.item() * batch_size
        total_samples += batch_size

    epoch_loss = running_loss / total_samples if total_samples > 0 else 0.0
    return epoch_loss
