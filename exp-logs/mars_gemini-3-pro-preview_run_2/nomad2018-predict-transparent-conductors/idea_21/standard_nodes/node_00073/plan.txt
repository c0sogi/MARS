To optimize the performance of the crystal property prediction model, I will implement the following targeted changes based on the provided lessons:

1.  **Architecture Upgrade (Receiver-Aware Message Passing):** I will replace the standard `CGConv` layer with a custom `ReceiverAwareCGConv` layer. This layer explicitly concatenates the receiver node features ($x_i$), source node features ($x_j$), and edge features ($e_{ij}$) before the linear transformation and gating mechanism. This addresses the limitation of standard implementations that often ignore the receiver's state in the non-linear transformation, allowing for better modeling of pairwise chemical interactions (Cite `solution_lesson_node_00072`).

2.  **Residual Connection Optimization:** I will modify the residual connection in the `InteractionBlock` to use a learnable scaling factor initialized to identity: $h_{l+1} = \text{Activation}(\text{BatchNorm}(\text{Message} + (1 + \epsilon) \cdot h_l))$, where $\epsilon$ is initialized to 0. This ensures an active identity path at the start of training, facilitating better gradient flow and convergence compared to the previous initialization (Cite `solution_lesson_node_00060`, `solution_lesson_node_00069`).

3.  **Removal of Global Skip Connection:** I will remove the "Stoichiometry-Skip" connection (concatenating the initial node embedding global pool to the final readout). Lessons indicate that with proper residual connections in the backbone, this explicit skip is redundant and can degrade performance by bypassing the deep feature refinement (Cite `solution_lesson_node_00069`). The model will rely on the GNN backbone to preserve necessary stoichiometric information.

4.  **Training Schedule:** I will increase the number of epochs in `runfile.py` from 60 to 120 to allow the model, now stabilized by the improved residual connections, to converge more fully.

class GaussianSmearing(nn.Module):
    """
    Expands distances using a set of Gaussian Radial Basis Functions.
    """

    def __init__(self, start=0.0, stop=5.0, num_gaussians=50):
        super().__init__()
        offset = torch.linspace(start, stop, num_gaussians)
        # The width (gamma) is determined by the spacing between centers
        self.coeff = -0.5 / (offset[1] - offset[0]).item() ** 2
        self.register_buffer("offset", offset)

    def forward(self, dist):
        # dist: (E, 1) or (E,)
        # Returns: (E, num_gaussians)
        dist = dist.view(-1, 1) - self.offset.view(1, -1)
        return torch.exp(self.coeff * torch.pow(dist, 2))


class InteractionBlock(nn.Module):
    """
    Interaction Block performing Gated Graph Convolution with a learnable
    scalar residual connection, Batch Normalization, and Softplus activation.
    """

    def __init__(self, hidden_dim, dropout_rate):
        super().__init__()
        # CGConv from torch_geometric implements the crystal graph convolution
        # We disable internal batch_norm to apply it explicitly as per architecture design
        self.conv = CGConv(channels=hidden_dim, dim=hidden_dim, batch_norm=False)
        self.bn = nn.BatchNorm1d(hidden_dim)
        self.dropout_rate = dropout_rate

        # Learnable scalar epsilon for the residual connection, initialized to 0
        self.epsilon = nn.Parameter(torch.tensor(0.0))

    def forward(self, x, edge_index, edge_attr):
        # x: (N, hidden_dim)
        # edge_index: (2, E)
        # edge_attr: (E, hidden_dim)

        # PyG's CGConv returns: x + sum(messages)
        z = self.conv(x, edge_index, edge_attr)

        # We want the update: messages + (1 + epsilon) * x
        # Since z = messages + x, we add epsilon * x to z
        out = z + self.epsilon * x

        # Apply BatchNorm, Softplus, and Dropout
        out = self.bn(out)
        out = F.softplus(out)
        out = F.dropout(out, p=self.dropout_rate, training=self.training)

        return out

    Expands distances using a set of Gaussian Radial Basis Functions.
    """

    def __init__(self, start=0.0, stop=5.0, num_gaussians=50):

        self.coeff = -0.5 / (offset[1] - offset[0]).item() ** 2
        self.register_buffer("offset", offset)

    def forward(self, dist):
        # dist: (E, 1) or (E,)
        # Returns: (E, num_gaussians)
        dist = dist.view(-1, 1) - self.offset.view(1, -1)
        return torch.exp(self.coeff * torch.pow(dist, 2))


class ReceiverAwareCGConv(MessagePassing):
    """
    Custom Crystal Graph Convolution that explicitly includes receiver node features.
    Cite solution_lesson_node_00072
    """
    def __init__(self, channels, dim):
        super().__init__(aggr='add')
        self.lin = nn.Linear(2 * channels + dim, 2 * channels)

    def forward(self, x, edge_index, edge_attr):
        return self.propagate(edge_index, x=x, edge_attr=edge_attr)

    def message(self, x_i, x_j, edge_attr):
        # x_i: receiver, x_j: source
        z = torch.cat([x_i, x_j, edge_attr], dim=-1)
        z = self.lin(z)
        f, s = z.chunk(2, dim=-1)
        return torch.sigmoid(f) * F.softplus(s)


class InteractionBlock(nn.Module):
    """
    Interaction Block performing Gated Graph Convolution with a learnable
    scalar residual connection, Batch Normalization, and Softplus activation.
    """

    def __init__(self, hidden_dim, dropout_rate):
        super().__init__()
        # Use custom Receiver-Aware CGConv
        self.conv = ReceiverAwareCGConv(channels=hidden_dim, dim=hidden_dim)
        self.bn = nn.BatchNorm1d(hidden_dim)
        self.dropout_rate = dropout_rate

        # Learnable scalar epsilon for the residual connection, initialized to 0
        # This results in (1 + epsilon) * x scaling
        self.epsilon = nn.Parameter(torch.tensor(0.0))

    def forward(self, x, edge_index, edge_attr):
        # x: (N, hidden_dim)
        # edge_index: (2, E)
        # edge_attr: (E, hidden_dim)

        # Message passing
        m = self.conv(x, edge_index, edge_attr)

        # Learnable Residual Scaling: (1 + epsilon) * x + m
        # Cite solution_lesson_node_00060, solution_lesson_node_00069
        out = m + (1.0 + self.epsilon) * x

        # Apply BatchNorm, Softplus, and Dropout
        out = self.bn(out)
        out = F.softplus(out)
        out = F.dropout(out, p=self.dropout_rate, training=self.training)

        return out
