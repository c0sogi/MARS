def get_aux_samples(base_paths):
    """Scans auxiliary folders (valid/devel) for CSV labels and video files."""
    samples = []
    for base_pattern in base_paths:
        search_path = os.path.join(INPUT_DIR, base_pattern)
        directories = glob.glob(search_path)
        
        for d in directories:
            if not os.path.isdir(d):
                continue
            
            # Find CSV files
            csv_files = glob.glob(os.path.join(d, "*.csv"))
            for csv_file in csv_files:
                try:
                    # Read CSV (assuming no header, col 0: ID, col 1: labels)
                    df = pd.read_csv(csv_file, header=None, dtype=str)
                    if df.shape[1] < 2:
                        continue
                        
                    for _, row in df.iterrows():
                        s_id = str(row[0])
                        s_labels = str(row[1])
                        
                        if pd.isna(s_id) or pd.isna(s_labels):
                            continue
                            
                        # Extract file number from ID (e.g. valid01_1 -> 1)
                        if "_" in s_id:
                            file_num = s_id.split("_")[-1]
                            filename = f"K_{file_num}.avi"
                        else:
                            filename = f"{s_id}.avi"
                            
                        rel_dir = os.path.relpath(d, INPUT_DIR)
                        rgb_path = os.path.join(rel_dir, filename)
                        
                        # Clean labels
                        clean_labels = []
                        for l in s_labels.split():
                            if l.strip().isdigit():
                                clean_labels.append(l.strip())
                        
                        samples.append({
                            "sample_id": s_id,
                            "rgb_path": rgb_path,
                            "depth_path": "",
                            "audio_path": "",
                            "user_path": "",
                            "data_path": "",
                            "labels": " ".join(clean_labels),
                            "num_frames": 0,
                            "source": "aux"
                        })
                except Exception as e:
                    print(f"Error reading CSV {csv_file}: {e}")
    return samples
