class GatedTCNBlock(nn.Module):
    """
    Gated Activation Unit for Temporal Convolutional Networks.
    Computes Z = tanh(W_f * X) * sigmoid(W_g * X).
    Includes a residual connection.
    """

    def __init__(self, channels, kernel_size, dilation, dropout):
        super(GatedTCNBlock, self).__init__()
        # Calculate padding to maintain temporal dimension
        # For kernel_size 3: padding = dilation
        self.padding = (kernel_size - 1) * dilation // 2

        # Filter convolution (W_f)
        self.filter_conv = nn.Conv1d(
            in_channels=channels,
            out_channels=channels,
            kernel_size=kernel_size,
            dilation=dilation,
            padding=self.padding,
        )

        # Gate convolution (W_g)
        self.gate_conv = nn.Conv1d(
            in_channels=channels,
            out_channels=channels,
            kernel_size=kernel_size,
            dilation=dilation,
            padding=self.padding,
        )

        self.dropout = nn.Dropout(dropout)

    def forward(self, x):
        # x: (Batch, Channels, Time)

        # Filter: tanh(W_f * x)
        f = torch.tanh(self.filter_conv(x))

        # Gate: sigmoid(W_g * x)
        g = torch.sigmoid(self.gate_conv(x))

        # Element-wise multiplication (Gating)
        out = f * g

        out = self.dropout(out)

        # Residual connection
        return x + out

class DilatedResidualBlock(nn.Module):
    """
    Standard Dilated Residual Block with Instance Normalization.
    Structure: DilatedConv -> InstanceNorm -> ReLU -> Dropout -> Conv1x1
    Cite Lesson 00063: Instance Normalization improves convergence.
    """

    def __init__(self, channels, kernel_size, dilation, dropout):
        super(DilatedResidualBlock, self).__init__()
        self.padding = (kernel_size - 1) * dilation // 2

        self.conv_dilated = nn.Conv1d(
            in_channels=channels,
            out_channels=channels,
            kernel_size=kernel_size,
            dilation=dilation,
            padding=self.padding,
        )
        
        self.norm = nn.InstanceNorm1d(channels, affine=True)
        self.relu = nn.ReLU()
        self.dropout = nn.Dropout(dropout)
        self.conv_1x1 = nn.Conv1d(channels, channels, 1)

    def forward(self, x):
        out = self.conv_dilated(x)
        out = self.norm(out)
        out = self.relu(out)
        out = self.dropout(out)
        out = self.conv_1x1(out)
        return x + out
