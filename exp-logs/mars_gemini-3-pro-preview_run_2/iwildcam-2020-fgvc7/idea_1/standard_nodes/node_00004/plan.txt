class ResNetClassifier(nn.Module):
    """
    A ResNet-based classifier with option to fine-tune backbone.
    """

    def __init__(self, num_classes=Config.NUM_CLASSES, fine_tune=True):
        super(ResNetClassifier, self).__init__()

        # Load pre-trained ResNet50
        # Using the new weights parameter as per torchvision >= 0.13
        weights = models.ResNet50_Weights.DEFAULT
        self.backbone = models.resnet50(weights=weights)

        # Freeze backbone if not fine-tuning
        if not fine_tune:
            for param in self.backbone.parameters():
                param.requires_grad = False

        # ResNet50 fc in_features is 2048
        in_features = self.backbone.fc.in_features
        self.backbone.fc = nn.Linear(in_features, num_classes)

    def forward(self, x):
        return self.backbone(x)

class AnimalClassifier(nn.Module):
    """
    EfficientNetV2-S based classifier.
    """

    def __init__(self, num_classes=Config.NUM_CLASSES, fine_tune=True):
        super(AnimalClassifier, self).__init__()

        # Load pre-trained EfficientNetV2-S
        weights = models.EfficientNet_V2_S_Weights.DEFAULT
        self.backbone = models.efficientnet_v2_s(weights=weights)

        # Freeze backbone if not fine-tuning
        if not fine_tune:
            for param in self.backbone.parameters():
                param.requires_grad = False

        # EfficientNetV2 classifier head is a Sequential(Dropout, Linear)
        # We replace it to match our number of classes
        # The last layer in the classifier sequence is the Linear layer
        in_features = self.backbone.classifier[-1].in_features
        
        # Rebuild classifier with Dropout and Linear
        self.backbone.classifier = nn.Sequential(
            nn.Dropout(p=0.3, inplace=True),
            nn.Linear(in_features, num_classes)
        )

    def forward(self, x):
        return self.backbone(x)
