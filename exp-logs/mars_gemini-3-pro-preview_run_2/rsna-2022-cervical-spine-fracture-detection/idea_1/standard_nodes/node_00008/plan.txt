class UnifiedTransform:
    """
    Applies the same random geometric transformations to all slices in a volume
    to preserve anatomical consistency (3D Augmentation).
    """

    def __call__(self, video):
        # video shape: (S, C, H, W) where C=1
        # F.affine and F.hflip can process a batch of images (S, ...) with the same parameters.

        # Random Horizontal Flip
        if random.random() < 0.5:
            video = F.hflip(video)

        # Random Affine (Rotation, Translation, Scale)
        # We sample parameters once and apply to the whole volume
        angle = random.uniform(-15, 15)
        translate_x = random.uniform(-0.1, 0.1) * video.shape[-1]
        translate_y = random.uniform(-0.1, 0.1) * video.shape[-2]
        scale = random.uniform(0.9, 1.1)

        video = F.affine(
            video,
            angle=angle,
            translate=[translate_x, translate_y],
            scale=scale,
            shear=0,
        )
        return video
