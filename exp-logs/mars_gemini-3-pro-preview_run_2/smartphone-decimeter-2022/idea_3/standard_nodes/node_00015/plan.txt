class TemporalStream(nn.Module):
    def __init__(self, input_channels, hidden_dim=64):
        super(TemporalStream, self).__init__()
        self.conv1 = nn.Conv1d(input_channels, 32, kernel_size=3, padding=1)
        self.bn1 = nn.BatchNorm1d(32)
        self.relu = nn.ReLU()
        self.conv2 = nn.Conv1d(32, hidden_dim, kernel_size=3, padding=1)
        self.bn2 = nn.BatchNorm1d(hidden_dim)
        self.gap = nn.AdaptiveAvgPool1d(1)
        self.flatten = nn.Flatten()

    def forward(self, x):
        # x shape: (Batch, Window, Channels) -> Permute to (Batch, Channels, Window)
        x = x.permute(0, 2, 1)
        x = self.relu(self.bn1(self.conv1(x)))
        x = self.relu(self.bn2(self.conv2(x)))
        x = self.gap(x)
        x = self.flatten(x)
        return x

            nn.ReLU(),

class DSTResNet(nn.Module):
    def __init__(self, dynamic_features, static_features, window_size, hidden_dim=64):
        super(DSTResNet, self).__init__()
        self.dynamic_features = dynamic_features
        self.static_features = static_features

        # Sub-modules
        self.temporal_stream = TemporalStream(dynamic_features, hidden_dim)
        self.spatial_stream = SpatialStream(static_features, hidden_dim // 2)

        # Fusion Head
        fusion_input_dim = hidden_dim + (hidden_dim // 2)
        self.fusion_head = nn.Sequential(
            nn.Linear(fusion_input_dim, 64),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.Linear(64, 32),
            nn.ReLU(),
            nn.Linear(32, 2),  # Output: Delta East, Delta North
        )

    def forward(self, x):
        # x shape: (Batch, Window, TotalFeatures)
        # Assuming last 'static_features' columns are static, rest are dynamic

        # Split features
        # Dynamic: All timesteps, first N columns
        x_dynamic = x[:, :, : -self.static_features]

        # Static: Center timestep, last M columns
        center_idx = x.shape[1] // 2
        x_static = x[:, center_idx, -self.static_features :]

        # Pass through streams
        t_out = self.temporal_stream(x_dynamic)
        s_out = self.spatial_stream(x_static)

        # Fusion
        fused = torch.cat([t_out, s_out], dim=1)
        out = self.fusion_head(fused)

        return out
