The issue is that the inference pipeline relies on a static metadata file (`test_metadata.csv`) to determine the dimensions of the test images for resizing predictions. In the evaluation environment, the test set is replaced with a hidden dataset ("hidden test set") containing different images with potentially different dimensions. Because the code uses the static metadata instead of scanning the actual input files, it attempts to resize predictions to incorrect dimensions or misses files entirely, leading to the `InvalidSubmissionError: all input arrays must have the same shape` when the grader attempts to stack the predictions (Cite debug_lesson_7, debug_lesson_2).

To fix this, I will:
1.  Modify `library/dataset.py` to implement a `scan_test_directory` function that dynamically walks the `input/test` directory to build the test metadata (IDs, file paths, and dimensions) at runtime.
2.  Update `get_test_loader` in `library/dataset.py` to use this dynamic scanning function instead of reading the static CSV.
3.  Update `library/inference.py` to retrieve the image dimensions from the dynamically generated dataframe (via `test_loader.dataset.df`) rather than the static CSV, ensuring predictions are resized to the correct original dimensions of the hidden test images.

def scan_test_directory():
    """
    Dynamically scans the test directory to build metadata.
    This ensures we handle the hidden test set correctly during submission.
    """
    data = []
    test_dir = os.path.join(Config.INPUT_DIR, "test")

    if not os.path.exists(test_dir):
        return pd.DataFrame(
            columns=[
                "id",
                "case",
                "day",
                "slice",
                "file_path",
                "img_width",
                "img_height",
            ]
        )

    for root, dirs, files in os.walk(test_dir):
        for file in files:
            if file.endswith(".png"):
                # Filename: slice_0001_360_310_1.50_1.50.png
                parts = file.replace(".png", "").split("_")
                if len(parts) < 4:
                    continue

                slice_id = parts[1]
                width = int(parts[2])
                height = int(parts[3])

                # Parent: case123_day20
                parent = os.path.basename(os.path.dirname(root))
                try:
                    c_str, d_str = parent.split("_")
                    case_num = int(c_str.replace("case", ""))
                    day_num = int(d_str.replace("day", ""))
                except ValueError:
                    continue

                img_id = f"{parent}_slice_{slice_id}"
                rel_path = os.path.relpath(os.path.join(root, file), Config.INPUT_DIR)

                data.append(
                    {
                        "id": img_id,
                        "case": case_num,
                        "day": day_num,
                        "slice": int(slice_id),
                        "file_path": rel_path,
                        "img_width": width,
                        "img_height": height,
                    }
                )

    df = pd.DataFrame(data)
    if not df.empty:
        df = df.sort_values(["case", "day", "slice"]).reset_index(drop=True)
    return df
