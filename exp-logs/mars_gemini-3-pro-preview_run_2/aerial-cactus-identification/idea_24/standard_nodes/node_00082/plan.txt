class Res2NeXtBlock(nn.Module):
    """
    Custom block integrating Res2Net hierarchical connections within a ResNeXt
    grouped convolution topology.
    """

    def __init__(
        self,
        in_planes,
        planes,
        stride=1,
        cardinality=32,
        base_width=4,
        scale=4,
        reduction=16,
    ):
        super(Res2NeXtBlock, self).__init__()

        # Calculate bottleneck width based on ResNeXt formula
        # width = floor(planes * (base_width/64)) * cardinality
        width = math.floor(planes * (base_width / 64.0)) * cardinality

        self.width = width
        self.scale = scale
        self.stride = stride
        self.width_per_scale = width // scale
        self.cardinality = cardinality

        if width % scale != 0:
            raise ValueError(f"Width {width} must be divisible by scale {scale}")

        # Conv1: 1x1 Projection
        self.conv1 = nn.Conv2d(in_planes, width, kernel_size=1, bias=False)
        self.bn1 = nn.BatchNorm2d(width)

        # Conv2: 3x3 (Res2Net + Grouped)
        # We create a list of convolutions.
        # If stride=1: We use hierarchical connections (scale-1 convs).
        # If stride>1: We process all splits in parallel to handle resolution changes robustly (scale convs).

        groups_per_conv = cardinality // scale
        self.convs = nn.ModuleList()

        num_convs = scale - 1

        for _ in range(num_convs):
            self.convs.append(
                nn.Conv2d(
                    self.width_per_scale,
                    self.width_per_scale,
                    kernel_size=3,
                    stride=stride,
                    padding=1,
                    groups=groups_per_conv,
                    bias=False,
                )
            )
        self.bns = nn.ModuleList(
            [nn.BatchNorm2d(self.width_per_scale) for _ in range(num_convs)]
        )

        # If stride > 1, the first split also needs downsampling
        if stride > 1:
            self.pool = nn.AvgPool2d(kernel_size=3, stride=stride, padding=1)

        # Conv3: 1x1 Projection
        self.conv3 = nn.Conv2d(width, planes, kernel_size=1, bias=False)
        self.bn3 = nn.BatchNorm2d(planes)

        # SE Block
        self.se = SEBlock(planes, reduction)

        # Shortcut connection
        self.shortcut = nn.Sequential()
        if stride != 1 or in_planes != planes:
            self.shortcut = nn.Sequential(
                nn.Conv2d(in_planes, planes, kernel_size=1, stride=stride, bias=False),
                nn.BatchNorm2d(planes),
            )

    def forward(self, x):
        residual = x

        out = F.relu(self.bn1(self.conv1(x)))

        # Split feature map into 'scale' chunks
        spx = torch.split(out, self.width_per_scale, 1)

        y = []

        if self.stride == 1:
            # Hierarchical connections
            # First split is identity
            sp = spx[0]
            y.append(sp)
            prev_y = sp

            for i, (conv, bn) in enumerate(zip(self.convs, self.bns)):
                # Input to conv is current split + previous output
                inp = spx[i + 1] + prev_y
                out_conv = F.relu(bn(conv(inp)))
                y.append(out_conv)
                prev_y = out_conv
        else:
            # Parallel processing for downsampling layers
            # Process first split with pool
            y.append(self.pool(spx[0]))

            # Process others with strided convs
            for i, (conv, bn) in enumerate(zip(self.convs, self.bns)):
                out_conv = F.relu(bn(conv(spx[i + 1])))
                y.append(out_conv)

        # Concatenate scales
        out = torch.cat(y, 1)

        out = self.bn3(self.conv3(out))
        out = self.se(out)

        out += self.shortcut(residual)
        out = F.relu(out)
        return out

class SEBasicBlock(nn.Module):
    """
    Standard ResNet BasicBlock with Squeeze-and-Excitation.
    """

    def __init__(self, in_planes, planes, stride=1, reduction=16):
        super(SEBasicBlock, self).__init__()
        self.conv1 = nn.Conv2d(
            in_planes, planes, kernel_size=3, stride=stride, padding=1, bias=False
        )
        self.bn1 = nn.BatchNorm2d(planes)
        self.conv2 = nn.Conv2d(
            planes, planes, kernel_size=3, stride=1, padding=1, bias=False
        )
        self.bn2 = nn.BatchNorm2d(planes)
        self.se = SEBlock(planes, reduction)

        self.shortcut = nn.Sequential()
        if stride != 1 or in_planes != planes:
            self.shortcut = nn.Sequential(
                nn.Conv2d(in_planes, planes, kernel_size=1, stride=stride, bias=False),
                nn.BatchNorm2d(planes),
            )

    def forward(self, x):
        out = F.relu(self.bn1(self.conv1(x)))
        out = self.bn2(self.conv2(out))
        out = self.se(out)
        out += self.shortcut(x)
        out = F.relu(out)
        return out
