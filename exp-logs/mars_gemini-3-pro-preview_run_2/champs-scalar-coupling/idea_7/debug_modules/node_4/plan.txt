# Atomic number mapping (H, C, N, O, F) - prevalent in this dataset
# We map them to 0-based indices for embedding layers
ATOM_MAP = {1: 0, 6: 1, 7: 2, 8: 3, 9: 4}
ATOM_SYMBOL_TO_Z = {"H": 1, "C": 6, "N": 7, "O": 8, "F": 9}


class MolecularGraphDataset(Dataset):
    @staticmethod
    def read_xyz(path):
        """Reads an XYZ file and returns atom symbols and coordinates."""
        with open(path, 'r') as f:
            lines = f.readlines()
        # Line 0: num atoms, Line 1: comment, Line 2+: atom x y z
        atoms = []
        coords = []
        for line in lines[2:]:
            parts = line.strip().split()
            if len(parts) >= 4:
                atoms.append(parts[0])
                coords.append([float(parts[1]), float(parts[2]), float(parts[3])])
        return atoms, np.array(coords, dtype=np.float32)
