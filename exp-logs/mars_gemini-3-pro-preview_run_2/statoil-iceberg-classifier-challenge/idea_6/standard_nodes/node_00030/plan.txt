def apply_tta(model, imgs, incs):
    """
    Applies Test Time Augmentation (8 transforms: 4 rotations + flips).
    Returns averaged predictions tensor.
    """
    preds = []
    # 1. Original
    preds.append(model(imgs, incs))
    # 2. Rot90
    preds.append(model(torch.rot90(imgs, 1, [2, 3]), incs))
    # 3. Rot180
    preds.append(model(torch.rot90(imgs, 2, [2, 3]), incs))
    # 4. Rot270
    preds.append(model(torch.rot90(imgs, 3, [2, 3]), incs))
    
    # Flips
    imgs_h = torch.flip(imgs, [3])
    # 5. HFlip
    preds.append(model(imgs_h, incs))
    # 6. HFlip + Rot90
    preds.append(model(torch.rot90(imgs_h, 1, [2, 3]), incs))
    # 7. HFlip + Rot180
    preds.append(model(torch.rot90(imgs_h, 2, [2, 3]), incs))
    # 8. HFlip + Rot270
    preds.append(model(torch.rot90(imgs_h, 3, [2, 3]), incs))

    # Stack and mean
    preds = torch.stack(preds, dim=0)
    return torch.mean(preds, dim=0)
