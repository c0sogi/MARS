I will implement the **Spatially-Aware Hybrid Convolutional Network (SAHCN)**, a custom shallow CNN architecture, replacing the previous ResNet-18 transfer learning approach. To support this architectural change and optimize performance:
1.  **Training Configuration**: I will increase `NUM_EPOCHS` to 100 and `PATIENCE` to 20 to allow the model sufficient time to converge, as the use of geometric augmentation makes the optimization landscape more complex (Cite `solution_lesson_node_00009`). 2.  **Augmentation Efficiency**: I will update the `TrainTransform` to use native PyTorch random operations (`torch.randint`, `torch.rand`) instead of NumPy, ensuring better integration with the data loader's multiprocessing and adhering to best practices for tensor-native pipelines (Cite `solution_lesson_node_00011`).

import torch
import torch.nn as nn
import timm
from library.config import Config


class RHTN(nn.Module):
    """
    Residual Hybrid Transfer Network (RHTN).

    This architecture fuses a visual branch based on a pre-trained ResNet-18 backbone
    with a scalar metadata branch processing the incidence angle.
    """

    def __init__(self):
        super(RHTN, self).__init__()

        # ==========================================
        # 1. Visual Branch (ResNet-18 Backbone)
        # ==========================================
        # We use timm to create the backbone.
        # in_chans=3: Expects 3-channel input (Band 1, Band 2, Mean).
        # num_classes=0: Removes the final fully connected classification layer.
        # global_pool='avg': Applies Global Average Pooling to the features.
        self.backbone = timm.create_model(
            Config.BACKBONE_NAME,
            pretrained=Config.PRETRAINED,
            in_chans=Config.IN_CHANNELS,
            num_classes=0,
            global_pool="avg",
        )

        # ==========================================
        # 2. Metadata Branch (Incidence Angle)
        # ==========================================
        # Processes the scalar incidence angle.
        self.meta_mlp = nn.Sequential(
            nn.Linear(Config.METADATA_INPUT_DIM, Config.METADATA_HIDDEN_DIM),
            nn.BatchNorm1d(Config.METADATA_HIDDEN_DIM),
            nn.ReLU(),
        )

        # ==========================================
        # 3. Fusion Head
        # ==========================================
        # Concatenates visual features (512) and metadata features (64).
        fusion_input_dim = Config.BACKBONE_OUT_DIM + Config.METADATA_HIDDEN_DIM

        self.fusion_head = nn.Sequential(
            nn.Linear(fusion_input_dim, Config.FUSION_HIDDEN_DIM),
            nn.BatchNorm1d(Config.FUSION_HIDDEN_DIM),
            nn.ReLU(),
            nn.Dropout(Config.DROPOUT_RATE),
            nn.Linear(Config.FUSION_HIDDEN_DIM, Config.NUM_CLASSES),
        )

    def forward(self, x_img, x_meta):
        """
        Forward pass of the RHTN.

        Args:
            x_img (torch.Tensor): Image tensor of shape (Batch, 3, 75, 75).
            x_meta (torch.Tensor): Metadata tensor of shape (Batch, 1).

        Returns:
            torch.Tensor: Logits of shape (Batch, 1).
        """
        # 1. Process Image
        # Output shape: (Batch, 512)
        img_features = self.backbone(x_img)

        # 2. Process Metadata
        # Output shape: (Batch, 64)
        meta_features = self.meta_mlp(x_meta)

        # 3. Feature Fusion
        # Concatenate along the feature dimension (dim=1)
        combined_features = torch.cat((img_features, meta_features), dim=1)

        # 4. Classification
        # Output shape: (Batch, 1)
        # We return logits. Sigmoid should be applied during inference or
        # handled by BCEWithLogitsLoss during training.
        logits = self.fusion_head(combined_features)

        return logits

import torch
import torch.nn as nn
from library.config import Config


class SAHCN(nn.Module):
    """
    Spatially-Aware Hybrid Convolutional Network (SAHCN).
    
    Custom shallow CNN architecture optimized for small SAR images (Cite solution_lesson_node_00016).
    """

    def __init__(self):
        super(SAHCN, self).__init__()

        # ==========================================
        # 1. Visual Branch (Custom 4-Layer CNN)
        # ==========================================
        self.conv_layers = nn.Sequential(
            # Block 1
            nn.Conv2d(Config.IN_CHANNELS, 64, kernel_size=3, padding=1),
            nn.BatchNorm2d(64),
            nn.ReLU(),
            nn.MaxPool2d(2, 2),
            
            # Block 2
            nn.Conv2d(64, 128, kernel_size=3, padding=1),
            nn.BatchNorm2d(128),
            nn.ReLU(),
            nn.MaxPool2d(2, 2),
            
            # Block 3
            nn.Conv2d(128, 128, kernel_size=3, padding=1),
            nn.BatchNorm2d(128),
            nn.ReLU(),
            nn.MaxPool2d(2, 2),
            
            # Block 4
            nn.Conv2d(128, 64, kernel_size=3, padding=1),
            nn.BatchNorm2d(64),
            nn.ReLU(),
            nn.MaxPool2d(2, 2)
        )
        
        # Flatten size calculation:
        # 75 -> 37 -> 18 -> 9 -> 4
        self.flatten_dim = 64 * 4 * 4

        # ==========================================
        # 2. Metadata Branch (Incidence Angle)
        # ==========================================
        self.meta_mlp = nn.Sequential(
            nn.Linear(Config.METADATA_INPUT_DIM, Config.METADATA_HIDDEN_DIM),
            nn.BatchNorm1d(Config.METADATA_HIDDEN_DIM),
            nn.ReLU(),
        )

        # ==========================================
        # 3. Fusion Head
        # ==========================================
        fusion_input_dim = self.flatten_dim + Config.METADATA_HIDDEN_DIM

        self.fusion_head = nn.Sequential(
            nn.Linear(fusion_input_dim, Config.FUSION_HIDDEN_DIM),
            nn.BatchNorm1d(Config.FUSION_HIDDEN_DIM),
            nn.ReLU(),
            nn.Dropout(Config.DROPOUT_RATE),
            nn.Linear(Config.FUSION_HIDDEN_DIM, Config.NUM_CLASSES),
        )

    def forward(self, x_img, x_meta):
        # 1. Process Image
        x = self.conv_layers(x_img)
        img_features = x.view(x.size(0), -1)

        # 2. Process Metadata
        meta_features = self.meta_mlp(x_meta)

        # 3. Feature Fusion
        combined_features = torch.cat((img_features, meta_features), dim=1)

        # 4. Classification
        logits = self.fusion_head(combined_features)

        return logits
