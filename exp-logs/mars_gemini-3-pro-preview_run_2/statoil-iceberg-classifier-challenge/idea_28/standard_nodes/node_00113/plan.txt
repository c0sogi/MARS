class DIDPBlock(nn.Module):
    """
    Delayed-Integration Dual-Pyramid Block.
    Structure: Conv(Wide) -> BN -> ReLU -> CBAM -> DualPooling.
    """

    def __init__(self, in_channels, out_channels):
        super(DIDPBlock, self).__init__()

        # Wide Convolution: 3x3 Convolution
        # This performs integration/compression at the start of the block
        self.conv = nn.Conv2d(
            in_channels, out_channels, kernel_size=3, padding=1, bias=False
        )
        self.bn = nn.BatchNorm2d(out_channels)
        self.relu = nn.ReLU(inplace=True)

        # Pre-Pooling Attention (CBAM)
        self.cbam = CBAM(out_channels)

        # Dual-Stream Pooling (Expands channels 2x)
        self.pool = DualPooling()

    def forward(self, x):
        x = self.conv(x)
        x = self.bn(x)
        x = self.relu(x)

        # Refine features at high resolution before pooling
        x = self.cbam(x)

        # Pool and expand channels
        x = self.pool(x)
        return x


class DIDPNet(nn.Module):
    """
    Delayed-Integration Dual-Pyramid Network (DIDP-Net).

    Features:
    - Split-Branch Topology (Visual + Metadata).
    - Sustained Width Backbone (128 filters).
    - Delayed Integration (Compression at start of new receptive field).
    - Quadrant-Based Readout.
    - High Dropout Regularization.
    """

    def __init__(
        self, backbone_filters=Config.BACKBONE_FILTERS, dropout_rate=Config.DROPOUT_RATE
    ):
        super(DIDPNet, self).__init__()

        self.backbone_filters = backbone_filters

        # ==========================
        # Visual Branch
        # ==========================

        # Stage 1: Input (3) -> 128 filters.
        # Output after DualPooling: 128 * 2 = 256 channels.
        # Size: 75 -> 37
        self.stage1 = DIDPBlock(3, self.backbone_filters)

        # Stage 2: Input (256) -> 128 filters.
        # Output after DualPooling: 256 channels.
        # Size: 37 -> 18
        self.stage2 = DIDPBlock(self.backbone_filters * 2, self.backbone_filters)

        # Stage 3: Input (256) -> 128 filters.
        # Output after DualPooling: 256 channels.
        # Size: 18 -> 9
        self.stage3 = DIDPBlock(self.backbone_filters * 2, self.backbone_filters)

        # Stage 4: Input (256) -> 128 filters.
        # Output after DualPooling: 256 channels.
        # Size: 9 -> 4
        self.stage4 = DIDPBlock(self.backbone_filters * 2, self.backbone_filters)

        # Quadrant-Based Readout
        # Adaptive Max Pooling to 2x2 grid
        self.quadrant_pool = nn.AdaptiveMaxPool2d((2, 2))

        # Flattened Visual Vector: 256 channels * 2 * 2 = 1024
        self.visual_dim = (self.backbone_filters * 2) * 2 * 2

        # ==========================
        # Metadata Branch
        # ==========================
        # Process incidence angle
        self.meta_mlp = nn.Sequential(
            nn.Linear(1, 16),
            nn.BatchNorm1d(16),
            nn.ReLU(inplace=True),
            nn.Linear(16, 32),
            nn.BatchNorm1d(32),
            nn.ReLU(inplace=True),
        )
        self.meta_dim = 32

        # ==========================
        # Fusion Head
        # ==========================
        fusion_dim = self.visual_dim + self.meta_dim

        self.classifier = nn.Sequential(
            nn.Linear(fusion_dim, 512),
            nn.BatchNorm1d(512),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout_rate),
            nn.Linear(512, 1),
        )

        # Initialize weights
        self._init_weights()

    def _init_weights(self):
        for m in self.modules():
            if isinstance(m, nn.Conv2d):
                nn.init.kaiming_normal_(m.weight, mode="fan_out", nonlinearity="relu")
            elif isinstance(m, nn.BatchNorm2d):
                nn.init.constant_(m.weight, 1)
                nn.init.constant_(m.bias, 0)
            elif isinstance(m, nn.Linear):
                nn.init.normal_(m.weight, 0, 0.01)
                nn.init.constant_(m.bias, 0)

    def forward(self, x, angle):
        """
        Args:
            x (torch.Tensor): Image input (N, 3, 75, 75)
            angle (torch.Tensor): Incidence angle input (N,) or (N, 1)
        """
        # --- Visual Forward ---
        x = self.stage1(x)
        x = self.stage2(x)
        x = self.stage3(x)
        x = self.stage4(x)

        # Quadrant Readout (N, 256, 4, 4) -> (N, 256, 2, 2)
        x = self.quadrant_pool(x)
        x = x.view(x.size(0), -1)  # Flatten -> (N, 1024)

        # --- Metadata Forward ---
        if angle.dim() == 1:
            angle = angle.view(-1, 1)

        m = self.meta_mlp(angle)

        # --- Fusion ---
        combined = torch.cat([x, m], dim=1)

        # --- Classification ---
        # Returns logits
        out = self.classifier(combined)

        return out

class DPCBlock(nn.Module):
    """
    Dual-Pooling Contracted Block.
    Structure: Conv(3x3) -> BN -> ReLU -> CBAM -> DualPooling.
    """

    def __init__(self, in_channels, out_channels):
        super(DPCBlock, self).__init__()

        # Convolution handles compression/expansion from previous layer
        self.conv = nn.Conv2d(
            in_channels, out_channels, kernel_size=3, padding=1, bias=False
        )
        self.bn = nn.BatchNorm2d(out_channels)
        self.relu = nn.ReLU(inplace=True)

        # Pre-Pooling Attention (CBAM) - Cite solution_lesson_node_00061
        self.cbam = CBAM(out_channels)

        # Dual-Stream Pooling (Expands channels 2x) - Cite solution_lesson_node_00070
        self.pool = DualPooling()

    def forward(self, x):
        x = self.conv(x)
        x = self.bn(x)
        x = self.relu(x)
        x = self.cbam(x)
        x = self.pool(x)
        return x


class DPCNet(nn.Module):
    """
    Dual-Pooling Contracted Network (DPCNet).
    
    Features:
    - Contracting Architecture (Preserve Spatial, Reduce Channels) - Cite solution_lesson_node_00112
    - Dual-Stream Pooling (Max+Min) - Cite solution_lesson_node_00070
    - CBAM Attention - Cite solution_lesson_node_00061
    - High Dropout (0.5) - Cite solution_lesson_node_00077
    """

    def __init__(
        self, backbone_filters=Config.BACKBONE_FILTERS, dropout_rate=Config.DROPOUT_RATE
    ):
        super(DPCNet, self).__init__()

        # Initial filters = 64
        self.filters = backbone_filters

        # ==========================
        # Visual Branch
        # ==========================

        # Stage 1: Input (3) -> 64 filters.
        # Output after DualPooling: 64 * 2 = 128 channels.
        # Size: 75 -> 37
        self.stage1 = DPCBlock(3, self.filters)

        # Stage 2: Input (128) -> 128 filters.
        # Output after DualPooling: 128 * 2 = 256 channels.
        # Size: 37 -> 18
        self.stage2 = DPCBlock(self.filters * 2, self.filters * 2)

        # Stage 3: Input (256) -> 128 filters.
        # Output after DualPooling: 128 * 2 = 256 channels.
        # Size: 18 -> 9
        self.stage3 = DPCBlock(self.filters * 4, self.filters * 2)

        # Stage 4: Input (256) -> 64 filters (Contraction).
        # Output after DualPooling: 64 * 2 = 128 channels.
        # Size: 9 -> 4
        self.stage4 = DPCBlock(self.filters * 4, self.filters)

        # Flattened Visual Vector: 128 channels * 4 * 4 = 2048
        # We preserve the 4x4 spatial grid rather than pooling to 2x2.
        self.visual_dim = (self.filters * 2) * 4 * 4

        # ==========================
        # Metadata Branch
        # ==========================
        self.meta_mlp = nn.Sequential(
            nn.Linear(1, 16),
            nn.BatchNorm1d(16),
            nn.ReLU(inplace=True),
            nn.Linear(16, 32),
            nn.BatchNorm1d(32),
            nn.ReLU(inplace=True),
        )
        self.meta_dim = 32

        # ==========================
        # Fusion Head
        # ==========================
        fusion_dim = self.visual_dim + self.meta_dim

        self.classifier = nn.Sequential(
            nn.Linear(fusion_dim, 512),
            nn.BatchNorm1d(512),
            nn.ReLU(inplace=True),
            nn.Dropout(dropout_rate),
            nn.Linear(512, 1),
        )

        self._init_weights()

    def _init_weights(self):
        for m in self.modules():
            if isinstance(m, nn.Conv2d):
                nn.init.kaiming_normal_(m.weight, mode="fan_out", nonlinearity="relu")

                nn.init.constant_(m.bias, 0)

                nn.init.normal_(m.weight, 0, 0.01)
                nn.init.constant_(m.bias, 0)

    def forward(self, x, angle):

        x = self.stage2(x)
        x = self.stage3(x)
        x = self.stage4(x)

        # --- Fusion ---

        # --- Classification ---
