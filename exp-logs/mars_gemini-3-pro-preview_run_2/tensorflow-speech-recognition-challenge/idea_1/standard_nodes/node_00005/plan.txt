To optimize the performance of the Speech Command Recognition task, I will implement a strategy based on **transfer learning with a vision backbone** and **spectrogram augmentation**, as recommended in the provided lessons.

### Strategy Outline
1.  **Model Architecture Upgrade (Cite `solution_lesson_node_00004`)**: I will replace the custom 4-layer `SimpleConvNet` with a **ResNet18** model pretrained on ImageNet. I will adapt the first convolutional layer to accept 1-channel input (spectrograms) by averaging the weights of the original 3-channel RGB filters. This allows the model to leverage robust feature extractors learned from a massive visual dataset.
2.  **Data Augmentation (Cite `solution_lesson_node_00004`)**: I will introduce **SpecAugment** (Frequency Masking and Time Masking) into the training data pipeline. This regularization technique randomly masks blocks of the spectrogram, forcing the model to learn distributed features and preventing it from overfitting to specific frequency triggers or temporal artifacts.
3.  **Hyperparameter Tuning**: I will increase the number of training epochs from 15 to 20 to allow the deeper ResNet model sufficient time to converge, and increase the number of data loader workers to 8 to maximize throughput on the provided vCPUs.
4.  **Submission Logic**: I will update the execution script to conditionally generate the submission file only if the validation accuracy exceeds the specified threshold of `0.9438916297349277`.

class SimpleConvNet(nn.Module):
    """
    A simple 4-block Convolutional Neural Network for Audio Classification.
    Input: (Batch, 1, Freq, Time) -> Log Mel-Spectrograms
    Output: (Batch, Num_Classes) -> Logits
    """

    def __init__(self, num_classes=Config.NUM_CLASSES):
        super(SimpleConvNet, self).__init__()

        # Block 1: 1 -> 16 channels
        self.block1 = nn.Sequential(
            nn.Conv2d(1, 16, kernel_size=3, padding=1),
            nn.BatchNorm2d(16),
            nn.ReLU(),
            nn.MaxPool2d(kernel_size=2, stride=2),
        )

        # Block 2: 16 -> 32 channels
        self.block2 = nn.Sequential(
            nn.Conv2d(16, 32, kernel_size=3, padding=1),
            nn.BatchNorm2d(32),
            nn.ReLU(),
            nn.MaxPool2d(kernel_size=2, stride=2),
        )

        # Block 3: 32 -> 64 channels
        self.block3 = nn.Sequential(
            nn.Conv2d(32, 64, kernel_size=3, padding=1),
            nn.BatchNorm2d(64),
            nn.ReLU(),
            nn.MaxPool2d(kernel_size=2, stride=2),
        )

        # Block 4: 64 -> 128 channels
        self.block4 = nn.Sequential(
            nn.Conv2d(64, 128, kernel_size=3, padding=1),
            nn.BatchNorm2d(128),
            nn.ReLU(),
            nn.MaxPool2d(kernel_size=2, stride=2),
        )

        # Global Average Pooling
        self.global_pool = nn.AdaptiveAvgPool2d(1)

        # Classifier
        self.fc = nn.Linear(128, num_classes)

    def forward(self, x):
        x = self.block1(x)
        x = self.block2(x)
        x = self.block3(x)
        x = self.block4(x)

        x = self.global_pool(x)
        x = x.view(x.size(0), -1)  # Flatten: (Batch, 128)
        x = self.fc(x)
        return x

import torchvision.models as models

class AudioResNet18(nn.Module):
    """
    ResNet18-based model adapted for Audio Classification (Spectrograms).
    Cite solution_lesson_node_00004
    """

    def __init__(self, num_classes=Config.NUM_CLASSES):
        super(AudioResNet18, self).__init__()

        # Load pretrained ResNet18
        weights = models.ResNet18_Weights.IMAGENET1K_V1
        self.model = models.resnet18(weights=weights)

        # Adapt first layer for 1-channel input (Spectrogram)
        # We average the weights of the original 3 RGB channels
        original_conv1 = self.model.conv1
        self.model.conv1 = nn.Conv2d(
            1,
            original_conv1.out_channels,
            kernel_size=original_conv1.kernel_size,
            stride=original_conv1.stride,
            padding=original_conv1.padding,
            bias=original_conv1.bias,
        )
        with torch.no_grad():
            self.model.conv1.weight.data = torch.mean(
                original_conv1.weight.data, dim=1, keepdim=True
            )

        # Adapt classifier
        num_ftrs = self.model.fc.in_features
        self.model.fc = nn.Linear(num_ftrs, num_classes)

    def forward(self, x):
        return self.model(x)
