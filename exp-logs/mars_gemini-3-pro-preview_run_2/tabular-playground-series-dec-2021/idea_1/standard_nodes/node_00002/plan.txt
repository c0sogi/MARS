def add_features(df):
    """
    Applies feature engineering to the dataset.
    Cite solution_lesson_node_00001: Addressing weak sub-domains via interaction features.
    """
    df = df.copy()

    # 1. Geometry Features
    # Euclidean distance to hydrology
    if "Horizontal_Distance_To_Hydrology" in df.columns and "Vertical_Distance_To_Hydrology" in df.columns:
        df["Euclidean_Distance_To_Hydrology"] = np.sqrt(
            df["Horizontal_Distance_To_Hydrology"] ** 2
            + df["Vertical_Distance_To_Hydrology"] ** 2
        )

    # Elevation relative to hydrology
    if "Elevation" in df.columns and "Vertical_Distance_To_Hydrology" in df.columns:
        df["Elevation_Hydrology_Diff"] = (
            df["Elevation"] - df["Vertical_Distance_To_Hydrology"]
        )

    # Mean distance to amenities
    if "Horizontal_Distance_To_Roadways" in df.columns and "Horizontal_Distance_To_Fire_Points" in df.columns:
        df["Mean_Distance_To_Amenities"] = (
            df["Horizontal_Distance_To_Roadways"]
            + df["Horizontal_Distance_To_Fire_Points"]
        ) / 2

    # 2. Interaction Features (Wilderness_Area interactions)
    # These areas were identified as high-error sources in failure analysis.
    # We interact them with high-importance features (Elevation, Roadways).
    wilderness_cols = [c for c in df.columns if "Wilderness_Area" in c]
    
    for wa in wilderness_cols:
        if "Elevation" in df.columns:
            df[f"Elevation_x_{wa}"] = df["Elevation"] * df[wa]
        if "Horizontal_Distance_To_Roadways" in df.columns:
            df[f"Roadways_x_{wa}"] = df["Horizontal_Distance_To_Roadways"] * df[wa]

    return df
