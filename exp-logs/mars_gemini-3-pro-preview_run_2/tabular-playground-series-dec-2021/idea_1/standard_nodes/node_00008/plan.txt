def engineer_features(df):
    """
    Performs feature engineering:
    - Condenses one-hot encoded Soil_Type and Wilderness_Area columns into single categorical columns.
    """
    # Soil_Type: Convert 40 binary columns to one integer column
    soil_cols = [c for c in df.columns if c.startswith("Soil_Type")]
    if soil_cols:
        # argmax returns the index of the max value (1).
        # We assume rows are valid one-hot (one 1, rest 0s).
        df["Soil_Type"] = df[soil_cols].values.argmax(axis=1).astype(np.int32)

    # Wilderness_Area: Convert 4 binary columns to one integer column
    wild_cols = [c for c in df.columns if c.startswith("Wilderness_Area")]
    if wild_cols:
        df["Wilderness_Area"] = df[wild_cols].values.argmax(axis=1).astype(np.int32)

    return df
