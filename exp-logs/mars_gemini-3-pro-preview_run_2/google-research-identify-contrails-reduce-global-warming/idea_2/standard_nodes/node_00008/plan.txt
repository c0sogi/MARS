class UNet(nn.Module):
    """
    Standard U-Net Architecture with ResNet18 Encoder.
    """

    def __init__(self, in_channels=N_CHANNELS, n_classes=N_CLASSES):
        super().__init__()
        self.encoder = ResNet18Encoder(in_channels)

        # Encoder output channels:
        # x0: 64 (Scale 1/2)
        # x1: 64 (Scale 1/4)
        # x2: 128 (Scale 1/8)
        # x3: 256 (Scale 1/16)
        # x4: 512 (Scale 1/32)

        # Decoder
        # Up 1: x4 (512) -> x3 (256). Concat -> 512+256 = 768
        self.up1 = nn.ConvTranspose2d(512, 256, kernel_size=2, stride=2)
        self.conv1 = ConvBlock(256 + 256, 256)

        # Up 2: x3 (256) -> x2 (128). Concat -> 256+128 = 384
        self.up2 = nn.ConvTranspose2d(256, 128, kernel_size=2, stride=2)
        self.conv2 = ConvBlock(128 + 128, 128)

        # Up 3: x2 (128) -> x1 (64). Concat -> 128+64 = 192
        self.up3 = nn.ConvTranspose2d(128, 64, kernel_size=2, stride=2)
        self.conv3 = ConvBlock(64 + 64, 64)

        # Up 4: x1 (64) -> x0 (64). Concat -> 64+64 = 128
        self.up4 = nn.ConvTranspose2d(64, 64, kernel_size=2, stride=2)
        self.conv4 = ConvBlock(64 + 64, 64)

        # Final Convolution
        self.final_conv = nn.Conv2d(64, n_classes, kernel_size=1)

    def forward(self, x):
        # Encoder
        x0, x1, x2, x3, x4 = self.encoder(x)

        # Decoder
        # Block 1
        d1 = self.up1(x4)
        d1 = torch.cat([d1, x3], dim=1)
        d1 = self.conv1(d1)

        # Block 2
        d2 = self.up2(d1)
        d2 = torch.cat([d2, x2], dim=1)
        d2 = self.conv2(d2)

        # Block 3
        d3 = self.up3(d2)
        d3 = torch.cat([d3, x1], dim=1)
        d3 = self.conv3(d3)

        # Block 4
        d4 = self.up4(d3)
        d4 = torch.cat([d4, x0], dim=1)
        d4 = self.conv4(d4)

        # Final projection (Scale 1/2 -> 128x128)
        logits = self.final_conv(d4)

        # Upsample to full resolution (Scale 1/1 -> 256x256)
        logits = F.interpolate(
            logits, scale_factor=2, mode="bilinear", align_corners=True
        )

        return logits
