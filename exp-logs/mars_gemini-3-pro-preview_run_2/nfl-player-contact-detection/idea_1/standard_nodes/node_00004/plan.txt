class FocalLoss(nn.Module):
    """
    Focal Loss for imbalanced classification.
    Cite solution_lesson_node_00003
    """
    def __init__(self, alpha=0.25, gamma=2.0, reduction='mean'):
        super(FocalLoss, self).__init__()
        self.alpha = alpha
        self.gamma = gamma
        self.reduction = reduction

    def forward(self, inputs, targets):
        p = inputs
        p = torch.clamp(p, min=1e-8, max=1.0 - 1e-8)
        loss = -self.alpha * (1 - p)**self.gamma * targets * torch.log(p) \
               - (1 - self.alpha) * p**self.gamma * (1 - targets) * torch.log(1 - p)
        
        if self.reduction == 'mean':
            return loss.mean()
        elif self.reduction == 'sum':
            return loss.sum()
        else:
            return loss

def train_one_epoch(dataloader, model, optimizer, device, pos_weight):
    """
    Performs backpropagation for one epoch using a weighted BCELoss.

    Args:
        dataloader: PyTorch DataLoader for training data.
        model: The KinematicMLP model.
        optimizer: The optimizer.
        device: 'cuda' or 'cpu'.
        pos_weight: Tensor containing the weight for the positive class.

    Returns:
        float: Average training loss for the epoch.
    """
    model.train()
    running_loss = 0.0
    dataset_size = 0

    # Using BCELoss with manual weighting to handle class imbalance
    # Model outputs probabilities via Sigmoid
    loss_fn = nn.BCELoss(reduction="none")

    for batch in dataloader:
        features = batch["features"].to(device)
        targets = batch["target"].to(device).unsqueeze(1)
        batch_size = features.size(0)

        optimizer.zero_grad()

        outputs = model(features)

        # Calculate weighted loss
        loss_unreduced = loss_fn(outputs, targets)
        weights = targets * (pos_weight - 1) + 1
        loss = (loss_unreduced * weights).mean()

        loss.backward()
        optimizer.step()
