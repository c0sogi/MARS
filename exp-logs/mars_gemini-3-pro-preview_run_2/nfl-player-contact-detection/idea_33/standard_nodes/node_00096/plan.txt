def calculate_iou(df, suffix):
    """Calculates Intersection over Union for bounding boxes."""
    # P1
    l1 = df[f"left{suffix}_1"]
    t1 = df[f"top{suffix}_1"]
    r1 = l1 + df[f"width{suffix}_1"]
    b1 = t1 + df[f"height{suffix}_1"]

    # P2
    l2 = df[f"left{suffix}_2"]
    t2 = df[f"top{suffix}_2"]
    r2 = l2 + df[f"width{suffix}_2"]
    b2 = t2 + df[f"height{suffix}_2"]

    # Intersection
    x_left = np.maximum(l1, l2)
    y_top = np.maximum(t1, t2)
    x_right = np.minimum(r1, r2)
    y_bottom = np.minimum(b1, b2)

    w_int = np.maximum(0, x_right - x_left)
    h_int = np.maximum(0, y_bottom - y_top)
    area_int = w_int * h_int

    # Union
    area1 = df[f"width{suffix}_1"] * df[f"height{suffix}_1"]
    area2 = df[f"width{suffix}_2"] * df[f"height{suffix}_2"]
    area_union = area1 + area2 - area_int

    # IoU
    iou = area_int / (area_union + 1e-6)
    return iou

        # 2. Relative Speed (Scalar Diff)
        df[f"relative_speed{suffix}"] = df[f"speed{suffix}_1"] - df[f"speed{suffix}_2"]

        # 3. Closing Speed (Vector Projected) - Cite Lesson 00005, 00007
        # Convert direction (deg) to radians. NFL: 0=Y, 90=X.
        # vx = s * sin(theta), vy = s * cos(theta)
        dir1_rad = np.deg2rad(df[f"direction{suffix}_1"])
        vx1 = df[f"speed{suffix}_1"] * np.sin(dir1_rad)
        vy1 = df[f"speed{suffix}_1"] * np.cos(dir1_rad)

        dir2_rad = np.deg2rad(df[f"direction{suffix}_2"])
        vx2 = df[f"speed{suffix}_2"] * np.sin(dir2_rad)
        vy2 = df[f"speed{suffix}_2"] * np.cos(dir2_rad)

        dvx = vx1 - vx2
        dvy = vy1 - vy2
        
        # Dot product of relative velocity and relative position
        dot = dvx * dx + dvy * dy
        # Closing speed is negative rate of distance change.
        # Clamping denominator for stability (Lesson 00007)
        df[f"closing_speed{suffix}"] = -(dot) / np.maximum(dist, 1e-6)
