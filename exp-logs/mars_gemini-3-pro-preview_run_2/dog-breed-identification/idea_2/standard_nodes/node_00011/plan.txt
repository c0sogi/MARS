class DualBackboneExtractor:
    """
    Handles loading of frozen backbones and feature extraction with TTA.
    """

    def __init__(self):
        self.device = config.DEVICE
        print(f"Initializing DualBackboneExtractor on {self.device}...")

        # Load Backbone A: CNN
        print(f"Loading CNN: {config.MODEL_CNN}")
        self.cnn = timm.create_model(config.MODEL_CNN, pretrained=True, num_classes=0)
        self.cnn.to(self.device)
        self.cnn.eval()

        # Load Backbone B: ViT
        print(f"Loading ViT: {config.MODEL_VIT}")
        self.vit = timm.create_model(config.MODEL_VIT, pretrained=True, num_classes=0)
        self.vit.to(self.device)
        self.vit.eval()

        # Freeze parameters
        for param in self.cnn.parameters():
            param.requires_grad = False
        for param in self.vit.parameters():
            param.requires_grad = False

    def extract_features(self, dataloader):
        """
        Iterates over the dataloader, applies TTA, and extracts fused embeddings.

        Args:
            dataloader: PyTorch DataLoader.

        Returns:
            tuple: (embeddings, labels, ids)
                   embeddings: np.ndarray of shape (N, dim_cnn + dim_vit)
                   labels: np.ndarray of shape (N,)
                   ids: np.ndarray of shape (N,)
        """
        all_embeddings = []
        all_labels = []
        all_ids = []

        with torch.no_grad():
            for batch in dataloader:
                # Unpack batch
                cnn_imgs = batch["cnn_img"].to(self.device)
                vit_imgs = batch["vit_img"].to(self.device)
                labels = batch["label"].numpy()
                ids = batch["id"]  # tuple of strings

                # --- Test Time Augmentation (Horizontal Flip) ---

                # CNN TTA
                # Create flipped versions
                cnn_imgs_flip = torch.flip(cnn_imgs, dims=[3])  # [B, C, H, W], flip W
                # Concatenate along batch dimension for efficient processing: [2B, C, H, W]
                cnn_batch = torch.cat([cnn_imgs, cnn_imgs_flip], dim=0)

                # Forward pass CNN
                cnn_feats_batch = self.cnn(cnn_batch)  # [2B, D_cnn]

                # Split back
                batch_size = cnn_imgs.shape[0]
                cnn_feats_orig = cnn_feats_batch[:batch_size]
                cnn_feats_flip = cnn_feats_batch[batch_size:]

                # Average embeddings
                cnn_feats = (cnn_feats_orig + cnn_feats_flip) / 2.0

                # ViT TTA
                vit_imgs_flip = torch.flip(vit_imgs, dims=[3])
                vit_batch = torch.cat([vit_imgs, vit_imgs_flip], dim=0)

                # Forward pass ViT
                vit_feats_batch = self.vit(vit_batch)  # [2B, D_vit]

                vit_feats_orig = vit_feats_batch[:batch_size]
                vit_feats_flip = vit_feats_batch[batch_size:]

                vit_feats = (vit_feats_orig + vit_feats_flip) / 2.0

                # --- Fusion ---
                # Concatenate CNN and ViT features: [B, D_cnn + D_vit]
                fused_feats = torch.cat([cnn_feats, vit_feats], dim=1)

                # Store results
                all_embeddings.append(fused_feats.cpu().numpy())
                all_labels.append(labels)
                all_ids.extend(ids)

        if len(all_embeddings) > 0:
            all_embeddings = np.vstack(all_embeddings)
            all_labels = np.concatenate(all_labels)
            all_ids = np.array(all_ids)
        else:
            all_embeddings = np.array([])
            all_labels = np.array([])
            all_ids = np.array([])

        return all_embeddings, all_labels, all_ids
