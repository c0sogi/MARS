    def apply_spec_augment(self, spec):
        """
        Applies Frequency and Time Masking to the spectrogram.
        Cite solution_lesson_node_00011, solution_lesson_node_00013
        """
        # spec: (1, n_mels, time)
        _, n_mels, n_steps = spec.shape

        # Frequency Masking
        f = Config.FREQ_MASK_PARAM
        f_val = np.random.randint(0, f + 1)
        if f_val > 0:
            f0 = np.random.randint(0, n_mels - f_val + 1)
            # Mask with minimum value (silence in dB)
            spec[:, f0 : f0 + f_val, :] = spec.min()

        # Time Masking
        t = Config.TIME_MASK_PARAM
        t_val = np.random.randint(0, t + 1)
        if t_val > 0:
            t0 = np.random.randint(0, n_steps - t_val + 1)
            spec[:, :, t0 : t0 + t_val] = spec.min()

        return spec
