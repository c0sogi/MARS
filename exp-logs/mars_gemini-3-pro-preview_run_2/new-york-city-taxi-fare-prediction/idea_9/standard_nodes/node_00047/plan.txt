    """
    Adds Haversine and Manhattan distance features to the dataframe.

    """
    print("Computing geometric features...")
    
    # Coordinates
    lat1 = df["pickup_latitude"]
    lon1 = df["pickup_longitude"]
    lat2 = df["dropoff_latitude"]
    lon2 = df["dropoff_longitude"]
    
    # Manhattan distance (Euclidean approximation in degrees)
    df["dist_manhattan"] = np.abs(lat2 - lat1) + np.abs(lon2 - lon1)
    
    # Haversine distance
    R = 6371  # Earth radius in km
    phi1 = np.radians(lat1)
    phi2 = np.radians(lat2)
    dphi = np.radians(lat2 - lat1)
    dlambda = np.radians(lon2 - lon1)

    c = 2 * np.arctan2(np.sqrt(a), np.sqrt(1 - a))
    
    df["dist_haversine"] = R * c
