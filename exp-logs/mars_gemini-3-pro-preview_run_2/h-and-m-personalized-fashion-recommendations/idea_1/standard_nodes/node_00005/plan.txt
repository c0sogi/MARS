def load_and_filter_data(load_cached_data=True):
    """
    Loads training and validation data.
    Merges the provided disjoint metadata files and performs a Time-based Split.
    Cite solution_lesson_node_00003: "Never use a disjoint user split... use a Time-based Split".
    Filters training data to keep only the last HISTORY_WEEKS of transactions.
    """
    train_split_path = WORKING_DIR / "time_split_train.parquet"
    val_split_path = WORKING_DIR / "time_split_val.parquet"

    WORKING_DIR.mkdir(parents=True, exist_ok=True)

    # 1. Try to load time-split data from cache
    if load_cached_data and train_split_path.exists() and val_split_path.exists():
        print("Loading time-split data from cache...")
        train_df = pd.read_parquet(train_split_path)
        val_df = pd.read_parquet(val_split_path)
        return train_df, val_df

    # 2. Load raw data and merge
    print("Loading raw metadata and merging for time-split...")
    df1 = pd.read_parquet(TRAIN_DATA_PATH)
    df2 = pd.read_parquet(VAL_DATA_PATH)
    
    full_df = pd.concat([df1, df2], axis=0)
    full_df["t_dat"] = pd.to_datetime(full_df["t_dat"])

    # 3. Perform Time-based Split (Last 7 days for validation)
    max_date = full_df["t_dat"].max()
    split_date = max_date - pd.Timedelta(days=7)
    
    print(f"Splitting data at {split_date}...")
    val_df = full_df[full_df["t_dat"] > split_date].copy()
    train_all = full_df[full_df["t_dat"] <= split_date].copy()

    # 4. Filter Training Data History
    print(f"Filtering training data to last {HISTORY_WEEKS} weeks...")
    cutoff_date = split_date - pd.Timedelta(weeks=HISTORY_WEEKS)
    train_df = train_all[train_all["t_dat"] > cutoff_date].copy()

    # 5. Save to cache
    train_df.to_parquet(train_split_path, index=False)
    val_df.to_parquet(val_split_path, index=False)
    print("Time-split data saved to cache.")

    return train_df, val_df
