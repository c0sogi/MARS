    @staticmethod
    def calculate_angular_velocity(df: pd.DataFrame) -> pd.DataFrame:
        """
        Calculates Angular Velocity (Cite Lesson 27).
        """
        logger.info("Calculating Angular Velocity...")

        # Ensure data is sorted
        df = df.sort_values(by=["game_play", "nfl_player_id", "step"]).copy()

        # Fill NaNs
        df["orientation"] = df["orientation"].fillna(0.0)

        grp = df.groupby(["game_play", "nfl_player_id"])

        def calc_ang_vel(x):
            d = x.diff()
            # Correct for wrap around 0/360
            d = np.where(d > 180, d - 360, d)
            d = np.where(d < -180, d + 360, d)
            return d / 0.1  # 10Hz

        df["angular_velocity"] = grp["orientation"].transform(calc_ang_vel).fillna(0.0)

        return df

    @staticmethod
    def create_rolling_features(
        df: pd.DataFrame, features: list, window: int
    ) -> pd.DataFrame:
        """
        Creates rolling statistics (Mean, Std) for Macro Context (Cite Lesson 31).
        """
        logger.info(f"Creating Rolling Features for {features} with window {window}...")

        # Determine grouping key
        if "contact_id" in df.columns:
            group_cols = ["contact_id"]
        elif "nfl_player_id_2" in df.columns:
            group_cols = ["game_play", "nfl_player_id_1", "nfl_player_id_2"]
        else:
            group_cols = ["game_play", "nfl_player_id"]

        grouper = df.groupby(group_cols)[features]

        # Calculate Rolling Mean
        df_mean = grouper.transform(
            lambda x: x.rolling(window=window, center=True, min_periods=1).mean()
        )
        df_mean.columns = [f"{c}_roll_mean" for c in features]

        # Calculate Rolling Std
        df_std = grouper.transform(
            lambda x: x.rolling(
                window=window, center=True, min_periods=1
            ).std().fillna(0)
        )
        df_std.columns = [f"{c}_roll_std" for c in features]

        return pd.concat([df, df_mean, df_std], axis=1)
