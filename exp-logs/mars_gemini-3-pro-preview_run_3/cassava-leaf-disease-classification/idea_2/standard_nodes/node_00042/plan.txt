def train_one_epoch(epoch, model, train_loader, optimizer, device, criterion):
    """
    Performs one training epoch with MixUp and CutMix augmentation.

    Args:
        epoch (int): Current epoch number.
        model (torch.nn.Module): The model to train.
        train_loader (DataLoader): DataLoader for training data.
        optimizer (Optimizer): The optimizer.
        device (torch.device): The device to run on.
        criterion (loss): The loss function.

    Returns:
        float: Average loss for the epoch.
    """
    model.train()
    summary_loss = AverageMeter()
    start_time = time.time()

    for step, (images, labels) in enumerate(train_loader):
        images = images.to(device)
        labels = labels.to(device)
        batch_size = images.size(0)

        # Randomly decide which augmentation to apply based on CFG probabilities
        # We treat the probabilities as a partition of the [0, 1] interval.
        # 0 -> mixup_prob : MixUp
        # mixup_prob -> mixup_prob + cutmix_prob : CutMix
        # mixup_prob + cutmix_prob -> 1 : No augmentation

        r = np.random.rand()

        if r < CFG.mixup_prob:
            # Apply MixUp
            images, y_a, y_b, lam = mixup_data(images, labels, alpha=1.0)
            outputs = model(images)
            loss = lam * criterion(outputs, y_a) + (1 - lam) * criterion(outputs, y_b)
        elif r < (CFG.mixup_prob + CFG.cutmix_prob):
            # Apply CutMix
            images, y_a, y_b, lam = cutmix_data(images, labels, alpha=1.0)

            loss = lam * criterion(outputs, y_a) + (1 - lam) * criterion(outputs, y_b)
        else:
            # Normal training
            outputs = model(images)
            loss = criterion(outputs, labels)

        loss.backward()
        optimizer.step()
