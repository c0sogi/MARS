def compute_custom_loss(logits, labels, criterion):
    """
    Computes the hierarchical loss:
    Loss = Mean(BCE(subtypes)) + BCE(Max(subtypes), overall)
    
    Cite solution_lesson_node_00005: Optimizing Differentiable Aggregates
    Cite solution_lesson_node_00008: Implicit Weighting via Loss Reduction
    """
    # Separate labels
    # labels shape: (Batch, 8) -> 0-6: C1-C7, 7: patient_overall
    labels_c1_c7 = labels[:, :7]
    labels_overall = labels[:, 7]

    # 1. Subtype Loss (C1-C7)
    # logits shape: (Batch, 7)
    loss_subtypes = criterion(logits, labels_c1_c7)

    # 2. Overall Loss (patient_overall)
    # Derive overall logit via max pooling over subtypes
    logits_overall, _ = torch.max(logits, dim=1)
    loss_overall = criterion(logits_overall, labels_overall)

    # Total loss
    # Implicit weighting: loss_subtypes is mean of 7, loss_overall is mean of 1.
    # Summing them gives 1:7 effective weight ratio.
    total_loss = loss_subtypes + loss_overall
    return total_loss
