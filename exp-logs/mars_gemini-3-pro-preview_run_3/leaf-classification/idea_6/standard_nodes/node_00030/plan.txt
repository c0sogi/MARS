class BaggedLDAPipeline(BaseEstimator, ClassifierMixin):
    """
    Homogeneous Bagged Ensemble of SingleLDAPipeline.
    """
    def __init__(self):
        self.n_estimators = Config.N_ESTIMATORS
        self.estimators_ = []
        self.classes_ = None

    def fit(self, X_dino, X_conv, X_tab, y):
        self.classes_ = np.unique(y)
        n_samples = X_dino.shape[0]
        self.estimators_ = []

        print(f"Fitting BaggedLDAPipeline with {self.n_estimators} estimators...")

        for i in range(self.n_estimators):
            # Ensure diversity in bootstrap sampling
            np.random.seed(Config.SEED + i)
            indices = np.random.choice(n_samples, size=n_samples, replace=True)
            
            estimator = SingleLDAPipeline()
            estimator.fit(
                X_dino[indices], 
                X_conv[indices], 
                X_tab[indices], 
                y[indices]
            )
            self.estimators_.append(estimator)
            
        return self

    def predict_proba(self, X_dino, X_conv, X_tab):
        if not self.estimators_:
            raise RuntimeError("Model not fitted")
            
        avg_probs = None
        for estimator in self.estimators_:
            probs = estimator.predict_proba(X_dino, X_conv, X_tab)
            if avg_probs is None:
                avg_probs = probs
            else:
                avg_probs += probs
                
        avg_probs /= len(self.estimators_)
        return avg_probs
