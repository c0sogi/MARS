def predict_proba_aligned(pipeline, X, n_classes):
    """
    Predicts probabilities and aligns them to the full class set.
    If the model was trained on a subset of classes (e.g. in debug mode),
    this expands the output to shape (N, n_classes) with zeros for missing classes.

    Args:
        pipeline: Fitted sklearn pipeline.
        X (np.ndarray): Input features.
        n_classes (int): Total expected number of classes.

    Returns:
        np.ndarray: Probability matrix of shape (N, n_classes).
    """
    probs = pipeline.predict_proba(X)
    # The classifier is the last step named 'classifier'
    model_classes = pipeline.named_steps['classifier'].classes_

    # If dimensions match, return directly
    if probs.shape[1] == n_classes:
        return probs

    # Otherwise expand
    full_probs = np.zeros((probs.shape[0], n_classes))
    # model_classes contains the integer indices of the classes the model learned
    # We map the partial predictions to their correct columns
    full_probs[:, model_classes] = probs
    return full_probs
