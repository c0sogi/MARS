    def shift(arr, num, fill_value=0):
        result = np.empty_like(arr)
        if num > 0:
            result[:, :num] = fill_value
            result[:, num:] = arr[:, :-num]
        elif num < 0:
            result[:, num:] = fill_value
            result[:, :num] = arr[:, -num:]
        else:
            result[:] = arr
