To optimize the model's performance, I will implement a feature engineering strategy based on **Trend-Texture Decomposition** and **Shift-Invariant Temporal Aggregation**, as supported by the provided lessons.

### Strategy

1.  **Trend-Texture Decomposition (Cite `solution_lesson_node_00049`):**
    -   Replace the Wavelet decomposition with a **Savitzky-Golay filter** to separate the signal into a low-frequency **Trend** and high-frequency **Texture** (residuals).
    -   This allows the model to learn distinct physical properties: trajectory (from Trend) and signal complexity/roughness (from Texture).

2.  **Kinematics & Granular Quantiles (Cite `solution_lesson_node_00021`, `solution_lesson_node_00059`):**
    -   Calculate **Velocity** and **Acceleration** (derivatives) specifically on the smoothed **Trend** component to avoid noise amplification.
    -   Extract a dense grid of **Quantiles** (1% to 99%) for Trend, Velocity, Acceleration, and Texture. This captures the shape of the distribution better than abstract metrics.

3.  **Shift-Invariant Temporal Features (Cite `solution_lesson_node_00050`):**
    -   Modify the temporal windowing logic. Instead of flattening window statistics into distinct features (e.g., `win0_mean`, `win1_mean`), which overfits to specific positions, I will **aggregate** them (e.g., `std(window_means)`, `mean(window_rms)`). This ensures the features describe the *evolution* and *volatility* of the signal regardless of the start time.

def extract_features(df):
    """
    Implements the Dense-Quantile Wavelet Decomposition Strategy.
    Extracts Kinematic Quantiles, Texture Energy, Raw Intensity, and Spectral features.
    """
    # 1. Imputation: Fill NaNs with sensor mean to preserve DC offset
    df = df.fillna(df.mean())

    features = {}

    # Pre-calculate window size for temporal flattening
    n_samples = len(df)
    window_size = n_samples // NUM_WINDOWS

    for sensor in df.columns:
        sig = df[sensor].values

        # --- View 3: Raw Intensity (Outlier Preservation) ---
        features[f"{sensor}_min"] = np.min(sig)
        features[f"{sensor}_max"] = np.max(sig)
        features[f"{sensor}_ptp"] = np.ptp(sig)

        # --- View 5: Flattened Temporal Windows (Time Arrow) ---
        # Truncate signal to be divisible by NUM_WINDOWS
        n_trim = n_samples - (n_samples % NUM_WINDOWS)
        sig_trim = sig[:n_trim]
        sig_reshaped = sig_trim.reshape(NUM_WINDOWS, -1)

        # Vectorized computation
        win_means = np.mean(sig_reshaped, axis=1)
        win_rms = np.sqrt(np.mean(sig_reshaped**2, axis=1))

        for w in range(NUM_WINDOWS):
            features[f"{sensor}_w{w}_mean"] = win_means[w]
            features[f"{sensor}_w{w}_rms"] = win_rms[w]

        # --- Multi-Resolution Analysis (Wavelets) ---
        if pywt is not None:
            try:
                coeffs = pywt.wavedec(sig, WAVELET_TYPE, level=DECOMPOSITION_LEVEL)
                cA = coeffs[0]  # Approximation (Trend)
                cDs = coeffs[1:]  # Details (Texture/Noise)

                # View 1: Kinematic Quantiles on Approximation
                # Compute Velocity (1st deriv) and Acceleration (2nd deriv) of trend
                vel = np.diff(cA)
                acc = np.diff(vel)

                # Extract dense quantiles
                for q in QUANTILES:
                    features[f"{sensor}_cA_q{q}"] = np.quantile(cA, q)
                    features[f"{sensor}_cA_vel_q{q}"] = np.quantile(vel, q)
                    features[f"{sensor}_cA_acc_q{q}"] = np.quantile(acc, q)

                # View 2: Texture Energy on Details
                for i, cD in enumerate(cDs):
                    level_idx = DECOMPOSITION_LEVEL - i
                    # Energy (RMS)
                    features[f"{sensor}_cD{level_idx}_rms"] = np.sqrt(np.mean(cD**2))
                    # Entropy (Distribution complexity)
                    p = np.abs(cD) / (np.sum(np.abs(cD)) + 1e-9)
                    features[f"{sensor}_cD{level_idx}_entropy"] = entropy(p)
            except Exception:
                # Fail gracefully if wavelet computation errors out
                pass

        # --- View 4: Structural Spectral Features ---
        f, Pxx = welch(sig, fs=SAMPLING_RATE, nperseg=256)
        for band_name, (low_f, high_f) in PSD_BANDS.items():
            # Boolean mask for the band
            idx = np.logical_and(f >= low_f, f <= high_f)
            if np.sum(idx) > 0:
                features[f"{sensor}_psd_{band_name}"] = np.sum(Pxx[idx])
            else:
                features[f"{sensor}_psd_{band_name}"] = 0.0

    return features

def extract_features(df):
    """
    Implements the Trend-Texture Decomposition Strategy with Shift-Invariant Temporal Features.
    """

    n_samples = len(df)
    window_size = n_samples // NUM_WINDOWS

    for sensor in df.columns:
        sig = df[sensor].values

        # --- View 1: Decomposition (Cite solution_lesson_node_00049) ---
        # Separate Trend (Low Freq) and Texture (High Freq)
        trend = savgol_filter(sig, window_length=SAVGOL_WINDOW, polyorder=SAVGOL_POLY)
        texture = sig - trend

        # --- View 2: Kinematics on Trend (Cite solution_lesson_node_00021) ---
        # Velocity and Acceleration on the smoothed trend
        vel = np.diff(trend)
        acc = np.diff(vel)

        # --- View 3: Distributional Statistics (Cite solution_lesson_node_00059) ---
        # Quantiles for Trend, Velocity, Acceleration
        for q in QUANTILES:

            features[f"{sensor}_vel_q{q}"] = np.quantile(vel, q)
            features[f"{sensor}_acc_q{q}"] = np.quantile(acc, q)
            # For Texture, we also capture quantiles to understand noise distribution
            features[f"{sensor}_txt_q{q}"] = np.quantile(texture, q)

        # Texture Moments (Cite solution_lesson_node_00049)
        features[f"{sensor}_txt_rms"] = np.sqrt(np.mean(texture**2))
        
        # Raw Extremes (Cite solution_lesson_node_00031)
        features[f"{sensor}_min"] = np.min(sig)
        features[f"{sensor}_max"] = np.max(sig)

        # --- View 4: Shift-Invariant Temporal Features (Cite solution_lesson_node_00050) ---

        n_trim = n_samples - (n_samples % NUM_WINDOWS)
