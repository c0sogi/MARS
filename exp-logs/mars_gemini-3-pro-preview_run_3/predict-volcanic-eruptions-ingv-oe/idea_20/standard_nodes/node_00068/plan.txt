The following implementation strategy optimizes the solution by refining the feature engineering pipeline based on the provided lessons.

1.  **Trend-Texture Decomposition (Cite {solution_lesson_node_00049})**: We will explicitly decompose the sensor signal into a low-frequency `Trend` (using Savitzky-Golay) and a high-frequency `Texture` (Residual).
    -   **Kinematic Features**: Calculated on the `Trend` component (including the Trend itself, Velocity, and Acceleration). We apply granular quantiles to these to capture the shape of the deformation (Cite {solution_lesson_node_00059}).
    -   **Texture Features**: We implement a new `compute_texture_features` function to calculate robust moments (RMS, MAD, Skewness, Kurtosis) on the residual signal, capturing the "roughness" without the noise-amplifying effects of derivatives on raw data.

2.  **Shift-Invariant Temporal Statistics (Cite {solution_lesson_node_00050})**: We modify `compute_spectral_features` to stop flattening window statistics (e.g., `w0_rms`, `w1_rms`). Instead, we aggregate them (Mean, Std, Min, Max of window RMS) to ensure the features are invariant to the specific start time of the segment.

3.  **Hyperparameter Tuning**:
    -   **Learning Rate**: Increased from 0.005 to 0.01 to improve convergence speed while maintaining stability.
    -   **Num Leaves**: Increased from 63 to 64 to match the capacity of the previous best solution.
    -   **SG Derivatives**: Updated configuration to include the 0-th derivative (Trend) in the kinematic generation.

def compute_kinematic_features(signal, sensor_name, params):
    """
    Computes features based on Savitzky-Golay derivatives (Velocity, Acceleration).
    """
    feats = {}
    quantiles = params["quantiles"]

    # Parameters for SG filter
    window = params["sg_window_length"]
    poly = params["sg_polyorder"]

    # Iterate through requested derivatives (e.g., 1=Velocity, 2=Acceleration)
    for deriv in params["sg_derivs"]:
        # Apply filter
        try:
            deriv_sig = scipy.signal.savgol_filter(
                signal, window_length=window, polyorder=poly, deriv=deriv
            )
        except Exception:
            # Fallback if signal is too short (unlikely given 60k rows)
            deriv_sig = np.zeros_like(signal)

        suffix = "vel" if deriv == 1 else "acc" if deriv == 2 else f"deriv{deriv}"

        # Dense Quantiles
        q_vals = np.quantile(deriv_sig, quantiles)
        for q, val in zip(quantiles, q_vals):
            feats[f"{sensor_name}_{suffix}_q{int(q*100):02d}"] = val

        # Range of motion
        feats[f"{sensor_name}_{suffix}_range"] = np.max(deriv_sig) - np.min(deriv_sig)

    return feats

def compute_texture_features(signal, sensor_name, params):
    """
    Computes robust statistical moments on the Texture (Residual) signal.
    Cite {solution_lesson_node_00049}: Moments on Texture.
    """
    feats = {}
    
    # Robust amplitude metrics
    feats[f"{sensor_name}_txt_rms"] = np.sqrt(np.mean(signal**2))
    feats[f"{sensor_name}_txt_mad"] = np.mean(np.abs(signal - np.mean(signal)))
    
    # Shape metrics
    feats[f"{sensor_name}_txt_skew"] = scipy.stats.skew(signal)
    feats[f"{sensor_name}_txt_kurt"] = scipy.stats.kurtosis(signal)
    
    return feats


def compute_spectral_features(signal, sensor_name, params):
    """
    Computes Raw Intensity stats, PSD Band Power, and Aggregated Windowed RMS.
    Cite {solution_lesson_node_00050}: Shift Invariance via Aggregation.
    """
    feats = {}
    fs = params["sampling_rate"]

    # --- 1. Raw Intensity ---
    feats[f"{sensor_name}_raw_min"] = np.min(signal)
    feats[f"{sensor_name}_raw_max"] = np.max(signal)
    feats[f"{sensor_name}_raw_p2p"] = np.max(signal) - np.min(signal)

    # --- 2. PSD Band Power ---
    # Welch's method
    freqs, psd = scipy.signal.welch(signal, fs=fs, nperseg=min(len(signal), 256))

    for low, high in params["band_freqs"]:
        idx = np.logical_and(freqs >= low, freqs <= high)
        if np.sum(idx) > 0:
            band_power = np.mean(psd[idx])
        else:
            band_power = 0.0
        feats[f"{sensor_name}_psd_{low}-{high}Hz"] = band_power

    # --- 3. Windowed Statistics (Aggregated) ---
    n_windows = params["n_windows"]
    windows = np.array_split(signal, n_windows)
    
    rms_list = []
    for w_data in windows:
        if len(w_data) > 0:
            rms_list.append(np.sqrt(np.mean(w_data**2)))
        else:
            rms_list.append(0.0)

    # Aggregate to preserve shift invariance
    feats[f"{sensor_name}_win_rms_mean"] = np.mean(rms_list)
    feats[f"{sensor_name}_win_rms_std"] = np.std(rms_list)
    feats[f"{sensor_name}_win_rms_min"] = np.min(rms_list)
    feats[f"{sensor_name}_win_rms_max"] = np.max(rms_list)

    return feats
