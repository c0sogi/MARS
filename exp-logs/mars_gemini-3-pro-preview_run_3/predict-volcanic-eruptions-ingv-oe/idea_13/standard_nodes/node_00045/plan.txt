    Operates on Stream A (Raw).

def extract_kinematics(x_smooth):
    """
    View 2: Extracts Robust Kinematics (Velocity, Acceleration).
    Operates on Stream B (Smoothed).
    """
    # Velocity (1st Derivative)
    vel = np.gradient(x_smooth)
    vel_feats = np.concatenate(
        [np.quantile(vel, [0.01, 0.05, 0.95, 0.99]), [np.mean(vel), np.std(vel)]]
    )

    # Acceleration (2nd Derivative)
    acc = np.gradient(vel)
    acc_feats = np.concatenate(
        [np.quantile(acc, [0.01, 0.05, 0.95, 0.99]), [np.mean(acc), np.std(acc)]]
    )

    return np.concatenate([vel_feats, acc_feats])


def extract_wavelet(x):
    """
    View 3: Extracts Multi-Resolution Texture features.
    Operates on Stream A (Raw).

    Note: Since PyWavelets (pywt) is not available in the provided environment,
    we utilize Short-Time Fourier Transform (STFT) statistics as a proxy to
    capture time-localized frequency (texture) information.
    """
    # Compute STFT
    f, t, Zxx = signal.stft(x, fs=SAMPLING_RATE, nperseg=256)
    mag = np.abs(Zxx)

    # Extract statistics of the magnitude spectrogram across time
    mean_mag = np.mean(mag, axis=1)  # Average magnitude per frequency bin
    std_mag = np.std(mag, axis=1)  # Fluctuation of magnitude per frequency bin

    # Summarize into global texture descriptors
    feats = [
        np.mean(mean_mag),  # Global average energy
        np.std(mean_mag),  # Spread of energy across frequencies
        np.mean(std_mag),  # Average temporal fluctuation
        np.max(mag),  # Peak transient energy
    ]
    return np.array(feats)

def extract_temporal_windows(x):
    """
    View 5: Extracts Flattened Temporal Windows (RMS, Mean).
    Operates on Stream A (Raw).
    """
    wins = np.array_split(x, NUM_TEMPORAL_WINDOWS)
    feats = []
    for w in wins:
        feats.append(np.sqrt(np.mean(w**2)))  # RMS
        feats.append(np.mean(w))  # Mean
    return np.array(feats)


def extract_spatial(df_smooth):

    View 6: Extracts Spatial Fingerprinting (Pairwise Correlations).
    Operates on Stream B (Smoothed) across all sensors.

    # Compute correlation matrix
    corr_matrix = df_smooth.corr().abs().fillna(0)

    # Extract upper triangle (excluding diagonal)
    n_sensors = df_smooth.shape[1]
    upper_indices = np.triu_indices(n_sensors, k=1)
    return corr_matrix.values[upper_indices]
