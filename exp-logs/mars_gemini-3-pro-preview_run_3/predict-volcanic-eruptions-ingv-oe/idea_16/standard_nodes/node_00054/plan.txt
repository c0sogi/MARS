def extract_spectral_features(data, prefix, fs=100):
    """
    Computes PSD and extracts spectral features including band powers.
    Cite solution_lesson_node_00007 (Band Power vs Generic)
    Cite solution_lesson_node_00049 (Domain-relevant bands)
    """
    # Compute Periodogram (fs=100Hz based on 60000 samples / 600 seconds)
    f, Pxx = signal.periodogram(data, fs=fs)

    features = {}
    
    # Dominant Frequency & Max Power
    features[f"{prefix}_spec_max_power"] = np.max(Pxx)
    features[f"{prefix}_spec_dom_freq"] = f[np.argmax(Pxx)]

    # Spectral Centroid
    sum_pxx = np.sum(Pxx)
    if sum_pxx > 0:
        features[f"{prefix}_spec_centroid"] = np.sum(f * Pxx) / sum_pxx
    else:
        features[f"{prefix}_spec_centroid"] = 0.0

    # Band Powers
    # Seismic bands: Low (0.1-3Hz), Mid (3-10Hz), High (10-20Hz), Ultra (20Hz+)
    bands = {
        'band_low': (0.1, 3),
        'band_mid': (3, 10),
        'band_high': (10, 20),
        'band_ultra': (20, 50)
    }
    
    for band_name, (low, high) in bands.items():
        mask = (f >= low) & (f < high)
        if np.any(mask):
            features[f"{prefix}_{band_name}"] = np.sum(Pxx[mask])
        else:
            features[f"{prefix}_{band_name}"] = 0.0

    return features
