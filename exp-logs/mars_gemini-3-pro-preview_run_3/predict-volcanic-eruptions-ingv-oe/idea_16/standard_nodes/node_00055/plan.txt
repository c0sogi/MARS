def extract_spectral_features(data, prefix):
    """
    Computes PSD using Welch's method and extracts Band Powers.
    Cite Lesson 00054 (Welch vs Periodogram) and Lesson 00007 (Band Power).
    """
    # Welch's method for robust PSD estimation
    # fs=100Hz (60k samples / 10 mins), nperseg=256 gives ~0.4Hz resolution
    f, Pxx = signal.welch(data, fs=100, nperseg=256)

    features = {}
    
    # Define Seismic Frequency Bands
    bands = {
        "band_0_2Hz": (0, 2),       # Long Period / Infrasound
        "band_2_5Hz": (2, 5),       # Volcanic Tremor
        "band_5_10Hz": (5, 10),     # High Freq
        "band_10_20Hz": (10, 20),   # Tectonic
        "band_20_50Hz": (20, 50)    # Noise / Very High Freq
    }
    
    total_power = np.sum(Pxx)
    features[f"{prefix}_spec_total_power"] = total_power
    features[f"{prefix}_spec_max_power"] = np.max(Pxx)
    
    if total_power > 0:
        # Band Powers
        for band_name, (low, high) in bands.items():
            idx = np.logical_and(f >= low, f < high)
            power_in_band = np.sum(Pxx[idx])
            features[f"{prefix}_spec_{band_name}"] = power_in_band
            features[f"{prefix}_spec_{band_name}_rel"] = power_in_band / total_power
            
        # Spectral Centroid
        features[f"{prefix}_spec_centroid"] = np.sum(f * Pxx) / total_power
        
        # Spectral Spread (Variance)
        centroid = features[f"{prefix}_spec_centroid"]
        features[f"{prefix}_spec_spread"] = np.sqrt(np.sum(((f - centroid) ** 2) * Pxx) / total_power)
    else:
        # Handle silence/zeros
        for band_name in bands:
            features[f"{prefix}_spec_{band_name}"] = 0.0
            features[f"{prefix}_spec_{band_name}_rel"] = 0.0
        features[f"{prefix}_spec_centroid"] = 0.0
        features[f"{prefix}_spec_spread"] = 0.0

    # Dominant Frequency
    features[f"{prefix}_spec_dom_freq"] = f[np.argmax(Pxx)]

    return features
