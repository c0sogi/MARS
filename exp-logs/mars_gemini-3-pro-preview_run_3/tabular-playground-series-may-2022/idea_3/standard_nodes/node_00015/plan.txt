class DenoisingAutoencoder(nn.Module):
    """
    Denoising Autoencoder for unsupervised pretraining.
    Consists of the Encoder backbone and a reconstruction Decoder.
    """
    def __init__(self, cont_dim, cat_cardinalities):
        super().__init__()
        self.encoder = Encoder(cont_dim, cat_cardinalities)
        
        # Decoder Body
        # Input is the output of the Encoder
        in_dim = self.encoder.output_dim
        layers = []
        
        for out_dim in Config.DECODER_LAYERS:
            layers.append(nn.Linear(in_dim, out_dim))
            layers.append(nn.BatchNorm1d(out_dim))
            layers.append(nn.ReLU())
            in_dim = out_dim
            
        self.decoder_body = nn.Sequential(*layers)
        
        # Reconstruction Heads
        # 1. Continuous Features
        self.head_cont = nn.Linear(in_dim, cont_dim)
        
        # 2. Categorical Features (one head per feature)
        self.heads_cat = nn.ModuleList()
        for card in cat_cardinalities:
            self.heads_cat.append(nn.Linear(in_dim, card))
            
    def forward(self, x_cont, x_cat):
        # Encode
        z = self.encoder(x_cont, x_cat)
        
        # Decode
        h = self.decoder_body(z)
        
        # Reconstruct
        rec_cont = self.head_cont(h)
        rec_cats = [head(h) for head in self.heads_cat]
        
        return rec_cont, rec_cats
