class GatedLinearUnit(nn.Module):
    """
    Gated Linear Unit (GLU) block.
    Consists of two parallel linear projections: a value branch and a gate branch.
    Output is the element-wise product of the value and the sigmoid-activated gate.
    Includes Batch Normalization and Dropout.
    """

    def __init__(self, in_features, out_features, dropout_rate):
        super(GatedLinearUnit, self).__init__()
        self.linear_val = nn.Linear(in_features, out_features)
        self.linear_gate = nn.Linear(in_features, out_features)
        self.bn = nn.BatchNorm1d(out_features)
        self.dropout = nn.Dropout(dropout_rate)

    def forward(self, x):
        # Value branch
        val = self.linear_val(x)
        # Gate branch
        gate = self.linear_gate(x)
        # Gating mechanism
        out = val * torch.sigmoid(gate)
        # Regularization
        out = self.bn(out)
        out = self.dropout(out)
        return out


class GatedFunnelMLP(nn.Module):
    """
    Gated Funnel MLP Architecture.
    Combines high-capacity embeddings for categorical features with continuous features.
    Processes the combined input through a funnel of Gated Linear Units.
    """

        super(GatedFunnelMLP, self).__init__()

            num_embeddings = vocab_sizes[col]
            self.embeddings.append(nn.Embedding(num_embeddings, self.emb_dim))

        input_dim = cont_dim + total_cat_dim

class MLPBlock(nn.Module):
    """
    Standard MLP Block: Linear -> BatchNorm -> ReLU -> Dropout
    Cite solution_lesson_node_00019: Simpler blocks often generalize better than GLU on this data.
    """

    def __init__(self, in_features, out_features, dropout_rate):
        super(MLPBlock, self).__init__()
        self.linear = nn.Linear(in_features, out_features)
        self.bn = nn.BatchNorm1d(out_features)
        self.activation = nn.ReLU()
        self.dropout = nn.Dropout(dropout_rate)

    def forward(self, x):
        x = self.linear(x)
        x = self.bn(x)
        x = self.activation(x)
        x = self.dropout(x)
        return x
