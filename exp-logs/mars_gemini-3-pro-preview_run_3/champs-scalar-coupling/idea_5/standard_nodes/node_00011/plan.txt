class AuxScaler:
    """
    Handles Standardization for Auxiliary Tasks (Shielding, Charges).
    """

    def __init__(self):
        self.device = Config.get_device()
        self.shield_mean = None
        self.shield_std = None
        self.charge_mean = None
        self.charge_std = None

    def fit(self):
        print("Computing auxiliary target statistics...")
        # Shielding
        try:
            df_s = pd.read_csv(Config.MAGNETIC_SHIELDING_CSV)
            s_cols = ["XX", "YX", "ZX", "XY", "YY", "ZY", "XZ", "YZ", "ZZ"]
            vals_s = df_s[s_cols].values
            self.shield_mean = torch.tensor(
                vals_s.mean(axis=0), dtype=torch.float32, device=self.device
            )
            self.shield_std = torch.tensor(
                vals_s.std(axis=0), dtype=torch.float32, device=self.device
            )
        except Exception as e:
            print(f"Warning: Could not fit shielding scaler: {e}")
            self.shield_mean = torch.zeros(9, device=self.device)
            self.shield_std = torch.ones(9, device=self.device)

        # Charges
        try:
            df_c = pd.read_csv(Config.MULLIKEN_CHARGES_CSV)
            vals_c = df_c["mulliken_charge"].values
            self.charge_mean = torch.tensor(
                vals_c.mean(), dtype=torch.float32, device=self.device
            )
            self.charge_std = torch.tensor(
                vals_c.std(), dtype=torch.float32, device=self.device
            )
        except Exception as e:
            print(f"Warning: Could not fit charge scaler: {e}")
            self.charge_mean = torch.tensor(0.0, device=self.device)
            self.charge_std = torch.tensor(1.0, device=self.device)

    def transform_shielding(self, y):
        return (y - self.shield_mean) / (self.shield_std + 1e-6)

    def transform_charge(self, y):
        return (y - self.charge_mean) / (self.charge_std + 1e-6)
