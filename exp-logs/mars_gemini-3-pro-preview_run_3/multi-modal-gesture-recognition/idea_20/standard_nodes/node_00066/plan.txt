The use of separable convolutions decoupled the channel mixing from temporal aggregation, which severely hurt the model's ability to model inter-class dependencies over time.

Additionally, I will reduce the `HIDDEN_DIM` from 96 back to 64 to avoid over-parameterization on this small dataset (Cite `solution_lesson_node_00053`). The architecture will remain an Iterative Refinement Network (MS-TCN style) with Gated Activations (Cite `solution_lesson_node_00032`), but with standard convolutional blocks to restore the receptive field quality and segmentation performance.

class SeparableGatedConvBlock(nn.Module):
    """
    A parameter-efficient dilated convolution block using Depthwise Separable Convolutions
    and Gated Activations (WaveNet style).
    Structure:
        Input -> Depthwise Conv (Dilated) -> Pointwise Conv (2x Channels)
              -> Split (Content, Gate) -> Tanh(Content) * Sigmoid(Gate)
              -> Dropout -> Residual + Input
    """

    def __init__(self, in_channels, out_channels, kernel_size, dilation, dropout):
        super(SeparableGatedConvBlock, self).__init__()

        # Padding to maintain sequence length (assuming odd kernel_size)
        padding = (kernel_size - 1) * dilation // 2

        # 1. Depthwise Convolution: Captures temporal context per channel
        # groups=in_channels makes it depthwise
        self.depthwise = nn.Conv1d(
            in_channels,
            in_channels,
            kernel_size,
            stride=1,
            padding=padding,
            dilation=dilation,
            groups=in_channels,
        )

        # 2. Pointwise Convolution: Projects to 2x channels for Gated Activation
        # Maps in_channels -> 2 * out_channels
        self.pointwise = nn.Conv1d(in_channels, 2 * out_channels, kernel_size=1)

        # Dropout
        self.dropout = nn.Dropout(dropout)

        # Residual connection handling
        self.residual_proj = None
        if in_channels != out_channels:
            self.residual_proj = nn.Conv1d(in_channels, out_channels, kernel_size=1)

    def forward(self, x):
        # x: (Batch, Channels, Time)
        residual = x

        # Depthwise
        out = self.depthwise(x)

        # Pointwise
        out = self.pointwise(out)

        # Gated Activation Unit
        # Split into Content (P) and Gate (Q)
        P, Q = out.chunk(2, dim=1)
        out = torch.tanh(P) * torch.sigmoid(Q)

        # Dropout
        out = self.dropout(out)

        # Residual Addition
        if self.residual_proj is not None:
            residual = self.residual_proj(residual)

        return out + residual

class GatedConvBlock(nn.Module):
    """
    Standard Dilated Convolution block with Gated Activations (WaveNet style).
    Replaces Depthwise Separable Convolutions to ensure proper channel mixing during
    temporal aggregation (Cite solution_lesson_node_00065).
    Structure:
        Input -> Dilated Conv (2x Channels) -> Split (Content, Gate)
              -> Tanh(Content) * Sigmoid(Gate) -> 1x1 Conv -> Dropout -> Residual + Input
    """

    def __init__(self, in_channels, out_channels, kernel_size, dilation, dropout):
        super(GatedConvBlock, self).__init__()

        # Padding to maintain sequence length
        padding = (kernel_size - 1) * dilation // 2

        # 1. Dilated Convolution: Maps in_channels -> 2 * out_channels
        # Standard convolution (groups=1) ensures channel mixing (Cite solution_lesson_node_00065)
        self.conv_dilated = nn.Conv1d(
            in_channels,
            2 * out_channels,
            kernel_size,
            stride=1,
            padding=padding,
            dilation=dilation,
        )

        # 2. 1x1 Convolution for output projection (after gating)
        self.conv_1x1 = nn.Conv1d(out_channels, out_channels, kernel_size=1)

        # Dropout
        self.dropout = nn.Dropout(dropout)

        # Residual connection handling
        self.residual_proj = None
        if in_channels != out_channels:
            self.residual_proj = nn.Conv1d(in_channels, out_channels, kernel_size=1)

    def forward(self, x):
        # x: (Batch, Channels, Time)
        residual = x

        # Dilated Conv
        out = self.conv_dilated(x)

        # Gated Activation Unit
        # Split into Content (P) and Gate (Q)
        P, Q = out.chunk(2, dim=1)
        out = torch.tanh(P) * torch.sigmoid(Q)

        # Output Projection
        out = self.conv_1x1(out)

        # Dropout
        out = self.dropout(out)

        # Residual Addition
        if self.residual_proj is not None:
            residual = self.residual_proj(residual)

        return out + residual
