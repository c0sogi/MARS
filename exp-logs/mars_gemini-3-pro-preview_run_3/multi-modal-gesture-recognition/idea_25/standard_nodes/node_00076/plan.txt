My implementation strategy focuses on rectifying the architectural and methodological flaws identified in the previous solution by applying lessons learned from earlier iterations.

1.  **Architectural Correction (Cite Lesson 00072)**: I will replace the `NRGSNet` (which used additive residual correction on logits) with `KC_IRN` (Kinematically-Consistent Iterative Refinement Network). This new architecture implements **Independent Probability Refinement**, where each refinement stage consumes the softmax probabilities of the previous stage and outputs fresh logits. This decouples the noise correction process and prevents the propagation of "jitter" seen in the residual approach. I will also reduce the TCN hidden channel size to 64 to avoid over-parameterization (Cite Lesson 00053).

2.  **Model Selection Optimization (Cite Lesson 00075)**: I will modify the training loop in `library/trainer.py` to perform **full-sequence validation** at the end of every epoch. Instead of relying on frame-wise validation loss (which correlates poorly with the task metric), early stopping and model checkpointing will now be driven directly by the **Levenshtein Error Rate**.

3.  **Regularization (Cite Lesson 00055)**: I will introduce `WEIGHT_DECAY = 1e-4` to the optimizer configuration to improve generalization on this small dataset.

4.  **Refactoring**: I will move the Levenshtein calculation logic to `library/inference.py` to share it between the trainer (for early stopping) and the final evaluation script.

class NRGSNet(nn.Module):
    """
    Normalized Residual Gated-Sawtooth Network.
    3-Stage Cascade:
    1. Gated Bi-GRU Encoder -> L1
    2. Normalized Residual TCN -> L2 = LogSoftmax(L1 + TCN(L1))
    3. Normalized Residual TCN -> L3 = LogSoftmax(L2 + TCN(L2))
    """

    def __init__(self):
        super(NRGSNet, self).__init__()

        # Calculate Input Dimension
        # Kinematics: 20 joints * 3 (Pos, Vel, Acc) * 3 coords = 180
        # Audio: 13 MFCC
        self.input_dim = (NUM_JOINTS * 3 * JOINTS_DIM) + N_MFCC

        # --- Stage 1: Gated High-Capacity Kinematic Encoder ---
        self.gated_input = GatedInput(self.input_dim)

        # Bi-GRU: Hidden size is per direction, so output is 2 * HIDDEN_SIZE
        self.gru = nn.GRU(
            input_size=self.input_dim,
            hidden_size=HIDDEN_SIZE,
            num_layers=2,
            batch_first=True,
            bidirectional=True,
            dropout=DROPOUT,
        )

        # Projection to classes
        self.stage1_fc = nn.Linear(HIDDEN_SIZE * 2, NUM_CLASSES)

        # --- Stage 2: Normalized Residual Sawtooth Refinement ---
        # Input: Log-probs (NUM_CLASSES channels)
        # Internal TCN width: 256 (same as GRU output capacity)
        self.stage2_tcn = SawtoothTCN(
            num_inputs=NUM_CLASSES,
            num_channels=HIDDEN_SIZE * 2,
            kernel_size=KERNEL_SIZE,
            dropout=DROPOUT,
        )

        # --- Stage 3: Independent Normalized Residual Refinement ---
        self.stage3_tcn = SawtoothTCN(
            num_inputs=NUM_CLASSES,
            num_channels=HIDDEN_SIZE * 2,
            kernel_size=KERNEL_SIZE,
            dropout=DROPOUT,
        )

    def forward(self, x):
        # x: (Batch, Time, Input_Dim)

        # --- Stage 1 ---
        # Gating
        x_gated = self.gated_input(x)

        # GRU Backbone
        gru_out, _ = self.gru(x_gated)

        # Initial Prediction
        logits_1 = self.stage1_fc(gru_out)  # (B, T, C)
        log_probs_1 = F.log_softmax(logits_1, dim=2)

        # --- Stage 2 ---
        # Permute for TCN: (B, T, C) -> (B, C, T)
        tcn_in_1 = log_probs_1.permute(0, 2, 1)

        # Predict additive correction to logits
        correction_1 = self.stage2_tcn(tcn_in_1)  # (B, C, T)

        # Normalized Residual: LogSoftmax(L1 + Correction)
        # We treat L1 as the base logits for the next stage
        logits_2 = tcn_in_1 + correction_1
        log_probs_2 = F.log_softmax(logits_2, dim=1)  # Softmax over channels

        # --- Stage 3 ---
        tcn_in_2 = log_probs_2  # (B, C, T)

        correction_2 = self.stage3_tcn(tcn_in_2)

        logits_3 = tcn_in_2 + correction_2
        log_probs_3 = F.log_softmax(logits_3, dim=1)

        # Permute back to (B, T, C) for output
        out_1 = log_probs_1
        out_2 = log_probs_2.permute(0, 2, 1)
        out_3 = log_probs_3.permute(0, 2, 1)

        return out_1, out_2, out_3

class KC_IRN(nn.Module):
    """
    Kinematically-Consistent Iterative Refinement Network.
    Implements Independent Probability Refinement (Cite Lesson 00072).
    3-Stage Cascade:
    1. Gated Bi-GRU Encoder -> Logits1
    2. TCN(Softmax(Logits1)) -> Logits2
    3. TCN(Softmax(Logits2)) -> Logits3
    """

    def __init__(self):
        super(KC_IRN, self).__init__()

        # Calculate Input Dimension
        self.input_dim = (NUM_JOINTS * 3 * JOINTS_DIM) + N_MFCC

        # --- Stage 1: Gated High-Capacity Kinematic Encoder ---
        self.gated_input = GatedInput(self.input_dim)

        self.gru = nn.GRU(
            input_size=self.input_dim,
            hidden_size=HIDDEN_SIZE,
            num_layers=2,
            batch_first=True,
            bidirectional=True,
            dropout=DROPOUT,
        )

        self.stage1_fc = nn.Linear(HIDDEN_SIZE * 2, NUM_CLASSES)

        # --- Stage 2: Independent Refinement ---
        # Input: Probabilities (NUM_CLASSES channels)
        # Output: New Logits (NUM_CLASSES channels)
        self.stage2_tcn = SawtoothTCN(
            num_inputs=NUM_CLASSES,
            num_channels=TCN_HIDDEN_CHANNELS,
            kernel_size=KERNEL_SIZE,
            dropout=DROPOUT,
        )

        # --- Stage 3: Independent Refinement ---
        self.stage3_tcn = SawtoothTCN(
            num_inputs=NUM_CLASSES,
            num_channels=TCN_HIDDEN_CHANNELS,
            kernel_size=KERNEL_SIZE,
            dropout=DROPOUT,
        )

    def forward(self, x):
        # x: (Batch, Time, Input_Dim)

        # --- Stage 1 ---
        x_gated = self.gated_input(x)
        gru_out, _ = self.gru(x_gated)
        logits_1 = self.stage1_fc(gru_out)  # (B, T, C)

        # --- Stage 2 ---
        # Pass Softmax Probabilities (Cite Lesson 00072)
        probs_1 = F.softmax(logits_1, dim=2)
        tcn_in_1 = probs_1.permute(0, 2, 1)  # (B, C, T)
        
        logits_2_tcn = self.stage2_tcn(tcn_in_1) # (B, C, T)
        logits_2 = logits_2_tcn.permute(0, 2, 1) # (B, T, C)

        # --- Stage 3 ---
        probs_2 = F.softmax(logits_2, dim=2)
        tcn_in_2 = probs_2.permute(0, 2, 1)
        
        logits_3_tcn = self.stage3_tcn(tcn_in_2)
        logits_3 = logits_3_tcn.permute(0, 2, 1)

        # Return LogSoftmax for NLLLoss
        return (
            F.log_softmax(logits_1, dim=2),
            F.log_softmax(logits_2, dim=2),
            F.log_softmax(logits_3, dim=2)
        )

def validate_epoch(model, dataloader, criterion, device):
    model.eval()
    running_loss = 0.0
    correct_preds = 0
    total_preds = 0

    with torch.no_grad():
        for inputs, labels in dataloader:
            inputs = inputs.to(device)
            labels = labels.to(device)

            outputs = model(inputs)
            loss = criterion(outputs, labels)

            running_loss += loss.item() * inputs.size(0)

            final_output = outputs[-1]
            _, predicted = torch.max(final_output, 2)

            correct_preds += (predicted == labels).sum().item()
            total_preds += labels.numel()

    epoch_loss = running_loss / len(dataloader.dataset)
    epoch_acc = correct_preds / total_preds
    return epoch_loss, epoch_acc

def validate_full_sequences(model, device):

    Performs validation on full sequences and computes Levenshtein Error Rate.

    """
    model.eval()

        gt_labels = []

                pass

        skeleton = data_utils.load_robust_mat(data_path, load_cached_data=True)

            total_gestures += len(gt_labels)
            continue

        audio = data_utils.extract_audio_mfcc(audio_path, num_frames, load_cached_data=True)

        total_distance += dist
        total_gestures += len(gt_labels)
