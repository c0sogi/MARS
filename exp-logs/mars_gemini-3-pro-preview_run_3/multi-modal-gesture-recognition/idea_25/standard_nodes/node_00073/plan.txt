To optimize the solution, I will implement a **Kinematically-Consistent Iterative Refinement Network (KC-IRN)**, reverting the architectural changes that caused the performance degradation in the previous submission. Specifically, I will:

1.  **Architecture**: Replace the "Normalized Residual Gated-Sawtooth Network" with **KC-IRN**. This involves changing the refinement stages to accept **normalized probabilities** (Softmax output) from the previous stage as input features, rather than learning additive corrections to logits. This directly addresses **Lesson 00072** (Independent Probability Refinement vs. Additive Logit Correction) and **Lesson 00025** (Iterative Refinement).
2.  **Regularization**: Introduce **Weight Decay (1e-4)** in the optimizer configuration. **Lesson 00055** highlights that weight decay is critical for MS-TCN architectures on this dataset to prevent overfitting, which was observed in the previous run.
3.  **Refinement Independence**: Ensure each refinement stage has independent parameters (already present) and operates on the probability distribution of the previous stage to "clean" the segmentation, acting as a learned smoothing filter.

These changes target the massive over-segmentation and overfitting observed in the previous failure analysis.

class NRGSNet(nn.Module):
    """
    Normalized Residual Gated-Sawtooth Network.
    3-Stage Cascade:
    1. Gated Bi-GRU Encoder -> L1
    2. Normalized Residual TCN -> L2 = LogSoftmax(L1 + TCN(L1))
    3. Normalized Residual TCN -> L3 = LogSoftmax(L2 + TCN(L2))
    """

    def __init__(self):
        super(NRGSNet, self).__init__()

        # Calculate Input Dimension
        # Kinematics: 20 joints * 3 (Pos, Vel, Acc) * 3 coords = 180
        # Audio: 13 MFCC
        self.input_dim = (NUM_JOINTS * 3 * JOINTS_DIM) + N_MFCC

        # --- Stage 1: Gated High-Capacity Kinematic Encoder ---
        self.gated_input = GatedInput(self.input_dim)

        # Bi-GRU: Hidden size is per direction, so output is 2 * HIDDEN_SIZE
        self.gru = nn.GRU(
            input_size=self.input_dim,
            hidden_size=HIDDEN_SIZE,
            num_layers=2,
            batch_first=True,
            bidirectional=True,
            dropout=DROPOUT,
        )

        # Projection to classes
        self.stage1_fc = nn.Linear(HIDDEN_SIZE * 2, NUM_CLASSES)

        # --- Stage 2: Normalized Residual Sawtooth Refinement ---
        # Input: Log-probs (NUM_CLASSES channels)
        # Internal TCN width: 256 (same as GRU output capacity)
        self.stage2_tcn = SawtoothTCN(
            num_inputs=NUM_CLASSES,
            num_channels=HIDDEN_SIZE * 2,
            kernel_size=KERNEL_SIZE,
            dropout=DROPOUT,
        )

        # --- Stage 3: Independent Normalized Residual Refinement ---
        self.stage3_tcn = SawtoothTCN(
            num_inputs=NUM_CLASSES,
            num_channels=HIDDEN_SIZE * 2,
            kernel_size=KERNEL_SIZE,
            dropout=DROPOUT,
        )

    def forward(self, x):
        # x: (Batch, Time, Input_Dim)

        # --- Stage 1 ---
        # Gating
        x_gated = self.gated_input(x)

        # GRU Backbone
        gru_out, _ = self.gru(x_gated)

        logits_1 = self.stage1_fc(gru_out)  # (B, T, C)
        log_probs_1 = F.log_softmax(logits_1, dim=2)

        out_2 = log_probs_2.permute(0, 2, 1)
        out_3 = log_probs_3.permute(0, 2, 1)

    Kinematically-Consistent Iterative Refinement Network (KC-IRN).
    Implements Independent Probability Refinement (Cite solution_lesson_node_00072).
    3-Stage Cascade:
    1. Gated Bi-GRU Encoder -> Probabilities 1
    2. TCN(Probabilities 1) -> Probabilities 2
    3. TCN(Probabilities 2) -> Probabilities 3
    """

    def __init__(self):
        super(KC_IRN, self).__init__()

        # Calculate Input Dimension
        # Kinematics: 20 joints * 3 (Pos, Vel, Acc) * 3 coords = 180
        # Audio: 13 MFCC

            num_channels=HIDDEN_SIZE * 2,
            kernel_size=KERNEL_SIZE,
            dropout=DROPOUT,
        )

        # --- Stage 3: Independent Refinement ---
        self.stage3_tcn = SawtoothTCN(
            num_inputs=NUM_CLASSES,
            num_channels=HIDDEN_SIZE * 2,
            kernel_size=KERNEL_SIZE,
            dropout=DROPOUT,
        )

    def forward(self, x):

        # --- Stage 1 ---
        # Gating
        x_gated = self.gated_input(x)

        # GRU Backbone
        gru_out, _ = self.gru(x_gated)

        # Initial Prediction
        logits_1 = self.stage1_fc(gru_out)  # (B, T, C)
        log_probs_1 = F.log_softmax(logits_1, dim=2)

        # --- Stage 2 ---

        probs_1 = torch.exp(log_probs_1)

        logits_2 = self.stage2_tcn(tcn_in_1)  # (B, C, T)

        # --- Stage 3 ---
        probs_2 = torch.exp(log_probs_2)
        tcn_in_2 = probs_2  # (B, C, T)

        logits_3 = self.stage3_tcn(tcn_in_2)
        log_probs_3 = F.log_softmax(logits_3, dim=1)

        # Permute back to (B, T, C) for output
        out_1 = log_probs_1
        out_2 = log_probs_2.permute(0, 2, 1)
        out_3 = log_probs_3.permute(0, 2, 1)

        return out_1, out_2, out_3
