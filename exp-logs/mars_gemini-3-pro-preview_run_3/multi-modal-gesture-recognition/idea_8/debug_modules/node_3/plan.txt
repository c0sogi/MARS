import os
import random
import numpy as np
import torch
from itertools import groupby
from nltk import edit_distance

def set_seed(seed=42):
    """Sets the random seed for reproducibility."""
    random.seed(seed)
    np.random.seed(seed)
    torch.manual_seed(seed)
    torch.cuda.manual_seed_all(seed)
    torch.backends.cudnn.deterministic = True
    torch.backends.cudnn.benchmark = False
    os.environ["PYTHONHASHSEED"] = str(seed)

def run_length_encoding(preds):
    """
    Converts frame-wise predictions to a sequence of gesture IDs.
    Collapses consecutive duplicates and removes background class (0).
    """
    # Collapse consecutive duplicates
    collapsed = [k for k, g in groupby(preds)]
    # Remove background (0)
    sequence = [x for x in collapsed if x != 0]
    return sequence

def compute_levenshtein_score(predictions, ground_truth):
    """
    Computes the Levenshtein score (error rate).
    Sum of distances / Total number of ground truth gestures.
    """
    total_dist = 0
    total_len = 0
    for key, pred_seq in predictions.items():
        gt_seq = ground_truth.get(key, [])
        # edit_distance from nltk calculates Levenshtein distance
        dist = edit_distance(pred_seq, gt_seq)
        total_dist += dist
        total_len += len(gt_seq)
    
    if total_len == 0:
        return 0.0
    return total_dist / total_len

def save_submission(sequences, path):
    """
    Saves predictions to a CSV file in the format: SessionID,label1,label2,...
    """
    with open(path, "w") as f:
        for sid, seq in sequences.items():
            line = str(sid)
            if seq:
                # Cite debug_lesson_5: Handle empty collections to avoid trailing commas
                line += "," + ",".join(map(str, seq))
            f.write(line + "\n")
