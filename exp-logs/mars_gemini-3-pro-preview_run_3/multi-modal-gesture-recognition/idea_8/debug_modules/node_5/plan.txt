import os
import random
import numpy as np
import torch

def set_seed(seed=42):
    random.seed(seed)
    np.random.seed(seed)
    torch.manual_seed(seed)
    if torch.cuda.is_available():
        torch.cuda.manual_seed_all(seed)
    torch.backends.cudnn.deterministic = True
    torch.backends.cudnn.benchmark = False
    os.environ["PYTHONHASHSEED"] = str(seed)

def run_length_encoding(frame_preds):
    """
    Collapses consecutive duplicate labels and removes background (0).
    """
    sequence = []
    if len(frame_preds) == 0:
        return sequence
    
    prev = frame_preds[0]
    if prev != 0:
        sequence.append(int(prev))
    
    for curr in frame_preds[1:]:
        if curr != prev:
            if curr != 0:
                sequence.append(int(curr))
            prev = curr
            
    return sequence

def _levenshtein(seq1, seq2):
    size_x = len(seq1) + 1
    size_y = len(seq2) + 1
    matrix = np.zeros((size_x, size_y))
    for x in range(size_x):
        matrix[x, 0] = x
    for y in range(size_y):
        matrix[0, y] = y
    for x in range(1, size_x):
        for y in range(1, size_y):
            if seq1[x-1] == seq2[y-1]:
                matrix[x,y] = matrix[x-1, y-1]
            else:
                matrix[x,y] = min(
                    matrix[x-1,y] + 1,
                    matrix[x-1,y-1] + 1,
                    matrix[x,y-1] + 1
                )
    return matrix[size_x-1, size_y-1]

def compute_levenshtein_score(predictions, ground_truth):
    total_dist = 0
    total_len = 0
    for sid, gt in ground_truth.items():
        pred = predictions.get(sid, [])
        total_dist += _levenshtein(pred, gt)
        total_len += len(gt)
    
    return total_dist / total_len if total_len > 0 else 0.0
