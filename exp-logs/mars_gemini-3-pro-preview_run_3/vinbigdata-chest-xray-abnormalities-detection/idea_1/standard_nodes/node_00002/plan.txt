def get_yolo_bbox(row, img_w, img_h):
    """

    """
    # Original coordinates

    ymin = row["y_min"]
    xmax = row["x_max"]
    ymax = row["y_max"]

    # Calculate center and dimensions
    box_w = xmax - xmin
    box_h = ymax - ymin
