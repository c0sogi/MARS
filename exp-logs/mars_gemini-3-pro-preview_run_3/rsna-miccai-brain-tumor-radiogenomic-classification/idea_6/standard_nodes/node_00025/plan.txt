Based on the provided lessons, the previous "DualStreamSiameseNet" architecture (Siamese Late Fusion) significantly underperformed compared to "Early Fusion" approaches (Cite solution_lesson_node_00018). Additionally, using a higher slice count (32) with splitting introduced overfitting and broke spatial continuity (Cite solution_lesson_node_00013, solution_lesson_node_00016).

I will optimize the solution by:
1.  **Reverting to Early Fusion**: Replacing the Siamese network with a single 2.5D EfficientNet-B0 that accepts a stacked volume as input.
2.  **Optimizing Slice Count**: Reducing the slice count to 16 uniformly sampled slices to balance information capture with regularization, matching the "Current Best" configuration (Cite solution_lesson_node_00003, solution_lesson_node_00013).
3.  **Data Processing**: Updating the data pipeline to produce a single 64-channel input tensor (16 slices × 4 modalities) instead of splitting into even/odd streams, preserving global volumetric normalization (Cite solution_lesson_node_00015).

class DualStreamSiameseNet(nn.Module):
    """
    A Siamese 2.5D Convolutional Neural Network for MGMT promoter methylation prediction.

    Architecture:
    - Backbone: EfficientNet-B0 (Shared weights)
    - Input: Two streams (Even slices, Odd slices), each with 64 channels.
    - Fusion: Late fusion via concatenation of Global Average Pooled features.
    - Head: Single linear layer for binary classification.
    """

    def __init__(self):
        super(DualStreamSiameseNet, self).__init__()

        # Initialize the shared backbone
        # We use EfficientNet-B0 pre-trained on ImageNet.
        # in_chans=64 adapts the first layer to accept our 16 slices * 4 modalities input.
        # num_classes=0 removes the classification head, returning the pooled feature vector.
        self.backbone = timm.create_model(
            "efficientnet_b0",
            pretrained=True,
            in_chans=Config.INPUT_CHANNELS,
            num_classes=0,
            global_pool="avg",
        )

        # Get the number of output features from the backbone (e.g., 1280 for EfficientNet-B0)
        num_features = self.backbone.num_features

        # Fusion Head
        # We concatenate the feature vectors from both streams, so input dim is * 2.
        # Output is 1 logit (for BCEWithLogitsLoss).
        self.fc = nn.Linear(num_features * 2, 1)

    def forward(self, x_even, x_odd):
        """
        Forward pass for the Siamese Network.

        Args:
            x_even (torch.Tensor): Input tensor for Even stream. Shape (B, 64, H, W).
            x_odd (torch.Tensor): Input tensor for Odd stream. Shape (B, 64, H, W).

        Returns:
            torch.Tensor: Logits of shape (B, 1).
        """
        # Pass both streams through the shared backbone
        # Output shape: (Batch_Size, num_features)
        feat_even = self.backbone(x_even)
        feat_odd = self.backbone(x_odd)

        # Late Fusion: Concatenate features
        # Output shape: (Batch_Size, num_features * 2)
        combined = torch.cat([feat_even, feat_odd], dim=1)

        # Final Classification
        # Output shape: (Batch_Size, 1)
        logits = self.fc(combined)

        return logits

class BraTSDataset(Dataset):
    def __init__(self, X, y=None, ids=None):
        """
        Args:
            X (np.ndarray): Input data of shape (N, 2, 64, 256, 256).
                            Dimension 1 contains [Even_Stream, Odd_Stream].
            y (np.ndarray, optional): Labels of shape (N,).
            ids (np.ndarray, optional): BraTS21IDs of shape (N,).
        """
        self.X = X
        self.y = y
        self.ids = ids

    def __len__(self):
        return len(self.X)

    def __getitem__(self, idx):
        # X[idx] is (2, 64, 256, 256)
        # 0 -> Even Stream, 1 -> Odd Stream
        even_stream = self.X[idx, 0]
        odd_stream = self.X[idx, 1]

        # Convert to torch tensors
        even_tensor = torch.from_numpy(even_stream).float()
        odd_tensor = torch.from_numpy(odd_stream).float()

        if self.y is not None:
            label = torch.tensor(self.y[idx], dtype=torch.float32)
            return even_tensor, odd_tensor, label
        else:
            # For inference, return the ID to map predictions
            patient_id = self.ids[idx] if self.ids is not None else ""
            return even_tensor, odd_tensor, patient_id


def process_patient(row, img_size, num_slices):
    """
    Processes a single patient's data:
    1. Loads full volumes for all 4 modalities.
    2. Normalizes each modality globally.
    3. Samples 32 slices uniformly.
    4. Interleaves into Even/Odd streams.
    """
    modalities = ["flair", "t1w", "t1wce", "t2w"]
    patient_slices = []

    for mod in modalities:
        paths = row.get(f"{mod}_paths", [])

        # Load all available slices for the modality
        volume_imgs = []
        if paths is not None and len(paths) > 0:
            for p in paths:
                img = read_dicom_file(p)
                volume_imgs.append(img)

        # Handle missing data or empty folders
        if len(volume_imgs) == 0:
            # Create a dummy volume with 1 slice to allow code to proceed
            volume = np.zeros((1, img_size, img_size), dtype=np.float32)
        else:
            volume = np.stack(volume_imgs)  # (D, H, W)

        # Global Volumetric Normalization (per modality)
        # This preserves inter-slice contrast
        volume = normalize_min_max(volume)

        # High-Density Uniform Sampling (10%-90% depth)
        indices = get_depth_indices(len(volume), num_slices)
        sampled_volume = volume[indices]  # (32, H, W)

        patient_slices.append(sampled_volume)

    # Stack modalities: (4, 32, H, W) -> Transpose to (32, 4, H, W)
    # We organize by slice index first to facilitate even/odd splitting
    stack = np.stack(patient_slices, axis=1)  # (32, 4, H, W)

    # Deterministic Interleaving
    # Even Indices: 0, 2, ..., 30
    # Odd Indices: 1, 3, ..., 31
    even_indices = np.arange(0, num_slices, 2)
    odd_indices = np.arange(1, num_slices, 2)

    even_vol = stack[even_indices]  # (16, 4, H, W)
    odd_vol = stack[odd_indices]  # (16, 4, H, W)

    # Flatten to Channels: (16, 4, H, W) -> (64, H, W)
    # We use reshape to merge the slice and modality dimensions
    even_stream = even_vol.reshape(-1, img_size, img_size)
    odd_stream = odd_vol.reshape(-1, img_size, img_size)

    # Stack streams: (2, 64, H, W)
    return np.stack([even_stream, odd_stream])

def train_epoch(model, loader, optimizer, criterion, device):
    """
    Trains the model for one epoch.

    Args:
        model: The PyTorch model.
        loader: DataLoader for training data.
        optimizer: The optimizer.
        criterion: The loss function.
        device: The device to run on (cpu or cuda).

    Returns:
        float: Average training loss for the epoch.
    """
    model.train()
    running_loss = 0.0

    for batch_idx, (x_even, x_odd, labels) in enumerate(loader):
        x_even = x_even.to(device)
        x_odd = x_odd.to(device)
        labels = labels.to(device).unsqueeze(1)  # (B, 1)

        optimizer.zero_grad()

        outputs = model(x_even, x_odd)
        loss = criterion(outputs, labels)

        loss.backward()
        optimizer.step()

        running_loss += loss.item() * x_even.size(0)

    epoch_loss = running_loss / len(loader.dataset)
    return epoch_loss
