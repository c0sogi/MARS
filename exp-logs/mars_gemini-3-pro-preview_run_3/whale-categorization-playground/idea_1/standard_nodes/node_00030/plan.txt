Implementation Strategy:
1.  **Architecture Upgrade (Cite Lesson 00010, 00029)**: Replace the DenseNet121 backbone with **EfficientNet-B2**, which has shown superior performance in fine-grained tasks due to better parameter efficiency and attention mechanisms. Implement a **Linear -> BatchNorm** neck (without activation) to preserve the embedding manifold structure.
2.  **Loss Function Transition (Cite Lesson 00008)**: Switch from Triplet Margin Loss to **ArcFace (ArcMarginProduct)**. ArcFace optimizes samples against learnable class prototypes on a hypersphere, which is more stable and sample-efficient than mining triplets in high-cardinality, few-shot datasets.
3.  **Optimization Tuning (Cite Lesson 00019, 00028)**: Set the learning rate to **3e-4** and batch size to **16**. This balance ensures sufficient gradient updates per epoch to minimize the difficult angular margin loss without destabilizing the sensitive EfficientNet backbone.
4.  **Data Handling**: Convert the training dataset from a Triplet sampler to a standard Classification sampler (`WhaleArcFaceDataset`) that returns `(image, label)`. Filter out `new_whale` during training to focus on learning robust features for known identities.
5.  **Inference Adjustment**: Update the model loading logic to `strict=False` during inference, allowing the backbone to be loaded without the training-specific ArcFace head.

        # DenseNet uses 'classifier' instead of 'fc'

import math
import torch
import torch.nn as nn
import torch.nn.functional as F
from torchvision import models
from library.config import Config


class ArcMarginProduct(nn.Module):
    """
    ArcFace output layer.
    Cite solution_lesson_node_00008
    """
    def __init__(self, in_features, out_features, s=30.0, m=0.50, easy_margin=False):
        super(ArcMarginProduct, self).__init__()
        self.in_features = in_features
        self.out_features = out_features
        self.s = s
        self.m = m
        self.weight = nn.Parameter(torch.FloatTensor(out_features, in_features))
        nn.init.xavier_uniform_(self.weight)

        self.cos_m = math.cos(m)
        self.sin_m = math.sin(m)
        self.th = math.cos(math.pi - m)
        self.mm = math.sin(math.pi - m) * m
        self.easy_margin = easy_margin

    def forward(self, input, label):
        # --------------------------- cos(theta) & phi(theta) ---------------------------
        cosine = F.linear(F.normalize(input), F.normalize(self.weight))
        sine = torch.sqrt((1.0 - torch.pow(cosine, 2)).clamp(0, 1))
        phi = cosine * self.cos_m - sine * self.sin_m
        if self.easy_margin:
            phi = torch.where(cosine > 0, phi, cosine)
        else:
            phi = torch.where(cosine > self.th, phi, cosine - self.mm)
        # --------------------------- convert label to one-hot ---------------------------
        one_hot = torch.zeros(cosine.size(), device=input.device)
        one_hot.scatter_(1, label.view(-1, 1).long(), 1)
        # -------------torch.where(out_i = {x_i if condition_i else y_i}) -------------
        output = (one_hot * phi) + ((1.0 - one_hot) * cosine)
        output *= self.s
        return output


class WhaleEmbeddingNet(nn.Module):
    """
    EfficientNet-B2 backbone with ArcFace Head.
    Cite solution_lesson_node_00010: EfficientNet-B2 with Attention.
    Cite solution_lesson_node_00008: ArcFace.
    """

    def __init__(self, num_classes=None, embedding_dim=None):
        super(WhaleEmbeddingNet, self).__init__()

        if embedding_dim is None:
            embedding_dim = Config.EMBEDDING_DIM

        # Load pre-trained EfficientNet-B2
        self.backbone = models.efficientnet_b2(weights="DEFAULT")
        
        # Remove the classifier
        self.backbone.classifier = nn.Identity()
        
        # EfficientNet-B2 last channel size is 1408
        num_ftrs = 1408 
        
        # Pooling
        self.pooling = nn.AdaptiveAvgPool2d(1)

        # Neck: BN -> Dropout -> Linear -> BN (Cite solution_lesson_node_00029)
        self.bn1 = nn.BatchNorm1d(num_ftrs)
        self.dropout = nn.Dropout(0.5)
        self.fc = nn.Linear(num_ftrs, embedding_dim)
        self.bn2 = nn.BatchNorm1d(embedding_dim)

        # ArcFace Head
        if num_classes is not None:
            self.arcface = ArcMarginProduct(
                embedding_dim, 
                num_classes, 
                s=Config.ARC_S, 
                m=Config.ARC_M
            )
        else:
            self.arcface = None

    def forward(self, x, labels=None):
        # Backbone features
        x = self.backbone.features(x)
        x = self.pooling(x)
        x = torch.flatten(x, 1)
        
        # Neck
        x = self.bn1(x)
        x = self.dropout(x)
        x = self.fc(x)
        features = self.bn2(x)
        
        # If training with labels, return ArcFace logits
        if labels is not None and self.arcface is not None:
            return self.arcface(features, labels)
        
        # Otherwise return normalized embeddings for inference
        return F.normalize(features, p=2, dim=1)

    def get_embedding(self, x):
        return self.forward(x, labels=None)

class WhaleTripletsDataset(Dataset):
    """
    Dataset for training a Triplet Network.
    Generates triplets of images: (Anchor, Positive, Negative)
    """

    def __init__(self, csv_file, subset_size=None, transform=None):
        self.df = pd.read_csv(csv_file)
        self.transform = transform

        # 1. Identify Anchors: Must be known whales (not new_whale)
        self.known_whales = self.df[self.df["Id"] != "new_whale"].reset_index(drop=True)

        if subset_size is not None:
            self.known_whales = self.known_whales.head(subset_size)

        # 2. Group all data by Id for efficient sampling
        self.all_groups = self.df.groupby("Id")["file_path"].apply(list).to_dict()
        self.all_ids = list(self.all_groups.keys())

    def __len__(self):
        return len(self.known_whales)

    def __getitem__(self, idx):
        # 1. Anchor
        row = self.known_whales.iloc[idx]
        anchor_id = row["Id"]
        anchor_path = os.path.join(Config.INPUT_DIR, row["file_path"])

        # 2. Positive (Same ID)
        pos_candidates = self.all_groups[anchor_id]
        pos_path_rel = np.random.choice(pos_candidates)
        pos_path = os.path.join(Config.INPUT_DIR, pos_path_rel)

        # 3. Negative (Different ID)
        while True:
            neg_id = np.random.choice(self.all_ids)
            if neg_id != anchor_id:
                break
        neg_candidates = self.all_groups[neg_id]
        neg_path_rel = np.random.choice(neg_candidates)
        neg_path = os.path.join(Config.INPUT_DIR, neg_path_rel)

        # Load Images
        anchor_img = load_and_preprocess_image(
            anchor_path, height=Config.IMG_HEIGHT, width=Config.IMG_WIDTH
        )
        pos_img = load_and_preprocess_image(
            pos_path, height=Config.IMG_HEIGHT, width=Config.IMG_WIDTH
        )
        neg_img = load_and_preprocess_image(
            neg_path, height=Config.IMG_HEIGHT, width=Config.IMG_WIDTH
        )

        # Apply Transforms
        if self.transform:
            anchor_img = self.transform(anchor_img)
            pos_img = self.transform(pos_img)
            neg_img = self.transform(neg_img)

        return anchor_img, pos_img, neg_img

def train_one_epoch(model, dataloader, criterion, optimizer, device):
    """
    Trains the model for one epoch using Triplet Loss.
    """
    model.train()
    running_loss = 0.0
    dataset_size = len(dataloader.dataset)

    for anchor, pos, neg in dataloader:
        anchor = anchor.to(device)
        pos = pos.to(device)
        neg = neg.to(device)

        optimizer.zero_grad()

        # Forward pass
        emb_a = model(anchor)
        emb_p = model(pos)
        emb_n = model(neg)

        loss = criterion(emb_a, emb_p, emb_n)

        loss.backward()
        optimizer.step()

        running_loss += loss.item() * anchor.size(0)

    epoch_loss = running_loss / dataset_size
    return epoch_loss
