import torch
import torch.nn as nn
import torch.nn.functional as F

class FocalLoss(nn.Module):
    """
    Focal Loss for addressing class imbalance.
    FL(p_t) = -alpha * (1 - p_t)^gamma * log(p_t)
    
    This implementation assumes alpha is handled implicitly or not used (set to 1),
    focusing primarily on the hard example mining property via gamma.
    """
    def __init__(self, gamma=2.0, reduction='mean'):
        super(FocalLoss, self).__init__()
        self.gamma = gamma
        self.reduction = reduction

    def forward(self, inputs, targets):
        """
        Args:
            inputs: Logits of shape (Batch, Num_Classes)
            targets: Ground truth labels of shape (Batch)
        """
        # Calculate standard Cross Entropy Loss (log(p_t))
        # reduction='none' so we can apply the focal weight per sample
        ce_loss = F.cross_entropy(inputs, targets, reduction='none')
        
        # p_t = exp(-ce_loss)
        pt = torch.exp(-ce_loss)
        
        # Focal term: (1 - p_t)^gamma
        focal_term = (1 - pt) ** self.gamma
        
        # Final Focal Loss
        loss = focal_term * ce_loss

        if self.reduction == 'mean':
            return loss.mean()
        elif self.reduction == 'sum':
            return loss.sum()
        else:
            return loss
