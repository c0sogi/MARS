class AttentionAugmentedResBiGRU(nn.Module):
    """
    Deep Residual BiGRU with a Convolutional Stem and Self-Attention Refinement.

    Architecture:
    1. Input (N, 107, 14)
    2. Conv1d Stem -> Projects to dense features
    3. Stack of Residual BiGRU layers -> Captures sequential context
    4. Transformer Encoder (Self-Attention) -> Refines global context
    5. Linear Head -> Predicts 5 targets
    """

    def __init__(self):
        super(AttentionAugmentedResBiGRU, self).__init__()

        # ------------------------------------------------------------------
        # 1. Convolutional Stem
        # ------------------------------------------------------------------
        # Input: (N, 14, 107) after permute
        # Output: (N, STEM_FILTERS, 107)
        self.stem = nn.Sequential(
            nn.Conv1d(
                in_channels=Config.INPUT_DIM,
                out_channels=Config.STEM_FILTERS,
                kernel_size=Config.STEM_KERNEL_SIZE,
                padding=Config.STEM_KERNEL_SIZE // 2,  # Maintain sequence length
            ),
            nn.GELU(),
            nn.Dropout(0.1),
        )

        # ------------------------------------------------------------------
        # 2. Deep Residual BiGRU Backbone
        # ------------------------------------------------------------------
        self.rnn_layers = nn.ModuleList()

        # First layer input dimension is the stem output dimension
        input_dim = Config.STEM_FILTERS

        for _ in range(Config.RNN_LAYERS):
            self.rnn_layers.append(
                ResidualBiGRU(
                    input_dim=input_dim,
                    hidden_dim=Config.RNN_HIDDEN_DIM,
                    dropout=Config.RNN_DROPOUT,
                )
            )
            # Subsequent layers take the output of the previous BiGRU (2 * hidden)
            input_dim = Config.RNN_HIDDEN_DIM * 2

        # ------------------------------------------------------------------
        # 3. Self-Attention Refinement
        # ------------------------------------------------------------------
        # The input to attention is the output of the last BiGRU layer
        rnn_output_dim = Config.RNN_HIDDEN_DIM * 2

        encoder_layer = nn.TransformerEncoderLayer(
            d_model=rnn_output_dim,
            nhead=Config.ATTN_HEADS,
            dim_feedforward=rnn_output_dim * 2,  # Standard expansion
            dropout=Config.ATTN_DROPOUT,
            activation="gelu",
            batch_first=True,
        )

        self.attention = nn.TransformerEncoder(
            encoder_layer, num_layers=Config.ATTN_LAYERS
        )

        # ------------------------------------------------------------------
        # 4. Output Head
        # ------------------------------------------------------------------
        self.head = nn.Linear(rnn_output_dim, Config.OUTPUT_DIM)

    def forward(self, x):
        """
        Args:
            x (torch.Tensor): Input tensor of shape (Batch, Seq_Len, Input_Dim)
                              e.g., (N, 107, 14)
        Returns:
            torch.Tensor: Output tensor of shape (Batch, Seq_Len, Output_Dim)
                          e.g., (N, 107, 5)
        """
        # 1. Stem (Conv1d expects Channel first)
        # x: (N, L, C) -> (N, C, L)
        x = x.transpose(1, 2)
        x = self.stem(x)
        # x: (N, C, L) -> (N, L, C)
        x = x.transpose(1, 2)

        # 2. Residual BiGRU Backbone
        for layer in self.rnn_layers:
            x = layer(x)

        # 3. Self-Attention Refinement
        x = self.attention(x)

        # 4. Head
        out = self.head(x)

        return out
