def build_augmented_features(df):
    """
    Constructs the spatially augmented feature tensor for the entire dataframe.

    Args:
        df (pd.DataFrame): Dataframe containing 'sequence', 'structure', 'predicted_loop_type'.

    Returns:
        np.array: Shape (N_samples, seq_len, 28).
    """
    n_samples = len(df)
    seq_len = Config.SEQ_LENGTH
    input_dim = 14  # Base dimension

    # Final container
    # Augmented dim = Base(14) + Paired_Base(14) = 28
    X = np.zeros((n_samples, seq_len, input_dim * 2), dtype=np.float32)

    sequences = df["sequence"].values
    structures = df["structure"].values
    loop_types = df["predicted_loop_type"].values

    for idx in range(n_samples):
        seq = sequences[idx]
        struct = structures[idx]
        loop = loop_types[idx]

        # 1. Get Base Features (L, 14)
        base_feats = encode_sequence_features(seq, struct, loop)

        # 2. Get Pairing Map
        pairs = get_structure_pairs(struct)

        # 3. Construct Paired Features
        pair_feats = np.zeros_like(base_feats)

        # Vectorized gathering of paired features
        # Identify indices that are paired
        paired_indices = np.where(pairs != -1)[0]
        partner_indices = pairs[paired_indices]

        # Assign features: pair_feats[i] = base_feats[partner_indices[i]]
        if len(paired_indices) > 0:
            pair_feats[paired_indices] = base_feats[partner_indices]

        # 4. Concatenate [Base, Paired] -> (L, 28)
        X[idx] = np.concatenate([base_feats, pair_feats], axis=1)

    return X
