To optimize the solution, I will implement a **Convolutional Recurrent Neural Network (CRNN)** architecture with **Learnable Embeddings**. This approach improves upon the previous solution by:

1.  **Using Embeddings**: Replacing one-hot encoding with dense vector embeddings allows the model to learn semantic similarities between nucleotide types and structural elements.
2.  **Adding 1D Convolution**: A convolutional layer before the GRU acts as a feature extractor for local motifs (e.g., specific k-mer patterns), which are critical in RNA biology. This effectively utilizes the "Locality" inductive bias (Cite solution_lesson_node_00002).
3.  **Preserving Sequence Dimension**: The architecture continues to process the sequence without flattening, maintaining the "Translation Invariance" bias (Cite solution_lesson_node_00002).

I will modify the configuration to support integer-based indexing for embeddings, update the dataset processing to produce these indices, and redesign the model architecture.

class RNA_GRU(nn.Module):
    """
    CRNN (Convolutional Recurrent Neural Network) with Embeddings.
    Cite solution_lesson_node_00002: Preserves sequence dimension and locality.
    Uses Embeddings for categorical inputs and Conv1d for local motif extraction.
    """

    def __init__(
        self,
        hidden_dim=Config.RNN_HIDDEN_DIM,
        num_layers=Config.RNN_LAYERS,
        dropout=Config.RNN_DROPOUT,
        num_targets=Config.NUM_TARGETS,
    ):
        super(RNA_GRU, self).__init__()

        # Embeddings
        self.emb_seq = nn.Embedding(Config.VOCAB_SIZE_SEQ, Config.EMBED_DIM_SEQ)
        self.emb_struct = nn.Embedding(Config.VOCAB_SIZE_STRUCT, Config.EMBED_DIM_STRUCT)
        self.emb_loop = nn.Embedding(Config.VOCAB_SIZE_LOOP, Config.EMBED_DIM_LOOP)

        # Convolutional Layer
        # Input: (Batch, Channels, Seq_Len) -> Conv1d -> (Batch, Hidden, Seq_Len)
        self.conv = nn.Conv1d(
            in_channels=Config.EMBED_TOTAL_DIM,
            out_channels=hidden_dim,
            kernel_size=3,
            padding=1,
        )
        self.relu = nn.ReLU()

        # GRU Layer
        self.gru = nn.GRU(
            input_size=hidden_dim,
            hidden_size=hidden_dim,
            num_layers=num_layers,
            batch_first=True,
            bidirectional=True,
            dropout=dropout if num_layers > 1 else 0,
        )

        # Fully Connected Layer
        # Bidirectional results in 2 * hidden_dim
        self.fc = nn.Linear(hidden_dim * 2, num_targets)

    def forward(self, x):
        # x: (Batch, Seq_Len, 3) where 3 are indices for [Seq, Struct, Loop]
        
        # 1. Embeddings
        seq_emb = self.emb_seq(x[:, :, 0])        # (Batch, Seq_Len, Dim_Seq)
        struct_emb = self.emb_struct(x[:, :, 1])  # (Batch, Seq_Len, Dim_Struct)
        loop_emb = self.emb_loop(x[:, :, 2])      # (Batch, Seq_Len, Dim_Loop)

        # Concatenate embeddings
        x_emb = torch.cat([seq_emb, struct_emb, loop_emb], dim=2) # (Batch, Seq_Len, Total_Dim)

        # 2. Convolution
        # Permute for Conv1d: (Batch, Total_Dim, Seq_Len)
        x_conv = x_emb.permute(0, 2, 1)
        x_conv = self.relu(self.conv(x_conv))
        
        # Permute back for GRU: (Batch, Seq_Len, Hidden)
        x_rnn_in = x_conv.permute(0, 2, 1)

        # 3. Recurrent Layers
        out, _ = self.gru(x_rnn_in)
        
        # 4. Output Head
        out = self.fc(out)
        
        return out
