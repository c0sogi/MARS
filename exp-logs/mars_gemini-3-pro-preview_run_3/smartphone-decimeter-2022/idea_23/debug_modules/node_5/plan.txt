def ecef_to_wgs84(x, y, z):
    """
    Convert ECEF Cartesian coordinates to WGS84 geodetic coordinates.
    Uses Ferrari's solution.

    Args:
        x (float or np.array): ECEF X coordinate in meters.
        y (float or np.array): ECEF Y coordinate in meters.
        z (float or np.array): ECEF Z coordinate in meters.

    Returns:
        tuple: (lat, lon, alt) in degrees and meters.
    """
    # Second eccentricity squared
    ep2 = (WGS84_A**2 - WGS84_B**2) / WGS84_B**2

    p = np.sqrt(x**2 + y**2)
    th = np.arctan2(WGS84_A * z, WGS84_B * p)

    lon = np.arctan2(y, x)

    num = z + ep2 * WGS84_B * np.sin(th)**3
    den = p - WGS84_E2 * WGS84_A * np.cos(th)**3
    lat = np.arctan2(num, den)

    sin_lat = np.sin(lat)
    N = WGS84_A / np.sqrt(1 - WGS84_E2 * sin_lat**2)

    # Altitude calculation
    alt = p / np.cos(lat) - N

    return np.degrees(lat), np.degrees(lon), alt
