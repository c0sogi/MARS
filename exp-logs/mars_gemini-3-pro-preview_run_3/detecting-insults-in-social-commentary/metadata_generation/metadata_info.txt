# Python Script for Metadata Generation
```
import os
import pandas as pd
import numpy as np
from sklearn.model_selection import StratifiedShuffleSplit


def generate_metadata():
    # Define paths
    INPUT_DIR = "./input"
    METADATA_DIR = "./metadata"
    TRAIN_PATH = os.path.join(INPUT_DIR, "train.csv")
    TEST_PATH = os.path.join(INPUT_DIR, "test.csv")

    # Create metadata directory
    os.makedirs(METADATA_DIR, exist_ok=True)

    # Load datasets
    # The dataset description mentions unicode-escaped text in double quotes.
    # Pandas read_csv handles standard CSV quoting (double quotes) by default.
    print("Loading raw data...")
    try:
        df_train_full = pd.read_csv(TRAIN_PATH)
        df_test = pd.read_csv(TEST_PATH)
    except Exception as e:
        print(f"Error reading CSV files: {e}")
        raise

    # Verify columns
    # Based on sample_submission_null.csv, expected columns are Insult, Date, Comment
    # We check if these exist.
    required_train_cols = {"Insult", "Comment"}
    if not required_train_cols.issubset(df_train_full.columns):
        # Fallback: if headers are missing or different, try to infer or raise error.
        # Given the problem description, we assume standard headers or specific order.
        # If 'Insult' is missing but we have 3 cols, we might map them.
        # But usually Kaggle-style CSVs have headers.
        print(
            f"Warning: Expected columns {required_train_cols} not found. Available: {df_train_full.columns}"
        )
        # If strictly required columns are missing, we can't proceed safely without assumptions.
        # Let's assume the dataset provided follows the sample_submission format.
        raise ValueError(
            f"Train dataset missing required columns: {required_train_cols - set(df_train_full.columns)}"
        )

    # Split training data into Train and Validation
    # Requirement: 80:20 split, Stratified, Random State 42
    print("Splitting data...")
    sss = StratifiedShuffleSplit(n_splits=1, test_size=0.2, random_state=42)

    # We split based on the 'Insult' label
    y = df_train_full["Insult"].values

    train_idx, val_idx = next(sss.split(df_train_full, y))

    df_train = df_train_full.iloc[train_idx].copy()
    df_val = df_train_full.iloc[val_idx].copy()

    # Save metadata
    # For text CSV data, the metadata is effectively the split CSVs themselves.
    # We do not have external files to reference with paths.
    print("Saving metadata...")
    train_meta_path = os.path.join(METADATA_DIR, "train.csv")
    val_meta_path = os.path.join(METADATA_DIR, "val.csv")
    test_meta_path = os.path.join(METADATA_DIR, "test.csv")

    df_train.to_csv(train_meta_path, index=False)
    df_val.to_csv(val_meta_path, index=False)
    df_test.to_csv(test_meta_path, index=False)

    print("Metadata generation complete.")

    # ==========================================
    # Validation Checks
    # ==========================================
    print("\nRunning validation checks...")

    # 1. Load generated metadata
    d_train = pd.read_csv(train_meta_path)
    d_val = pd.read_csv(val_meta_path)
    d_test = pd.read_csv(test_meta_path)

    # 2. Print Summary Statistics
    print("-" * 30)
    print("Summary Statistics:")
    print(f"Train shape: {d_train.shape}")
    print(f"Val shape:   {d_val.shape}")
    print(f"Test shape:  {d_test.shape}")

    print("\nClass Distribution (Train):")
    print(d_train["Insult"].value_counts(normalize=True))
    print("\nClass Distribution (Val):")
    print(d_val["Insult"].value_counts(normalize=True))
    print("-" * 30)

    # 3. Check file paths
    # The prompt asks: "If the metadata contains file paths, programmatically check..."
    # Our metadata contains text content directly, not paths to external files.
    # Therefore, this check is not applicable (missing file ratio = 0).
    # We explicitly note this to satisfy the requirement logic.
    print(
        "No external file paths in metadata (text data contained in CSV). Skipping file path resolution check."
    )

    # 4. Verify Validation Split
    print("Verifying split requirements...")

    # Check 1: Ratio
    total_train_val = len(d_train) + len(d_val)
    val_ratio = len(d_val) / total_train_val
    print(f"Validation Ratio: {val_ratio:.4f} (Expected ~0.2)")

    # Allow a small margin of error due to integer division
    assert 0.19 < val_ratio < 0.21, "Validation split ratio is not approximately 0.2"

    # Check 2: Stratification
    # We check if the mean of the target label is similar in train and val
    train_mean = d_train["Insult"].mean()
    val_mean = d_val["Insult"].mean()
    diff = abs(train_mean - val_mean)

    print(f"Train Label Mean: {train_mean:.4f}")
    print(f"Val Label Mean:   {val_mean:.4f}")
    print(f"Difference:       {diff:.4f}")

    # Stratification should keep distributions very close.
    # With ~4000 samples, difference should be small (e.g., < 0.02).
    if diff > 0.02:
        raise AssertionError(
            "Stratified sampling failed: Label distribution differs significantly between train and val."
        )

    print("All validation checks passed successfully.")


if __name__ == "__main__":
    generate_metadata()

```

### 1. `./metadata/train.csv`

*   **Content and Purpose**:
    *   This file contains the **training subset** of the original labeled dataset, representing approximately 80% of the available data (3,157 rows).
    *   It is generated using a stratified shuffle split to ensure the distribution of "Insult" vs. "Neutral" comments remains consistent with the original dataset.
    *   **Purpose**: This file is intended to be used for training machine learning models to detect insulting comments.

*   **Schema / Structure**:
    *   **Format**: CSV (Comma Separated Values)
    *   **Shape**: (3157, 3)
    *   **Columns**:
        1.  `Insult` (int64): The target label. `0` indicates a neutral comment, `1` indicates an insulting comment.
        2.  `Date` (object/string): The timestamp of the comment (e.g., `"20120618222256Z"`).
        3.  `Comment` (object/string): The unicode-escaped text content of the comment.
    *   **Example Row**:
        ```csv
        Insult,Date,Comment
        1,20120529142300Z,"You are an idiot."
        ```

*   **Loading Method**:
    ```python
    import pandas as pd
    df_train = pd.read_csv("./metadata/train.csv")
    ```

### 2. `./metadata/val.csv`

*   **Content and Purpose**:
    *   This file contains the **validation subset** of the original labeled dataset, representing approximately 20% of the available data (790 rows).
    *   Like the training set, it is created via stratified sampling to maintain class balance (approx. 26.6% insulting comments).
    *   **Purpose**: This file is used to evaluate model performance during training, tune hyperparameters, and prevent overfitting.

*   **Schema / Structure**:
    *   **Format**: CSV
    *   **Shape**: (790, 3)
    *   **Columns**: Same as `train.csv`.
        1.  `Insult` (int64)
        2.  `Date` (object/string)
        3.  `Comment` (object/string)

*   **Loading Method**:


### 3. `./metadata/test.csv`

*   **Content and Purpose**:
    *   This file contains the **unlabeled test data** (2,647 rows).
    *   **Purpose**: This file provides the feature data (`Date` and `Comment`) for which the model must generate probability predictions (0 to 1) for the `Insult` label.

*   **Schema / Structure**:
    *   **Format**: CSV
    *   **Shape**: (2647, 2)
    *   **Columns**:
        1.  `Date` (object/string): The timestamp of the comment.
