class BCEDiceLoss(nn.Module):
    """
    Combined Binary Cross Entropy and Dice Loss for segmentation.

    The loss is calculated as: alpha * BCE + (1 - alpha) * DiceLoss
    """

    def __init__(self, bce_weight=0.5, smooth=1e-6):
        """
        Args:
            bce_weight (float): Weight assigned to BCE loss. (1 - bce_weight) is assigned to Dice.
            smooth (float): Smoothing factor to prevent division by zero in Dice calculation.
        """
        super(BCEDiceLoss, self).__init__()
        self.bce_weight = bce_weight
        self.smooth = smooth
        self.bce_loss = nn.BCELoss()

    def forward(self, preds, targets):
        """
        Args:
            preds (torch.Tensor): Predicted probabilities (after Sigmoid), shape (B, 1, H, W).
            targets (torch.Tensor): Ground truth binary masks, shape (B, 1, H, W).
        """
        # 1. BCE Loss
        bce = self.bce_loss(preds, targets)

        # 2. Dice Loss (Soft Dice for differentiability)
        preds_flat = preds.view(-1)
        targets_flat = targets.view(-1)

        intersection = (preds_flat * targets_flat).sum()
        union = preds_flat.sum() + targets_flat.sum()

        dice_score = (2.0 * intersection + self.smooth) / (union + self.smooth)
        dice_loss = 1.0 - dice_score

        # Combined Loss
        return self.bce_weight * bce + (1.0 - self.bce_weight) * dice_loss

class BCETverskyLoss(nn.Module):
    """
    Combined Binary Cross Entropy and Tversky Loss for segmentation.
    
    Tversky Loss allows weighting False Positives and False Negatives differently.
    For F0.5 score (Precision > Recall), we set alpha > beta to penalize FPs more.
    Cite Lesson 00009.
    """

    def __init__(self, alpha=0.7, beta=0.3, bce_weight=0.5, smooth=1e-6):
        """
        Args:
            alpha (float): Weight for False Positives.
            beta (float): Weight for False Negatives.
            bce_weight (float): Weight assigned to BCE loss.
            smooth (float): Smoothing factor.
        """
        super(BCETverskyLoss, self).__init__()
        self.alpha = alpha
        self.beta = beta
        self.bce_weight = bce_weight
        self.smooth = smooth
        self.bce_loss = nn.BCELoss()

    def forward(self, preds, targets):
        # 1. BCE Loss
        bce = self.bce_loss(preds, targets)

        # 2. Tversky Loss
        preds_flat = preds.view(-1)
        targets_flat = targets.view(-1)

        tp = (preds_flat * targets_flat).sum()
        fp = (preds_flat * (1 - targets_flat)).sum()
        fn = ((1 - preds_flat) * targets_flat).sum()

        tversky_index = (tp + self.smooth) / (tp + self.alpha * fp + self.beta * fn + self.smooth)
        tversky_loss = 1.0 - tversky_index

        # Combined Loss
        return self.bce_weight * bce + (1.0 - self.bce_weight) * tversky_loss
