import torch

import torch.nn.functional as F
import random

def get_mel_filters(sr, n_fft, n_mels):
    """
    Creates a Mel filterbank matrix using NumPy.
    """
    def lin_mel(f):
        return 2595.0 * np.log10(1.0 + f / 700.0)

    def mel_lin(m):
        return 700.0 * (10.0 ** (m / 2595.0) - 1.0)

    fmin = 0.0
    fmax = sr / 2.0
    n_freqs = n_fft // 2 + 1

    mel_min = lin_mel(fmin)
    mel_max = lin_mel(fmax)

    mel_points = np.linspace(mel_min, mel_max, n_mels + 2)
    hz_points = mel_lin(mel_points)
    
    # Map Hz to FFT bin indices
    bin_points = np.floor((n_fft + 1) * hz_points / sr).astype(int)

    filters = np.zeros((n_mels, n_freqs))

    for i in range(n_mels):
        start = bin_points[i]
        center = bin_points[i + 1]
        end = bin_points[i + 2]

        if center > start:
            filters[i, start:center] = (np.arange(start, center) - start) / (center - start)
        if end > center:
            filters[i, center:end] = (end - np.arange(center, end)) / (end - center)

    return filters


class MelSpectrogram(nn.Module):
    """
    Native PyTorch implementation of MelSpectrogram to avoid torchaudio dependency.
    """
    def __init__(self, sample_rate, n_fft, win_length, hop_length, n_mels):
        super().__init__()
        self.n_fft = n_fft
        self.win_length = win_length
        self.hop_length = hop_length
        
        # Create Mel Basis
        mel_basis = get_mel_filters(sample_rate, n_fft, n_mels)
        self.register_buffer("mel_basis", torch.from_numpy(mel_basis).float())
        self.register_buffer("window", torch.hann_window(win_length))

    def forward(self, x):
        # x: (Batch, Time)
        # STFT
        x_stft = torch.stft(
            x,
            n_fft=self.n_fft,
            hop_length=self.hop_length,
            win_length=self.win_length,
            window=self.window,
            center=True,
            return_complex=True,
        )
        
        # Power Spectrogram
        power_spec = x_stft.abs().pow(2.0)
        
        # Mel Spectrogram: (Batch, n_mels, Time)
        # Basis: (n_mels, n_freqs)
        # Spec: (Batch, n_freqs, Time)
        return torch.matmul(self.mel_basis, power_spec)


class AmplitudeToDB(nn.Module):
    """Native implementation of AmplitudeToDB."""
    def __init__(self, eps=1e-10):
        super().__init__()
        self.eps = eps

    def forward(self, x):
        return 10.0 * torch.log10(x + self.eps)


class TimeMasking(nn.Module):
    """Native implementation of TimeMasking."""
    def __init__(self, time_mask_param):
        super().__init__()
        self.time_mask_param = time_mask_param

    def forward(self, x):
        # x: (Batch, Channels, Freq, Time) or flattened
        # We assume input is flattened to (B*C, F, T) or (B, C, F, T)
        # For simplicity, we handle the last dimension as Time
        T = x.shape[-1]
        mask_len = random.randint(0, self.time_mask_param)
        t0 = random.randint(0, max(0, T - mask_len))
        
        # Clone to avoid in-place modification issues if any
        x_masked = x.clone()
        x_masked[..., t0 : t0 + mask_len] = 0
        return x_masked


class FrequencyMasking(nn.Module):
    """Native implementation of FrequencyMasking."""
    def __init__(self, freq_mask_param):
        super().__init__()
        self.freq_mask_param = freq_mask_param

    def forward(self, x):
        # x: (..., Freq, Time)
        F_dim = x.shape[-2]
        mask_len = random.randint(0, self.freq_mask_param)
        f0 = random.randint(0, max(0, F_dim - mask_len))
        
        x_masked = x.clone()
        x_masked[..., f0 : f0 + mask_len, :] = 0
        return x_masked
