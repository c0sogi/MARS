class FrequencyMasking(torch.nn.Module):
    def __init__(self, freq_mask_param):
        super().__init__()
        self.freq_mask_param = freq_mask_param

    def forward(self, specgram):
        """
        Args:
            specgram: (..., Freq, Time)
        """
        F = specgram.shape[-2]
        f = int(torch.rand(1).item() * self.freq_mask_param)
        if f == 0:
            return specgram
            
        f0 = int(torch.rand(1).item() * (F - f))

        masked_spec = specgram.clone()
        masked_spec[..., f0:f0+f, :] = 0
        return masked_spec
