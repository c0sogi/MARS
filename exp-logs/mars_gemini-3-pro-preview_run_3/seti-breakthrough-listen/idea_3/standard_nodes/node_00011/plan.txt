class Shallow3DCNN(nn.Module):
    """
    A lightweight 3D CNN designed to capture spatiotemporal patterns in cadence snippets.
    Input shape: (Batch, 1, Depth=6, Height=256, Width=256)
    """

    def __init__(self):
        super(Shallow3DCNN, self).__init__()

        # Block 1: Preserve depth, pool spatial
        self.layer1 = nn.Sequential(
            nn.Conv3d(in_channels=1, out_channels=32, kernel_size=3, padding=1),
            nn.BatchNorm3d(32),
            nn.ReLU(inplace=True),
            nn.MaxPool3d(kernel_size=(1, 2, 2), stride=(1, 2, 2)),
        )

        # Block 2: Preserve depth, pool spatial
        self.layer2 = nn.Sequential(
            nn.Conv3d(in_channels=32, out_channels=64, kernel_size=3, padding=1),
            nn.BatchNorm3d(64),
            nn.ReLU(inplace=True),
            nn.MaxPool3d(kernel_size=(1, 2, 2), stride=(1, 2, 2)),
        )

        # Block 3: Preserve depth, pool spatial
        self.layer3 = nn.Sequential(
            nn.Conv3d(in_channels=64, out_channels=128, kernel_size=3, padding=1),
            nn.BatchNorm3d(128),
            nn.ReLU(inplace=True),
            nn.MaxPool3d(kernel_size=(1, 2, 2), stride=(1, 2, 2)),
        )

        # Block 4: Pool depth and spatial
        self.layer4 = nn.Sequential(
            nn.Conv3d(in_channels=128, out_channels=256, kernel_size=3, padding=1),
            nn.BatchNorm3d(256),
            nn.ReLU(inplace=True),
            nn.MaxPool3d(kernel_size=(2, 2, 2), stride=(2, 2, 2)),
        )

        # Head
        self.avgpool = nn.AdaptiveAvgPool3d((1, 1, 1))
        self.fc = nn.Linear(256, 1)

        x = self.layer2(x)

        x = self.layer4(x)

def predict(model, loader, device):
    """
    Generates predictions using TTA (Test Time Augmentation).
    TTA Strategy: Average prediction of original and horizontally flipped (frequency reversed) input.
    """
    model.eval()
    preds = []

    with torch.no_grad():
        for images, _ in loader:
            images = images.to(device)

            # Forward pass 1: Original
            out_orig = model(images)
            prob_orig = torch.sigmoid(out_orig)

            # Forward pass 2: Horizontal Flip (Frequency dimension is last: W)
            # Input shape is (B, 1, 6, H, W)
            images_flip = torch.flip(images, dims=[-1])
            out_flip = model(images_flip)
            prob_flip = torch.sigmoid(out_flip)

            # Average probabilities
            avg_prob = (prob_orig + prob_flip) / 2.0

            preds.append(avg_prob.cpu().numpy())

    return np.concatenate(preds)
