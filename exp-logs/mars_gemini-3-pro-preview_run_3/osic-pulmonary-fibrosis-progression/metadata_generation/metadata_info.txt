# Python Script for Metadata Generation
```
import os
import pandas as pd
import numpy as np
from sklearn.model_selection import GroupShuffleSplit

# Configuration
INPUT_DIR = "./input"
METADATA_DIR = "./metadata"
RANDOM_STATE = 42
VAL_SIZE = 0.2


def generate_metadata():
    """Generates metadata files for train, val, and test sets."""
    print("Generating metadata...")
    os.makedirs(METADATA_DIR, exist_ok=True)

    # Load raw data
    train_df = pd.read_csv(os.path.join(INPUT_DIR, "train.csv"))
    test_df = pd.read_csv(os.path.join(INPUT_DIR, "test.csv"))

    # Add image paths
    # The dataset structure is input/train/<PatientID> and input/test/<PatientID>
    # We store the relative path to the directory containing the DICOM slices
    train_df["image_path"] = train_df["Patient"].apply(
        lambda x: os.path.join("train", x)
    )
    test_df["image_path"] = test_df["Patient"].apply(lambda x: os.path.join("test", x))

    # Split training data into train and validation sets
    # We must split by Patient ID (Group Sampling) to avoid data leakage
    splitter = GroupShuffleSplit(
        n_splits=1, test_size=VAL_SIZE, random_state=RANDOM_STATE
    )
    groups = train_df["Patient"]

    train_idx, val_idx = next(splitter.split(train_df, groups=groups))

    train_meta = train_df.iloc[train_idx].copy()
    val_meta = train_df.iloc[val_idx].copy()
    test_meta = test_df.copy()

    # Save metadata
    train_meta.to_csv(os.path.join(METADATA_DIR, "train.csv"), index=False)
    val_meta.to_csv(os.path.join(METADATA_DIR, "val.csv"), index=False)
    test_meta.to_csv(os.path.join(METADATA_DIR, "test.csv"), index=False)

    print("Metadata generation complete.")


def check_file_paths(df, dataset_name):
    """Checks if a sample of file paths in the dataframe exist."""
    paths = df["image_path"].values
    n_total = len(paths)
    n_check = min(1000, n_total)

    # Randomly sample paths
    rng = np.random.default_rng(RANDOM_STATE)
    sample_paths = rng.choice(paths, size=n_check, replace=False)

    missing_count = 0
    sample_missing = []

    for rel_path in sample_paths:
        full_path = os.path.join(INPUT_DIR, rel_path)
        # Check if the directory exists
        if not os.path.exists(full_path):
            missing_count += 1
            if len(sample_missing) < 5:
                sample_missing.append(rel_path)

    ratio = missing_count / n_check
    print(
        f"[{dataset_name}] Checked {n_check}/{n_total} paths. Missing ratio: {ratio:.4f}"
    )

    if ratio > 0.5:
        print(f"[{dataset_name}] Example missing paths: {sample_missing}")
        raise FileNotFoundError(
            f"[{dataset_name}] Missing file ratio {ratio:.4f} exceeds threshold 0.5"
        )


def validate_metadata():
    """Performs validation checks on the generated metadata."""
    print("\nValidating metadata...")

    # Load generated metadata
    train_meta = pd.read_csv(os.path.join(METADATA_DIR, "train.csv"))
    val_meta = pd.read_csv(os.path.join(METADATA_DIR, "val.csv"))
    test_meta = pd.read_csv(os.path.join(METADATA_DIR, "test.csv"))

    # 1. Summary Statistics
    print("--- Summary Statistics ---")
    print(
        f"Train: {len(train_meta)} rows, {train_meta['Patient'].nunique()} unique patients"
    )
    print(
        f"Val:   {len(val_meta)} rows, {val_meta['Patient'].nunique()} unique patients"
    )
    print(
        f"Test:  {len(test_meta)} rows, {test_meta['Patient'].nunique()} unique patients"
    )

    print(f"Train FVC Mean: {train_meta['FVC'].mean():.2f}")
    print(f"Val FVC Mean:   {val_meta['FVC'].mean():.2f}")
    print(f"Test FVC Mean:  {test_meta['FVC'].mean():.2f}")

    # 2. Verify Split Logic
    print("\n--- Verifying Split ---")
    train_patients = set(train_meta["Patient"].unique())
    val_patients = set(val_meta["Patient"].unique())

    # Check for intersection
    intersection = train_patients.intersection(val_patients)
    if intersection:
        raise AssertionError(
            f"Data leakage detected! Patients found in both train and val: {intersection}"
        )

    # Check split ratio
    total_patients = len(train_patients) + len(val_patients)
    actual_val_ratio = len(val_patients) / total_patients
    print(
        f"Actual Validation Ratio (by patient): {actual_val_ratio:.4f} (Target: {VAL_SIZE})"
    )

    # 3. Check File Paths
    print("\n--- Verifying File Paths ---")
    check_file_paths(train_meta, "Train")
    check_file_paths(val_meta, "Val")
    check_file_paths(test_meta, "Test")

    print("\nAll validation checks passed successfully.")


if __name__ == "__main__":
    generate_metadata()
    validate_metadata()

```

# Document
Based on the analysis of the Python script and its execution output, here is the documentation for the files generated in the `./metadata` directory.

### 1. `./metadata/train.csv`

*   **Content and Purpose**:
    *   This file contains the **training subset** of the original dataset. It includes clinical metadata, baseline and follow-up FVC measurements, and relative paths to the corresponding CT scan directories.
    *   It is generated by performing a Group Shuffle Split on the original `train.csv`, ensuring that approximately 80% of the unique patients are assigned to this set.
    *   **Purpose**: This file is used to train the machine learning models. The `image_path` column allows data loaders to locate the directory containing the DICOM files for each patient.
*   **Schema / Structure**:
    *   **Format**: CSV (Comma Separated Values)
    *   **Rows**: 1109 rows (representing individual visits/measurements).
    *   **Columns**:
        *   `Patient` (String): Unique ID for the patient (e.g., `ID00007637202177411956430`).
        *   `Weeks` (Integer): Relative number of weeks pre/post the baseline CT.
        *   `FVC` (Integer): The recorded lung capacity in ml (Target variable).
        *   `Percent` (Float): Approximated FVC as a percent of typical FVC.
        *   `Age` (Integer): Patient age.
        *   `Sex` (String): Patient sex (e.g., `Male`, `Female`).
        *   `SmokingStatus` (String): Smoking history (e.g., `Ex-smoker`, `Never smoked`).
        *   `image_path` (String): Relative path to the patient's DICOM directory (e.g., `train/ID00007637202177411956430`).
*   **Loading Method**:
    *   Load using pandas:
        ```python
        import pandas as pd
        train_df = pd.read_csv("./metadata/train.csv")
        ```

### 2. `./metadata/val.csv`

*   **Content and Purpose**:
    *   This file contains the **validation subset** of the original dataset. Like the training set, it includes clinical metadata and FVC measurements.
    *   It represents approximately 20% of the unique patients (32 patients) from the original training data. Crucially, there is no patient overlap between `train.csv` and `val.csv` to prevent data leakage.
    *   **Purpose**: This file is used to evaluate model performance during training (e.g., for early stopping or hyperparameter tuning) on unseen patients.
*   **Schema / Structure**:
    *   **Format**: CSV
    *   **Rows**: 285 rows.
    *   **Columns**: Identical to `./metadata/train.csv`.
        *   `Patient`, `Weeks`, `FVC`, `Percent`, `Age`, `Sex`, `SmokingStatus`, `image_path`.
*   **Loading Method**:
    *   Load using pandas:
        ```python
        import pandas as pd
        val_df = pd.read_csv("./metadata/val.csv")
        ```

### 3. `./metadata/test.csv`

*   **Content and Purpose**:
    *   This file contains the **test set** metadata. Unlike the training/validation sets which contain full histories, this file typically contains the baseline measurement information for the test patients.
    *   **Purpose**: This file is used to generate the final submission predictions. The model must use the baseline data provided here (and the associated CT scans found via `image_path`) to predict FVC values for future weeks.
*   **Schema / Structure**:
    *   **Format**: CSV
    *   **Rows**: 18 rows (18 unique patients).
    *   **Columns**:
        *   `Patient`, `Weeks`, `FVC`, `Percent`, `Age`, `Sex`, `SmokingStatus`.
        *   `image_path` (String): Relative path to the test patient's DICOM directory (e.g., `test/ID00419637202311204720264`).
*   **Loading Method**:
    *   Load using pandas:
        ```python
        import pandas as pd
        test_df = pd.read_csv("./metadata/test.csv")
        ```
