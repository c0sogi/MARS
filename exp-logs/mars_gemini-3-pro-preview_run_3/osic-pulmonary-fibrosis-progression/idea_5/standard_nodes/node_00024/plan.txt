My strategy is to optimize the model architecture by addressing the limitations identified in the failure analysis and lessons learned. Specifically, I will replace the "Wide-and-Deep" architecture with a "Late Fusion" architecture. The previous model isolated the temporal variable (`Weeks`) and `Baseline FVC` in a linear "Wide" branch, which prevented the network from learning multiplicative interactions between disease severity and progression rate (Cite solution_lesson_node_00022). I will modify the model to concatenate these features with other clinical metadata early in a shared MLP branch. Furthermore, I will ensure that the high-dimensional image features are fused with the tabular features only at the final layer to prevent them from overwhelming the strong tabular signals (Cite solution_lesson_node_00023). Finally, I will increase the training epochs to 50 to ensure sufficient convergence for the uncertainty estimation head (Cite solution_lesson_node_00011).

class WideAndDeepNet(nn.Module):
    """
    Multi-Scale Wide-and-Deep 2.5D Network.
    Fuses a Deep branch (Image + Metadata) with a Wide branch (Linear Baseline Trend).
    """

    def __init__(self):
        super().__init__()

        # --- Image Branch ---
        self.image_encoder = MultiScaleEfficientNet()
        self.img_vec_dim = Config.PROJECTION_DIM * 2

        # --- Tabular Embeddings (Deep Branch) ---
        self.sex_embed = nn.Embedding(2, Config.EMBED_DIMS["Sex"])
        self.smoke_embed = nn.Embedding(3, Config.EMBED_DIMS["SmokingStatus"])

        # Deep Branch Input Dimension
        # Image (128) + Sex (2) + Smoke (3) + Age (1)
        self.deep_input_dim = (
            self.img_vec_dim
            + Config.EMBED_DIMS["Sex"]
            + Config.EMBED_DIMS["SmokingStatus"]
            + 1
        )

        # Latent Dimension for Fusion
        self.latent_dim = 128

        # Deep Branch MLP (No BatchNorm as per strategy)
        self.deep_mlp = nn.Sequential(
            nn.Linear(self.deep_input_dim, self.latent_dim),
            nn.ReLU(),
            nn.Linear(self.latent_dim, self.latent_dim),
        )

        # --- Wide Branch (Linear Trend) ---
        # Input: Baseline FVC (1) + Weeks (1) = 2
        # No Activation, No Norm, No Dropout
        self.wide_layer = nn.Linear(2, self.latent_dim)

        # --- Heads ---
        # Project summed latent vector to mu and sigma
        self.head_mu = nn.Linear(self.latent_dim, 1)
        self.head_sigma = nn.Linear(self.latent_dim, 1)

    def forward(self, img, weeks, base_fvc, age, sex, smoke):
        # --- Image Branch ---
        img_vec = self.image_encoder(img)  # (B, 128)

        # --- Deep Branch ---
        sex_vec = self.sex_embed(sex)  # (B, 2)
        smoke_vec = self.smoke_embed(smoke)  # (B, 3)
        # age is (B, 1)

        deep_in = torch.cat([img_vec, sex_vec, smoke_vec, age], dim=1)
        deep_out = self.deep_mlp(deep_in)  # (B, Latent)

        # --- Wide Branch ---
        # wide inputs: (B, 2)
        wide_in = torch.cat([base_fvc, weeks], dim=1)
        wide_out = self.wide_layer(wide_in)  # (B, Latent)

        # --- Fusion ---
        # Sum outputs of Wide and Deep branches
        fused = deep_out + wide_out

        # --- Output ---
        mu = self.head_mu(fused)
        sigma_raw = self.head_sigma(fused)

        # Sigma must be positive. Softplus + small epsilon for stability.
        sigma = F.softplus(sigma_raw) + 1e-3

        return mu, sigma

class LateFusionNet(nn.Module):
    """
    Late Fusion Network.
    Separates Image and Tabular processing until the final layer.
    Tabular branch mixes temporal (Weeks) and static features (Baseline, Age, Sex, Smoke).
    Cite solution_lesson_node_00022: Mixes temporal and static features early.
    Cite solution_lesson_node_00023: Late fusion of modalities.
    """

    def __init__(self):
        super().__init__()
        # Image Branch
        self.image_encoder = MultiScaleEfficientNet()
        self.img_dim = Config.PROJECTION_DIM * 2  # 128

        # Tabular Branch
        self.sex_embed = nn.Embedding(2, Config.EMBED_DIMS["Sex"])
        self.smoke_embed = nn.Embedding(3, Config.EMBED_DIMS["SmokingStatus"])

        # Inputs: Sex(2) + Smoke(3) + Age(1) + Weeks(1) + BaseFVC(1)
        self.tab_input_dim = (
            Config.EMBED_DIMS["Sex"]
            + Config.EMBED_DIMS["SmokingStatus"]
            + 1
            + 1
            + 1
        )
        self.tab_hidden_dim = 128

        self.tab_mlp = nn.Sequential(
            nn.Linear(self.tab_input_dim, self.tab_hidden_dim),
            nn.ReLU(),
            nn.Linear(self.tab_hidden_dim, self.tab_hidden_dim),
            nn.ReLU(),
        )

        # Fusion
        self.fusion_dim = self.img_dim + self.tab_hidden_dim

        self.head_mu = nn.Linear(self.fusion_dim, 1)
        self.head_sigma = nn.Linear(self.fusion_dim, 1)

    def forward(self, img, weeks, base_fvc, age, sex, smoke):
        # Image
        img_feat = self.image_encoder(img)  # (B, 128)

        # Tabular
        s_emb = self.sex_embed(sex)
        sm_emb = self.smoke_embed(smoke)

        # Concatenate all tabular features (Cite solution_lesson_node_00022)
        tab_in = torch.cat([s_emb, sm_emb, age, weeks, base_fvc], dim=1)
        tab_feat = self.tab_mlp(tab_in)  # (B, 128)

        # Late Fusion (Cite solution_lesson_node_00023)
        fused = torch.cat([img_feat, tab_feat], dim=1)

        mu = self.head_mu(fused)
        sigma = F.softplus(self.head_sigma(fused)) + 1e-3

        return mu, sigma
