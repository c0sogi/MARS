My optimization strategy involves reverting the "Wide-and-Deep" architecture change that caused performance degradation (Cite `solution_lesson_node_00022`) and returning to a "Fused" architecture where all features (Image, Metadata, Temporal) are concatenated early. This allows the MLP to learn multiplicative interactions between the disease severity (from images/baseline) and the progression time (`Weeks`), which were hindered by the additive wide branch.

I will maintain the successful "Content-Adaptive 2.5D Multi-View" image processing (Cite `solution_lesson_node_00015`) and the target standardization (Cite `solution_lesson_node_00006`). I will also keep the extended training schedule (35 epochs) to ensure the uncertainty head has sufficient time to converge (Cite `solution_lesson_node_00011`).

    Fuses a Deep branch (Image + Metadata) with a Wide branch (Linear Baseline Trend).
    """

        super().__init__()

        # --- Image Branch ---
        self.image_encoder = MultiScaleEfficientNet()
        self.img_vec_dim = Config.PROJECTION_DIM * 2

        # --- Tabular Embeddings (Deep Branch) ---
        self.sex_embed = nn.Embedding(2, Config.EMBED_DIMS["Sex"])
        self.smoke_embed = nn.Embedding(3, Config.EMBED_DIMS["SmokingStatus"])

        # Deep Branch Input Dimension
        # Image (128) + Sex (2) + Smoke (3) + Age (1)
        self.deep_input_dim = (

            + Config.EMBED_DIMS["SmokingStatus"]

        self.latent_dim = 128

class FusedNet(nn.Module):
    """
    Fused Multimodal Network.
    Concatenates all features (Image + Metadata + Temporal) into a single MLP.
    Cite solution_lesson_node_00022: Avoid isolating time variables in additive branches.
    """

    def __init__(self):
        super().__init__()

        # --- Image Branch ---
        self.image_encoder = MultiScaleEfficientNet()
        self.img_vec_dim = Config.PROJECTION_DIM * 2

        # --- Tabular Embeddings ---
        self.sex_embed = nn.Embedding(2, Config.EMBED_DIMS["Sex"])
        self.smoke_embed = nn.Embedding(3, Config.EMBED_DIMS["SmokingStatus"])

        # Input Dimension
        # Image (128) + Sex (2) + Smoke (3) + Age (1) + BaseFVC (1) + Weeks (1)
        self.input_dim = (
            self.img_vec_dim
            + Config.EMBED_DIMS["Sex"]
            + Config.EMBED_DIMS["SmokingStatus"]
            + 1  # Age
            + 1  # BaseFVC
            + 1  # Weeks
        )

        # Latent Dimension
        self.latent_dim = 128

        # MLP (No BatchNorm as per solution_lesson_node_00010)
        self.mlp = nn.Sequential(
            nn.Linear(self.input_dim, self.latent_dim),
            nn.ReLU(),
            nn.Linear(self.latent_dim, self.latent_dim),
            nn.ReLU(),
        )

        # --- Heads ---
        self.head_mu = nn.Linear(self.latent_dim, 1)
        self.head_sigma = nn.Linear(self.latent_dim, 1)

    def forward(self, img, weeks, base_fvc, age, sex, smoke):
        # --- Feature Extraction ---
        img_vec = self.image_encoder(img)  # (B, 128)
        sex_vec = self.sex_embed(sex)  # (B, 2)
        smoke_vec = self.smoke_embed(smoke)  # (B, 3)

        # --- Concatenation ---
        # Concatenate all features early to allow interaction learning
        x = torch.cat([img_vec, sex_vec, smoke_vec, age, base_fvc, weeks], dim=1)

        # --- MLP ---
        feat = self.mlp(x)

        # --- Output ---
        mu = self.head_mu(feat)
        sigma_raw = self.head_sigma(feat)

        # Sigma must be positive. Softplus + small epsilon for stability.
        sigma = F.softplus(sigma_raw) + 1e-3

        return mu, sigma
