Based on the analysis of the previous solution and the provided lessons, I will switch the architecture from the "Origin-Corrected Parametric Network" (OCP-Net) to a "Context-Aware Point-Wise Network" (CAP-Net). The OCP-Net failed because it enforced a rigid linear trajectory assumption that degraded over time, as evidenced by the high correlation between error and `Weeks` (Cite Lesson 00037). The CAP-Net will treat `Time` as a direct input feature to the dense layers, allowing the model to learn non-linear progression and uncertainty dynamics. Additionally, I will increase the hidden layer dimension to 512 and the training epochs to 50 to ensure sufficient capacity and convergence for the fine-tuning task (Cite Lesson 00027).

class OCPNet(nn.Module):
    """
    Origin-Corrected Parametric Network.
    Predicts patient-specific linear trajectory parameters and dynamic uncertainty.
    """

    def __init__(self):
        super(OCPNet, self).__init__()

        # 1. Image Branch
        self.image_encoder = ImageEncoder(projection_dim=Config.PROJECTION_DIM)

        # 2. Fusion & MLP
        # Input: Image Projection (128) + Tabular Features (4)
        input_dim = Config.PROJECTION_DIM + Config.N_TABULAR_FEATURES
        hidden_dim = 64

        self.fusion_mlp = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
        )

        # 3. Parametric Heads
        # Trajectory Head: alpha, beta, gamma
        self.traj_head = nn.Linear(hidden_dim, 3)

        # Uncertainty Head: delta_base, delta_growth
        self.unc_head = nn.Linear(hidden_dim, 2)

    def forward(self, images, tabular, t_rel):
        """
        Args:
            images: (Batch, 3, H, W)
            tabular: (Batch, 4) -> [Base_FVC_Norm, Age_Norm, Sex, Smoke]
            t_rel: (Batch, 1) -> Relative time scaled

        Returns:
            mu: (Batch, 1) -> Predicted normalized FVC mean
            sigma: (Batch, 1) -> Predicted normalized FVC uncertainty
        """
        # 1. Image Features
        img_embed = self.image_encoder(images)  # (Batch, 128)

        # 2. Fusion
        # Concatenate image embeddings and raw tabular features
        # Tabular features are NOT passed through BN/LN to preserve magnitude info
        combined = torch.cat([img_embed, tabular], dim=1)  # (Batch, 132)

        # 3. Shared Representation
        hidden = self.fusion_mlp(combined)  # (Batch, 64)

        # 4. Predict Parameters
        traj_params = self.traj_head(hidden)  # (Batch, 3)
        unc_params = self.unc_head(hidden)  # (Batch, 2)

        # Extract specific parameters
        alpha = traj_params[:, 0:1]  # Autoregressive coefficient
        beta = traj_params[:, 1:2]  # Offset
        gamma = traj_params[:, 2:3]  # Slope

        delta_base = unc_params[:, 0:1]
        delta_growth = unc_params[:, 1:2]

        # 5. Calculate Trajectory (Mean)
        # mu(t) = alpha * Base + beta + gamma * t_rel
        # Note: Base FVC is the first column of tabular input
        base_fvc_norm = tabular[:, 0:1]
        mu = alpha * base_fvc_norm + beta + gamma * t_rel

        # 6. Calculate Uncertainty (Sigma)
        # sigma(t) = softplus(delta_base) + softplus(delta_growth) * |t_rel|
        # Enforces positive uncertainty that grows (or changes) with time distance
        sigma = F.softplus(delta_base) + F.softplus(delta_growth) * torch.abs(t_rel)

        return mu, sigma

class CAPNet(nn.Module):
    """
    Context-Aware Point-Wise Network (CAP-Net).
    Predicts FVC and Uncertainty for a specific time point using fused features.
    Cite solution_lesson_node_00037: Point-Wise Conditioning vs. Parametric Trajectory Prediction.
    """

    def __init__(self):
        super(CAPNet, self).__init__()

        # 1. Image Branch
        self.image_encoder = ImageEncoder(projection_dim=Config.PROJECTION_DIM)

        # 2. Fusion & MLP
        # Input: Image Projection (128) + Tabular Features (4) + Time (1)
        # Cite solution_lesson_node_00037: Include Time as an explicit input feature.
        input_dim = Config.PROJECTION_DIM + Config.N_TABULAR_FEATURES + 1
        
        # Cite solution_lesson_node_00027: Increase capacity (Hidden Dimension: 256 -> 512)
        hidden_dim = 512

        self.fusion_mlp = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
        )

        # 3. Output Head
        # Predicts mu and sigma directly
        self.head = nn.Linear(hidden_dim, 2)

    def forward(self, images, tabular, t_rel):
        """
        Args:
            images: (Batch, 3, H, W)
            tabular: (Batch, 4) -> [Base_FVC_Norm, Age_Norm, Sex, Smoke]
            t_rel: (Batch, 1) -> Relative time scaled

        Returns:
            mu: (Batch, 1) -> Predicted normalized FVC mean
            sigma: (Batch, 1) -> Predicted normalized FVC uncertainty
        """
        # 1. Image Features
        img_embed = self.image_encoder(images)  # (Batch, 128)

        # 2. Fusion
        # Concatenate image embeddings, raw tabular features, and time
        # Cite solution_lesson_node_00037: Feed Time into dense layers
        combined = torch.cat([img_embed, tabular, t_rel], dim=1)

        # 3. Shared Representation
        hidden = self.fusion_mlp(combined)

        # 4. Predict Mu and Sigma
        out = self.head(hidden)
        
        mu = out[:, 0:1]
        
        # Cite solution_lesson_node_00014: Do not enforce domain-specific clipping in architecture
        # Use simple softplus + epsilon for stability
        sigma = F.softplus(out[:, 1:2]) + 1e-6

        return mu, sigma
