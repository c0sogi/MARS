def frequency_wise_norm(spectrogram):
    """
    Applies per-instance frequency-wise standardization.
    Input: (C, F, T)
    Output: (C, F, T)
    """
    # Calculate mean and std along time axis (dim=2)
    mean = spectrogram.mean(dim=2, keepdim=True)
    std = spectrogram.std(dim=2, keepdim=True)

    # Normalize with epsilon for stability
    return (spectrogram - mean) / (std + 1e-6)


def min_max_norm(spectrogram):
    """
    Applies instance-level Min-Max normalization.
    Input: (C, F, T)
    Output: (C, F, T) scaled to [0, 1]
    """
    max_val = spectrogram.max()
    min_val = spectrogram.min()
    return (spectrogram - min_val) / (max_val - min_val + 1e-6)
