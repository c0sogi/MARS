    def compute_sublattice_embeddings(self, atoms):
        """
        Computes Sublattice-Resolved Embeddings using MatGL.
        Pools node embeddings separately for Anion (O) and Cation (Al, Ga, In) sublattices.
        """
        features = {}

        # Default empty features (embedding dim 64 is standard for M3GNet)
        emb_dim = 64
        prefixes = ["anion_mean", "anion_std", "cation_mean", "cation_std"]
        for p in prefixes:
            for i in range(emb_dim):
                features[f"emb_{p}_{i}"] = 0.0

        if self.model is None or self.graph_converter is None:
            return features

        try:
            # Convert to graph
            graph, state_attr = self.graph_converter.get_graph(atoms)
            graph = graph.to(self.device)
            state_attr = state_attr.to(self.device)

            # Run model forward pass
            # M3GNet forward returns a tuple/list. We need to inspect it or use internal layers.
            # Standard M3GNet: forward(g, state_attr, l_g) -> predictions
            # However, we want node embeddings.
            # We can manually pass through the embedding and encoder layers.

            # 1. Embedding
            node_types = graph.ndata["node_type"]
            bond_dist = graph.edata["bond_dist"]
            bond_types = graph.edata[
                "edge_type"
            ]  # M3GNet uses this if available, or just dist

            # M3GNet implementation details vary, but usually:
            # node_feat = model.embedding(node_types)
            # edge_feat = model.rbf(bond_dist) ...

            # To be safe and compatible with the installed package version, we try to access the last layer's node features
            # if the model exposes a method for it, or we rely on the fact that many GNN implementations
            # store the last node features in the graph structure during forward.

            # Let's try to run the encoder directly if possible.
            # Assuming model.encoder exists.

            # Prepare inputs for encoder
            # The exact call depends on matgl version.
            # In 1.1.3, M3GNet forward calls:
            # node_feat, edge_feat, state_feat = self.embedding(node_types, bond_types, state_attr)
            # node_feat = self.encoder(graph, node_feat, edge_feat, state_feat)

            node_feat, edge_feat, state_feat = self.model.embedding(
                node_types, bond_dist, state_attr
            )
            node_feat = self.model.encoder(graph, node_feat, edge_feat, state_feat)

            # node_feat is now the tensor of node embeddings [N_atoms, Dim]
            node_emb = node_feat.detach().cpu().numpy()

            # Sublattice Pooling
            symbols = np.array(atoms.get_chemical_symbols())

            # Anion (Oxygen)
            anion_mask = symbols == "O"
            if np.any(anion_mask):
                anion_embs = node_emb[anion_mask]
                anion_mean = np.mean(anion_embs, axis=0)
                anion_std = np.std(anion_embs, axis=0)
                for i in range(emb_dim):
                    features[f"emb_anion_mean_{i}"] = anion_mean[i]
                    features[f"emb_anion_std_{i}"] = anion_std[i]

            # Cation (Al, Ga, In)
            cation_mask = np.isin(symbols, list(self.cation_symbols))
            if np.any(cation_mask):
                cation_embs = node_emb[cation_mask]
                cation_mean = np.mean(cation_embs, axis=0)
                cation_std = np.std(cation_embs, axis=0)
                for i in range(emb_dim):
                    features[f"emb_cation_mean_{i}"] = cation_mean[i]
                    features[f"emb_cation_std_{i}"] = cation_std[i]

        except Exception as e:
            # Fallback if model structure is different or inference fails
            # print(f"Embedding extraction failed: {e}")
            pass

        return features
