def compute_voronoi_fingerprints(atoms):
    """
    Computes topological features based on Voronoi tessellation of the crystal structure.
    Aggregates Voronoi volume and coordination number by chemical element.

    Args:
        atoms (ase.Atoms): The atomic structure.

    Returns:
        dict: Aggregated Voronoi statistics (mean, std) for each element.
    """
    # Jiggle atoms slightly to avoid degenerate vertices in perfect crystals
    atoms_jit = atoms.copy()
    atoms_jit.rattle(stdev=1e-4, seed=42)

    # Create supercell
    supercell = atoms_jit.repeat(VORONOI_SUPERCELL_REPEAT)
    positions = supercell.get_positions()

    # Determine indices of the central unit cell
    # Repeat order in ASE is z, y, x (slowest to fastest) or x, y, z depending on version,
    # but standard 3x3x3 puts the center image at index 13 (0-26 flattened).
    # Center image corresponds to translation (1, 1, 1).
    # Grid size is 3x3x3 = 27 cells.
    # Index offset = (1*9 + 1*3 + 1) = 13.
    n_atoms = len(atoms)
    center_start = 13 * n_atoms
    center_end = 14 * n_atoms
    target_indices = list(range(center_start, center_end))

    features = {}

    try:
        # Compute Voronoi Tessellation
        vor = Voronoi(positions)

        # Storage for metrics
        element_volumes = defaultdict(list)
        element_coordination = defaultdict(list)

        symbols = supercell.get_chemical_symbols()

        for i, atom_idx in enumerate(target_indices):
            element = symbols[atom_idx]
            region_idx = vor.point_region[atom_idx]
            region_vertices_indices = vor.regions[region_idx]

            # Check if region is finite (no -1 in indices and not empty)
            if -1 in region_vertices_indices or len(region_vertices_indices) == 0:
                continue

            vertices = vor.vertices[region_vertices_indices]

            # Calculate Volume
            try:
                hull = ConvexHull(vertices)
                vol = hull.volume
            except QhullError:
                vol = 0.0

            # Calculate Coordination Number (Number of faces of the polyhedra)
            # In Voronoi, number of faces corresponds to number of neighbors sharing a boundary.
            # We can count how many ridge_points involve this atom_idx.
            # ridge_points is a list of [p1, p2] indices.
            coordination = 0
            for p1, p2 in vor.ridge_points:
                if p1 == atom_idx or p2 == atom_idx:
                    coordination += 1

            element_volumes[element].append(vol)
            element_coordination[element].append(coordination)

        # Aggregate statistics
        for elem in ATOM_TYPES:
            vols = element_volumes.get(elem, [])
            coords = element_coordination.get(elem, [])

            if vols:
                features[f"Voronoi_Vol_Mean_{elem}"] = np.mean(vols)
                features[f"Voronoi_Vol_Std_{elem}"] = np.std(vols)
            else:
                features[f"Voronoi_Vol_Mean_{elem}"] = 0.0
                features[f"Voronoi_Vol_Std_{elem}"] = 0.0

            if coords:
                features[f"Voronoi_Coord_Mean_{elem}"] = np.mean(coords)
                features[f"Voronoi_Coord_Std_{elem}"] = np.std(coords)
            else:
                features[f"Voronoi_Coord_Mean_{elem}"] = 0.0
                features[f"Voronoi_Coord_Std_{elem}"] = 0.0

    except Exception as e:
        # Fallback in case of tessellation failure
        for elem in ATOM_TYPES:
            features[f"Voronoi_Vol_Mean_{elem}"] = 0.0
            features[f"Voronoi_Vol_Std_{elem}"] = 0.0
            features[f"Voronoi_Coord_Mean_{elem}"] = 0.0
            features[f"Voronoi_Coord_Std_{elem}"] = 0.0

    return features

def compute_local_environment_moments(atoms):
    """
    Computes Chemically-Resolved Local Environment Moments (CR-LEM).
    Calculates mean and std of Coordination Number (CN) and Bond Angle Variance
    for each element type within a radial cutoff.
    
    Cite {solution_lesson_node_00037}
    Cite {solution_lesson_node_00042}
    """
    # Ensure PBC is set for neighbor_list to work correctly
    if not any(atoms.pbc):
        atoms.pbc = True
        
    # Get neighbor list
    # i: center atom indices, j: neighbor atom indices
    # D: distance vectors (vector from i to j)
    i_indices, j_indices, D_vectors = neighbor_list('ijD', atoms, LEM_CUTOFF)
    
    # Calculate lengths of distance vectors
    distances = np.linalg.norm(D_vectors, axis=1)
    
    # Normalize vectors for angle calculation
    # Handle division by zero if any distance is 0 (should exclude self-interaction by default)
    with np.errstate(divide='ignore', invalid='ignore'):
        norm_vectors = D_vectors / distances[:, None]
        
    n_atoms = len(atoms)
    chemical_symbols = atoms.get_chemical_symbols()
    
    # 1. Coordination Numbers
    # Count occurrences of each i index to get coordination number per atom
    cn_counts = np.bincount(i_indices, minlength=n_atoms)
    
    # 2. Bond Angle Variance
    # Build adjacency list of normalized vectors for each center atom
    adj_vectors = [[] for _ in range(n_atoms)]
    for k, center_idx in enumerate(i_indices):
        adj_vectors[center_idx].append(norm_vectors[k])
        
    angle_variances = np.zeros(n_atoms)
    
    for idx in range(n_atoms):
        vecs = np.array(adj_vectors[idx])
        k_neighbors = len(vecs)
        
        if k_neighbors < 2:
            angle_variances[idx] = 0.0
            continue
            
        # Compute pairwise dot products between all neighbors
        # Shape (k, k)
        dots = np.dot(vecs, vecs.T)
        
        # Clip for numerical stability in arccos
        dots = np.clip(dots, -1.0, 1.0)
        
        # Get angles (in degrees)
        angles = np.degrees(np.arccos(dots))
        
        # Extract unique off-diagonal elements (upper triangle)
        # indices of upper triangle, k=1 excludes diagonal
        iu = np.triu_indices(k_neighbors, k=1)
        unique_angles = angles[iu]
        
        # Calculate variance of bond angles
        if len(unique_angles) > 0:
            angle_variances[idx] = np.var(unique_angles)
        else:
            angle_variances[idx] = 0.0
        
    # 3. Aggregate by Element
    features = {}
    
    # Map symbols to atom indices
    elem_map = defaultdict(list)
    for idx, sym in enumerate(chemical_symbols):
        elem_map[sym].append(idx)
        
    for elem in ATOM_TYPES:
        indices = elem_map.get(elem, [])
        
        if indices:
            cns = cn_counts[indices]
            vars = angle_variances[indices]
            
            features[f"LEM_CN_Mean_{elem}"] = np.mean(cns)
            features[f"LEM_CN_Std_{elem}"] = np.std(cns)
            features[f"LEM_AngVar_Mean_{elem}"] = np.mean(vars)
            features[f"LEM_AngVar_Std_{elem}"] = np.std(vars)
        else:
            features[f"LEM_CN_Mean_{elem}"] = 0.0
            features[f"LEM_CN_Std_{elem}"] = 0.0
            features[f"LEM_AngVar_Mean_{elem}"] = 0.0
            features[f"LEM_AngVar_Std_{elem}"] = 0.0
            
    return features
