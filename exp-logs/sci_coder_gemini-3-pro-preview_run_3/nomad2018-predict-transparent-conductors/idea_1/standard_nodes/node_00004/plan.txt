from sklearn.linear_model import RidgeCV
from library.config import Config

class RidgeRegressorWrapper:
    """
    Wrapper for training and predicting with Ridge Regression models for
    formation energy and bandgap energy. Uses RidgeCV for built-in
    hyperparameter tuning of the regularization strength (alpha).
    """

    def __init__(self):
        # Initialize independent models for each target
        # RidgeCV performs efficient Leave-One-Out Cross-Validation (or K-Fold)
        # to select the best alpha from the provided list.
        self.model_formation = RidgeCV(
            alphas=Config.RIDGE_ALPHAS,
            cv=Config.CV_FOLDS,
            scoring="neg_mean_squared_error",
        )
        self.model_bandgap = RidgeCV(
            alphas=Config.RIDGE_ALPHAS,
            cv=Config.CV_FOLDS,
            scoring="neg_mean_squared_error",
        )
        self.is_fitted = False

    def train(self, X_train, y_train):
        """
        Trains the Ridge regression models.

        Args:
            X_train (pd.DataFrame or np.ndarray): Feature matrix.
            y_train (pd.DataFrame or np.ndarray): Target matrix with columns
                                                  [formation_energy, bandgap_energy].
        """
        # Ensure inputs are numpy arrays
        if isinstance(X_train, pd.DataFrame):
            X_train = X_train.values
        if isinstance(y_train, pd.DataFrame):
            y_train = y_train.values

        # Apply log transformation if configured to match RMSLE objective
        if Config.APPLY_LOG_TARGET:
            y_train_trans = log_transform(y_train)
        else:
            y_train_trans = y_train

        print("Training Formation Energy Model...")
        self.model_formation.fit(X_train, y_train_trans[:, 0])
        print(f"Formation Energy - Best Alpha: {self.model_formation.alpha_}")

        print("Training Bandgap Energy Model...")
        self.model_bandgap.fit(X_train, y_train_trans[:, 1])
        print(f"Bandgap Energy - Best Alpha: {self.model_bandgap.alpha_}")

        self.is_fitted = True

        # Calculate training metrics on the training set itself
        # Note: RidgeCV does not refit on the full set by default in some versions,
        # but standard sklearn RidgeCV.fit() does refit on the best alpha.
        y_pred_trans = np.column_stack(
            [self.model_formation.predict(X_train), self.model_bandgap.predict(X_train)]
        )
