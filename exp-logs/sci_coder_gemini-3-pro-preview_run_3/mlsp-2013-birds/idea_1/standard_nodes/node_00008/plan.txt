def train_epoch(model, loader, criterion, optimizer, device, mixup_alpha=0.0):
    """
    Trains the model for one epoch, optionally using Mixup.
    """
    model.train()
    running_loss = 0.0

    for inputs, labels in loader:
        inputs = inputs.to(device)
        labels = labels.to(device)

        optimizer.zero_grad()

        # Apply Mixup (Cite solution_lesson_node_00007)
        if mixup_alpha > 0:
            lam = np.random.beta(mixup_alpha, mixup_alpha)
            index = torch.randperm(inputs.size(0)).to(device)

            mixed_inputs = lam * inputs + (1 - lam) * inputs[index]
            mixed_labels = lam * labels + (1 - lam) * labels[index]

            logits = model(mixed_inputs)
            loss = criterion(logits, mixed_labels)
        else:
            logits = model(inputs)
            loss = criterion(logits, labels)

        loss.backward()
        optimizer.step()

        running_loss += loss.item() * inputs.size(0)

    return epoch_loss
