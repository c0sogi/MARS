class TiledDetector:
    def __init__(self, model, device, input_size=1024, stride=768, conf_thresh=0.3):
        self.model = model
        self.device = device
        self.input_size = input_size
        self.stride = stride
        self.conf_thresh = conf_thresh

        self.transform = A.Compose(
            [A.Normalize(mean=Config.MEAN, std=Config.STD), ToTensorV2()]
        )

    def detect(self, image):
        """
        Runs tiled detection on a single full-resolution image.
        Returns tensor of shape (N, 5) -> [x1, y1, x2, y2, score] in global coords.
        """
        h, w, _ = image.shape

        pad_h = 0
        pad_w = 0

        # Calculate padding and steps
        if h > self.input_size:
            y_steps = np.arange(0, h - self.input_size + 1, self.stride)
            if y_steps[-1] + self.input_size < h:
                y_steps = np.append(y_steps, h - self.input_size)
        else:
            y_steps = np.array([0])
            pad_h = self.input_size - h

        if w > self.input_size:
            x_steps = np.arange(0, w - self.input_size + 1, self.stride)
            if x_steps[-1] + self.input_size < w:
                x_steps = np.append(x_steps, w - self.input_size)
        else:
            x_steps = np.array([0])
            pad_w = self.input_size - w

        # Apply padding
        img_padded = image.copy()
        if pad_h > 0 or pad_w > 0:
            img_padded = np.pad(
                image,
                ((0, pad_h), (0, pad_w), (0, 0)),
                mode="constant",
                constant_values=0,
            )

        tiles = []
        coords = []  # (y, x)

        # Extract tiles
        for y in y_steps:
            for x in x_steps:
                tile = img_padded[y : y + self.input_size, x : x + self.input_size]
                tiles.append(tile)
                coords.append((y, x))

        # Batch process tiles
        all_detections = []
        batch_size = Config.DETECTOR_BATCH_SIZE

        for i in range(0, len(tiles), batch_size):
            batch_tiles = tiles[i : i + batch_size]
            batch_coords = coords[i : i + batch_size]

            # Preprocess
            batch_tensors = []
            for t in batch_tiles:
                aug = self.transform(image=t)
                batch_tensors.append(aug["image"])

            batch_input = torch.stack(batch_tensors).to(self.device)

            with torch.no_grad():
                hm, wh, reg = self.model(batch_input)

            # Decode
            batch_dets = decode_centernet(
                hm,
                wh,
                reg,
                stride=Config.DETECTOR_STRIDE,
                score_thresh=self.conf_thresh,
            )

            # Map back to global
            for j, dets in enumerate(batch_dets):
                if dets is None:
                    continue

                y_offset, x_offset = batch_coords[j]

                # Adjust coordinates
                dets[:, 0] += x_offset
                dets[:, 1] += y_offset
                dets[:, 2] += x_offset
                dets[:, 3] += y_offset

                # Clip to original image size
                dets[:, 0] = torch.clamp(dets[:, 0], 0, w)
                dets[:, 1] = torch.clamp(dets[:, 1], 0, h)
                dets[:, 2] = torch.clamp(dets[:, 2], 0, w)
                dets[:, 3] = torch.clamp(dets[:, 3], 0, h)

                all_detections.append(dets)

        if not all_detections:
            return torch.empty((0, 5), device=self.device)

        return torch.cat(all_detections, dim=0)

class LetterboxDetector:
    """
    Detector that resizes the full image to input_size (preserving aspect ratio),
    pads to square, detects, and maps coordinates back.
    Cite solution_lesson_node_00021, solution_lesson_node_00022
    """
    def __init__(self, model, device, input_size=1024, conf_thresh=0.3):
        self.model = model
        self.device = device
        self.input_size = input_size
        self.conf_thresh = conf_thresh
        self.transform = A.Compose(
            [A.Normalize(mean=Config.MEAN, std=Config.STD), ToTensorV2()]
        )

    def detect(self, image):
        # image is RGB numpy array
        h, w, _ = image.shape
        
        # 1. Letterbox Resize
        scale = self.input_size / max(h, w)
        new_h, new_w = int(h * scale), int(w * scale)
        
        img_resized = cv2.resize(image, (new_w, new_h))
        
        # Pad to square (bottom-right padding)
        pad_h = self.input_size - new_h
        pad_w = self.input_size - new_w
        
        img_padded = np.pad(
            img_resized,
            ((0, pad_h), (0, pad_w), (0, 0)),
            mode="constant",
            constant_values=0
        )
        
        # 2. Inference
        aug = self.transform(image=img_padded)
        img_tensor = aug["image"].unsqueeze(0).to(self.device)
        
        with torch.no_grad():
            hm, wh, reg = self.model(img_tensor)
            
        detections = decode_centernet(
            hm, wh, reg, 
            stride=Config.DETECTOR_STRIDE, 
            score_thresh=self.conf_thresh
        )
        
        dets = detections[0] # Batch size 1
        
        if dets is None:
            return torch.empty((0, 5), device=self.device)
            
        # 3. Map back to original coordinates
        # dets: [x1, y1, x2, y2, score]
        
        # Remove padding (coords are in padded space)
        # Since we padded bottom-right, coords in valid area are just coords.
        # But we need to check if they fall in padded region (unlikely with valid mask but possible)
        
        # Scale back
        dets[:, 0] /= scale
        dets[:, 1] /= scale
        dets[:, 2] /= scale
        dets[:, 3] /= scale
        
        # Clip
        dets[:, 0] = torch.clamp(dets[:, 0], 0, w)
        dets[:, 1] = torch.clamp(dets[:, 1], 0, h)
        dets[:, 2] = torch.clamp(dets[:, 2], 0, w)
        dets[:, 3] = torch.clamp(dets[:, 3], 0, h)
        
        return dets
