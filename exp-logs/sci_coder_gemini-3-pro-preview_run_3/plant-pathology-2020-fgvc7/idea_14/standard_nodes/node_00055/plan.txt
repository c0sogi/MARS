def _resolve_model_name(model_name):
    # Cite debug_lesson_3: Query Registry Keys Programmatically Instead of Guessing
    available_models = timm.list_models()
    if model_name in available_models:
        return model_name
    
    print(f"Model '{model_name}' not found in registry. Attempting to resolve...")
    # Try finding matches with suffix (e.g., .in1k)
    matches = fnmatch.filter(available_models, f"{model_name}*")
    if not matches:
        # Try broader search
        matches = fnmatch.filter(available_models, f"*{model_name}*")
    
    if matches:
        # Pick the shortest match (usually the base version without extra tags)
        new_name = min(matches, key=len)
        print(f"Resolved '{model_name}' to '{new_name}'")
        return new_name
