def _resolve_model_name(model_name):
    # Cite debug_lesson_3: Query Registry Keys Programmatically Instead of Guessing
    available_models = timm.list_models()
    if model_name in available_models:
        return model_name

    print(f"Model '{model_name}' not found in registry. Attempting to resolve...")
    # Try finding matches with suffix (e.g., .in1k)
    matches = fnmatch.filter(available_models, f"{model_name}*")
    if not matches:
        # Try broader search
        matches = fnmatch.filter(available_models, f"*{model_name}*")

    if matches:
        # Pick the shortest match (usually the base version without extra tags)
        new_name = min(matches, key=len)
        print(f"Resolved '{model_name}' to '{new_name}'")
        return new_name

    # Fallback: Relaxed search for architecture components (Cite debug_lesson_3)
    # Handles cases like 'maxvit_small_tf_384' -> 'maxvit_rmlp_small_rw_384'
    parts = model_name.split("_")
    # Filter out specific implementation tags that might differ

        matches = fnmatch.filter(available_models, pattern)

            return new_name

    # Cite debug_lesson_4: Fail Fast on Empty Registry Lookups
    raise RuntimeError(
        f"Could not resolve valid model name for '{model_name}' in timm registry."
