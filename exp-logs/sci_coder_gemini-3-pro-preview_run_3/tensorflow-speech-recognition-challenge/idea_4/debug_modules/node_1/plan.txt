class CustomMelSpectrogram(nn.Module):
    def __init__(self, sample_rate, n_fft, win_length, hop_length, n_mels):
        super().__init__()
        self.n_fft = n_fft
        self.win_length = win_length
        self.hop_length = hop_length
        
        # Create Mel Filterbank
        f_min = 0.0
        f_max = sample_rate / 2.0
        
        def hz_to_mel(hz): return 2595.0 * np.log10(1.0 + hz / 700.0)
        def mel_to_hz(mel): return 700.0 * (10.0**(mel / 2595.0) - 1.0)
        
        mel_min = hz_to_mel(f_min)
        mel_max = hz_to_mel(f_max)
        mel_points = np.linspace(mel_min, mel_max, n_mels + 2)
        hz_points = mel_to_hz(mel_points)
        bin_points = np.floor((n_fft + 1) * hz_points / sample_rate).astype(int)
        
        fbank = np.zeros((n_mels, n_fft // 2 + 1))
        for i in range(n_mels):
            for j in range(bin_points[i], bin_points[i+1]):
                fbank[i, j] = (j - bin_points[i]) / (bin_points[i+1] - bin_points[i])
            for j in range(bin_points[i+1], bin_points[i+2]):
                fbank[i, j] = (bin_points[i+2] - j) / (bin_points[i+2] - bin_points[i+1])
        
        self.register_buffer('mel_basis', torch.from_numpy(fbank).float())
        self.register_buffer('window', torch.hann_window(win_length))
        
    def forward(self, x):
        # x: (Batch, Time)
        stft = torch.stft(
            x, 
            n_fft=self.n_fft, 
            hop_length=self.hop_length, 
            win_length=self.win_length,
            window=self.window.to(x.device),
            center=True,
            return_complex=True
        )
        power_spec = stft.abs().pow(2.0)
        return torch.matmul(self.mel_basis.to(x.device), power_spec)
