def get_dataloaders(batch_size=config.BATCH_SIZE, num_workers=2):
    """
    Prepares DataLoaders with balanced sampling for training.
    """
    # 1. Load Metadata
    train_df = pd.read_csv(config.TRAIN_CSV)
    val_df = pd.read_csv(config.VAL_CSV)
    test_df = pd.read_csv(config.TEST_CSV)

    # 2. Prepare Weighted Sampler for Training
    # We use the full dataset (no dropping unknown) and balance via sampling
    # Cite solution_lesson_node_00022: Maximize Negative Class Diversity via Weighted Sampling
    
    # Calculate weights per class
    label_counts = train_df["label"].value_counts()
    # Weight = 1.0 / count
    class_weights = {label: 1.0 / count for label, count in label_counts.items()}
    
    # Assign weight to each sample
    sample_weights = [class_weights[label] for label in train_df["label"]]
    sample_weights = torch.DoubleTensor(sample_weights)
    
    # Create Sampler
    # Replacement=True allows oversampling of minority classes (silence, commands)
    # and "undersampling" (probabilistically) of majority class (unknown) in a single epoch
    sampler = WeightedRandomSampler(
        weights=sample_weights,
        num_samples=len(train_df),
        replacement=True
    )

    print(f"Train Dataset Size: {len(train_df)}")
    print(f"  Unique Labels: {len(label_counts)}")

    # We pass the full train_df

    test_dataset = SpeechDataset(test_df, is_train=False)

    # 4. Create DataLoaders
    # Note: shuffle must be False when using a sampler

        train_dataset,

        test_dataset,

        shuffle=False,
        num_workers=num_workers,
        pin_memory=True,
    )
