class MelSpectrogram(nn.Module):
    """
    Pure PyTorch implementation of MelSpectrogram to avoid torchaudio dependency.
    """
    def __init__(self, sample_rate, n_fft, win_length, hop_length, n_mels, fmin=0.0, fmax=None):
        super().__init__()
        self.sample_rate = sample_rate
        self.n_fft = n_fft
        self.win_length = win_length
        self.hop_length = hop_length
        self.n_mels = n_mels
        self.fmin = fmin
        self.fmax = fmax if fmax is not None else float(sample_rate) / 2

        # Create Mel Basis and Window
        mel_basis = self._create_mel_basis()
        window = torch.hann_window(win_length)
        
        # Register as buffers to ensure they move to GPU with the module
        self.register_buffer("mel_basis", mel_basis)
        self.register_buffer("window", window)

    def _create_mel_basis(self):
        # Create Mel filterbank matrix using numpy
        # Formula: m = 2595 * log10(1 + f / 700)
        def hz_to_mel(hz):
            return 2595.0 * np.log10(1.0 + hz / 700.0)
        
        def mel_to_hz(mel):
            return 700.0 * (10.0**(mel / 2595.0) - 1.0)
        
        mel_min = hz_to_mel(self.fmin)
        mel_max = hz_to_mel(self.fmax)
        
        mel_points = np.linspace(mel_min, mel_max, self.n_mels + 2)
        hz_points = mel_to_hz(mel_points)
        
        # FFT bins
        n_freqs = self.n_fft // 2 + 1
        bin_points = np.floor((self.n_fft + 1) * hz_points / self.sample_rate).astype(int)
        
        filters = np.zeros((self.n_mels, n_freqs))
        
        for i in range(1, self.n_mels + 1):
            left = bin_points[i - 1]
            center = bin_points[i]
            right = bin_points[i + 1]
            
            if center > left:
                filters[i - 1, left:center] = (np.arange(left, center) - left) / (center - left)
            if right > center:
                filters[i - 1, center:right] = (right - np.arange(center, right)) / (right - center)
                
        return torch.from_numpy(filters).float()

    def forward(self, x):
        # x: (Batch, Time)
        if self.window.device != x.device:
            self.window = self.window.to(x.device)
            
        # STFT: (Batch, Freq, Frames) - Complex
        stft = torch.stft(
            x, 
            n_fft=self.n_fft, 
            hop_length=self.hop_length, 
            win_length=self.win_length, 
            window=self.window, 
            center=True, 
            return_complex=True
        )
        
        # Power Spectrogram: |STFT|^2
        power_spec = stft.abs().pow(2.0)
        
        # Mel Spectrogram: (Batch, n_mels, Frames)
        # Basis: (n_mels, n_freqs) @ (Batch, n_freqs, Frames) -> (Batch, n_mels, Frames)
        mel_spec = torch.matmul(self.mel_basis, power_spec)
        
        return mel_spec


class AmplitudeToDB(nn.Module):
    """
    Pure PyTorch implementation of AmplitudeToDB.
    """
    def __init__(self, top_db=80.0):
        super().__init__()
        self.top_db = top_db
        self.amin = 1e-10
        self.multiplier = 10.0

    def forward(self, x):
        # x: Power spectrogram
        x_db = self.multiplier * torch.log10(torch.clamp(x, min=self.amin))
        
        if self.top_db is not None:
            x_db = torch.maximum(x_db, x_db.max() - self.top_db)
            
        return x_db
