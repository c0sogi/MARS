class GaussianNoiseLayer(nn.Module):
    """
    Custom layer that injects Gaussian noise into the input tensor during training.
    Acts as a regularizer by smoothing the decision boundary.
    """

    def __init__(self, sigma=0.1):
        """
        Args:
            sigma (float): Standard deviation of the Gaussian noise.
        """
        super().__init__()
        self.sigma = sigma

    def forward(self, x):
        if self.training and self.sigma > 0:
            noise = torch.randn_like(x) * self.sigma
            return x + noise
        return x

        dropout_rate=Config.DROPOUT_RATE,
        noise_sigma=Config.NOISE_SIGMA,
    ):
        """
        Args:
            vocab_sizes (list[int]): List containing the vocabulary size for each categorical feature.
            num_cont_features (int): Number of continuous input features.
            embed_dim (int): Dimension of the embedding vectors.
            hidden_layers (list[int]): List defining the width of each hidden layer.
            dropout_rate (float): Probability of dropout.
            noise_sigma (float): Standard deviation for the Gaussian Noise layer.
        """
        super().__init__()

        # 1. Entity Embeddings for Categorical Features
        # We use a ModuleList to hold an embedding layer for each categorical column
        self.embeddings = nn.ModuleList(
            [
                nn.Embedding(num_embeddings=v, embedding_dim=embed_dim)
                for v in vocab_sizes
            ]
        )

        # Input Dim = (Num_Categorical * Embed_Dim) + Num_Continuous

        # 2. Gaussian Noise Injection Layer
        # Applied immediately after feature concatenation
        self.noise_layer = GaussianNoiseLayer(sigma=noise_sigma)

        # 3. Funnel MLP Backbone
        layers = []
        in_dim = total_input_dim

            layers.append(nn.Linear(in_dim, h_dim))
            layers.append(nn.ReLU())
            layers.append(nn.Dropout(dropout_rate))
            in_dim = h_dim

        self.mlp = nn.Sequential(*layers)

        # 4. Output Head
        # Projects the final hidden representation to a single logit
        self.head = nn.Linear(in_dim, 1)

        Args:
            cat_x (torch.Tensor): Categorical features (Batch, Num_Cat_Cols). LongTensor.

            torch.Tensor: Logits (Batch, 1).
