class GaussianNoiseLayer(nn.Module):
    """
    Custom layer that injects Gaussian noise into the input tensor during training.
    Acts as a regularizer by smoothing the decision boundary.
    """

    def __init__(self, sigma=0.1):
        """
        Args:
            sigma (float): Standard deviation of the Gaussian noise.
        """
        super().__init__()
        self.sigma = sigma

    def forward(self, x):
        if self.training and self.sigma > 0:
            noise = torch.randn_like(x) * self.sigma
            return x + noise
        return x
