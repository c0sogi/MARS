    def compute_kinematics(
        self, skeleton: np.ndarray
    ) -> Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
        """
        Computes Relative Pos, Bone Vectors, Velocity, Acceleration.
        skeleton: (T, J, 3)
        """
        # 1. Root Relative (HipCenter at index 0 is root)
        root = skeleton[:, 0:1, :]  # (T, 1, 3)
        rel_pos = skeleton - root

        # 2. Bone Vectors
        bone_vecs = []
        for p, c in Config.BONE_PAIRS:
            # Vector from parent to child
            bv = skeleton[:, c, :] - skeleton[:, p, :]
            bone_vecs.append(bv)
        bone_vecs = np.stack(bone_vecs, axis=1)  # (T, NumBones, 3)

        # 3. Velocity (First difference)
        vel = np.zeros_like(rel_pos)
        vel[1:] = rel_pos[1:] - rel_pos[:-1]

        # 4. Acceleration (Second difference)
        acc = np.zeros_like(vel)
        acc[1:] = vel[1:] - vel[:-1]

        return rel_pos, bone_vecs, vel, acc
