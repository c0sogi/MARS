class GestureDataset(Dataset):
    """
    PyTorch Dataset for Gesture Recognition.
    Supports both windowed data (training) and full sequences (validation/testing).
    """

    def __init__(self, features, labels=None, sample_ids=None, augment=False):
        """
        Args:
            features (list of np.ndarray): List of feature arrays.
            labels (list of np.ndarray, optional): List of label arrays.
            sample_ids (list of str, optional): List of sample IDs.
            augment (bool): Whether to apply geometric augmentation.
        """
        self.features = features
        self.labels = labels
        self.sample_ids = sample_ids
        self.augment = augment

    def augment_skeleton(self, x):
        """
        Applies random rotation (Y-axis) and scaling to skeleton features.
        Skeleton features are the first 180 columns: [Pos(60), Vel(60), Acc(60)].
        """
        # x shape: (T, 193)
        skel = x[:, :180].copy()
        audio = x[:, 180:]

        # Random Rotation around Y (-15 to +15 degrees approx)
        theta = np.random.uniform(-0.26, 0.26)
        c, s = np.cos(theta), np.sin(theta)
        # Rotation matrix for [x, y, z]
        rot_mat = np.array([[c, 0, s], [0, 1, 0], [-s, 0, c]])

        # Random Scaling (0.9 to 1.1)
        scale = np.random.uniform(0.9, 1.1)

        # Apply to Pos, Vel, Acc blocks
        # Each block is 60 columns (20 joints * 3 coords)
        for block_start in [0, 60, 120]:
            block = skel[:, block_start : block_start + 60]
            # Reshape to (N_points, 3)
            reshaped = block.reshape(-1, 3)
            # Rotate: v' = v @ R.T
            rotated = np.dot(reshaped, rot_mat.T)
            # Scale
            scaled = rotated * scale
            # Reshape back
            skel[:, block_start : block_start + 60] = scaled.reshape(block.shape)

        return np.concatenate([skel, audio], axis=1)

    def __len__(self):
        return len(self.features)

    def __getitem__(self, idx):
        x_np = self.features[idx]

        # Apply augmentation if enabled
        if self.augment:
            x_np = self.augment_skeleton(x_np)

        # Convert to torch tensors
        x = torch.from_numpy(x_np).float()

        item = {"features": x}

        if self.labels is not None:
            y = torch.from_numpy(self.labels[idx]).long()
            item["labels"] = y

        if self.sample_ids is not None:
            item["sample_id"] = self.sample_ids[idx]

        return item
