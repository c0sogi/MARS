class DSPRNet(nn.Module):
    """
    Dual-Stream Point-Wise Residual Network.
    Cite Lesson 00052: Dual-Stream Residuals for Strong Autoregressive Signals.
    """

    def __init__(self):
        super(DSPRNet, self).__init__()
        self.image_encoder = ImageEncoder()

        # Stream A: Deep Interaction (Image + Clinical)
        # Input: Image(64) + Clinical(6) = 70
        self.deep_stream = nn.Sequential(
            nn.Linear(Config.IMG_EMBED_DIM + Config.CLINICAL_INPUT_DIM, 128),
            nn.ReLU(),
            nn.Linear(128, 64),
        )

        # Stream B: Linear Residual (Baseline + Time)
        # Input: Baseline_FVC(1) + Relative_Time(1) = 2
        # Cite Lesson 00060: Over-Parameterization of Linear Baselines (Project to Latent)
        self.linear_stream = nn.Linear(2, 64)

        # Shared Head for Prediction
        # Cite Lesson 00055: Shared fusion head ensures uncertainty sees strong priors
        self.head = nn.Linear(64, 2)

    def forward(self, image, clinical):
        """
        Args:
            image (torch.Tensor): (B, 3, 260, 260)
            clinical (torch.Tensor): (B, 6)
        Returns:
            mu (torch.Tensor): Predicted FVC
            sigma (torch.Tensor): Predicted Confidence
        """
        # 1. Image Feature Extraction
        img_embed = self.image_encoder(image)  # (B, 64)

        # 2. Stream A: Deep Interaction
        # Concatenate Image + All Clinical Features
        deep_input = torch.cat([img_embed, clinical], dim=1)
        h_deep = self.deep_stream(deep_input)  # (B, 64)

        # 3. Stream B: Linear Residual
        # Extract Baseline_FVC (idx 0) and Relative_Time (idx 5)
        # clinical columns: [Baseline_FVC, Baseline_Percent, Age, Sex_Code, Smoking_Code, Relative_Time]
        linear_input = clinical[:, [0, 5]]
        h_linear = self.linear_stream(linear_input)  # (B, 64)

        # 4. Latent Summation
        # Cite Lesson 00052: Sum outputs in latent space
        h_final = h_deep + h_linear

        # 5. Prediction Head
        out = self.head(h_final)

        mu = out[:, 0]
        raw_sigma = out[:, 1]

        # Uncertainty Constraint
        # Cite Lesson 00014: Do not enforce domain-specific clipping in architecture
        sigma = F.softplus(raw_sigma) + 1e-6

        return mu, sigma
