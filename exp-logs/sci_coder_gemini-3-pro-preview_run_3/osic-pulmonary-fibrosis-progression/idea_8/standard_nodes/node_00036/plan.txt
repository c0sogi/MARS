def compute_patient_slopes(df):
    """
    Computes the standardized slope (gamma) for each patient using OLS.
    Returns a DataFrame with Patient and Slope.
    """
    # Z-score normalize target and time for slope calculation
    # to match the model's latent space expectations
    df_norm = df.copy()
    df_norm["FVC_scaled"] = (df["FVC"] - STATS["FVC_MEAN"]) / STATS["FVC_STD"]
    df_norm["Weeks_scaled"] = (df["Weeks"] - STATS["WEEKS_MEAN"]) / STATS["WEEKS_STD"]

    slopes = []
    patients = df_norm["Patient"].unique()

    for p in patients:
        p_data = df_norm[df_norm["Patient"] == p]
        if len(p_data) < 2:
            slope = 0.0
        else:
            X = p_data["Weeks_scaled"].values.reshape(-1, 1)

            reg = LinearRegression().fit(X, y)

        slopes.append({"Patient": p, "Slope": slope})

class TabularEncoder(nn.Module):
    """
    Tabular branch processing clinical metadata.
    Explicitly excludes Batch/Layer Norm to preserve magnitude signals of Baseline FVC.
    Input: (B, 7) -> [Base_FVC, Base_Percent, Base_Age, Sex, Smoke_Ex, Smoke_Never, Smoke_Current]
    Output: (B, EMBEDDING_DIM)
    """

    def __init__(self, input_dim=7, embedding_dim=EMBEDDING_DIM):
        super(TabularEncoder, self).__init__()
        self.mlp = nn.Sequential(
            nn.Linear(input_dim, 256),
            nn.ReLU(),
            nn.Linear(256, embedding_dim),
            nn.ReLU(),
        )

    def forward(self, x):
        return self.mlp(x)

    Time-Aware Parametric Network (TAP-Net).

            nn.ReLU(),

            nn.ReLU(),
            nn.Linear(256, 5),

    def forward(self, image, tabular):
        """
        Args:
            image: (B, 3, H, W)
            tabular: (B, 7)

        Returns:
            params: (B, 5) containing raw parameters:
                0: alpha (Baseline Autoregressive Coefficient)
                1: beta (Intercept Adjustment)
                2: gamma (Rate of Decline/Slope)
                3: delta_base (Baseline Uncertainty)

        img_emb = self.image_encoder(image)
        tab_emb = self.tabular_encoder(tabular)

        # Concatenate embeddings
        combined = torch.cat([img_emb, tab_emb], dim=1)

        # Predict parameters
        params = self.head(combined)

        return params

        """
