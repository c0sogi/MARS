I will implement the **Dual-Stream Point-Wise Residual Network (DSPRNet)** architecture. This approach separates the modeling of the strong linear autoregressive signal (Baseline FVC and Time) from the complex non-linear interactions (Image features and other metadata). By summing these streams in a latent space, we allow the model to robustly capture the dominant trend while focusing its deep capacity on image-based residuals (Cite solution_lesson_node_00052). I will also ensure the linear stream is over-parameterized (projected to latent space) to avoid optimization bottlenecks (Cite solution_lesson_node_00060). Additionally, I will increase the training epochs to 25 to accommodate the slower convergence of uncertainty estimation (Cite solution_lesson_node_00011).

class ClinicalStreamA(nn.Module):
    """
    Stream A: Over-Parameterized Clinical Anchor.
    Learns the expected clinical trajectory based on tabular data.
    """

    def __init__(self):
        super(ClinicalStreamA, self).__init__()
        # Inputs: Baseline_FVC, Baseline_Percent, Age, Sex, Smoking, Relative_Weeks
        input_dim = 6

        self.net = nn.Sequential(
            nn.Linear(input_dim, Config.CLINICAL_HIDDEN_DIM),  # 128
            nn.ReLU(),
            nn.Linear(Config.CLINICAL_HIDDEN_DIM, Config.LATENT_DIM),  # 64
        )

    def forward(self, x):
        return self.net(x)

class CCVRNet(nn.Module):
    """
    Cascaded Clinical-Visual Residual Network.
    """

    def __init__(self):
        super(CCVRNet, self).__init__()
        self.visual_branch = VisualBranch()
        self.stream_a = ClinicalStreamA()
        self.stream_b = VisualStreamB()

        # Shared Head: Projects H_final to mu and sigma
        self.head = nn.Linear(Config.LATENT_DIM, 2)

    def forward(self, image, tabular):
        """
        Args:
            image: Tensor (Batch, 3, H, W)
            tabular: Tensor (Batch, 6)
        Returns:
            mu: Predicted FVC (scaled)
            sigma: Predicted Uncertainty (scaled)
        """
        # 1. Get Visual Embedding (E_img)
        img_embed = self.visual_branch(image)  # (B, 64)

        # 2. Get Clinical Anchor Embedding (H_clin) - Stream A
        clin_embed = self.stream_a(tabular)  # (B, 64)

        # 3. Get Visual Interaction Embedding (H_inter) - Stream B
        # Cascade: Condition visual processing on clinical state
        combined_input = torch.cat([img_embed, clin_embed], dim=1)  # (B, 128)
        inter_embed = self.stream_b(combined_input)  # (B, 64)

        # 4. Residual Fusion
        # H_final = StreamA + StreamB
        h_final = clin_embed + inter_embed  # (B, 64)

        # 5. Prediction Head
        out = self.head(h_final)  # (B, 2)

        mu = out[:, 0]
        raw_sigma = out[:, 1]

        # Uncertainty Constraint
        # Apply softplus to ensure positivity, add epsilon for stability
        sigma = F.softplus(raw_sigma) + 1e-6

        return mu, sigma

class DSPRNet(nn.Module):
    """
    Dual-Stream Point-Wise Residual Network (Cite solution_lesson_node_00052).
    Combines a deep interaction stream (Image + Tabular) with a linear residual stream (Baseline + Time).
    """

    def __init__(self):
        super(DSPRNet, self).__init__()
        self.visual_branch = VisualBranch()

        # Stream A: Deep Interaction (Image + Full Tabular)
        # Tabular encoder
        self.tab_encoder = nn.Linear(6, Config.LATENT_DIM)  # 6 -> 64

        # Interaction MLP
        self.deep_mlp = nn.Sequential(
            nn.Linear(Config.IMG_EMBED_DIM + Config.LATENT_DIM, 128),
            nn.ReLU(),
            nn.Linear(128, Config.LATENT_DIM),
        )

        # Stream B: Linear Residual (Baseline FVC + Weeks)
        # Projected to latent space (Cite solution_lesson_node_00060)
        self.linear_stream = nn.Linear(2, Config.LATENT_DIM)

        # Shared Head
        self.head = nn.Linear(Config.LATENT_DIM, 2)

    def forward(self, image, tabular):
        # 1. Visual Features
        img_embed = self.visual_branch(image)  # (B, 64)

        # 2. Stream A: Deep Interaction
        tab_embed = F.relu(self.tab_encoder(tabular))  # (B, 64)
        deep_in = torch.cat([img_embed, tab_embed], dim=1)
        deep_latent = self.deep_mlp(deep_in)  # (B, 64)

        # 3. Stream B: Linear Residual
        # Inputs: Baseline_FVC (idx 0) and Relative_Weeks (idx 5)
        linear_input = tabular[:, [0, 5]]
        linear_latent = self.linear_stream(linear_input)  # (B, 64)

        # 4. Fusion (Summation in latent space)
        h_final = deep_latent + linear_latent

        # 5. Prediction Head
        out = self.head(h_final)
        mu = out[:, 0]
        sigma = F.softplus(out[:, 1]) + 1e-6

        return mu, sigma
