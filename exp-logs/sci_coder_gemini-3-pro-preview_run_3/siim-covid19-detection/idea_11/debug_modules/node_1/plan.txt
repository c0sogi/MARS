def collate_fn(batch):
    """
    Custom collate function for the DataLoader.
    Handles variable number of boxes per image.
    Stacks images into a NestedTensor.
    """
    batch_zip = list(zip(*batch))
    images = batch_zip[0]  # tuple of tensors
    targets = list(batch_zip[1])  # tuple of dicts

    # Stack images: (B, C, H, W)
    # They are already resized to Config.IMG_SIZE in dataset
    tensor = torch.stack(images, dim=0)

    # Create mask: (B, H, W)
    # False indicates valid pixel.
    # Since we don't have padding info here, we assume all valid.
    # (The black padding from letterbox is treated as valid pixels with value 0)
    mask = torch.zeros(
        (tensor.shape[0], tensor.shape[2], tensor.shape[3]), dtype=torch.bool
    )

    return NestedTensor(tensor, mask), targets
