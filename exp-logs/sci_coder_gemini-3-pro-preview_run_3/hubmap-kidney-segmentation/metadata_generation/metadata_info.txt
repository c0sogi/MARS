import os
import pandas as pd
import numpy as np
from sklearn.model_selection import GroupShuffleSplit


def generate_metadata():
    # Define input and output directories
    INPUT_DIR = "./input"
    METADATA_DIR = "./metadata"

    os.makedirs(METADATA_DIR, exist_ok=True)

    # Load raw CSVs
    train_rle_path = os.path.join(INPUT_DIR, "train.csv")
    info_path = os.path.join(INPUT_DIR, "HuBMAP-20-dataset_information.csv")
    sample_sub_path = os.path.join(INPUT_DIR, "sample_submission.csv")

    # Read data
    # train.csv contains 'id' and 'encoding' (RLE)
    train_rle_df = pd.read_csv(train_rle_path)

    # dataset_info contains 'image_file', 'patient_number', etc.
    info_df = pd.read_csv(info_path)

    # sample_submission contains 'id' for test set
    test_df = pd.read_csv(sample_sub_path)

    # Preprocess info_df to get 'id' from 'image_file' (remove extension)
    info_df["id"] = info_df["image_file"].apply(lambda x: os.path.splitext(x)[0])

    # --- Process Training Data ---
    # Merge RLE data with info to get patient numbers and other metadata
    # We use an inner join to ensure we have metadata for all training samples
    full_train_df = pd.merge(train_rle_df, info_df, on="id", how="inner")

    # Construct relative file paths for training data
    # Based on file structure, training images are in input/train/
    full_train_df["image_path"] = full_train_df["id"].apply(
        lambda x: os.path.join("input", "train", f"{x}.tiff")
    )
    full_train_df["json_path"] = full_train_df["id"].apply(
        lambda x: os.path.join("input", "train", f"{x}.json")
    )
    full_train_df["anatomical_json_path"] = full_train_df["id"].apply(
        lambda x: os.path.join("input", "train", f"{x}-anatomical-structure.json")
    )

    # --- Split Train/Val ---
    # We use GroupShuffleSplit to split by patient_number to avoid data leakage
    # 80:20 split, random_state=42
    gss = GroupShuffleSplit(n_splits=1, train_size=0.8, random_state=42)

    groups = full_train_df["patient_number"]
    train_idx, val_idx = next(gss.split(full_train_df, groups=groups))

    train_metadata = full_train_df.iloc[train_idx].copy()
    val_metadata = full_train_df.iloc[val_idx].copy()

    # Add split label for clarity (optional, but good for merged debugging)
    train_metadata["split"] = "train"
    val_metadata["split"] = "validation"

    # --- Process Test Data ---
    # Merge test IDs with info if available, otherwise just use IDs
    # Note: dataset_info might contain test images too.
    test_metadata = pd.merge(test_df[["id"]], info_df, on="id", how="left")

    # Construct relative file paths for test data
    # Based on file structure, test images are in input/test/
    test_metadata["image_path"] = test_metadata["id"].apply(
        lambda x: os.path.join("input", "test", f"{x}.tiff")
    )
    # Test set usually doesn't have the ground truth glomerulus json provided in the same way or it's hidden,
    # but the anatomical structure is often provided.
    test_metadata["anatomical_json_path"] = test_metadata["id"].apply(
        lambda x: os.path.join("input", "test", f"{x}-anatomical-structure.json")
    )
    # The main json might exist in the folder (as per file listing), so we add it, but it might not be used for prediction.
    test_metadata["json_path"] = test_metadata["id"].apply(
        lambda x: os.path.join("input", "test", f"{x}.json")
    )

    test_metadata["split"] = "test"

    # --- Save Metadata ---
    train_save_path = os.path.join(METADATA_DIR, "train.csv")
    val_save_path = os.path.join(METADATA_DIR, "val.csv")
    test_save_path = os.path.join(METADATA_DIR, "test.csv")

    train_metadata.to_csv(train_save_path, index=False)
    val_metadata.to_csv(val_save_path, index=False)
    test_metadata.to_csv(test_save_path, index=False)

    print(f"Metadata saved to {METADATA_DIR}")

    return train_save_path, val_save_path, test_save_path

Based on the analysis of the provided Python script and its execution, here is the detailed documentation for the files generated in the `./metadata` directory.

### 1. `./metadata/train.csv`

*   **Content and Purpose**:
    *   This file contains the metadata and ground truth information for the training subset of the dataset.
    *   It includes the Run-Length Encoded (RLE) segmentation masks for functional tissue units (FTUs), patient demographics, tissue characteristics, and file paths for images and annotations.
    *   **Purpose**: This file is the primary index for training machine learning models. It allows data loaders to map image IDs to their corresponding TIFF images, JSON annotations, and target masks while referencing patient IDs to ensure group-based splitting.

*   **Schema / Structure**:
    *   **Format**: CSV (Comma Separated Values)
    *   **Key Columns**:
        *   `id`: (String) Unique identifier for the image (e.g., `afa5e8098`).
        *   `encoding`: (String) RLE-encoded binary mask of the glomeruli (target).
        *   `patient_number`: (Integer) Unique ID for the patient; used to group data for cross-validation.
        *   `image_path`: (String) Relative path to the TIFF image file starting from the input directory (e.g., `train/afa5e8098.tiff`).
        *   `json_path`: (String) Relative path to the detailed JSON segmentation file (e.g., `train/afa5e8098.json`).
        *   `anatomical_json_path`: (String) Relative path to the anatomical structure segmentation file (e.g., `train/afa5e8098-anatomical-structure.json`).
        *   **Demographics & Metadata**: `age`, `sex`, `race`, `ethnicity`, `weight_kilograms`, `bmi_kg/m^2`, `laterality`, `percent_cortex`, `percent_medulla`, `width_pixels`, `height_pixels`.

*   **Loading Method**:
    ```python
    import pandas as pd
    df_train = pd.read_csv("./metadata/train.csv")
    
    # Example access to a full image path
    import os
    full_image_path = os.path.join("./input", df_train.iloc[0]["image_path"])
    ```

### 2. `./metadata/val.csv`

*   **Content and Purpose**:
    *   This file contains the metadata for the validation subset of the dataset.
    *   It is structurally identical to `train.csv` but consists of a distinct set of patients (determined by `patient_number`) held out from the training process.
    *   **Purpose**: Used to evaluate model performance during training. The split ensures that the model is validated on tissue samples from patients it has never seen before, preventing data leakage.

*   **Schema / Structure**:
    *   **Format**: CSV
    *   **Key Columns**: Same as `train.csv` (`id`, `encoding`, `patient_number`, `image_path`, etc.).
    *   **Note**: The split ratio defined in the script is 80% training and 20% validation.

*   **Loading Method**:
    ```python
    import pandas as pd
    df_val = pd.read_csv("./metadata/val.csv")
    ```

### 3. `./metadata/test.csv`

*   **Content and Purpose**:
    *   This file contains metadata for the test set images for which predictions must be generated.
    *   It includes file paths and image IDs but does not typically rely on the `encoding` column for input (though the column structure may exist if merged from source info).
    *   **Purpose**: Used by the inference pipeline to locate test images and their corresponding anatomical structure files to generate submission predictions.

*   **Schema / Structure**:
    *   **Format**: CSV
    *   **Key Columns**:
        *   `id`: (String) Unique identifier for the test image.
        *   `image_path`: (String) Relative path to the test TIFF image (e.g., `test/0486052bb.tiff`).
        *   `anatomical_json_path`: (String) Relative path to the anatomical structure JSON file.
        *   `json_path`: (String) Relative path to the JSON file (if available).
        *   **Metadata**: May contain demographic or image dimension columns (`width_pixels`, `height_pixels`, etc.) derived from the dataset information file.
