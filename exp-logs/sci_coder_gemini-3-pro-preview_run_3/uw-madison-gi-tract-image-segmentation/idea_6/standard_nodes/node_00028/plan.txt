class CurriculumLoss(nn.Module):
    """
    Wrapper class that implements the curriculum learning strategy and handles Deep Supervision.
    """

    def __init__(self, config):
        super(CurriculumLoss, self).__init__()
        self.config = config

        # Phase 1 Loss: BCE + Tversky (Dice)
        self.bce_tversky = BCETverskyLoss(alpha=0.5, beta=0.5, bce_weight=0.5)

        # Phase 2 Loss: Boundary
        self.boundary = BoundaryLoss()

        # Weight for boundary loss in the refinement phase
        # Kept small because SDF values can be significant
        self.boundary_weight = 0.05

    def forward(self, preds, targets, epoch):
        """
        Args:
            preds: Model output. Can be a tensor (B, C, H, W) or a list of tensors (Deep Supervision).
            targets: Ground truth mask (B, C, H, W).
            epoch: Current training epoch (0-indexed).
        """
        # Handle Deep Supervision (List of outputs)
        if isinstance(preds, (list, tuple)):
            loss = 0
            # We average the loss across all decoder levels
            # Alternatively, we could weight deeper layers higher, but simple average is robust
            for p in preds:
                loss += self._compute_single_loss(p, targets, epoch)
            return loss / len(preds)
        else:
            return self._compute_single_loss(preds, targets, epoch)

    def _compute_single_loss(self, pred, target, epoch):
        # Resize target if prediction shape doesn't match (e.g., lower resolution decoder outputs)
        if pred.shape[2:] != target.shape[2:]:
            target = F.interpolate(target, size=pred.shape[2:], mode="nearest")

        base_loss = self.bce_tversky(pred, target)

        if epoch >= self.config.WARMUP_EPOCHS:
            bd_loss = self.boundary(pred, target)
            return base_loss + self.boundary_weight * bd_loss

        return base_loss
