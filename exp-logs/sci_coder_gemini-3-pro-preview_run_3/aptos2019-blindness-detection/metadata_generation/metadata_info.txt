# Python Script for Metadata Generation
```
import os
import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split


def main():
    # Configuration
    INPUT_DIR = "./input"
    METADATA_DIR = "./metadata"
    RANDOM_STATE = 42
    VAL_SIZE = 0.2

    # Create metadata directory
    os.makedirs(METADATA_DIR, exist_ok=True)

    print("Loading raw data...")
    # Load raw csvs
    train_df_orig = pd.read_csv(os.path.join(INPUT_DIR, "train.csv"))
    test_df_orig = pd.read_csv(os.path.join(INPUT_DIR, "test.csv"))

    # Add relative file paths
    # Note: The requirement is paths relative to ./input
    train_df_orig["file_path"] = train_df_orig["id_code"].apply(
        lambda x: os.path.join("train_images", f"{x}.png")
    )
    test_df_orig["file_path"] = test_df_orig["id_code"].apply(
        lambda x: os.path.join("test_images", f"{x}.png")
    )

    # Perform Stratified Split
    print("Splitting data into train and validation sets...")
    train_df, val_df = train_test_split(
        train_df_orig,
        test_size=VAL_SIZE,
        stratify=train_df_orig["diagnosis"],
        random_state=RANDOM_STATE,
        shuffle=True,
    )

    # Save metadata
    print("Saving metadata...")
    train_path = os.path.join(METADATA_DIR, "train.csv")
    val_path = os.path.join(METADATA_DIR, "val.csv")
    test_path = os.path.join(METADATA_DIR, "test.csv")

    train_df.to_csv(train_path, index=False)
    val_df.to_csv(val_path, index=False)
    test_df_orig.to_csv(test_path, index=False)

    print("Metadata generation complete.")

    # ==========================================
    # Verification Steps
    # ==========================================
    print("\nStarting verification...")

    # Reload datasets
    df_train = pd.read_csv(train_path)
    df_val = pd.read_csv(val_path)
    df_test = pd.read_csv(test_path)

    # 1. Summary Statistics
    print("\n--- Summary Statistics ---")
    print(f"Train set size: {len(df_train)}")
    print(f"Validation set size: {len(df_val)}")
    print(f"Test set size: {len(df_test)}")

    print("\nTrain Class Distribution:")
    print(df_train["diagnosis"].value_counts(normalize=True).sort_index())

    print("\nValidation Class Distribution:")
    print(df_val["diagnosis"].value_counts(normalize=True).sort_index())

    # 2. Check File Paths
    def check_paths(df, name):
        print(f"\nChecking file paths for {name}...")
        # Sample up to 1000 paths
        n_sample = min(1000, len(df))
        sample_paths = (
            df["file_path"].sample(n=n_sample, random_state=RANDOM_STATE).tolist()
        )

        missing_count = 0
        missing_samples = []

        for rel_path in sample_paths:
            # Construct full path based on requirement that metadata paths are relative to ./input
            full_path = os.path.join(INPUT_DIR, rel_path)
            if not os.path.exists(full_path):
                missing_count += 1
                if len(missing_samples) < 5:
                    missing_samples.append(rel_path)

        missing_ratio = missing_count / n_sample
        print(
            f"Missing file ratio for {name}: {missing_ratio:.4f} ({missing_count}/{n_sample})"
        )

        if missing_ratio > 0.5:
            print("Sample of missing paths:")
            for p in missing_samples:
                print(f"  {p}")
            raise FileNotFoundError(
                f"Error: More than 50% of file paths in {name} are missing."
            )

    check_paths(df_train, "Train")
    check_paths(df_val, "Validation")
    check_paths(df_test, "Test")

    # 3. Verify Stratification
    print("\nVerifying stratification...")
    train_dist = df_train["diagnosis"].value_counts(normalize=True).sort_index()
    val_dist = df_val["diagnosis"].value_counts(normalize=True).sort_index()

    # Calculate maximum absolute difference in class proportions
    # Since classes 0-4 should exist, we align them
    all_classes = sorted(train_df_orig["diagnosis"].unique())

    for c in all_classes:
        t_prop = train_dist.get(c, 0)
        v_prop = val_dist.get(c, 0)
        diff = abs(t_prop - v_prop)

        # We allow a small tolerance. With small datasets, exact stratification isn't always possible,
        # but with ~3000 samples, it should be close.
        # Tolerance of 0.05 (5%) is generous but sufficient to catch non-stratified random splits on imbalanced data.
        if diff > 0.05:
            raise AssertionError(
                f"Stratification failed for class {c}. Train prop: {t_prop:.4f}, Val prop: {v_prop:.4f}, Diff: {diff:.4f}"
            )

    print("Stratification verification passed.")
    print("\nAll checks passed successfully.")


if __name__ == "__main__":
    main()

```

# Document
Based on the provided Python script and its execution output, here is the documentation for the files generated in the `./metadata` directory.

### 1. `./metadata/train.csv`

*   **Content and Purpose:**
    *   **Content:** This file contains the training subset of the dataset, representing 80% (2,636 samples) of the original labeled data. It is the result of a stratified split, ensuring the class distribution of the `diagnosis` label mirrors the original dataset. It includes the original ID, the diagnostic label, and the computed relative file path to the image.
    *   **Purpose:** This file is intended to be used as the primary input for training the machine learning model. The `file_path` column allows data loaders to locate the specific image files within the `./input/train_images` directory.
*   **Schema / Structure:**
    *   **Format:** CSV (Comma Separated Values)
    *   **Columns:**
        *   `id_code` (String): The unique identifier for the image (e.g., `000c1434d8d7`).
        *   `diagnosis` (Int64): The severity of diabetic retinopathy on a scale of 0 to 4.
        *   `file_path` (String): The relative path to the image file starting from the input root (e.g., `train_images/000c1434d8d7.png`).
    *   **Example Row:**
        ```text
        id_code,diagnosis,file_path
        000c1434d8d7,2,train_images/000c1434d8d7.png
        ```
*   **Loading Method:**
    *   Load with pandas: `pd.read_csv("./metadata/train.csv")`

### 2. `./metadata/val.csv`

*   **Content and Purpose:**
    *   **Content:** This file contains the validation subset of the dataset, representing the remaining 20% (659 samples) of the original labeled data. Like the training set, it is stratified to maintain class distribution balance.
    *   **Purpose:** This file is used to evaluate the model's performance during the training process (e.g., at the end of every epoch). It helps in tuning hyperparameters and monitoring for overfitting by providing a set of data the model has not seen during the gradient update steps.
*   **Schema / Structure:**
    *   **Format:** CSV
    *   **Columns:**
        *   `id_code` (String): The unique identifier for the image.
        *   `diagnosis` (Int64): The severity of diabetic retinopathy (0-4).
        *   `file_path` (String): The relative path to the image file (e.g., `train_images/19e350c7c83c.png`).
    *   **Example Row:**
        ```text
        id_code,diagnosis,file_path
        19e350c7c83c,0,train_images/19e350c7c83c.png
        ```
*   **Loading Method:**
    *   Load with pandas: `pd.read_csv("./metadata/val.csv")`

### 3. `./metadata/test.csv`

*   **Content and Purpose:**
    *   **Content:** This file contains metadata for the 367 images in the public test set. It maps the test image IDs to their specific file paths.
    *   **Purpose:** This file is used during the inference phase. The model uses the `file_path` to load images, predicts the diagnosis, and associates the prediction with the `id_code` to generate the final `submission.csv`.
*   **Schema / Structure:**
    *   **Format:** CSV
    *   **Columns:**
        *   `id_code` (String): The unique identifier for the test image.
        *   `file_path` (String): The relative path to the image file (e.g., `test_images/001639a390f0.png`).

    *   **Example Row:**
        ```text
        id_code,file_path
        001639a390f0,test_images/001639a390f0.png
        ```
*   **Loading Method:**
    *   Load with pandas: `pd.read_csv("./metadata/test.csv")`
