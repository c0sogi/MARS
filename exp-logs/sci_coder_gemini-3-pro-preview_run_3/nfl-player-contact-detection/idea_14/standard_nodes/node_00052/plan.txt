1.  **Reverting the Cascade Architecture**: Removing the dependency of Stream B on Stream A's predictions, as previous lessons indicated this introduced noise.

    def _add_kinematic_derivatives(self, df):
        """
        Adds higher-order kinematic derivatives (Jerk, Angular Velocity) and alignment features.
        Cite solution_lesson_node_00027: Enrich Kinematic Models with Pose-Motion Alignment and Higher-Order Derivatives.
        """
        # Ensure sorting for diff calculations
        # Assumes df is for a single stream (e.g. Stream B with p1)
        # We group by game_play and player_id
        
        # We only compute for p1 as Stream B focuses on p1 vs Ground (p2 is G)
        
        # Sort
        df = df.sort_values(["game_play", "nfl_player_id_1", "step"]).reset_index(drop=True)
        
        # Group
        grp = df.groupby(["game_play", "nfl_player_id_1"])
        
        # Jerk: Derivative of acceleration
        # acceleration is in yards/s^2. Step is 0.1s.
        if "acceleration_p1" in df.columns:
            df["jerk_p1"] = grp["acceleration_p1"].diff().fillna(0) / 0.1
            
        # Angular Velocity: Derivative of orientation
        # Use sin/cos diffs as proxies for angular velocity components to handle wrap-around.
        if "sin_orientation_p1" in df.columns:
            df["ang_vel_sin_p1"] = grp["sin_orientation_p1"].diff().fillna(0) / 0.1
            df["ang_vel_cos_p1"] = grp["cos_orientation_p1"].diff().fillna(0) / 0.1
            
        # Pose-Motion Alignment: Dot product of Orientation and Direction vectors
        if "sin_orientation_p1" in df.columns and "sin_direction_p1" in df.columns:
            df["cos_orient_dir_p1"] = (
                df["sin_orientation_p1"] * df["sin_direction_p1"] + 
                df["cos_orientation_p1"] * df["cos_direction_p1"]
            )
            
        return df

    def _add_pairwise_features(self, df):
        """
        Adds explicit relational features between P1 and P2.
        Cite solution_lesson_node_00051: Explicit Relational Features vs Implicit Raw Coordinates.
        """
        # Euclidean Distance
        df["dist_p1_p2"] = np.sqrt(
            (df["x_position_p1"] - df["x_position_p2"])**2 + 
            (df["y_position_p1"] - df["y_position_p2"])**2
        )
        
        # Relative Speed
        if "speed_p1" in df.columns and "speed_p2" in df.columns:
            df["rel_speed"] = np.abs(df["speed_p1"] - df["speed_p2"])
            
        return df

            preds.rename(columns={"contact": "prob"}, inplace=True)

        split_ids = preds["contact_id"].str.split("_", expand=True)

    def _generate_oof_stream_a(self, X, y, ids, groups, load_cached_data=True):
        """
        Generates Out-of-Fold predictions for Stream A to be used as context for Stream B.
        Uses GroupKFold to prevent leakage across plays.
        """
        # Determine valid number of splits based on available groups
        unique_groups = np.unique(groups)
        n_groups = len(unique_groups)
        n_splits = min(5, n_groups)

        if n_splits < 2:
            raise ValueError(
                f"Insufficient groups ({n_groups}) for Cross Validation. Need at least 2."
            )

        if n_splits < 5:
            print(
                f"Adjusting n_splits from 5 to {n_splits} due to limited groups (Debug Mode)."
            )

        cache_config = {
            "task": "oof_generation_stream_a",
            "n_splits": n_splits,
            "seed": Config.SEED,
            "debug": Config.DEBUG,
        }
        cache_path = get_cache_path(
            self.working_dir, "oof_preds_stream_a", cache_config, "parquet"
        )

        if load_cached_data and os.path.exists(cache_path):
            print(f"Loading cached OOF predictions from {cache_path}")
            return pd.read_parquet(cache_path)

        print(
            f"Generating OOF predictions for Stream A ({n_splits}-Fold GroupKFold)..."
        )

        # Initialize storage for OOF predictions
        oof_preds = []

        # GroupKFold Split
        gkf = GroupKFold(n_splits=n_splits)

        # We need to reconstruct a DataFrame to map predictions back to contact_ids
        # X is a DataFrame, y is numpy array, ids is numpy array

        fold = 1
        for train_idx, val_idx in gkf.split(X, y, groups=groups):
            print(f"  Processing Fold {fold}/{n_splits}...")

            # Slice data
            X_train_fold = X.iloc[train_idx]
            y_train_fold = y[train_idx]
            X_val_fold = X.iloc[val_idx]
            # y_val_fold = y[val_idx] # Not needed for prediction
            ids_val_fold = ids[val_idx]

            # Train Model
            model = XGBWrapper(Config.get_xgb_params("A"), model_path=None)
            model.fit(X_train_fold, y_train_fold)

            # Predict
            probs = model.predict(X_val_fold)

            # Store results
            fold_df = pd.DataFrame({"contact_id": ids_val_fold, "prob": probs})
