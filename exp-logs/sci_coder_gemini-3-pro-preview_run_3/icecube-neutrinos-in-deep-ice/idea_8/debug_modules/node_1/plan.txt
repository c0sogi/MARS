def angular_error(pred, true_azimuth, true_zenith):
    """
    Computes the mean angular error between predicted vectors and true angles.
    """
    # Pred is vector (x,y,z)
    pred = F.normalize(pred, p=2, dim=1)

    sa = np.sin(true_azimuth)
    ca = np.cos(true_azimuth)
    sz = np.sin(true_zenith)
    cz = np.cos(true_zenith)

    true_x = ca * sz
    true_y = sa * sz
    true_z = cz

    # Dot product
    dot = (
        pred[:, 0].cpu().numpy() * true_x
        + pred[:, 1].cpu().numpy() * true_y
        + pred[:, 2].cpu().numpy() * true_z
    )

    dot = np.clip(dot, -1.0, 1.0)
    return np.arccos(dot)
