def extract_wavelet_stats(signal):
    """
    View B (Texture): Extracts statistics from Discrete Wavelet Transform detail coefficients.
    """
    # Decompose signal
    # Using a level that ensures we get useful details.
    # For ~60k points, level 4 or 5 is reasonable. Defaulting to what works well for texture.
    coeffs = pywt.wavedec(signal, config.DWT_WAVELET, level=4)

    # coeffs[0] is approximation, coeffs[1:] are details (cD4, cD3, cD2, cD1)
    # We focus on detail coefficients for texture
    features = {}

    for i, coeff in enumerate(coeffs[1:], start=1):
        # Level index matches the detail level (inverted in output of wavedec usually,
        # but here we just index them sequentially as detail_1, detail_2 etc.)
        # wavedec returns [cA_n, cD_n, cD_n-1, ..., cD_1]
        level_name = f"dwt_detail_{len(coeffs)-i}"

        energy = np.sum(coeff**2)
        # Shannon Entropy of energy distribution
        # Normalize energy to treat as probability distribution
        p = (coeff**2) / (energy + 1e-10)
        entropy = -np.sum(p * np.log(p + 1e-10))

        features[f"{level_name}_energy"] = energy
        features[f"{level_name}_entropy"] = entropy
        features[f"{level_name}_std"] = np.std(coeff)
        features[f"{level_name}_skew"] = scipy.stats.skew(coeff)
        features[f"{level_name}_kurt"] = scipy.stats.kurtosis(coeff)

    return features
