def compute_wavelet_features(x):
    """
    Computes features from Discrete Wavelet Transform.
    """
    coeffs = pywt.wavedec(x, WAVELET_NAME, level=WAVELET_LEVELS)
    # coeffs is [cA_n, cD_n, cD_n-1, ..., cD_1]

    feat_dict = {}

    # Helper to get stats for a coefficient array
    def get_coeff_stats(c, name):
        energy = np.sum(c**2)
        # Entropy (Shannon) - handle negative values/zeros for log
        # Normalize to probability distribution
        p = np.abs(c)
        p_sum = np.sum(p)
        if p_sum > 0:
            p = p / p_sum
            # Add epsilon to avoid log(0)
            entropy = -np.sum(p * np.log(p + 1e-12))
        else:
            entropy = 0

        return {
            f"{name}_std": np.std(c),
            f"{name}_energy": energy,
            f"{name}_entropy": entropy,
        }

    # Approximation coefficients (Level n)
    feat_dict.update(get_coeff_stats(coeffs[0], f"wt_cA{WAVELET_LEVELS}"))

    # Detail coefficients
    for i, c in enumerate(coeffs[1:]):
        level = WAVELET_LEVELS - i
        feat_dict.update(get_coeff_stats(c, f"wt_cD{level}"))

    return feat_dict
