def compute_wavelet_features(signal):
    """
    Applies DWT (db4) and extracts energy and entropy from detail coefficients.
    """
    # Single level DWT
    cA, cD = pywt.dwt(signal, Config.WAVELET_NAME)

    # Energy of detail coefficients
    energy = np.sum(cD**2)

    # Entropy of detail coefficients (Shannon entropy of normalized energy)
    # Add epsilon to avoid log(0)
    p = cD**2 / (energy + 1e-10)
    entropy = -np.sum(p * np.log(p + 1e-10))

    return {"wavelet_energy": energy, "wavelet_entropy": entropy}
