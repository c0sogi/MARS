def extract_wavelet(series, prefix):
    """
    Applies Discrete Wavelet Transform and extracts energy/std from coefficients.
    """
    features = {}

    # Decompose
    # Level 3 decomposition with db4
    coeffs = pywt.wavedec(series, Config.WAVELET_TYPE, level=3)

    # coeffs[0] is approximation (cA3)
    # coeffs[1] is detail 3 (cD3), coeffs[2] is cD2, coeffs[3] is cD1
    names = ["cA3", "cD3", "cD2", "cD1"]

    for name, c in zip(names, coeffs):
        features[f"{prefix}_wav_{name}_energy"] = np.sum(c**2)
        features[f"{prefix}_wav_{name}_std"] = np.std(c)

    return features
