def get_texture_features(residuals, sensor_id):
    """
    Extracts texture statistics from DWT Detail Coefficients (View B).
    """
    feats = {}

    # Discrete Wavelet Transform
    # We use 'db4' and extract all coefficients
    coeffs = pywt.wavedec(residuals, SIGNAL_PARAMS.DWT_WAVELET)

    # Concatenate Detail Coefficients (exclude approximation coeffs[0])
    # This provides a comprehensive view of high-frequency texture
    if len(coeffs) > 1:
        details = np.concatenate(coeffs[1:])
    else:
        details = coeffs[0]  # Fallback if decomposition is shallow

    # Energy
    energy = np.sum(details**2)
    feats[f"{sensor_id}_txt_energy"] = energy

    # Energy Entropy
    # Normalize squared details to get a probability distribution
    if energy > 0:
        p = details**2 / energy
        # Add epsilon to avoid log(0)
        entropy = -np.sum(p * np.log(p + 1e-12))
    else:
        entropy = 0.0
    feats[f"{sensor_id}_txt_entropy"] = entropy

    # Higher order stats of texture
    feats[f"{sensor_id}_txt_skew"] = stats.skew(details)
    feats[f"{sensor_id}_txt_kurt"] = stats.kurtosis(details)

    return feats
