def compute_wavelet_energy(x):
    """
    Computes the energy of the signal using a Discrete Wavelet Transform (DWT).
    Implements a manual Haar Wavelet Transform (Level 1) to ensure robustness
    without relying on external packages like PyWavelets.

    Args:
        x (np.ndarray): Input signal (typically the Texture/Residual view).

    Returns:
        dict: Energy of detail coefficients (high freq) and approximation coefficients (low freq).
    """
    x_clean = fill_missing_values(x)

    # Need at least 2 samples
    if len(x_clean) < 2:
        return {"energy_detail": 0.0, "energy_approx": 0.0}

    # Ensure even length for vectorized operation
    if len(x_clean) % 2 != 0:
        x_proc = x_clean[:-1]
    else:
        x_proc = x_clean

    # Haar Transform Level 1
    # Detail coefficients (High Frequency / Texture): (x[2n] - x[2n+1]) / sqrt(2)
    # This captures the local "roughness" or high-frequency oscillation
    detail = (x_proc[::2] - x_proc[1::2]) / np.sqrt(2)

    # Approximation coefficients (Low Frequency / Trend): (x[2n] + x[2n+1]) / sqrt(2)
    approx = (x_proc[::2] + x_proc[1::2]) / np.sqrt(2)

    # Compute Energy (Sum of Squares)
    energy_detail = np.sum(detail**2)
    energy_approx = np.sum(approx**2)

    return {"energy_detail": energy_detail, "energy_approx": energy_approx}
