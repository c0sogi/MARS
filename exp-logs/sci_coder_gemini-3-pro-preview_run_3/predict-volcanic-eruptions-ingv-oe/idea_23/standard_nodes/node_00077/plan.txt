def compute_temporal_features(data, prefix):
    """
    Computes aggregated statistics over non-overlapping temporal windows.
    Ensures shift-invariance (Cite solution_lesson_node_00050).
    """
    features = {}
    # Split into windows
    windows = np.array_split(data, Config.NUM_TEMPORAL_WINDOWS)
    
    # Calculate RMS for each window
    win_rms = [np.sqrt(np.mean(w**2)) for w in windows]
    
    # Aggregate statistics
    features[f"{prefix}_win_mean"] = np.mean(win_rms)
    features[f"{prefix}_win_std"] = np.std(win_rms)
    features[f"{prefix}_win_min"] = np.min(win_rms)
    features[f"{prefix}_win_max"] = np.max(win_rms)
    features[f"{prefix}_win_range"] = np.ptp(win_rms)
    
    return features
