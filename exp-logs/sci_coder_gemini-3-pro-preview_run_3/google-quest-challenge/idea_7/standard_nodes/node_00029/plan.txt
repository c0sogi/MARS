The `torch.OutOfMemoryError` indicates that the physical batch size of 4 for the `deberta-v3-large` model exceeds the available GPU memory (approx. 16GB). To resolve this while maintaining the same optimization dynamics (effective batch size), I will reduce the physical batch size to 1 and increase the gradient accumulation steps to 4. This strategy decouples the memory footprint from the optimization batch size (Cite debug_lesson_1). I will apply this fix to the configuration and ensure both the DAPT trainer and the custom fine-tuning loop respect these settings.

def train_one_epoch(model, dataloader, optimizer, scheduler, device, epoch):
    model.train()
    total_loss = 0
    optimizer.zero_grad()

    for step, batch in enumerate(dataloader):
        input_ids = batch["input_ids"].to(device)
        attention_mask = batch["attention_mask"].to(device)
        q_mask = batch["q_mask"].to(device)
        a_mask = batch["a_mask"].to(device)
        labels = batch["labels"].to(device)

        logits, _, loss = model(input_ids, attention_mask, q_mask, a_mask, labels)

        loss = loss / Config.GRAD_ACCUM_STEPS
        loss.backward()

        if (step + 1) % Config.GRAD_ACCUM_STEPS == 0 or (step + 1) == len(dataloader):
            optimizer.step()
            if scheduler is not None:
                scheduler.step()
            optimizer.zero_grad()

        total_loss += loss.item() * Config.GRAD_ACCUM_STEPS

    avg_loss = total_loss / len(dataloader)
