    A wrapper around timm's ConvNeXt that replaces the standard
    Global Average Pooling (GAP) with Global Max Pooling (GMP).

    This is designed to preserve sparse anomaly signals (fractures)
    that might be washed out by averaging.
    """

    def __init__(self, backbone_name, pretrained=True):
        super().__init__()
        # global_pool='' and num_classes=0 ensures we get the spatial feature map (B, C, H, W)
        self.backbone = timm.create_model(
            backbone_name,
            pretrained=pretrained,
            num_classes=0,
            global_pool="",
            in_chans=3,
        )

        # Usually 768 for convnext_tiny
        self.num_features = self.backbone.num_features

    def forward(self, x):
