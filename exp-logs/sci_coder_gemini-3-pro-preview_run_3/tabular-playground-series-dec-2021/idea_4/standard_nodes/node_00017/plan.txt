Based on the lessons learned, I will optimize the solution by removing the DAE pretraining stage and improving feature engineering.

1.  **Remove DAE Pretraining (Cite `solution_lesson_node_00014`)**: The previous attempt showed that unsupervised pretraining with a Denoising Autoencoder (DAE) caused a performance regression and significantly increased runtime. I will revert to a direct supervised ResNet-MLP approach, which was the previous best performing method.
2.  **Explicit Geometric Features (Cite `solution_lesson_node_00016`)**: I will replace the generic `Sum_Distance` and `Mean_Distance` features with explicitly engineered geometric features: `Euclidean_Distance_To_Hydrology` (hypotenuse of horizontal and vertical distances) and `Hydrology_Elevation` (absolute elevation of the water source). The lesson indicates that models struggle to learn these non-linear geometric relationships from raw coordinates.
3.  **Architecture Update**: I will simplify the `ResNetClassifier` to accept the input dimension directly, removing the dependency on an encoder.

I will also update the `WORKING_DIR` to ensure fresh data processing with the new features.

class Encoder(nn.Module):
    """
    Encoder part of the DAE. Maps raw features to a latent representation.
    """

    def __init__(
        self, input_dim, hidden_dim=Config.HIDDEN_DIM, latent_dim=Config.LATENT_DIM
    ):
        super(Encoder, self).__init__()
        self.net = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.BatchNorm1d(hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, latent_dim),
            nn.BatchNorm1d(latent_dim),
            nn.ReLU(),
        )

    def forward(self, x):
        return self.net(x)


class Decoder(nn.Module):
    """
    Decoder part of the DAE. Reconstructs features from the latent representation.
    """

    def __init__(
        self,
        latent_dim=Config.LATENT_DIM,
        hidden_dim=Config.HIDDEN_DIM,
        output_dim=None,
    ):
        super(Decoder, self).__init__()
        if output_dim is None:
            raise ValueError("output_dim must be specified for Decoder")

        self.net = nn.Sequential(
            nn.Linear(latent_dim, hidden_dim),
            nn.BatchNorm1d(hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, output_dim),
        )

    def forward(self, x):
        return self.net(x)


class DenoisingAutoencoder(nn.Module):
    """
    Denoising Autoencoder (DAE) combining Encoder and Decoder.
    """

    def __init__(
        self, input_dim, hidden_dim=Config.HIDDEN_DIM, latent_dim=Config.LATENT_DIM
    ):
        super(DenoisingAutoencoder, self).__init__()
        self.encoder = Encoder(input_dim, hidden_dim, latent_dim)
        self.decoder = Decoder(latent_dim, hidden_dim, output_dim=input_dim)

    def forward(self, x):
        # x should be the noisy input during training
        z = self.encoder(x)
        x_recon = self.decoder(z)
        return x_recon, z

class DAEDataset(Dataset):
    """
    Dataset for Denoising Autoencoder Pretraining.
    Implements on-the-fly Swap Noise.
    """

    def __init__(self, data, noise_prob=Config.SWAP_NOISE_PROB):
        """
        Args:
            data (np.ndarray): The full feature matrix (Train + Test).
            noise_prob (float): Probability of swapping a feature value.
        """
        self.data = torch.from_numpy(data).float()
        self.noise_prob = noise_prob
        self.num_samples, self.num_features = self.data.shape

    def __len__(self):
        return self.num_samples

    def __getitem__(self, idx):
        # Get clean sample
        x_clean = self.data[idx]

        # Apply Swap Noise
        if self.noise_prob > 0:
            # Create a mask for features to swap
            mask = torch.rand(self.num_features) < self.noise_prob

            if mask.any():
                # For features to swap, pick values from random rows
                # We generate random row indices for all features for vectorized operation
                random_row_indices = torch.randint(
                    0, self.num_samples, (self.num_features,)
                )

                # Gather noise values: self.data[random_row_indices, feature_indices]
                # Note: Advanced indexing with tensors
                col_indices = torch.arange(self.num_features)
                noise_values = self.data[random_row_indices, col_indices]

                # Apply noise
                x_noisy = torch.where(mask, noise_values, x_clean)
            else:
                x_noisy = x_clean
        else:
            x_noisy = x_clean

        return x_noisy, x_clean

def calculate_dae_loss(recon, target):
    """
    Calculates the composite loss for DAE: MSE for continuous, BCE for binary.
    """
    # Determine split index
    # Continuous cols: 10 original + 2 engineered = 12
    num_cont = len(Config.CONTINUOUS_COLS) + 2

    # Split tensors
    recon_cont = recon[:, :num_cont]
    recon_bin = recon[:, num_cont:]

    target_cont = target[:, :num_cont]
    target_bin = target[:, num_cont:]

    # Define criteria
    mse_crit = nn.MSELoss()
    bce_crit = nn.BCEWithLogitsLoss()

    # Calculate losses
    loss_mse = mse_crit(recon_cont, target_cont)
    loss_bce = bce_crit(recon_bin, target_bin)

    # Weighted sum
    total_loss = (Config.MSE_WEIGHT * loss_mse) + (Config.BCE_WEIGHT * loss_bce)
    return total_loss, loss_mse.item(), loss_bce.item()


def train_dae(
    model,
    train_loader,
    val_loader,
    epochs=Config.EPOCHS_PRETRAIN,
    lr=Config.LR_PRETRAIN,
    device=Config.DEVICE,
):
    """
    Trains the Denoising Autoencoder.
    """
    model = model.to(device)
    optimizer = optim.Adam(model.parameters(), lr=lr)

    print("Starting DAE Pretraining...")

    for epoch in range(epochs):
        # --- Training ---
        model.train()
        train_loss = 0.0

        for batch in train_loader:
            # Handle different dataset returns
            if isinstance(batch, (list, tuple)):
                if len(batch) == 2:
                    x_noisy, x_clean = batch
                else:
                    x_noisy = batch[0]
                    x_clean = batch[0]
            else:
                x_noisy = batch
                x_clean = batch

            x_noisy = x_noisy.to(device)
            x_clean = x_clean.to(device)

            optimizer.zero_grad()

            # Forward pass (returns recon, latent)
            recon, _ = model(x_noisy)

            loss, _, _ = calculate_dae_loss(recon, x_clean)

            loss.backward()
            optimizer.step()

            train_loss += loss.item() * x_noisy.size(0)

        train_loss /= len(train_loader.dataset)

        # --- Validation ---
        model.eval()
        val_loss = 0.0
        val_mse = 0.0
        val_bce = 0.0

        with torch.no_grad():
            for batch in val_loader:
                # Determine input and target for validation
                # If batch is (x, y_label), use x for both recon input and target
                # If batch is (noisy, clean), use noisy input and clean target

                if isinstance(batch, (list, tuple)):
                    elem1, elem2 = batch
                    # Check if second element is target label (long/int)
                    if elem2.dtype == torch.long:
                        x_in = elem1
                        target = elem1
                    else:
                        x_in = elem1
                        target = elem2
                else:
                    x_in = batch
                    target = batch

                target = target.to(device)

                recon, _ = model(x_in)
                loss, mse, bce = calculate_dae_loss(recon, target)

                bs = x_in.size(0)
                val_loss += loss.item() * bs
                val_mse += mse * bs
                val_bce += bce * bs

        val_loss /= len(val_loader.dataset)
        val_mse /= len(val_loader.dataset)
        val_bce /= len(val_loader.dataset)

        print(f"Epoch {epoch+1}/{epochs}")

        print_metric("DAE Val Loss", val_loss)
        print_metric("DAE Val MSE", val_mse)
        print_metric("DAE Val BCE", val_bce)

    return model
