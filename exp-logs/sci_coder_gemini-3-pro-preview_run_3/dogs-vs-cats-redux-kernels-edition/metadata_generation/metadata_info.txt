# Python Script for Metadata Generation
```
import os
import glob
import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
import shutil

# Configuration
INPUT_DIR = "./input"
METADATA_DIR = "./metadata"
RANDOM_STATE = 42


def generate_metadata():
    """
    Scans the input directory, parses filenames, splits training data,
    and saves metadata CSVs to the metadata directory.
    """
    print("Starting metadata generation...")

    # Ensure metadata directory exists and is clean
    if os.path.exists(METADATA_DIR):
        shutil.rmtree(METADATA_DIR)
    os.makedirs(METADATA_DIR)

    # --- Process Training Data ---
    train_dir = os.path.join(INPUT_DIR, "train")
    # Get all jpg files
    train_files = glob.glob(os.path.join(train_dir, "*.jpg"))

    if not train_files:
        raise FileNotFoundError(f"No training files found in {train_dir}")

    train_data = []
    for fpath in train_files:
        filename = os.path.basename(fpath)
        # Expected format: class.id.jpg (e.g., cat.0.jpg)
        # Extract label from the first part of the filename
        parts = filename.split(".")
        if len(parts) < 3:
            continue  # Skip files that don't match expected pattern

        label_str = parts[0].lower()
        if label_str == "dog":
            label = 1
        elif label_str == "cat":
            label = 0
        else:
            continue  # Skip unknown labels

        # Path relative to ./input
        rel_path = os.path.join("train", filename)
        train_data.append({"filepath": rel_path, "label": label, "filename": filename})

    df_full = pd.DataFrame(train_data)
    print(f"Found {len(df_full)} training images.")

    if len(df_full) == 0:
        raise ValueError("No valid training images parsed.")

    # Split into Train and Validation (80:20, Stratified)
    # Stratify based on label to maintain class distribution
    train_df, val_df = train_test_split(
        df_full,
        test_size=0.2,
        stratify=df_full["label"],
        random_state=RANDOM_STATE,
        shuffle=True,
    )

    # Save Train/Val Metadata
    train_csv_path = os.path.join(METADATA_DIR, "train.csv")
    val_csv_path = os.path.join(METADATA_DIR, "val.csv")

    train_df.to_csv(train_csv_path, index=False)
    val_df.to_csv(val_csv_path, index=False)
    print(f"Saved train.csv ({len(train_df)} rows) and val.csv ({len(val_df)} rows).")

    # --- Process Test Data ---
    test_dir = os.path.join(INPUT_DIR, "test")
    test_files = glob.glob(os.path.join(test_dir, "*.jpg"))

    test_data = []
    for fpath in test_files:
        filename = os.path.basename(fpath)
        # Expected format: id.jpg (e.g., 1.jpg)
        parts = filename.split(".")
        try:
            # Extract numeric ID
            img_id = int(parts[0])
            rel_path = os.path.join("test", filename)
            test_data.append({"filepath": rel_path, "id": img_id})
        except ValueError:
            continue  # Skip files without numeric IDs

    test_df = pd.DataFrame(test_data)
    # Sort by ID for consistency
    test_df = test_df.sort_values("id")

    test_csv_path = os.path.join(METADATA_DIR, "test.csv")
    test_df.to_csv(test_csv_path, index=False)
    print(f"Saved test.csv ({len(test_df)} rows).")


def validate_metadata():
    """
    Loads generated metadata and performs statistical and integrity checks.
    """
    print("\nStarting validation checks...")

    # Load Metadata
    train_df = pd.read_csv(os.path.join(METADATA_DIR, "train.csv"))
    val_df = pd.read_csv(os.path.join(METADATA_DIR, "val.csv"))
    test_df = pd.read_csv(os.path.join(METADATA_DIR, "test.csv"))

    # 1. Summary Statistics
    print("\n--- Summary Statistics ---")
    print(f"Train Set: {len(train_df)} samples")
    print(
        f"Class Distribution (Train):\n{train_df['label'].value_counts(normalize=True)}"
    )

    print(f"\nValidation Set: {len(val_df)} samples")
    print(f"Class Distribution (Val):\n{val_df['label'].value_counts(normalize=True)}")

    print(f"\nTest Set: {len(test_df)} samples")

    # 2. File Existence Checks
    def check_paths(df, dataset_name):
        print(f"\nChecking file paths for {dataset_name}...")
        paths = df["filepath"].values

        # Randomly sample 1000 paths if dataset is larger
        if len(paths) > 1000:
            np.random.seed(RANDOM_STATE)  # Ensure reproducibility of check
            paths_to_check = np.random.choice(paths, 1000, replace=False)
        else:
            paths_to_check = paths

        missing_count = 0
        missing_examples = []

        for p in paths_to_check:
            # Construct full path: ./input/ + relative_path
            full_path = os.path.join(INPUT_DIR, p)
            if not os.path.exists(full_path):
                missing_count += 1
                if len(missing_examples) < 5:
                    missing_examples.append(p)

        if len(paths_to_check) == 0:
            print(f"Warning: No paths to check for {dataset_name}")
            return

        ratio = missing_count / len(paths_to_check)
        print(f"Missing File Ratio: {ratio:.4f}")

        if ratio > 0.5:
            print("Sample missing paths:", missing_examples)
            raise FileNotFoundError(
                f"Validation failed: >50% of files missing in {dataset_name} metadata."
            )

    check_paths(train_df, "Train")
    check_paths(val_df, "Validation")
    check_paths(test_df, "Test")

    # 3. Stratification Verification
    print("\nVerifying stratification...")
    if len(train_df) > 0 and len(val_df) > 0:
        train_mean = train_df["label"].mean()
        val_mean = val_df["label"].mean()

        diff = abs(train_mean - val_mean)
        print(f"Train Mean Label: {train_mean:.4f}")
        print(f"Val Mean Label: {val_mean:.4f}")
        print(f"Difference: {diff:.4f}")

        # Tolerance: 1% difference acceptable
        if diff > 0.01:
            raise AssertionError(
                f"Stratification check failed. Difference {diff:.4f} exceeds tolerance."
            )
    else:
        print("Skipping stratification check due to empty dataset.")

    print("\nAll validation checks passed.")


if __name__ == "__main__":
    try:
        generate_metadata()
        validate_metadata()
    except Exception as e:
        # Re-raise exception to ensure the script fails explicitly on error
        raise e

```

# Document
Based on the provided Python script and its execution output, here is the documentation for the files generated in the `./metadata` directory.

### 1. `./metadata/train.csv`

*   **Content and Purpose:**
    *   This file contains the metadata for the **training dataset**. It represents an 80% stratified split of the labeled images found in the `input/train/` directory.
    *   Its primary purpose is to provide the data loader with the relative file paths and ground truth labels required to train the machine learning model.
    *   **Class Mapping:** Based on the script logic, `cat` maps to `0` and `dog` maps to `1`.
    *   **Size:** Contains approximately 18,000 rows (as per the execution output).

*   **Schema / Structure:**
    *   **Format:** CSV (Comma Separated Values) with a header.
    *   **Columns:**
        *   `filepath` (String): The relative path to the image starting from the input directory (e.g., `train/cat.0.jpg`).
        *   `label` (Integer): The target classification label (`0` for cat, `1` for dog).
        *   `filename` (String): The original filename of the image (e.g., `cat.0.jpg`).
    *   **Example Row:**
        ```csv
        filepath,label,filename
        train/dog.402.jpg,1,dog.402.jpg
        ```

*   **Loading Method:**
    *   This file should be loaded using the Pandas library:
        ```python
        import pandas as pd
        train_df = pd.read_csv("./metadata/train.csv")
        ```
