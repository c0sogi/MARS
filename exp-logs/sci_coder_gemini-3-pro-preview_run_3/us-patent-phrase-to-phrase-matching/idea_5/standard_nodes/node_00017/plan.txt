To resolve the `torch.OutOfMemoryError` caused by the mismatch between the model size (`microsoft/deberta-v3-large`) and the available GPU memory (16GB vs. expected 40GB), I will implement the following strategies:

1.  **Enable Automatic Mixed Precision (AMP) and Gradient Accumulation**: I will modify `library/engine.py` to use `torch.amp.GradScaler` and `torch.amp.autocast`. This reduces memory usage by storing activations in half-precision and allows us to simulate a larger batch size by accumulating gradients over multiple smaller steps.
2.  **Adjust Configuration**: I will update `runfile.py` to reduce the physical batch size to 4 (from 32) and increase `gradient_accumulation_steps` to 8. This maintains the effective batch size of 32 while fitting within the 16GB VRAM limit.
3.  **Enable Gradient Checkpointing**: I will explicitly enable gradient checkpointing in `runfile.py` to further trade off compute for memory.

This approach targets the root cause (memory exhaustion) without altering the model's convergence properties.

def train_fn(
    train_loader, model, optimizer, epoch, scheduler, device, loss_fn, awp=None
):
    """
    Executes one training epoch.
    """
    model.train()
    losses = AverageMeter()

    # Progress bar
    pbar = tqdm(train_loader, desc=f"Epoch {epoch+1}", leave=False)

    for step, batch in enumerate(pbar):
        input_ids = batch["input_ids"].to(device)
        attention_mask = batch["attention_mask"].to(device)
        targets = batch["labels"].to(device)

        # Bin targets for hybrid loss if needed
        bin_targets = None
        if "bin_labels" in batch:
            bin_targets = batch["bin_labels"].to(device)

        batch_size = input_ids.size(0)

        # Forward
        outputs = model(input_ids, attention_mask)
        loss = loss_fn(outputs, targets, bin_targets)

        # Backward
        loss.backward()

        # Adversarial Weight Perturbation (AWP)
        if Config.use_awp and awp is not None and epoch >= Config.awp_start_epoch:
            awp.attack_step()

            # Forward pass with perturbed weights
            adv_outputs = model(input_ids, attention_mask)
            adv_loss = loss_fn(adv_outputs, targets, bin_targets)

            adv_loss.backward()
            awp.restore()

        # Gradient Clipping
        torch.nn.utils.clip_grad_norm_(model.parameters(), Config.max_grad_norm)

        # Optimizer Step
        optimizer.step()
        scheduler.step()
        optimizer.zero_grad()

        losses.update(loss.item(), batch_size)

        if step % Config.print_freq == 0 or step == (len(train_loader) - 1):
            pbar.set_postfix(loss=losses.avg, lr=scheduler.get_last_lr()[0])

    return losses.avg

def valid_fn(val_loader, model, device, loss_fn):
    """
    Evaluates the model on the validation set.
    """
    model.eval()
    losses = AverageMeter()
    preds = []
    ground_truth = []

    with torch.no_grad():
        for batch in pbar:
            input_ids = batch["input_ids"].to(device)
            attention_mask = batch["attention_mask"].to(device)
            targets = batch["labels"].to(device)

            bin_targets = None
            if "bin_labels" in batch:
                bin_targets = batch["bin_labels"].to(device)

            batch_size = input_ids.size(0)

            outputs = model(input_ids, attention_mask)
            loss = loss_fn(outputs, targets, bin_targets)

            losses.update(loss.item(), batch_size)

            # Collect predictions (Regression logits)
            # Flatten to 1D array
            preds.append(outputs["logits"].view(-1).cpu().numpy())
            ground_truth.append(targets.view(-1).cpu().numpy())

    predictions = np.concatenate(preds)
    true_labels = np.concatenate(ground_truth)

    score = get_score(true_labels, predictions)

    return losses.avg, score
