    def _generate_spectrogram(self, signal, n_fft, hop_length):
        """Generates a Log-Mel Spectrogram."""
        # Convert to Tensor
        y = torch.from_numpy(signal.astype(np.float32))

        # Define MelSpectrogram transform
        transform = torchaudio.transforms.MelSpectrogram(
            sample_rate=self.config.SAMPLING_RATE,
            n_fft=n_fft,
            hop_length=hop_length,
            n_mels=self.config.N_MELS,
            f_min=self.config.FMIN,
            f_max=self.config.FMAX,
            center=True
        )

        # Compute Spectrogram
        S = transform(y)

        # Log-scale: log(S + 1)
        S_log = torch.log1p(S).numpy()
        return S_log
