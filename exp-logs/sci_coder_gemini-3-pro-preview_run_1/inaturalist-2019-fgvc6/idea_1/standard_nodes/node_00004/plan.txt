Additionally, I will introduce **Label Smoothing (0.1)** to the loss function to prevent overfitting and improve generalization on this noisy, high-cardinality dataset.

def train_one_epoch(train_loader, model, criterion, optimizer, device, epoch):
    """
    Trains the model for one epoch.
    """
    model.train()
    losses = AverageMeter()
    top1 = AverageMeter()

        images = images.to(device)
        targets = targets.to(device)

        # Forward pass
        outputs = model(images)
        loss = criterion(outputs, targets)

        loss.backward()
        optimizer.step()

def train_one_epoch(train_loader, model, criterion, optimizer, device, epoch, scaler):
    """
    Trains the model for one epoch using AMP.
    """
    model.train()
    losses = AverageMeter()
    top1 = AverageMeter()

    for i, (images, targets) in enumerate(train_loader):
        images = images.to(device)
        targets = targets.to(device)

        # Forward pass with AMP
        with autocast():
            outputs = model(images)
            loss = criterion(outputs, targets)

        # Measure accuracy and record loss
        acc1 = accuracy(outputs, targets, topk=(1,))[0]
        losses.update(loss.item(), images.size(0))
        top1.update(acc1.item(), images.size(0))

        # Backward pass and optimize with scaler
        optimizer.zero_grad()
        scaler.scale(loss).backward()
        scaler.step(optimizer)
        scaler.update()
