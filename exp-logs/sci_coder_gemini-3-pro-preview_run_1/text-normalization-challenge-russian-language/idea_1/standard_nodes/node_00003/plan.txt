To improve the model's performance on "tail" classes (e.g., `MONEY`, `DECIMAL`) while maintaining high accuracy on frequent classes, I will implement a **Hybrid Neuro-Symbolic Architecture**.

**Strategy:**
1.  **Cache Invalidation**: I will update the `MODEL_STATS_PATH` in `library/config.py` to `ngram_neural_stats_v2.npy`. This ensures the pipeline re-runs the training phase instead of loading the old, non-neural statistics, directly addressing the silent failure observed in previous attempts (Cite solution_lesson_node_00002).
2.  **Neural Fallback**: I will create a new `library/neural_model.py` module containing a character-level Seq2Seq model (Bi-GRU Encoder-Decoder). This model will be trained on unique `(before, after)` pairs from the training set to learn generative normalization rules for complex tokens.
3.  **Hybrid Logic**: I will modify `library/ngram_model.py` to integrate this neural model. The system will first attempt the Hierarchical N-gram Lookup (Trigram -> Bigram -> Unigram) (Cite solution_lesson_node_00001). If the lookup fails (returns `NaN`), the system will check if the token contains digits. If it does, it will pass the token to the Neural Model for prediction; otherwise, it will fall back to Identity. This targeted fallback prevents the neural model from hallucinating on rare proper nouns while allowing it to solve unseen numerical expressions.

import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import Dataset, DataLoader
import numpy as np
import random
import logging
import pandas as pd

logger = logging.getLogger("NeuralModel")

class CharTokenizer:
    def __init__(self):
        self.char2id = {}
        self.id2char = {}
        self.pad_token = "<PAD>"
        self.sos_token = "<SOS>"
        self.eos_token = "<EOS>"
        self.unk_token = "<UNK>"
        self.special_tokens = [self.pad_token, self.sos_token, self.eos_token, self.unk_token]
    
    def fit(self, texts):
        unique_chars = set()
        for text in texts:
            unique_chars.update(str(text))
        
        # Sort for determinism
        sorted_chars = sorted(list(unique_chars))
        
        self.id2char = {i: token for i, token in enumerate(self.special_tokens)}
        for i, char in enumerate(sorted_chars):
            self.id2char[len(self.special_tokens) + i] = char
            
        self.char2id = {v: k for k, v in self.id2char.items()}
        
    def encode(self, text, add_special_tokens=True):
        text = str(text)
        ids = [self.char2id.get(c, self.char2id[self.unk_token]) for c in text]
        if add_special_tokens:
            ids = [self.char2id[self.sos_token]] + ids + [self.char2id[self.eos_token]]
        return ids
    
    def decode(self, ids):
        chars = []
        for i in ids:
            token = self.id2char.get(i, self.unk_token)
            if token == self.eos_token:
                break
            if token not in self.special_tokens:
                chars.append(token)
        return "".join(chars)
    
    def __len__(self):
        return len(self.char2id)

class Seq2SeqDataset(Dataset):
    def __init__(self, src_texts, tgt_texts, tokenizer):
        self.src_texts = src_texts
        self.tgt_texts = tgt_texts
        self.tokenizer = tokenizer
        
    def __len__(self):
        return len(self.src_texts)
    
    def __getitem__(self, idx):
        src = self.tokenizer.encode(self.src_texts[idx], add_special_tokens=False) # Encoder doesn't strictly need SOS/EOS but good practice
        tgt = self.tokenizer.encode(self.tgt_texts[idx], add_special_tokens=True)
        return torch.tensor(src), torch.tensor(tgt)

def collate_fn(batch):
    src_list, tgt_list = zip(*batch)
    src_lens = [len(x) for x in src_list]
    tgt_lens = [len(x) for x in tgt_list]
    
    # Pad
    src_padded = torch.nn.utils.rnn.pad_sequence(src_list, batch_first=True, padding_value=0)
    tgt_padded = torch.nn.utils.rnn.pad_sequence(tgt_list, batch_first=True, padding_value=0)
    
    return src_padded, tgt_padded, torch.tensor(src_lens), torch.tensor(tgt_lens)

class Seq2SeqModel(nn.Module):
    def __init__(self, vocab_size, embed_dim, hidden_dim, device):
        super().__init__()
        self.device = device
        self.embedding = nn.Embedding(vocab_size, embed_dim, padding_idx=0)
        self.encoder = nn.GRU(embed_dim, hidden_dim, batch_first=True, bidirectional=True)
        # Decoder input dim is embed_dim, hidden is hidden_dim * 2 (because of bidirectional encoder)
        self.decoder = nn.GRU(embed_dim, hidden_dim * 2, batch_first=True)
        self.fc = nn.Linear(hidden_dim * 2, vocab_size)
        self.dropout = nn.Dropout(0.2)
        
    def forward(self, src, tgt, src_lens):
        # src: [batch, src_len]
        # tgt: [batch, tgt_len]
        
        # Encoder
        embedded = self.dropout(self.embedding(src))
        packed = torch.nn.utils.rnn.pack_padded_sequence(embedded, src_lens.cpu(), batch_first=True, enforce_sorted=False)
        _, hidden = self.encoder(packed)
        
        # Reshape hidden: (2, batch, hidden) -> (1, batch, 2*hidden) for decoder
        hidden = torch.cat([hidden[-2], hidden[-1]], dim=1).unsqueeze(0)
        
        # Decoder (Teacher Forcing)
        # Embed target (exclude last token for input)
        tgt_input = tgt[:, :-1]
        dec_embedded = self.dropout(self.embedding(tgt_input))
        
        output, _ = self.decoder(dec_embedded, hidden)
        prediction = self.fc(output)
        return prediction

def train_neural_module(df_train, device_name="cpu", epochs=5, batch_size=256):
    logger.info(f"Training Neural Module on device: {device_name}")
    device = torch.device(device_name)
    
    # Filter for unique pairs
    # We prioritize pairs where before != after to learn normalization
    logger.info("Preparing data for neural training...")
    # Get unique pairs
    unique_pairs = df_train[["before", "after"]].drop_duplicates()
    
    # Focus on changed or digit-containing tokens + random sample of others
    # This is a heuristic to balance the dataset
    mask_changed = unique_pairs["before"] != unique_pairs["after"]
    mask_digit = unique_pairs["before"].str.contains(r'\d', na=False)
    
    important_pairs = unique_pairs[mask_changed | mask_digit]
    other_pairs = unique_pairs[~(mask_changed | mask_digit)]
    
    # Sample others to not overwhelm
    if len(other_pairs) > 50000:
        other_pairs = other_pairs.sample(50000, random_state=42)
        
    train_pairs = pd.concat([important_pairs, other_pairs])
    logger.info(f"Training on {len(train_pairs)} unique pairs.")
    
    src_texts = train_pairs["before"].astype(str).tolist()
    tgt_texts = train_pairs["after"].astype(str).tolist()
    
    # Tokenizer
    tokenizer = CharTokenizer()
    tokenizer.fit(src_texts + tgt_texts)
    
    # Dataset
    dataset = Seq2SeqDataset(src_texts, tgt_texts, tokenizer)
    dataloader = DataLoader(dataset, batch_size=batch_size, shuffle=True, collate_fn=collate_fn, num_workers=2)
    
    # Model
    model = Seq2SeqModel(len(tokenizer), 64, 256, device).to(device)
    optimizer = optim.Adam(model.parameters(), lr=0.001)
    criterion = nn.CrossEntropyLoss(ignore_index=0)
    
    # Training Loop
    model.train()
    for epoch in range(epochs):
        total_loss = 0
        for src, tgt, src_lens, _ in dataloader:
            src, tgt = src.to(device), tgt.to(device)
            
            optimizer.zero_grad()
            output = model(src, tgt, src_lens)
            
            # Output: [batch, seq_len-1, vocab]
            # Tgt: [batch, seq_len] -> remove SOS (first token) for label
            tgt_labels = tgt[:, 1:]
            
            # Flatten
            output = output.reshape(-1, len(tokenizer))
            tgt_labels = tgt_labels.reshape(-1)
            
            loss = criterion(output, tgt_labels)
            loss.backward()
            optimizer.step()
            
            total_loss += loss.item()
            
        logger.info(f"Epoch {epoch+1}/{epochs} - Loss: {total_loss / len(dataloader):.4f}")
        
    return model, tokenizer

def batch_predict_neural(model, tokenizer, texts, device_name="cpu"):
    device = torch.device(device_name)
    model.to(device)
    model.eval()
    
    preds = []
    
    dataset = Seq2SeqDataset(texts, [""]*len(texts), tokenizer)

            embedded = model.embedding(src)
            packed = torch.nn.utils.rnn.pack_padded_sequence(embedded, src_lens.cpu(), batch_first=True, enforce_sorted=False)
            _, hidden = model.encoder(packed)
            hidden = torch.cat([hidden[-2], hidden[-1]], dim=1).unsqueeze(0)

            input_token = torch.full((batch_size, 1), tokenizer.char2id[tokenizer.sos_token], device=device, dtype=torch.long)

            for _ in range(100): # max len

                output, hidden = model.decoder(dec_emb, hidden)

                    break
                
                input_token = top1
            
            # Decode to string
            for seq in decoded_seqs:
                preds.append(tokenizer.decode(seq))
                
    return preds
