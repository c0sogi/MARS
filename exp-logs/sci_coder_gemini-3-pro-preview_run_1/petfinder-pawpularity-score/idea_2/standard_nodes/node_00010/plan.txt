    """
    Extracts image embeddings and metadata features from the loader.
    Cite solution_lesson_node_00001: Decouple feature extraction from learning.
    Cite solution_lesson_node_00009: Feature-Space Augmentation Averaging.
    """
    model.eval()
    embeddings = []
    meta_features = []
    targets = []
    ids = []

    with torch.no_grad():
        for batch_data in loader:
            images = batch_data["image"].to(device)
            features = batch_data["features"]
            batch_ids = batch_data["id"]

            # 1. Original View
            img_emb_1 = model.backbone(images)
            if len(img_emb_1.shape) > 2:
                img_emb_1 = img_emb_1.mean(dim=[-2, -1])

            # 2. Flipped View (Horizontal Flip)
            # Images are (B, C, H, W). Flip on dim 3.
            images_flip = torch.flip(images, dims=[3])
            img_emb_2 = model.backbone(images_flip)
            if len(img_emb_2.shape) > 2:

            # Average embeddings for robust representation
            img_emb = (img_emb_1 + img_emb_2) / 2.0
