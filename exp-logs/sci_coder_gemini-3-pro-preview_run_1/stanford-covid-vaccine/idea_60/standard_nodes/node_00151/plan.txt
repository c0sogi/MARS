class VectorScaledResidualBlock(nn.Module):
    """
    Residual Block with BiGRU and Channel-Wise (Vector) Residual Scaling.
    Architecture: x -> LN -> BiGRU -> Dropout -> Scale -> + -> x
    """

    def __init__(self, hidden_dim, dropout=0.1):
        super().__init__()
        self.ln = nn.LayerNorm(hidden_dim)

        # BiGRU: Output dimension must match hidden_dim.
        # Since it's bidirectional, hidden_size per direction is hidden_dim // 2.
        self.gru = nn.GRU(
            hidden_dim, hidden_dim // 2, batch_first=True, bidirectional=True
        )

        self.dropout = nn.Dropout(dropout)

        # Vector Scaling: Learnable parameter per channel, initialized to 1.0
        # This allows the model to independently regulate feature importance in the residual stream.
        self.scale = nn.Parameter(torch.ones(hidden_dim))

    def forward(self, x):
        residual = x

        # Pre-LayerNorm
        out = self.ln(x)

        # BiGRU
        out, _ = self.gru(out)

        # Dropout
        out = self.dropout(out)

        # Vector Scaling
        # out: (B, L, D) * scale: (1, 1, D)
        out = out * self.scale.view(1, 1, -1)

        return residual + out
