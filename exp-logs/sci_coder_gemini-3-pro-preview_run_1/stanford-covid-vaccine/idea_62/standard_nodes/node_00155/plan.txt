class SequenceGatedBlock(nn.Module):
    """
    Residual Block with Pre-LayerNorm BiGRU and Sequence-Aware Gating.
    """

    def __init__(self, hidden_dim, seq_embed_dim, dropout):
        super(SequenceGatedBlock, self).__init__()

        self.norm = nn.LayerNorm(hidden_dim)

        # BiGRU: Input hidden_dim -> Output hidden_dim (2 * hidden_dim//2)
        self.gru = nn.GRU(
            input_size=hidden_dim,
            hidden_size=hidden_dim // 2,
            batch_first=True,
            bidirectional=True,
        )

        # Gating projection: Maps Atomic Sequence Embedding to Hidden Dim
        self.gate_proj = nn.Linear(seq_embed_dim, hidden_dim)

        self.dropout = nn.Dropout(dropout)

    def forward(self, x, seq_emb):
        """
        Args:
            x: Input tensor (batch, seq_len, hidden_dim)
            seq_emb: Atomic sequence embeddings (batch, seq_len, seq_embed_dim)
        """
        residual = x

        # Pre-LayerNorm
        x_norm = self.norm(x)

        # BiGRU
        rnn_out, _ = self.gru(x_norm)

        # Sequence Gating
        # Project sequence embedding to gate logits
        gate_logits = self.gate_proj(seq_emb)
        gate = torch.sigmoid(gate_logits)

        # Apply gate: Modulate RNN features by sequence context
        h_gated = rnn_out * gate

        # Dropout
        out = self.dropout(h_gated)

        # Residual Connection
        return residual + out
