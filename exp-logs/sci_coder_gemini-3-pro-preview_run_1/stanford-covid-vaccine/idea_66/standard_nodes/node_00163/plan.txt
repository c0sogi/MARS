class ProjectedBiLSTMBlock(nn.Module):
    """
    Inverted Bottleneck Block:
    Input (W) -> LN -> BiLSTM (Expand to 2*W) -> Linear (Compress to W) -> Dropout -> Residual
    """

    def __init__(self, width, dropout):
        super().__init__()
        self.layer_norm = nn.LayerNorm(width)

        # Expanded BiLSTM: Hidden size = Width.
        # Output size will be 2 * Width (768 if Width is 384).
        self.lstm = nn.LSTM(
            input_size=width, hidden_size=width, batch_first=True, bidirectional=True
        )

        # Compression Projection: 2 * Width -> Width
        self.projection = nn.Linear(width * 2, width)

        self.dropout = nn.Dropout(dropout)

    def forward(self, x):
        residual = x

        # 1. Pre-Norm
        out = self.layer_norm(x)

        # 2. Expanded BiLSTM
        out, _ = self.lstm(out)

        # 3. Compression Projection
        out = self.projection(out)

        # 4. Dropout
        out = self.dropout(out)

        # 5. Residual
        return residual + out

class ResidualBiLSTMBlock(nn.Module):
    """
    Standard Pre-LayerNorm Residual BiLSTM Block.
    Structure: Input -> LN -> BiLSTM -> Dropout -> Residual + Input
    Cite Lesson 135: Superiority of Pre-LayerNorm Residual Blocks.
    """

    def __init__(self, width, dropout):
        super().__init__()
        self.layer_norm = nn.LayerNorm(width)

        # Standard BiLSTM: Hidden size = Width // 2.
        # Output size will be Width (e.g., 192 * 2 = 384).
        self.lstm = nn.LSTM(
            input_size=width,
            hidden_size=width // 2,
            batch_first=True,
            bidirectional=True,
        )

        self.dropout = nn.Dropout(dropout)

    def forward(self, x):
        residual = x

        # 1. Pre-Norm
        out = self.layer_norm(x)

        # 2. BiLSTM
        out, _ = self.lstm(out)

        # 3. Dropout
        out = self.dropout(out)

        # 4. Residual
        return residual + out
