Based on the lessons learned from previous iterations, I will optimize the solution by reverting the "Zoneout-Regularized Wide-Stream" architecture to a standard "Wide-Stream Residual BiGRU" with specific hyperparameter adjustments that have historically yielded the best performance.

**Implementation Strategy:**

1.  **Configuration (`library/config.py`)**:
    *   **Hidden Dimension**: Reduce `HIDDEN_DIM` from 512 to 384. Lesson 00022 and 00070 indicate that 384 is the optimal capacity for this dataset size; 512 introduces instability and sensitivity to weight decay.
    *   **Regularization**: Remove `ZONEOUT_PROB` and replace it with `DROPOUT = 0.1`. Lesson 00076 explicitly states that inter-layer dropout outperforms recurrent state regularization (Zoneout) for this task.
    *   **Training Duration**: Restore `EPOCHS` to 20 to ensure the Cosine Annealing scheduler has sufficient time to converge (Cite Lesson 00051).

2.  **Layers (`library/layers.py`)**:
    *   **Embedding Optimization**: Modify `SinusoidalPairingEmbedding` to use the full embedding dimension for frequency encoding, rather than padding half with zeros. This improves feature utilization.
    *   **Block Architecture**: Replace `ZoneoutBiGRUBlock` with `ResidualBiGRUBlock`. This new block implements a standard BiGRU followed by a Dropout layer, aligning with the successful "Current Best" architecture described in the lessons.

3.  **Model (`library/model.py`)**:
    *   Update the `WideResBiGRU` class to utilize the new `ResidualBiGRUBlock` and the optimized hyperparameters.

4.  **Execution (`runfile.py`)**:
    *   Update the model instantiation to use the optimized `WideResBiGRU`.
    *   Remove the manual override of `Config.EPOCHS = 15` to allow the full 20-epoch training cycle defined in the config.

class ZoneoutBiGRUBlock(nn.Module):
    """
    A Bidirectional GRU block with Zoneout regularization.
    Unrolls the sequence to apply stochastic state updates during training.
    """

    def __init__(
        self,
        input_size: int,
        hidden_size: int,
        zoneout_prob: float = Config.ZONEOUT_PROB,
    ):
        super().__init__()
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.zoneout_prob = zoneout_prob

        # Forward and Backward Cells
        self.fwd_cell = nn.GRUCell(input_size, hidden_size)
        self.bwd_cell = nn.GRUCell(input_size, hidden_size)

    def forward(
        self, x: torch.Tensor, h_0: Optional[torch.Tensor] = None
    ) -> torch.Tensor:
        """
        Args:
            x (torch.Tensor): Input sequence of shape (Batch, Seq_Len, Input_Size).
            h_0 (torch.Tensor, optional): Initial hidden state (Batch, Hidden_Size * 2).

        Returns:
            torch.Tensor: Output sequence of shape (Batch, Seq_Len, Hidden_Size * 2).
        """
        batch_size, seq_len, _ = x.size()

        # Initialize hidden states
        if h_0 is None:
            h_fwd = torch.zeros(
                batch_size, self.hidden_size, device=x.device, dtype=x.dtype
            )
            h_bwd = torch.zeros(
                batch_size, self.hidden_size, device=x.device, dtype=x.dtype
            )
        else:
            # Assuming h_0 is concatenated [h_fwd, h_bwd]
            h_fwd = h_0[:, : self.hidden_size]
            h_bwd = h_0[:, self.hidden_size :]

        outputs_fwd = []
        outputs_bwd = []

        # Pre-generate masks for efficiency if training
        # We need masks for every step: (Seq_Len, Batch, Hidden_Size)
        use_zoneout = self.training and self.zoneout_prob > 0.0

        # Forward Pass
        for t in range(seq_len):
            x_t = x[:, t, :]
            h_next = self.fwd_cell(x_t, h_fwd)

            if use_zoneout:
                # Zoneout: h_t = d_t * h_next + (1 - d_t) * h_prev
                # d_t ~ Bernoulli(1 - prob) (1 means keep new, 0 means keep old)
                # We sample a mask where 1 = update, 0 = keep old
                mask = torch.bernoulli(torch.full_like(h_next, 1.0 - self.zoneout_prob))
                h_fwd = mask * h_next + (1 - mask) * h_fwd
            elif self.zoneout_prob > 0.0:
                # Evaluation: Expectation
                # h_t = (1 - p) * h_next + p * h_prev
                h_fwd = (1 - self.zoneout_prob) * h_next + self.zoneout_prob * h_fwd
            else:
                # Standard GRU update
                h_fwd = h_next

            outputs_fwd.append(h_fwd)

        # Backward Pass
        for t in range(seq_len - 1, -1, -1):
            x_t = x[:, t, :]
            h_next = self.bwd_cell(x_t, h_bwd)

            if use_zoneout:
                mask = torch.bernoulli(torch.full_like(h_next, 1.0 - self.zoneout_prob))
                h_bwd = mask * h_next + (1 - mask) * h_bwd
            elif self.zoneout_prob > 0.0:
                h_bwd = (1 - self.zoneout_prob) * h_next + self.zoneout_prob * h_bwd
            else:
                h_bwd = h_next

            outputs_bwd.insert(0, h_bwd)

        # Stack and Concatenate
        # outputs_fwd: List of (B, H) -> (B, L, H)
        out_fwd = torch.stack(outputs_fwd, dim=1)
        out_bwd = torch.stack(outputs_bwd, dim=1)

        # (B, L, 2*H)
        output = torch.cat([out_fwd, out_bwd], dim=2)

        return output

class ResidualBiGRUBlock(nn.Module):
    """
    Standard Bidirectional GRU block with Inter-Layer Dropout.
    Replaces Zoneout to improve local feature learning (Cite solution_lesson_node_00076).
    """

    def __init__(
        self,
        input_size: int,
        hidden_size: int,
        dropout: float = Config.DROPOUT,
    ):
        super().__init__()
        self.gru = nn.GRU(
            input_size=input_size,
            hidden_size=hidden_size,
            batch_first=True,
            bidirectional=True,
        )
        self.dropout = nn.Dropout(dropout)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x (torch.Tensor): Input sequence (Batch, Seq_Len, Input_Size).
        Returns:
            torch.Tensor: Output sequence (Batch, Seq_Len, Hidden_Size * 2).
        """
        # GRU Output: (B, L, 2*H_hidden)
        # We assume 2*H_hidden matches input_size for the residual stream
        out, _ = self.gru(x)
        return self.dropout(out)
