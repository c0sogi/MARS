class RNAModel(nn.Module):
    """
    Structural Shortcut Wide-Stream Residual BiGRU.
    Integrates sequence, loop, and distance embeddings, processes them with a deep
    residual BiGRU backbone, and uses a structural shortcut readout to fuse
    paired base features.
    """

    def __init__(self, config):
        super().__init__()
        self.config = config

        # 1. Embeddings
        self.seq_embed = nn.Embedding(4, config.EMBED_DIM)
        self.loop_embed = nn.Embedding(7, config.EMBED_DIM)
        self.dist_embed = SinusoidalPositionalEmbedding(config.EMBED_DIM)

        # Input Projection: Concatenate 3 embeddings -> Hidden Dim
        input_dim = config.EMBED_DIM * 3
        self.input_proj = nn.Linear(input_dim, config.HIDDEN_DIM)

        # 2. Recurrent Stem
        self.stem = nn.GRU(
            config.HIDDEN_DIM,
            config.HIDDEN_DIM // 2,
            num_layers=1,
            batch_first=True,
            bidirectional=True,
        )

        # 3. Backbone: Stack of Residual Blocks
        self.blocks = nn.ModuleList(
            [ResidualBiGRUBlock(config) for _ in range(config.N_LAYERS)]
        )

        # 4. Aggregation: Scalar Mixture
        # Weights for Stem + N_LAYERS blocks
        self.agg_weights = nn.Parameter(torch.zeros(config.N_LAYERS + 1))

        # 5. Structural Shortcut Readout
        # Projects concatenated local and partner states ([H; H] -> 2H) to targets
        self.readout = nn.Sequential(
            nn.Linear(config.HIDDEN_DIM * 2, config.HIDDEN_DIM),
            nn.ReLU(),
            nn.Dropout(config.DROPOUT),
            nn.Linear(config.HIDDEN_DIM, config.N_TARGETS),
        )

    def forward(self, batch):
        seq = batch["sequence"]
        loop = batch["loop_type"]
        dist = batch["distance"]
        pair_idx = batch["pair_index"]

        # --- Embeddings ---
        emb_seq = self.seq_embed(seq)
        emb_loop = self.loop_embed(loop)
        emb_dist = self.dist_embed(dist)

        # Concatenate and project
        x = torch.cat([emb_seq, emb_loop, emb_dist], dim=-1)
        x = self.input_proj(x)

        # --- Stem ---
        x, _ = self.stem(x)
        layer_outputs = [x]

        # --- Residual Blocks ---
        current_x = x
        for block in self.blocks:
            current_x = block(current_x)
            layer_outputs.append(current_x)

        # --- Aggregation ---
        # Compute softmax weights
        weights = F.softmax(self.agg_weights, dim=0)

        # Stack outputs: [N_layers+1, B, L, H]
        stacked_outputs = torch.stack(layer_outputs, dim=0)

        # Weighted sum: [B, L, H]
        agg_state = (stacked_outputs * weights.view(-1, 1, 1, 1)).sum(dim=0)

        # --- Structural Shortcut Readout ---
        B, L, H = agg_state.shape

        # Identify valid pairs (pair_index != -1)
        valid_mask = pair_idx != -1  # [B, L]

        # Create safe indices for gathering (replace -1 with 0)
        safe_pair_idx = pair_idx.clone()
        safe_pair_idx[~valid_mask] = 0

        # Expand indices for gather: [B, L, H]
        gather_idx = safe_pair_idx.unsqueeze(-1).expand(-1, -1, H)

        # Gather partner states
        partner_states = torch.gather(agg_state, 1, gather_idx)

        # Zero out partner states where there is no pair
        partner_states = partner_states * valid_mask.unsqueeze(-1).float()

        # Concatenate local state and partner state
        fused = torch.cat([agg_state, partner_states], dim=-1)  # [B, L, 2H]

        # Project to targets
        logits = self.readout(fused)  # [B, L, 3]

        return logits
