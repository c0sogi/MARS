class LayerScale(nn.Module):
    """
    LayerScale: A learnable diagonal scaling matrix for residual branches.
    Initialized to a small value (epsilon) to improve training stability of deep/wide networks.
    """

    def __init__(self, dim, init_value=1e-4):
        super().__init__()
        self.gamma = nn.Parameter(init_value * torch.ones(dim))

    def forward(self, x):
        return x * self.gamma


class ResidualBlock(nn.Module):
    """
    Wide-Stream Residual Block with Pre-LayerNorm and LayerScale stabilization.
    Maintains the full hidden dimension throughout the block.
    """

    def __init__(self, dim, dropout=0.1, layer_scale_init=1e-4):
        super().__init__()
        self.norm = nn.LayerNorm(dim)
        # BiGRU: Hidden size is dim // 2 per direction, so output is dim.
        self.bigru = nn.GRU(dim, dim // 2, batch_first=True, bidirectional=True)
        self.dropout = nn.Dropout(dropout)
        self.layer_scale = LayerScale(dim, init_value=layer_scale_init)

    def forward(self, x):
        residual = x

        # Pre-LayerNorm
        out = self.norm(x)

        # Wide BiGRU
        out, _ = self.bigru(out)

        # Regularization
        out = self.dropout(out)

        # Gating
        out = self.layer_scale(out)

        # Residual Connection
        return residual + out
