def resize_with_pad(image, target_size, interpolation=cv2.INTER_LINEAR, padding_value=0):
    """
    Resizes an image maintaining aspect ratio and pads to target size.
    Cite solution_lesson_node_00035.
    
    Args:
        image (np.array): Input image (H, W) or (H, W, C).
        target_size (tuple): (Target_H, Target_W).
        interpolation: OpenCv interpolation method.
        padding_value: Value to pad with.

    Returns:
        tuple: (padded_image, (top, bottom, left, right))
    """
    h, w = image.shape[:2]
    target_h, target_w = target_size
    
    scale = min(target_h / h, target_w / w)
    new_h, new_w = int(h * scale), int(w * scale)
    
    resized = cv2.resize(image, (new_w, new_h), interpolation=interpolation)
    
    delta_h = target_h - new_h
    delta_w = target_w - new_w
    top, bottom = delta_h // 2, delta_h - (delta_h // 2)
    left, right = delta_w // 2, delta_w - (delta_w // 2)
    
    if len(image.shape) == 3:
        padded = cv2.copyMakeBorder(
            resized, top, bottom, left, right, 
            cv2.BORDER_CONSTANT, value=[padding_value]*image.shape[2]
        )
    else:
        padded = cv2.copyMakeBorder(
            resized, top, bottom, left, right, 
            cv2.BORDER_CONSTANT, value=padding_value
        )
        
    return padded, (top, bottom, left, right)


def recover_original(padded_image, original_shape, target_shape, interpolation=cv2.INTER_NEAREST):
    """
    Reverses the resize_with_pad operation to get back to original dimensions.
    
    Args:
        padded_image (np.array): Image in target_shape.
        original_shape (tuple): (Orig_H, Orig_W).
        target_shape (tuple): (Target_H, Target_W) used during padding.
        interpolation: OpenCV interpolation method.

    Returns:
        np.array: Image resized back to original_shape.
    """
    orig_h, orig_w = original_shape
    target_h, target_w = target_shape
    
    scale = min(target_h / orig_h, target_w / orig_w)
    new_h, new_w = int(orig_h * scale), int(orig_w * scale)
    
    delta_h = target_h - new_h
    delta_w = target_w - new_w
    top = delta_h // 2
    left = delta_w // 2
    
    # Crop the padding
    cropped = padded_image[top:top+new_h, left:left+new_w]
    
    # Resize back to original
    recovered = cv2.resize(cropped, (orig_w, orig_h), interpolation=interpolation)
    
    return recovered
