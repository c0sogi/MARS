3.  **Update `library/evaluate.py`**: Replace the naive resizing in the evaluation and submission generation loops with `crop_resize_back` to ensure masks are mapped correctly back to the original image space.

def resize_pad(image, target_size, interpolation=cv2.INTER_LINEAR, padding_value=0):
    """
    Resizes image to target_size (H, W) maintaining aspect ratio and padding the rest.
    """
    h, w = image.shape[:2]
    target_h, target_w = target_size

    scale = min(target_w / w, target_h / h)
    new_w = int(w * scale)
    new_h = int(h * scale)

    resized = cv2.resize(image, (new_w, new_h), interpolation=interpolation)

    # Create canvas
    if len(image.shape) == 3:
        canvas = np.full(
            (target_h, target_w, image.shape[2]), padding_value, dtype=image.dtype
        )
        canvas[:new_h, :new_w, :] = resized
    else:
        canvas = np.full((target_h, target_w), padding_value, dtype=image.dtype)
        canvas[:new_h, :new_w] = resized

    return canvas


def crop_resize_back(img, original_size, interpolation=cv2.INTER_NEAREST):
    """
    Reverses resize_pad: crops valid region and resizes to original_size (H, W).
    img is assumed to be (target_h, target_w, ...)
    """
    target_h, target_w = img.shape[:2]
    orig_h, orig_w = original_size

    scale = min(target_w / orig_w, target_h / orig_h)
    valid_w = int(orig_w * scale)
    valid_h = int(orig_h * scale)

    cropped = img[:valid_h, :valid_w]
    return cv2.resize(cropped, (orig_w, orig_h), interpolation=interpolation)
