def engineer_keyword_features(df):
    """
    Generates explicit keyword counts for high-signal categories (Politeness, Need, Reciprocity).
    Explicit counts help tree models split on critical narrative elements.
    Cite solution_lesson_node_00013
    """
    if config.TEXT_COL in df.columns:
        # Lowercase for matching
        texts = df[config.TEXT_COL].fillna("").astype(str).str.lower()

        # 1. Politeness
        df["kw_please"] = texts.apply(lambda x: x.count("please"))
        df["kw_thanks"] = texts.apply(lambda x: x.count("thanks") + x.count("thank you") + x.count("appreciated"))

        # 2. Need / Urgency
        df["kw_money"] = texts.apply(lambda x: x.count("money") + x.count("broke") + x.count("rent") + x.count("bills"))
        df["kw_hungry"] = texts.apply(lambda x: x.count("hungry") + x.count("starving") + x.count("food") + x.count("eat"))

        # 3. Reciprocity
        df["kw_repay"] = texts.apply(lambda x: x.count("repay") + x.count("return") + x.count("pay it forward"))

    return df
