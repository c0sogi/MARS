class MultiScaleStem(nn.Module):
    """
    Multi-Scale Stem that aggregates features from multiple temporal receptive fields.
    """

    def __init__(self, in_channels, out_channels_per_branch, kernel_sizes):
        super(MultiScaleStem, self).__init__()
        self.branches = nn.ModuleList(
            [
                SingleScaleStem(in_channels, out_channels_per_branch, k)
                for k in kernel_sizes
            ]
        )

    def forward(self, x):
        # x: (B, T, C)
        # Concatenate outputs from all branches along the channel dimension
        outputs = [branch(x) for branch in self.branches]
        return torch.cat(outputs, dim=2)
