import numpy as np
import scipy.ndimage
from itertools import groupby
from library.config import Config


def levenshtein_distance(hyp, ref):
    """
    Computes the Levenshtein distance between two sequences.

    Args:
        hyp (list): Hypothesis sequence (predicted).
        ref (list): Reference sequence (ground truth).

    Returns:
        int: The edit distance.
    """
    m, n = len(hyp), len(ref)
    if m < n:
        return levenshtein_distance(ref, hyp)
    if n == 0:
        return m

    previous_row = range(n + 1)
    for i, c1 in enumerate(hyp):
        current_row = [i + 1]
        for j, c2 in enumerate(ref):
            insertions = previous_row[j + 1] + 1
            deletions = current_row[j] + 1
            substitutions = previous_row[j] + (c1 != c2)
            current_row.append(min(insertions, deletions, substitutions))
        previous_row = current_row

    return previous_row[n]


def decode_sequence(predictions, apply_median_filter=False):
    """
    Decodes a frame-wise prediction sequence into a list of gesture IDs.

    Steps:
    1. (Optional) Apply Median Filter to smooth noise.
    2. Collapse consecutive duplicates (Run-Length Encoding).
    3. Filter out segments shorter than Config.MIN_GESTURE_LENGTH.
    4. Remove background class (0).

    Args:
        predictions (np.ndarray): Array of shape (T,) containing class IDs.
        apply_median_filter (bool): Whether to apply smoothing.

    Returns:
        list: Ordered list of recognized gesture IDs.
    """
    if apply_median_filter:
        predictions = scipy.ndimage.median_filter(
            predictions, size=Config.MEDIAN_FILTER_KERNEL, mode="nearest"
        )

    decoded = []

    # Group consecutive identical values
    for k, g in groupby(predictions):
        # Class 0 is Background
        if k != 0:
            # Calculate length of the segment
            length = sum(1 for _ in g)

            # Filter out short, spurious gestures
            if length >= Config.MIN_GESTURE_LENGTH:
                decoded.append(int(k))

    return decoded
