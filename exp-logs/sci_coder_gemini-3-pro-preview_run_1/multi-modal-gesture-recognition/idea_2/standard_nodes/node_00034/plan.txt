import numpy as np
import scipy.ndimage
from library.config import Config


def levenshtein_distance(preds, targets):
    """
    Computes the Levenshtein distance between two sequences.
    Args:
        preds (list): List of predicted class IDs.
        targets (list): List of target class IDs.
    Returns:
        int: The edit distance.
    """
    n = len(preds)
    m = len(targets)

    if n == 0:
        return m
    if m == 0:
        return n

    # Initialize matrix
    d = np.zeros((n + 1, m + 1), dtype=int)

    for i in range(n + 1):
        d[i, 0] = i
    for j in range(m + 1):
        d[0, j] = j

    for i in range(1, n + 1):
        for j in range(1, m + 1):
            cost = 0 if preds[i - 1] == targets[j - 1] else 1
            d[i, j] = min(
                d[i - 1, j] + 1,  # deletion
                d[i, j - 1] + 1,  # insertion
                d[i - 1, j - 1] + cost,
            )  # substitution

    return d[n, m]


def decode_sequence(frame_predictions, apply_median_filter=True):
    """
    Decodes frame-wise predictions into a sequence of gesture IDs.
    Args:
        frame_predictions (np.array): Array of shape (T,) containing class IDs.
        apply_median_filter (bool): Whether to smooth predictions.
    Returns:
        list: Ordered list of gesture IDs (excluding background).
    """
    if len(frame_predictions) == 0:
        return []

    # 1. Smoothing
    if apply_median_filter:
        # Ensure kernel size is valid (odd)
        k = Config.MEDIAN_FILTER_KERNEL
        if k % 2 == 0:
            k += 1
        frame_predictions = scipy.ndimage.median_filter(
            frame_predictions, size=k, mode="nearest"
        )

    # 2. Run-Length Encoding to collapse repeats and filter by duration
    decoded_gestures = []

    if len(frame_predictions) == 0:
        return []

    current_label = frame_predictions[0]
    current_len = 1

    # Helper to add if valid
    def add_if_valid(label, length):
        # Class 0 is background/null
        if label != 0 and length >= Config.MIN_GESTURE_LENGTH:
            decoded_gestures.append(int(label))

    for i in range(1, len(frame_predictions)):
        label = frame_predictions[i]
        if label == current_label:
            current_len += 1
        else:
            add_if_valid(current_label, current_len)
            current_label = label
            current_len = 1

    # Add last
    add_if_valid(current_label, current_len)

    return decoded_gestures
