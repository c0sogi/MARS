def masked_global_average_pooling(x, lengths):
    """
    Computes Global Average Pooling while ignoring padded time steps.
    
    Args:
        x (torch.Tensor): (Batch, Time, Channels)
        lengths (torch.Tensor): (Batch,) containing valid sequence lengths
        
    Returns:
        torch.Tensor: (Batch, Channels)
    """
    # Create mask: (B, T) -> 1 for valid, 0 for pad
    mask = torch.arange(x.size(1), device=x.device)[None, :] < lengths[:, None]
    mask = mask.float().unsqueeze(2)  # (B, T, 1)

    # Sum valid steps
    sum_pooled = torch.sum(x * mask, dim=1)

    # Divide by actual lengths (clamp to avoid div by 0)
    lengths_safe = torch.clamp(lengths.float().unsqueeze(1), min=1.0)
    
    return sum_pooled / lengths_safe
