def get_class_weights(label2id):
    """
    Computes class weights for the training set to handle imbalance.
    Weights are inversely proportional to class frequency.
    """
    if not TRAIN_META_PATH.exists():
        return None
    
    df = pd.read_csv(TRAIN_META_PATH)
    
    # Filter valid categories
    valid_cats = set(label2id.keys())
    df = df[df["category_id"].isin(valid_cats)]
    
    # Map to label indices
    labels = df["category_id"].map(label2id)
    
    # Count frequencies
    counts = labels.value_counts().sort_index()
    
    # Create count array for all classes
    num_classes = len(label2id)
    count_array = np.zeros(num_classes)
    count_array[counts.index] = counts.values
    
    # Avoid division by zero
    count_array = np.maximum(count_array, 1)
    
    # Compute balanced weights: Total / (NumClasses * Count)
    total_samples = len(df)
    weights = total_samples / (num_classes * count_array)
    
    return torch.tensor(weights, dtype=torch.float32)
