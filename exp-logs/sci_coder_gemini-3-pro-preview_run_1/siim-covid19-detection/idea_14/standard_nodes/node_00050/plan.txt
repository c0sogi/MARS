def detection_collate(batch):
    """
    Custom collate function for object detection.
    Stacks images and fixed-size targets (masks, study_labels).
    Keeps variable-size targets (boxes, labels) as lists.
    """
    images = []
    targets = []
    for img, target in batch:
        images.append(img)
        targets.append(target)

    images = torch.stack(images, dim=0)

    batched_targets = {}
    # Keys that should be stacked (tensors of fixed size)
    stackable_keys = {"mask", "study_label"}

    # Assume all targets have same keys
    for key in targets[0].keys():
        if key in stackable_keys:
            batched_targets[key] = torch.stack([t[key] for t in targets], dim=0)
        else:
            batched_targets[key] = [t[key] for t in targets]

    return images, batched_targets
