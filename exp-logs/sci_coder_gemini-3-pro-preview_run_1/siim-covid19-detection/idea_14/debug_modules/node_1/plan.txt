from library.inference import run_inference


def detection_collate(batch):
    """
    Custom collate function for object detection.
    Stacks images and fixed-size targets, but keeps variable-size targets (boxes) as lists.
    """
    images = []
    targets = {
        "mask": [],
        "study_label": [],
        "boxes": [],
        "labels": [],
        "image_id": [],
        "study_id": [],
    }

    for img, tgt in batch:
        images.append(img)
        for k in targets.keys():
            targets[k].append(tgt[k])

    # Stack fields that have consistent shapes
    images = torch.stack(images, dim=0)
    targets["mask"] = torch.stack(targets["mask"], dim=0)
    targets["study_label"] = torch.stack(targets["study_label"], dim=0)

    # Leave 'boxes', 'labels', 'image_id', 'study_id' as lists
    return images, targets
