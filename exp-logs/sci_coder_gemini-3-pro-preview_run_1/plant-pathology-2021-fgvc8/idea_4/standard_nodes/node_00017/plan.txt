def inference(model, dataloader, device):
    """
    Generates predictions for the test set (or validation set).
    Supports TTA (Horizontal Flip) if configured.
    """
    model.eval()
    preds_all = []

    with torch.no_grad():
        for step, batch in enumerate(dataloader):
            # Handle both (image, target) from val_loader and (image) from test_loader
            if isinstance(batch, (tuple, list)):
                images = batch[0]
            else:
                images = batch
            
            images = images.to(device, non_blocking=True)
            
            # 1. Forward Pass (Original)
            with autocast(enabled=Config.USE_AMP):
                logits = model(images)
                probs = torch.sigmoid(logits)

            # 2. TTA (Horizontal Flip)
            if Config.USE_TTA:
                images_flip = torch.flip(images, dims=[3])  # Flip width (B, C, H, W)
                with autocast(enabled=Config.USE_AMP):
                    logits_flip = model(images_flip)
                    probs_flip = torch.sigmoid(logits_flip)
                
                # Average probabilities
                probs = (probs + probs_flip) / 2.0

            preds_all.append(probs.cpu().numpy())

    return np.concatenate(preds_all)
