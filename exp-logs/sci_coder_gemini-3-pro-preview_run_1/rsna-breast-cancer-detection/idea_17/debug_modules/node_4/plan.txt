from library.train import train

def cleanup():
    """
    Aggressive memory cleanup to handle persistent environments.
    Clears exception tracebacks and forces garbage collection.
    """
    # 1. Clear traceback references which might hold variables (Cite debug_lesson_18)
    if hasattr(sys, 'last_traceback'):
        sys.last_traceback = None
    if hasattr(sys, 'last_value'):
        sys.last_value = None
    if hasattr(sys, 'last_type'):
        sys.last_type = None
        
    # 2. Force Garbage Collection
    gc.collect()
    
    # 3. Clear CUDA Cache
    if torch.cuda.is_available():
        torch.cuda.empty_cache()
