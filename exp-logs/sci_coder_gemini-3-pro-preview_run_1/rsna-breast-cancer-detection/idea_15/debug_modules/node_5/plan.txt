# Aggressive cleanup of zombie GPU resources from previous runs (Cite debug_lesson_7, Cite debug_lesson_15)
def cleanup_gpu_resources():
    print("Cleaning up GPU resources...")
    
    # 1. Clear System Tracebacks (Cite debug_lesson_18)
    if hasattr(sys, "last_traceback"):
        sys.last_traceback = None
    
    # 2. Purge Global Variables (Cite debug_lesson_7)
    # We iterate over a list of keys to avoid runtime errors while modifying the dict
    for name in list(globals().keys()):
        # Skip internal variables and the cleanup function itself
        if name.startswith("_") or name in ["cleanup_gpu_resources", "gc", "torch", "sys"]:
            continue
        
        obj = globals()[name]
        
        # Check for PyTorch objects that hold GPU memory
        # We also check for Optimizers as they hold references to parameters (Cite debug_lesson_15)
        try:
            # Import optim locally to ensure availability for check
            import torch.optim
            if isinstance(obj, (torch.nn.Module, torch.Tensor, torch.optim.Optimizer)):
                print(f"Deleting zombie object: {name}")
                del globals()[name]
        except Exception:
            pass
            
    # 3. Garbage Collection & Empty Cache
    gc.collect()
    if torch.cuda.is_available():
        torch.cuda.empty_cache()
    print("GPU cleanup complete.")

# Execute cleanup immediately
