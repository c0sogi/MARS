    roundness = (4 * np.pi * area) / (perimeter**2) if perimeter > 0 else 0.0

    print("Preprocessing (Yeo-Johnson + StandardScaler)...")
    # 1. Yeo-Johnson
    pt = PowerTransformer(method="yeo-johnson", standardize=False)
    X_train_pt = pt.fit_transform(X_train)
    X_val_pt = pt.transform(X_val)
    X_test_pt = pt.transform(X_test)

    # 2. Standard Scaling
    ss = StandardScaler()
    X_train_scaled = ss.fit_transform(X_train_pt)
    X_val_scaled = ss.transform(X_val_pt)
    X_test_scaled = ss.transform(X_test_pt)

    # OAS Model Training
    print("Training OAS Linear Discriminant...")

    # Compute Class Means and Priors
    means = np.zeros((n_classes, X_train_scaled.shape[1]), dtype=np.float64)
    priors = np.zeros(n_classes, dtype=np.float64)

    for k in range(n_classes):
        X_k = X_train_scaled[y_train_enc == k]
        means[k] = np.mean(X_k, axis=0)
        priors[k] = len(X_k) / len(X_train_scaled)

    # Compute Residuals (Center data by class mean)
    residuals = np.zeros_like(X_train_scaled)
    for k in range(n_classes):
        mask = y_train_enc == k
        residuals[mask] = X_train_scaled[mask] - means[k]

    # Fit OAS to Residuals
    oas = OAS(assume_centered=True)
    oas.fit(residuals)
    precision = oas.precision_  # Inverse Covariance Matrix

    # Calculate Weights (W) and Bias (b) for Linear Discriminant
    # W = means @ precision (Shape: n_classes x n_features)
    W = np.dot(means, precision)

    quad_term = 0.5 * np.sum(means * W, axis=1)
    b = -quad_term + np.log(priors)

    Z_val = np.dot(X_val_scaled, W.T) + b
