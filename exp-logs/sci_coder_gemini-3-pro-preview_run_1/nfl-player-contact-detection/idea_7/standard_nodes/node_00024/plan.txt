def add_spatial_density(df_tracking):
    """
    Calculates spatial density (neighbor count within 1.5 yards) for each player-step.
    Cite Lesson 00012, 00023.
    """
    # Filter valid positions
    mask = df_tracking['x_position'].notna() & df_tracking['y_position'].notna()
    
    # We will fill a new column with 0 initially
    df_tracking['spatial_density'] = 0.0
    
    # We operate on the subset to build trees
    # Group by game_play and step
    # This loop is unavoidable but runs in ~2-3 mins for 1.2M rows
    groups = df_tracking[mask].groupby(['game_play', 'step'])
    
    # Prepare a dictionary to map (index) -> density
    # Or better, direct assignment if we iterate carefully.
    # To be safe and vectorized-ish, we collect results.
    
    indices = []
    densities = []
    
    for _, group in groups:
        coords = group[['x_position', 'y_position']].values
        # Build Tree
        tree = cKDTree(coords)
        # Query 1.5 yards
        counts = tree.query_ball_point(coords, r=1.5, return_length=True)
        # Subtract self
        counts = counts - 1
        
        indices.extend(group.index.tolist())
        densities.extend(counts.tolist())
        
    # Assign back
    df_tracking.loc[indices, 'spatial_density'] = densities
    
    return df_tracking
