def load_mip_slice(directory_path):
    """
    Computes the Maximum Intensity Projection (MIP) from a directory of DICOM files.
    Cite solution_lesson_node_00002.

    1. Lists and sorts .dcm files.
    2. Selects a central subset of slices to avoid noise at volume ends.
    3. Reads slices, resizes them to Config.IMG_SIZE.
    4. Computes pixel-wise maximum across the selected slices.

    Args:
        directory_path (str): Path to the directory containing .dcm files.

    Returns:
        np.ndarray: The MIP image array.
    """
    fallback_shape = (Config.IMG_SIZE, Config.IMG_SIZE)

    if not os.path.exists(directory_path):
        return np.zeros(fallback_shape, dtype=np.uint8)

    files = [f for f in os.listdir(directory_path) if f.endswith(".dcm")]

    if not files:
        return np.zeros(fallback_shape, dtype=np.uint8)

    # Sort files numerically
    files.sort(key=_extract_number)

    # Select middle 60% of slices to capture tumor while ignoring skull/neck noise
    n_files = len(files)
    start_idx = int(n_files * 0.2)
    end_idx = int(n_files * 0.8)
    
    if start_idx == end_idx:
        end_idx += 1
        
    selected_files = files[start_idx:end_idx]

    # Subsample if too many files to keep processing fast (max 64 slices)
    if len(selected_files) > 64:
        step = len(selected_files) // 64
        selected_files = selected_files[::step]

    imgs = []
    for f in selected_files:
        full_path = os.path.join(directory_path, f)
        img = read_dicom_file(full_path)
        
        if img is not None:
            # Resize to target size immediately to save memory and ensure alignment
            # Note: interpolation linear is fine for structural MRI
            try:
                if img.shape[:2] != fallback_shape:
                    img = cv2.resize(img, fallback_shape, interpolation=cv2.INTER_LINEAR)
                imgs.append(img)
            except Exception:
                continue

    if not imgs:
        return np.zeros(fallback_shape, dtype=np.uint8)

    # Stack and compute Max Intensity Projection
    stack = np.array(imgs)
    mip_img = np.max(stack, axis=0)

    return mip_img
