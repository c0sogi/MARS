    """
    Selects indices for Lower, Center, and Upper instances based on the median.
    Clamps indices to be within valid range [0, len-1].
    """
    n = len(file_list)
    if n == 0:
        return []

    mid = n // 2

    # Define indices: Median - Stride, Median, Median + Stride
    indices = [mid - stride, mid, mid + stride]

    # Clamp to valid range
    indices = [max(0, min(idx, n - 1)) for idx in indices]
