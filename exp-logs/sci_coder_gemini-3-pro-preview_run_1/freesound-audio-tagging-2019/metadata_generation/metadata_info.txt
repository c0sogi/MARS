# Python Script for Metadata Generation
```
import os
import pandas as pd
import numpy as np
import glob
from skmultilearn.model_selection import iterative_train_test_split

# Constants
INPUT_DIR = "./input"
METADATA_DIR = "./metadata"
RANDOM_STATE = 42

# Ensure metadata directory exists
os.makedirs(METADATA_DIR, exist_ok=True)


def generate_metadata():
    print("Loading raw data...")

    # Load sample submission to get the class list
    sub_df = pd.read_csv(os.path.join(INPUT_DIR, "sample_submission.csv"))
    classes = list(sub_df.columns[1:])  # Exclude fname
    print(f"Found {len(classes)} classes.")

    # Load training data
    train_curated = pd.read_csv(os.path.join(INPUT_DIR, "train_curated.csv"))
    train_noisy = pd.read_csv(os.path.join(INPUT_DIR, "train_noisy.csv"))

    # Data Cleaning
    # Files to remove from curated set
    corrupted_files = ["1d44b0bd.wav"]
    wrong_label_files = [
        "f76181c4.wav",
        "77b925c2.wav",
        "6a1f682a.wav",
        "c7db12aa.wav",
        "7752cc8a.wav",
    ]
    drop_files = set(corrupted_files + wrong_label_files)

    print(f"Curated samples before cleaning: {len(train_curated)}")
    train_curated = train_curated[~train_curated["fname"].isin(drop_files)].reset_index(
        drop=True
    )
    print(f"Curated samples after cleaning: {len(train_curated)}")

    # Prepare Multi-label Stratification
    # We need to turn string labels into a binary matrix for skmultilearn
    print("Preparing stratified split...")

    # Helper to vectorize labels
    def vectorize_labels(df, cls_list):
        matrix = np.zeros((len(df), len(cls_list)), dtype=int)
        for i, row in df.iterrows():
            lbls = row["labels"].split(",")
            for l in lbls:
                if l in cls_list:
                    idx = cls_list.index(l)
                    matrix[i, idx] = 1
        return matrix

    # Vectorize curated data
    y_curated = vectorize_labels(train_curated, classes)

    # Since iterative_train_test_split requires X to be an array, we use the dataframe index
    X_indices = train_curated.index.values.reshape(-1, 1)

    # Split curated into Train (80%) and Val (20%)
    # iterative_train_test_split returns X_train, y_train, X_test, y_test
    X_train_idx, y_train_split, X_val_idx, y_val_split = iterative_train_test_split(
        X_indices, y_curated, test_size=0.2
    )

    # Flatten indices
    train_idx = X_train_idx.flatten()
    val_idx = X_val_idx.flatten()

    # Create DataFrames
    df_curated_train = train_curated.iloc[train_idx].copy()
    df_curated_val = train_curated.iloc[val_idx].copy()

    # Add one-hot columns to DataFrames (optional but good for metadata)
    # We'll just stick to the required format, but having the binary targets is useful.
    # Let's add the binary targets to the metadata files for ease of loading.
    df_curated_train[classes] = y_train_split
    df_curated_val[classes] = y_val_split

    # Process Noisy Data
    # Noisy data goes purely into training
    y_noisy = vectorize_labels(train_noisy, classes)
    train_noisy[classes] = y_noisy

    # Assign Filepaths
    # Paths must be relative to ./input
    df_curated_train["filepath"] = df_curated_train["fname"].apply(
        lambda x: os.path.join("train_curated", x)
    )
    df_curated_val["filepath"] = df_curated_val["fname"].apply(
        lambda x: os.path.join("train_curated", x)
    )
    train_noisy["filepath"] = train_noisy["fname"].apply(
        lambda x: os.path.join("train_noisy", x)
    )

    # Combine for Final Training Set
    # Train = Curated_Train + Noisy
    final_train = pd.concat([df_curated_train, train_noisy], ignore_index=True)
    final_val = df_curated_val.reset_index(drop=True)

    # Process Test Data
    final_test = sub_df.copy()
    final_test["filepath"] = final_test["fname"].apply(
        lambda x: os.path.join("test", x)
    )
    # sample_submission already has the class columns initialized to 0

    # Save Metadata
    print("Saving metadata...")
    final_train.to_csv(os.path.join(METADATA_DIR, "train.csv"), index=False)
    final_val.to_csv(os.path.join(METADATA_DIR, "val.csv"), index=False)
    final_test.to_csv(os.path.join(METADATA_DIR, "test.csv"), index=False)

    return final_train, final_val, final_test, classes


def verify_metadata(train_df, val_df, test_df, classes):
    print("\n=== Verifying Metadata ===")

    # 1. Summary Statistics
    print(f"Train set shape: {train_df.shape}")
    print(f"Val set shape: {val_df.shape}")
    print(f"Test set shape: {test_df.shape}")

    # Check class distribution in Val vs Train (Curated part of train)
    # We only check if columns exist and have data
    label_counts_val = val_df[classes].sum().sum()
    print(f"Total positive labels in Validation: {label_counts_val}")

    # 2. File Path Verification
    def check_paths(df, name):
        print(f"Checking file paths for {name}...")
        sample_paths = (
            df["filepath"]
            .sample(n=min(1000, len(df)), random_state=RANDOM_STATE)
            .tolist()
        )
        missing_count = 0
        missing_samples = []

        for p in sample_paths:
            full_path = os.path.join(INPUT_DIR, p)
            if not os.path.exists(full_path):
                missing_count += 1
                if len(missing_samples) < 5:
                    missing_samples.append(p)

        missing_ratio = missing_count / len(sample_paths)
        print(f"Missing ratio for {name}: {missing_ratio:.4f}")

        if missing_ratio > 0.5:
            print("Sample missing paths:", missing_samples)
            raise FileNotFoundError(
                f"More than 50% of files missing in {name} dataset."
            )

    check_paths(train_df, "Train")
    check_paths(val_df, "Validation")
    check_paths(test_df, "Test")

    # 3. Validation Split Verification
    print("Verifying split requirements...")
    # Verify split size (approx 20% of curated)
    # Note: Train contains noisy data too, so we can't just do len(val)/len(train)
    # We reconstruct the curated size
    n_curated_train = train_df[
        train_df["filepath"].str.contains("train_curated")
    ].shape[0]
    n_curated_val = val_df.shape[0]
    total_curated = n_curated_train + n_curated_val
    split_ratio = n_curated_val / total_curated

    print(f"Curated Train count: {n_curated_train}")
    print(f"Curated Val count: {n_curated_val}")
    print(f"Split ratio (Val / Total Curated): {split_ratio:.4f}")

    # Allow small deviation due to iterative stratification rounding
    if not (0.18 < split_ratio < 0.22):
        raise AssertionError(
            f"Validation split ratio {split_ratio} is not close to 0.2"
        )

    # Verify no data leakage
    train_fnames = set(train_df["fname"])
    val_fnames = set(val_df["fname"])
    intersection = train_fnames.intersection(val_fnames)
    if len(intersection) > 0:
        raise AssertionError(
            f"Data leakage detected! {len(intersection)} files overlap between train and val."
        )

    print("Verification passed successfully.")


if __name__ == "__main__":
    try:
        train, val, test, classes = generate_metadata()
        verify_metadata(train, val, test, classes)
    except Exception as e:
        print(f"Error during execution: {e}")
        raise e

```

# Document
Based on the provided Python script and its execution output, here is the documentation for the files generated in the `./metadata` directory.

### 1. `./metadata/train.csv`

*   **Content and Purpose:**
    *   **Content:** This file contains the combined training dataset. It aggregates approximately 80% of the "curated" dataset (cleaned of known errors) and 100% of the "noisy" dataset. It includes the filename, the raw string labels, the relative file path to the audio file, and 80 one-hot encoded columns representing the specific audio tags (e.g., `Bark`, `Slam`).
    *   **Purpose:** This file is the primary input for training the machine learning model. The inclusion of both curated and noisy data allows the model to learn from a large volume of data while maintaining a core of high-quality labels.
*   **Schema / Structure:**
    *   **Rows:** 23,772
    *   **Columns:** 83
    *   **Column Details:**
        *   `fname` (String): The unique identifier for the audio file (e.g., `0006ae4e.wav`).
        *   `labels` (String): The ground truth labels as a comma-separated string (e.g., `"Bark"` or `"Walk_and_footsteps,Slam"`).
        *   `filepath` (String): The relative path to the audio file from the input root (e.g., `train_curated/0006ae4e.wav` or `train_noisy/00097e21.wav`).
        *   `[Class Columns]` (Integer): 80 columns named after specific audio categories (e.g., `Accelerating_and_revving_and_vroom`, `Accordion`, ... `Zipper_(clothing)`). Values are `1` if the label is present, `0` otherwise.
*   **Loading Method:**
    ```python
    import pandas as pd
    train_df = pd.read_csv("./metadata/train.csv")
    ```

### 2. `./metadata/val.csv`

*   **Content and Purpose:**
    *   **Content:** This file contains the validation dataset, consisting of the remaining ~20% of the "curated" dataset (cleaned of known errors). These samples were selected using iterative stratification to ensure the multi-label distribution closely matches the curated portion of the training set.
    *   **Purpose:** This file is used to evaluate the model's performance during training (e.g., calculating the Label-Ranking Average Precision metric). Since it contains only curated data, it serves as a high-quality benchmark to detect overfitting and guide hyperparameter tuning.
*   **Schema / Structure:**
    *   **Rows:** 1,007
    *   **Columns:** 83
    *   **Column Details:**
        *   `fname` (String): The unique identifier for the audio file.
        *   `labels` (String): The ground truth labels as a comma-separated string.
        *   `filepath` (String): The relative path to the audio file (e.g., `train_curated/0019ef41.wav`).
        *   `[Class Columns]` (Integer): 80 columns corresponding to the audio categories. Values are `1` if the label is present, `0` otherwise.
*   **Loading Method:**
    ```python
    import pandas as pd
    val_df = pd.read_csv("./metadata/val.csv")
    ```

### 3. `./metadata/test.csv`

*   **Content and Purpose:**
    *   **Content:** This file contains the list of test files for which predictions must be generated. It is derived from the `sample_submission.csv` file and includes the filename, the relative file path, and placeholders for the 80 class probabilities.
    *   **Purpose:** This file is used during the inference phase. The data loader iterates through these rows, loads the audio from the `filepath`, and the model populates the class columns with predicted probabilities for submission.
*   **Schema / Structure:**
    *   **Rows:** 3,361
    *   **Columns:** 82
    *   **Column Details:**
        *   `fname` (String): The unique identifier for the audio file (e.g., `4260ebea.wav`).
        *   `filepath` (String): The relative path to the audio file (e.g., `test/4260ebea.wav`).
        *   `[Class Columns]` (Integer/Float): 80 columns corresponding to the audio categories. In this file, they are initialized to `0` and are intended to be filled with predicted probabilities (0.0 to 1.0).
        *   *Note:* Unlike train/val, this file does **not** contain a `labels` column.
*   **Loading Method:**
    ```python
    import pandas as pd
    test_df = pd.read_csv("./metadata/test.csv")
    ```
