class GEGLUFeedForward(nn.Module):
    """
    Feed-Forward block with Gated Linear Unit (GEGLU) activation.
    Cite solution_lesson_node_00029
    """

    def __init__(self, d_model, d_ff, dropout):
        super().__init__()
        # Project to 2 * d_ff to create gate and value
        self.linear1 = nn.Linear(d_model, d_ff * 2)
        self.dropout = nn.Dropout(dropout)
        self.linear2 = nn.Linear(d_ff, d_model)

    def forward(self, x):
        x = self.linear1(x)
        x, gate = x.chunk(2, dim=-1)
        x = x * F.gelu(gate)
        x = self.dropout(x)
        x = self.linear2(x)
        return x


class DeGUTEncoderLayer(nn.Module):
    """
    Custom Transformer Encoder Layer using GEGLU FFN and Pre-LN.
    """

    def __init__(self, d_model, nhead, dim_feedforward, dropout):
        super().__init__()
        self.self_attn = nn.MultiheadAttention(
            d_model, nhead, dropout=dropout, batch_first=True
        )

        self.norm1 = nn.LayerNorm(d_model)
        self.norm2 = nn.LayerNorm(d_model)
        self.dropout1 = nn.Dropout(dropout)
        self.dropout2 = nn.Dropout(dropout)

        # GEGLU Block instead of standard FFN
        self.ffn = GEGLUFeedForward(d_model, dim_feedforward, dropout)

    def forward(self, src, src_mask=None, src_key_padding_mask=None):
        # Pre-LN architecture (Norm First)

        # Self Attention
        src2 = self.norm1(src)
        src2, _ = self.self_attn(
            src2,
            src2,
            src2,
            key_padding_mask=src_key_padding_mask,
            attn_mask=src_mask,
        )
        src = src + self.dropout1(src2)

        # FFN
        src2 = self.norm2(src)
        src2 = self.ffn(src2)
        src = src + self.dropout2(src2)

        return src
