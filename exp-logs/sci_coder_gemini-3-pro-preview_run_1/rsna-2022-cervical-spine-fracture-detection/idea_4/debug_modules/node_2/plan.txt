def prepare_segmentation_cache(metadata_df, load_cached_data=True):
    """
    Extracts 2D slices from 3D NIFTI segmentation files and saves them as .npy files.
    This ensures fast random access during training.
    """
    cache_dir = os.path.join(Config.CACHE_DIR, "segmentation_slices")

    # Check if cache exists and we want to load it
    if (
        load_cached_data
        and os.path.exists(cache_dir)
        and len(os.listdir(cache_dir)) > 0
    ):
        # Assume valid if directory exists and is not empty
        return cache_dir

    # Compute from scratch
    os.makedirs(cache_dir, exist_ok=True)

    # Filter for studies that have segmentation
    seg_studies = metadata_df[metadata_df["has_segmentation"] == True]

    print(f"Processing {len(seg_studies)} segmentation files for caching...")

    for _, row in seg_studies.iterrows():
        study_uid = row["StudyInstanceUID"]
        seg_path = row["segmentation_path"]

        if not seg_path or pd.isna(seg_path):
            continue

        full_seg_path = os.path.join(Config.INPUT_DIR, seg_path)
        if not os.path.exists(full_seg_path):
            continue

        try:
            # Load NIFTI
            nii = nib.load(full_seg_path)
            # Reorient to canonical (RAS+) to match standard DICOM orientation roughly
            nii = nib.as_closest_canonical(nii)
            data = nii.get_fdata()

            # Data shape is usually (H, W, D) or (W, H, D).
            # We assume the last dimension is the Z-axis (slices).
            # We also need to handle the fact that DICOMs might be inverted in Z.
            # However, matching slice by slice index is the standard heuristic if
            # DICOMs are sorted by ImagePositionPatient Z.
            # Here we save all slices by index.

            num_slices = data.shape[2]

            # Create study dir
            study_cache_dir = os.path.join(cache_dir, study_uid)
            os.makedirs(study_cache_dir, exist_ok=True)

            for z in range(num_slices):
                # Extract slice
                slice_mask = data[:, :, z]

                # NIFTI (X, Y) often maps to DICOM (Y, X) transposed.
                # We rotate/transpose to align with standard pydicom loading.
                # A common transform is rotation by 90 deg or transpose.
                # Without visual feedback, we assume a transpose aligns (W, H) -> (H, W).
                slice_mask = np.transpose(slice_mask)

                # Binarize: The task is spine localization.
                # Labels 1-7 (Cervical) and 8-19 (Thoracic).
                # We treat anything > 0 as "Spine".
                binary_mask = (slice_mask > 0).astype(np.uint8)

                # Save only if there is a mask to save space, or save all?
                # We save all to maintain indexing.
                save_path = os.path.join(study_cache_dir, f"{z}.npy")
                np.save(save_path, binary_mask)

        except Exception as e:
            print(f"Error processing segmentation for {study_uid}: {e}")

    return cache_dir

def prepare_segmentation_cache(metadata_df, load_cached_data=True):
    """
    Extracts 2D slices from 3D NIFTI segmentation files and saves them as .npy files.
    
    MODIFIED: Since nibabel is missing, we fallback to using Bounding Boxes 
    to generate binary masks for segmentation training.
    """
    cache_dir = os.path.join(Config.CACHE_DIR, "segmentation_slices")

    # Check if cache exists and we want to load it
    if (
        load_cached_data
        and os.path.exists(cache_dir)
        and len(os.listdir(cache_dir)) > 0
    ):
        return cache_dir

    # Compute from scratch
    os.makedirs(cache_dir, exist_ok=True)

    # Load Bounding Boxes
    if not os.path.exists(Config.TRAIN_BBOX_PATH):
        print("Warning: train_bounding_boxes.csv not found. Segmentation cache will be empty.")
        return cache_dir

    bbox_df = pd.read_csv(Config.TRAIN_BBOX_PATH)
    
    # Filter for studies present in the provided metadata
    # This ensures we only process relevant studies (e.g. in the demo subset)
    present_studies = set(metadata_df["StudyInstanceUID"].unique())
    bbox_df = bbox_df[bbox_df["StudyInstanceUID"].isin(present_studies)]

    print(f"Processing bounding boxes for {len(bbox_df['StudyInstanceUID'].unique())} studies...")

    for study_uid, group in bbox_df.groupby("StudyInstanceUID"):
        study_cache_dir = os.path.join(cache_dir, study_uid)
        os.makedirs(study_cache_dir, exist_ok=True)

        # We need to map 'slice_number' from CSV to the index in the sorted file list
        # because SegmentationDataset accesses files by index.
        
        # Get image directory from metadata
        # We assume the metadata has the correct path for this study
        study_meta = metadata_df[metadata_df["StudyInstanceUID"] == study_uid]
        if study_meta.empty:
            continue
            
        rel_path = study_meta.iloc[0]["image_path"]
        image_dir = os.path.join(Config.INPUT_DIR, rel_path)
        
        if not os.path.exists(image_dir):
            continue
            
        # List and sort DICOMs to establish index mapping
        dcm_files = glob.glob(os.path.join(image_dir, "*.dcm"))
        try:
            dcm_files.sort(key=lambda x: int(os.path.splitext(os.path.basename(x))[0]))
        except ValueError:
            dcm_files.sort()
            
        # Map slice_number (filename integer) to index (0..N)
        slice_num_to_idx = {}
        for idx, f in enumerate(dcm_files):
            try:
                # filename '100.dcm' -> 100
                fname_num = int(os.path.splitext(os.path.basename(f))[0])
                slice_num_to_idx[fname_num] = idx
            except ValueError:
                pass

        # Generate masks
        for _, row in group.iterrows():
            slice_num = int(row["slice_number"])
            
            # If this slice exists in the folder
            if slice_num in slice_num_to_idx:
                idx = slice_num_to_idx[slice_num]
                save_path = os.path.join(study_cache_dir, f"{idx}.npy")
                
                # Create mask (512x512)
                # BBox coords are x, y, width, height
                x, y, w, h = int(row["x"]), int(row["y"]), int(row["width"]), int(row["height"])
                
                # Load existing or create new
                if os.path.exists(save_path):
                    mask = np.load(save_path)
                else:
                    mask = np.zeros((Config.ORIGINAL_IMAGE_SIZE, Config.ORIGINAL_IMAGE_SIZE), dtype=np.uint8)
                
                # Draw rectangle (set to 1)
                # Ensure coords are within bounds
                x = max(0, min(x, Config.ORIGINAL_IMAGE_SIZE - 1))
                y = max(0, min(y, Config.ORIGINAL_IMAGE_SIZE - 1))
                w = min(w, Config.ORIGINAL_IMAGE_SIZE - x)
                h = min(h, Config.ORIGINAL_IMAGE_SIZE - y)
                
                if w > 0 and h > 0:
                    mask[y:y+h, x:x+w] = 1
                    np.save(save_path, mask)

    return cache_dir
