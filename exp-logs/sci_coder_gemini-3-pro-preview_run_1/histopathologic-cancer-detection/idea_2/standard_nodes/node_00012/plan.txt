class ModifiedDenseNet(nn.Module):
    """
    A DenseNet121-based architecture modified for small input resolutions (e.g., 48x48).

    Modifications:
    1. Input Stem: Replaces the standard 7x7 stride-2 conv and 3x3 stride-2 maxpool
       with a 3x3 stride-1 conv and Identity pooling. This preserves spatial resolution.
    2. Classifier: Replaces the ImageNet classifier with a binary classification head.
    """

    def __init__(self, pretrained=True):
        """
        Args:
            pretrained (bool): If True, loads ImageNet weights for the backbone.
        """
        super(ModifiedDenseNet, self).__init__()

        # Load the base DenseNet121 model
        if pretrained:
            weights = DenseNet121_Weights.IMAGENET1K_V1
        else:
            weights = None

        self.model = models.densenet121(weights=weights)

        # =====================================================================
        # Modification 1: Input Stem
        # =====================================================================
        # The standard DenseNet starts with:
        #   Conv2d(3, 64, kernel=7, stride=2, padding=3) -> 2x downsample
        #   MaxPool2d(kernel=3, stride=2, padding=1)     -> 2x downsample
        # Total downsample: 4x. For 48x48 input, this results in 12x12 features
        # entering the first block, which is too small.

        # Replace Conv0 with 3x3 stride-1
        self.model.features.conv0 = nn.Conv2d(
            in_channels=3,
            out_channels=64,
            kernel_size=3,
            stride=1,
            padding=1,
            bias=False,
        )

        # Replace Pool0 with Identity (no downsampling)
        self.model.features.pool0 = nn.Identity()

        # Modification 2: Classifier
        # =====================================================================

from torchvision.models import efficientnet_b0, EfficientNet_B0_Weights

class ModifiedEfficientNet(nn.Module):
    """
    An EfficientNet-B0 architecture modified for small input resolutions (e.g., 48x48).
    
    Modifications:
    1. Input Stem: Replaces the standard 3x3 stride-2 conv with a 3x3 stride-1 conv.
       This reduces the total downsampling factor from 32x to 16x, preserving
       spatial resolution for small inputs.
    2. Classifier: Replaces the ImageNet classifier with a binary classification head.
    """

    def __init__(self, pretrained=True):
        """
        Args:
            pretrained (bool): If True, loads ImageNet weights for the backbone.
        """
        super(ModifiedEfficientNet, self).__init__()

        # Load the base EfficientNet-B0 model
        if pretrained:
            weights = EfficientNet_B0_Weights.IMAGENET1K_V1
        else:
            weights = None

        self.model = efficientnet_b0(weights=weights)

        # =====================================================================
        # Modification 1: Input Stem
        # =====================================================================
        # The standard EfficientNet-B0 starts with:
        #   Conv2dNormActivation(..., stride=2) -> 2x downsample
        # We want to change this to stride=1 to keep more spatial details.
        
        # Access the first convolution layer
        # model.features[0] is Conv2dNormActivation
        # model.features[0][0] is the Conv2d layer
        old_conv = self.model.features[0][0]
        
        new_conv = nn.Conv2d(
            in_channels=old_conv.in_channels,
            out_channels=old_conv.out_channels,
            kernel_size=old_conv.kernel_size,
            stride=1,  # Change stride from 2 to 1
            padding=old_conv.padding,
            bias=False
        )
        
        # Copy pretrained weights to the new layer
        if pretrained:
            new_conv.weight.data = old_conv.weight.data
            
        self.model.features[0][0] = new_conv

        # =====================================================================
        # Modification 2: Classifier
        # =====================================================================
        # EfficientNet classifier is Sequential(Dropout, Linear)
        # self.model.classifier[1] is the Linear layer
        in_features = self.model.classifier[1].in_features
        self.model.classifier[1] = nn.Linear(in_features, 1)

    def forward(self, x):
        """
        Forward pass of the network.

        Args:
            x (torch.Tensor): Input tensor of shape (B, 3, H, W)

        Returns:
            torch.Tensor: Logits of shape (B, 1)
        """
        return self.model(x)
