    Combined Loss: Weighted sum of BCELoss and Dice Loss.
    Loss = BCE_WEIGHT * BCE + DICE_WEIGHT * (1 - Dice)
    """

    def __init__(self, bce_weight=0.5, dice_weight=0.5, smooth=1e-6):
        super(ContrailLoss, self).__init__()
        self.bce_weight = bce_weight
        self.dice_weight = dice_weight
        self.smooth = smooth
        self.bce = nn.BCELoss()

    def forward(self, y_pred, y_true):
        # BCE Loss
        bce_loss = self.bce(y_pred, y_true)

        # Dice Loss
        # We compute the soft dice coefficient for differentiability
        y_pred_f = y_pred.view(-1)

class ContrailLoss(nn.Module):
    """
    Combined Loss: Weighted sum of Focal Loss and Dice Loss.
    Addresses class imbalance where background dominates.
    """

    def __init__(self, focal_weight=1.0, dice_weight=1.0, alpha=0.25, gamma=2.0, smooth=1e-6):
        super(ContrailLoss, self).__init__()
        self.focal_weight = focal_weight
        self.dice_weight = dice_weight
        self.alpha = alpha
        self.gamma = gamma
        self.smooth = smooth

    def forward(self, y_pred, y_true):
        # Clamp predictions for numerical stability
        y_pred = torch.clamp(y_pred, 1e-7, 1.0 - 1e-7)

        # Focal Loss
        # Calculate BCE term
        bce = - y_true * torch.log(y_pred) - (1 - y_true) * torch.log(1 - y_pred)
        
        # Calculate weighting factors
        # pt = p if y=1 else 1-p
        pt = torch.where(y_true == 1, y_pred, 1 - y_pred)
        # alpha_t = alpha if y=1 else 1-alpha
        alpha_t = torch.where(y_true == 1, self.alpha, 1 - self.alpha)
        
        focal_loss = alpha_t * (1 - pt).pow(self.gamma) * bce
        focal_loss = focal_loss.mean()

        # Dice Loss
        y_pred_f = y_pred.view(-1)
        y_true_f = y_true.view(-1)
        intersection = (y_pred_f * y_true_f).sum()
        union = y_pred_f.sum() + y_true_f.sum()
        dice_score = (2.0 * intersection + self.smooth) / (union + self.smooth)
        dice_loss = 1.0 - dice_score

        return self.focal_weight * focal_loss + self.dice_weight * dice_loss
