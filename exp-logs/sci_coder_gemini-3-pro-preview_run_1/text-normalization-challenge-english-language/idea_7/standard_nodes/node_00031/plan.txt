import sys
import traceback
import torch
from torch.utils.data import DataLoader
from library.config import Config
from library.utils import seed_everything
from library.data import prepare_data, collate_tagger, collate_seq2seq
from library.models import AttentionBiLSTMTagger, TransformerSeq2Seq
from library.engine import TaggerEngine, Seq2SeqEngine, get_class_weights


def train_tagger_safe():
    """
    Wrapper function to train the Tagger model in a separate process.
    Catches exceptions and flushes stderr to ensure errors are visible.
    """
    try:
        # Redirect output to ensure it's captured
        sys.stdout.flush()
        sys.stderr.flush()
        
        print("Initializing Tagger Training Process...")
        seed_everything()
        
        # Load Data (using cache)
        vocab, _, train_d_tag, val_d_tag, _, _, _ = prepare_data(load_cached_data=True)
        
        # Create Loaders
        train_loader = DataLoader(
            train_d_tag,
            batch_size=Config.BATCH_SIZE,
            shuffle=True,
            collate_fn=collate_tagger,
            num_workers=Config.NUM_WORKERS,
            pin_memory=True,
        )
        val_loader = DataLoader(
            val_d_tag,
            batch_size=Config.BATCH_SIZE,
            shuffle=False,
            collate_fn=collate_tagger,
            num_workers=Config.NUM_WORKERS,
            pin_memory=True,
        )
        
        # Initialize Model
        num_tokens = len(vocab.token2id)
        num_chars = len(vocab.char2id)
        num_classes = len(vocab.class2id)
        
        model = AttentionBiLSTMTagger(num_tokens, num_chars, num_classes)
        
        # Get Class Weights
        class_weights = get_class_weights(train_d_tag, vocab, Config.DEVICE)
        
        # Initialize Engine
        engine = TaggerEngine(model, Config.DEVICE, train_loader, val_loader, class_weights)
        
        # Train
        engine.fit()
        
    except Exception:
        traceback.print_exc()
        sys.stderr.flush()
        sys.stdout.flush()
        sys.exit(1)
