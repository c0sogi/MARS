I will optimize the solution by reverting to a pure-text **DistilRoBERTa Dual-Encoder** architecture, removing the explicit metadata embeddings and Multi-Sample Dropout that hindered performance in the previous run.

**Strategy:**
1.  **Backbone Switch:** Replace `microsoft/deberta-v3-small` with `distilroberta-base`. Previous lessons indicated that on this small dataset, the simpler DistilRoBERTa model outperformed the metadata-enhanced DeBERTa model (Cite 00039, 00023).
2.  **Remove Metadata:** Eliminate the `category` and `host` embedding layers. Explicitly modeling these high-cardinality features introduced noise and overfitting compared to allowing the model to infer context from text (Cite 00039).
3.  **Simplify Regularization:** Remove Multi-Sample Dropout (MSD) and replace it with a standard Dropout ($p=0.1$). MSD introduced convergence latency that prevented the model from fitting the data within a short training budget (Cite 00042).
4.  **Optimize Training Schedule:** Increase training epochs from 3 to 5 (Cite 00011) and increase batch size to 16 (Cite 00012). This leverages the smaller memory footprint of DistilRoBERTa to improve generalization and convergence.

def encode_metadata(train_df, val_df, test_df, col_name):
    """
    Encodes a categorical column based on training data.
    Assigns a special index for unknown values in val/test.
    """
    # Get unique values from training set
    unique_vals = train_df[col_name].unique()

    # Create mapping: val -> idx
    # Reserve 0 for unknown/padding if needed, but here we just append unknown at the end
    val2idx = {v: i for i, v in enumerate(unique_vals)}

    # Index for unknown values
    unk_idx = len(unique_vals)

    def transform(df):
        # Map and fill NaNs/Unknowns with unk_idx
        return df[col_name].map(val2idx).fillna(unk_idx).astype(int)

    train_encoded = transform(train_df)
    val_encoded = transform(val_df)
    test_encoded = transform(test_df)

    # Return encoded series and the total number of embeddings needed (unique + 1 for unknown)
    return train_encoded, val_encoded, test_encoded, unk_idx + 1

class DebertaDualEncoder(nn.Module):
    """
    Metadata-Enhanced DeBERTa-v3-Small Dual-Encoder.
    Fuses text features from Question and Answer branches with categorical metadata.
    """

    def __init__(self, meta_dims):
        """
        Args:
            meta_dims (dict): Dictionary containing vocabulary sizes for metadata.
                              e.g., {'num_categories': 10, 'num_hosts': 50}
        """
        super(DebertaDualEncoder, self).__init__()

        # 1. Backbone
        self.config = AutoConfig.from_pretrained(config.model_name)
        self.backbone = AutoModel.from_pretrained(config.model_name, config=self.config)

        # Gradient Checkpointing for memory efficiency if needed (optional, but good for larger batches)
        # self.backbone.gradient_checkpointing_enable()

        self.hidden_size = self.config.hidden_size

        # 2. Pooling
        self.pooler = MeanMaxPooling()

        # 3. Metadata Embeddings
        # Embedding dimensions - can be tuned, using 32 as a reasonable default
        self.cat_emb_dim = 32
        self.host_emb_dim = 32

        self.category_embedding = nn.Embedding(
            meta_dims["num_categories"], self.cat_emb_dim
        )
        self.host_embedding = nn.Embedding(meta_dims["num_hosts"], self.host_emb_dim)

        # 4. Fusion Dimension Calculation
        # Text: Q_avg, Q_max, A_avg, A_max, Q_avg*A_avg, |Q_avg-A_avg| -> 6 vectors
        text_fusion_dim = self.hidden_size * 6
        meta_fusion_dim = self.cat_emb_dim + self.host_emb_dim
        total_fusion_dim = text_fusion_dim + meta_fusion_dim

        # 5. Fusion Normalization
        self.fusion_norm = nn.LayerNorm(total_fusion_dim)

        # 6. Non-Linear MLP Head
        # Structure: Linear -> GELU -> Dropout -> Linear (logits)
        # We use Multi-Sample Dropout for the final projection
        self.inter_dim = self.hidden_size  # Intermediate dimension

        self.head_fc1 = nn.Linear(total_fusion_dim, self.inter_dim)
        self.head_act = nn.GELU()
        self.head_norm = nn.LayerNorm(self.inter_dim)

        # Multi-Sample Dropout
        self.dropouts = nn.ModuleList([nn.Dropout(0.5) for _ in range(5)])

        # Final projection to 30 targets
        self.head_fc2 = nn.Linear(self.inter_dim, len(config.target_cols))

        # Initialize weights for new layers
        self._init_weights(self.category_embedding)
        self._init_weights(self.host_embedding)
        self._init_weights(self.fusion_norm)
        self._init_weights(self.head_fc1)
        self._init_weights(self.head_norm)
        self._init_weights(self.head_fc2)

    def _init_weights(self, module):
        """Initialize weights for new layers similar to transformers."""
        if isinstance(module, nn.Linear):
            module.weight.data.normal_(mean=0.0, std=self.config.initializer_range)
            if module.bias is not None:
                module.bias.data.zero_()
        elif isinstance(module, nn.Embedding):
            module.weight.data.normal_(mean=0.0, std=self.config.initializer_range)
            if module.padding_idx is not None:
                module.weight.data[module.padding_idx].zero_()
        elif isinstance(module, nn.LayerNorm):
            module.bias.data.zero_()
            module.weight.data.fill_(1.0)

    def forward(
        self,
        q_input_ids,
        q_attention_mask,
        a_input_ids,
        a_attention_mask,
        category,
        host,
    ):
        # --- Branch 1: Question ---
        q_out = self.backbone(input_ids=q_input_ids, attention_mask=q_attention_mask)
        # q_out.last_hidden_state: (B, L, H)
        q_avg, q_max = self.pooler(q_out.last_hidden_state, q_attention_mask)

        # --- Branch 2: Answer ---
        a_out = self.backbone(input_ids=a_input_ids, attention_mask=a_attention_mask)
        a_avg, a_max = self.pooler(a_out.last_hidden_state, a_attention_mask)

        # --- Interactions (on Avg Pooled features) ---
        # Element-wise product
        inter_prod = q_avg * a_avg
        # Absolute difference
        inter_diff = torch.abs(q_avg - a_avg)

        # --- Metadata Embeddings ---
        cat_emb = self.category_embedding(category)  # (B, 32)
        host_emb = self.host_embedding(host)  # (B, 32)

        # --- Fusion ---
        # Concatenate all features
        # [Q_avg, Q_max, A_avg, A_max, Prod, Diff, Cat, Host]
        fused_features = torch.cat(
            [q_avg, q_max, a_avg, a_max, inter_prod, inter_diff, cat_emb, host_emb],
            dim=1,
        )

        # Normalize fused vector
        fused_features = self.fusion_norm(fused_features)

        # --- MLP Head ---
        x = self.head_fc1(fused_features)
        x = self.head_act(x)
        x = self.head_norm(x)

        # Multi-Sample Dropout
        # Average the logits from multiple dropout masks
        logits = torch.mean(
            torch.stack(
                [self.head_fc2(dropout(x)) for dropout in self.dropouts], dim=0
            ),
            dim=0,
        )

        return logits
