def _transpose_and_gather_feat(feat, ind):
    """
    Transposes feature map and gathers features.
    Optimized to avoid large memory allocation from contiguous() on permuted tensor.
    Cite debug_lesson_3: Optimize Memory by Indexing Sparse Targets Before Reshaping.
    """
    # feat: (B, C, H, W)
    # ind: (B, K)
    batch, channel, height, width = feat.size()

    # View as (B, C, H*W) - usually no copy if already contiguous
    feat = feat.view(batch, channel, -1)

    # Expand indices to (B, C, K)
    # ind is (B, K) -> unsqueeze -> (B, 1, K) -> expand -> (B, C, K)
    ind = ind.unsqueeze(1).expand(batch, channel, ind.size(1))

    # Gather along spatial dimension (dim 2)
    feat = feat.gather(2, ind)  # Result: (B, C, K)

    # Permute to (B, K, C) - no copy
    feat = feat.permute(0, 2, 1)

    return feat
