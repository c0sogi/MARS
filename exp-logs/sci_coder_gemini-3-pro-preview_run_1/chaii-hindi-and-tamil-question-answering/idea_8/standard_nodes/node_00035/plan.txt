def get_optimizer(model, config):
    """
    Constructs the optimizer with differential learning rates and weight decay settings.
    Separates the backbone parameters from the head parameters using explicit module references.
    Cite solution_lesson_node_00033
    """
    no_decay = ["bias", "LayerNorm.weight"]
    optimizer_grouped_parameters = []

    # Helper to group parameters
    def add_params(module, lr):
        decay = []
        no_decay_list = []
        for n, p in module.named_parameters():
            if not p.requires_grad:
                continue
            if any(nd in n for nd in no_decay):
                no_decay_list.append(p)
            else:
                decay.append(p)
        
        optimizer_grouped_parameters.append({
            "params": decay,
            "weight_decay": config.weight_decay,
            "lr": lr
        })
        optimizer_grouped_parameters.append({
            "params": no_decay_list,
            "weight_decay": 0.0,
            "lr": lr
        })

    # Backbone parameters
    add_params(model.backbone, config.lr_backbone)

    # Head parameters (qa_outputs and relevance_classifier)
    add_params(model.qa_outputs, config.lr_head)
    add_params(model.relevance_classifier, config.lr_head)

    optimizer = torch.optim.AdamW(optimizer_grouped_parameters, eps=config.adam_epsilon)

    return optimizer
