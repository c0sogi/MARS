class CompositeBlock(nn.Module):
    """
    Uniform Composite Block with Curated Context Injection and Strict Identity Residuals.
    Maintains uniform capacity (Hidden Dim 512) throughout.
    """

    def __init__(self, hidden_dim, context_dim, dropout=0.1):
        super().__init__()
        self.hidden_dim = hidden_dim

        # Aligned Bi-LSTM
        # Input is hidden_state + context
        # Hidden dim is hidden_dim // 2 per direction -> output is hidden_dim
        # This prevents the information bottleneck seen in standard Bi-LSTMs
        self.lstm = nn.LSTM(
            input_size=hidden_dim + context_dim,
            hidden_size=hidden_dim // 2,
            num_layers=1,
            batch_first=True,
            bidirectional=True,
        )

        # Pointwise FFN (Expansion 2x)
        # We avoid 4x expansion to maintain stability without normalization
        ffn_dim = hidden_dim * Config.FFN_EXPANSION_FACTOR
        self.ffn = nn.Sequential(
            nn.Linear(hidden_dim, ffn_dim), nn.GELU(), nn.Linear(ffn_dim, hidden_dim)
        )

        self.dropout = nn.Dropout(dropout)

    def forward(self, x, context):
        # x: (Batch, Seq, Hidden)
        # context: (Batch, Seq, Context)

        # 1. Curated Context Injection
        # We re-inject the static/physics context at every block
        lstm_input = torch.cat([x, context], dim=-1)

        # 2. Aligned Bi-LSTM
        lstm_out, _ = self.lstm(lstm_input)

        # 3. Strict Identity Residual 1
        # Dimensions match (512 -> 512), no linear projection on skip
        # Dropout applied to residual branch only
        x = x + self.dropout(lstm_out)

        # 4. Pointwise FFN
        ffn_out = self.ffn(x)

        # 5. Strict Identity Residual 2
        # No LayerNorm, preserving absolute pressure magnitude
        x = x + self.dropout(ffn_out)

        return x

    Wide-State Composite Block with Explicit Context Injection.
    Supports dimension expansion (Bottleneck -> Wide) via residual projection.
    Cite solution_lesson_node_00055: Projection Shortcuts for Dimension-Mismatched Residuals.
    Cite solution_lesson_node_00044: Unnormalized Residual Connections.
    """

    def __init__(self, input_dim, hidden_dim, context_dim, dropout=0.1):
        super().__init__()
        self.input_dim = input_dim
        self.hidden_dim = hidden_dim

        # Bi-LSTM
        # Input: input_dim + context
        # Output: hidden_dim (hidden_dim // 2 * 2)
        # Cite solution_lesson_node_00055: Allow layer output to be 2*hidden_size (Wide State)
        self.lstm = nn.LSTM(
            input_size=input_dim + context_dim,
            hidden_size=hidden_dim // 2,
            num_layers=1,
            batch_first=True,
            bidirectional=True,
        )

        # Residual Projection
        # If input_dim != hidden_dim, project residual path
        self.use_projection = input_dim != hidden_dim
        if self.use_projection:
            self.res_proj = nn.Linear(input_dim, hidden_dim)

        # Pointwise FFN (Expansion 2x)
        # Cite solution_lesson_node_00052: Keep FFN expansion modest (2x) in unnormalized blocks
        ffn_dim = hidden_dim * Config.FFN_EXPANSION_FACTOR
        self.ffn = nn.Sequential(
            nn.Linear(hidden_dim, ffn_dim), nn.GELU(), nn.Linear(ffn_dim, hidden_dim)
        )

        self.dropout = nn.Dropout(dropout)

    def forward(self, x, context):
        # x: (Batch, Seq, input_dim)
        # context: (Batch, Seq, context_dim)

        # 1. Explicit Context Injection
        # Cite solution_lesson_node_00090: Deep Context Injection of raw features
        lstm_input = torch.cat([x, context], dim=-1)

        # 2. Bi-LSTM
        lstm_out, _ = self.lstm(lstm_input)  # (Batch, Seq, hidden_dim)

        res = x

        # 4. Pointwise FFN + Residual 2 (Channel)
        # Cite solution_lesson_node_00040: Interleave FFNs
        x = x + self.dropout(self.ffn(x))

        return x
