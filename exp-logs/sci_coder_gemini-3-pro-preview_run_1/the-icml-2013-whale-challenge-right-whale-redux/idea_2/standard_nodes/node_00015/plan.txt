Based on the provided lessons and failure analysis, the primary bottleneck in the previous solution was the use of a ResNet-18 backbone. While powerful for images, ResNet-18 downsamples the temporal dimension by a factor of 32, collapsing the short-duration right whale up-calls (approx. 0.5-1.0s) into just a few frames. This loss of temporal resolution hinders the BiGRU's ability to model the signal's evolution (Cite solution_lesson_node_00012).

To address this, I will replace the ResNet-18 backbone with a shallower, custom 2-layer CNN frontend. This "Simple CRNN" architecture will downsample the temporal dimension by only 4x, preserving the detailed time-frequency morphology required for accurate detection (Cite solution_lesson_node_00003). The configuration already includes the optimized SpecAugment time masking of 200ms (Cite solution_lesson_node_00011), so no changes are needed there.

class ResNetBiGRUAttention(nn.Module):
    """
    CRNN Architecture: ResNet-18 Backbone + BiGRU + Attention Pooling.
    """

    def __init__(self):
        super(ResNetBiGRUAttention, self).__init__()

        # ==========================================
        # 1. Deep Spectral Feature Extraction (ResNet-18)
        # ==========================================
        # Load standard ResNet18 structure
        resnet = models.resnet18(weights=None)

        # Modify first conv layer: 3 channels -> 1 channel
        # Original: kernel=7, stride=2, padding=3
        resnet.conv1 = nn.Conv2d(1, 64, kernel_size=7, stride=2, padding=3, bias=False)

        # Remove the classification head (AvgPool and FC)
        # We keep layers up to layer4
        self.backbone = nn.Sequential(*list(resnet.children())[:-2])

        # ==========================================
        # 2. Temporal Context Modeling (BiGRU)
        # ==========================================
        # ResNet-18 layer4 output has 512 channels
        self.feature_dim = 512

        self.gru = nn.GRU(
            input_size=self.feature_dim,
            hidden_size=Config.GRU_HIDDEN_DIM,
            num_layers=Config.GRU_NUM_LAYERS,
            batch_first=True,
            bidirectional=True,
            dropout=Config.DROPOUT if Config.GRU_NUM_LAYERS > 1 else 0,
        )

        # ==========================================
        # 3. Attention-Based Aggregation
        # ==========================================
        # BiGRU output dimension is hidden_dim * 2 (directions)
        self.gru_out_dim = Config.GRU_HIDDEN_DIM * 2
        self.attention = AttentionPooling(self.gru_out_dim)

        # ==========================================
        # 4. Classifier
        # ==========================================
        self.classifier = nn.Sequential(
            nn.Dropout(Config.DROPOUT), nn.Linear(self.gru_out_dim, Config.NUM_CLASSES)
        )

    def forward(self, x):
        # Input x: (Batch, 1, F, T) where F=64

        # 1. Extract Features
        # Output: (Batch, 512, F', T')
        # ResNet reduces dimensions by factor of 32.
        # F=64 -> F'=2. T depends on input duration.
        x = self.backbone(x)

        # 2. Prepare for RNN
        # Collapse Frequency dimension (Average Pooling) -> (Batch, 512, T')
        x = torch.mean(x, dim=2)

        # Permute to (Batch, T', 512) for GRU
        x = x.permute(0, 2, 1)

        # 3. Temporal Modeling
        # Output: (Batch, T', Hidden*2)
        x, _ = self.gru(x)

        # 4. Attention Aggregation
        # Output: (Batch, Hidden*2)
        x = self.attention(x)

        # 5. Classification
        # Output: (Batch, 1) - Logits
        logits = self.classifier(x)

        return logits

class SimpleCRNN(nn.Module):
    """
    CRNN Architecture: 2-Layer CNN + BiGRU + Attention Pooling.
    Replaces ResNet backbone to preserve temporal resolution (Cite solution_lesson_node_00012).
    """

    def __init__(self):
        super(SimpleCRNN, self).__init__()

        # ==========================================
        # 1. Shallow Spectral Feature Extraction
        # ==========================================
        # 2-Layer CNN to preserve temporal resolution (Cite solution_lesson_node_00003)
        self.cnn = nn.Sequential(
            # Layer 1
            nn.Conv2d(1, 32, kernel_size=3, stride=1, padding=1, bias=False),
            nn.BatchNorm2d(32),
            nn.ReLU(),
            nn.MaxPool2d(2, 2),  # Downsample F and T by 2

            # Layer 2
            nn.Conv2d(32, 64, kernel_size=3, stride=1, padding=1, bias=False),
            nn.BatchNorm2d(64),
            nn.ReLU(),
            nn.MaxPool2d(2, 2)   # Downsample F and T by 2
        )
        # Total downsampling: 4x

        # ==========================================
        # 2. Temporal Context Modeling (BiGRU)
        # ==========================================
        # Calculate RNN input dimension
        # F_in = 64. After two poolings (stride 2): 64 -> 32 -> 16.
        # Channels = 64.
        # Dim = 64 * 16 = 1024.
        self.rnn_input_dim = 64 * 16

        self.gru = nn.GRU(
            input_size=self.rnn_input_dim,
            hidden_size=Config.GRU_HIDDEN_DIM,
            num_layers=Config.GRU_NUM_LAYERS,
            batch_first=True,
            bidirectional=True,
            dropout=Config.DROPOUT if Config.GRU_NUM_LAYERS > 1 else 0,
        )

        # ==========================================
        # 3. Attention-Based Aggregation
        # ==========================================
        self.gru_out_dim = Config.GRU_HIDDEN_DIM * 2
        self.attention = AttentionPooling(self.gru_out_dim)

        # ==========================================
        # 4. Classifier
        # ==========================================
        self.classifier = nn.Sequential(
            nn.Dropout(Config.DROPOUT),
            nn.Linear(self.gru_out_dim, Config.NUM_CLASSES)
        )

    def forward(self, x):
        # Input x: (Batch, 1, F, T)

        # 1. Extract Features
        x = self.cnn(x)  # Output: (Batch, 64, 16, T/4)

        # 2. Prepare for RNN
        # Permute to (Batch, Time, Features)
        B, C, F, T = x.size()
        x = x.permute(0, 3, 1, 2).contiguous()  # (Batch, T, C, F)
        x = x.view(B, T, -1)  # (Batch, T, C*F) -> (Batch, T, 1024)

        # 3. Temporal Modeling
        x, _ = self.gru(x)

        # 4. Attention Aggregation
        x = self.attention(x)

        # 5. Classification
        logits = self.classifier(x)

        return logits
