I will optimize the solution by reverting the model architecture from the ResNet-18 backbone to a custom Convolutional Recurrent Neural Network (CRNN). This decision is based on **Lesson ID: solution_lesson_node_00012**, which established that standard computer vision backbones like ResNet excessively downsample the temporal dimension (from ~200 to ~6 frames), destroying the temporal evolution of short acoustic events. By switching to a shallower 2-layer CNN frontend with only 4x downsampling, we preserve sufficient temporal resolution (50 frames) for the Bi-GRU to effectively model the right whale up-calls. I will retain the SpecAugment settings (Time Masking = 20) which were proven effective in **Lesson ID: solution_lesson_node_00011**.

class ResNetBiGRUAttention(nn.Module):
    """
    CRNN Architecture: ResNet-18 Backbone + BiGRU + Attention Pooling.
    """

    def __init__(self):
        super(ResNetBiGRUAttention, self).__init__()

        # ==========================================
        # 1. Deep Spectral Feature Extraction (ResNet-18)
        # ==========================================
        # Load standard ResNet18 structure
        resnet = models.resnet18(weights=None)

        # Modify first conv layer: 3 channels -> 1 channel
        # Original: kernel=7, stride=2, padding=3
        resnet.conv1 = nn.Conv2d(1, 64, kernel_size=7, stride=2, padding=3, bias=False)

        # Remove the classification head (AvgPool and FC)
        # We keep layers up to layer4
        self.backbone = nn.Sequential(*list(resnet.children())[:-2])

        # ==========================================
        # 2. Temporal Context Modeling (BiGRU)
        # ==========================================
        # ResNet-18 layer4 output has 512 channels
        self.feature_dim = 512

        self.gru = nn.GRU(
            input_size=self.feature_dim,
            hidden_size=Config.GRU_HIDDEN_DIM,
            num_layers=Config.GRU_NUM_LAYERS,
            batch_first=True,
            bidirectional=True,
            dropout=Config.DROPOUT if Config.GRU_NUM_LAYERS > 1 else 0,
        )

        # ==========================================
        # 3. Attention-Based Aggregation
        # ==========================================
        # BiGRU output dimension is hidden_dim * 2 (directions)
        self.gru_out_dim = Config.GRU_HIDDEN_DIM * 2
        self.attention = AttentionPooling(self.gru_out_dim)

        # ==========================================
        # 4. Classifier
        # ==========================================
        self.classifier = nn.Sequential(
            nn.Dropout(Config.DROPOUT), nn.Linear(self.gru_out_dim, Config.NUM_CLASSES)
        )

    def forward(self, x):
        # Input x: (Batch, 1, F, T) where F=64

        # 1. Extract Features
        # Output: (Batch, 512, F', T')
        # ResNet reduces dimensions by factor of 32.
        # F=64 -> F'=2. T depends on input duration.
        x = self.backbone(x)

        # 2. Prepare for RNN
        # Collapse Frequency dimension (Average Pooling) -> (Batch, 512, T')
        x = torch.mean(x, dim=2)

        # Permute to (Batch, T', 512) for GRU
        x = x.permute(0, 2, 1)

        # 3. Temporal Modeling
        # Output: (Batch, T', Hidden*2)
        x, _ = self.gru(x)

        # 4. Attention Aggregation
        # Output: (Batch, Hidden*2)
        x = self.attention(x)

        # 5. Classification
        # Output: (Batch, 1) - Logits
        logits = self.classifier(x)

        return logits

class CRNN(nn.Module):
    """
    Custom CRNN Architecture: 2-Layer CNN + BiGRU + Attention Pooling.
    Cite solution_lesson_node_00012: Preserves temporal resolution better than ResNet.
    """

    def __init__(self):
        super(CRNN, self).__init__()

        # ==========================================
        # 1. CNN Frontend
        # ==========================================
        # Layer 1
        self.conv1 = nn.Conv2d(1, 32, kernel_size=3, stride=1, padding=1)
        self.bn1 = nn.BatchNorm2d(32)
        self.relu = nn.ReLU()
        self.pool1 = nn.MaxPool2d(2, 2)  # F: 64->32, T: 200->100

        # Layer 2
        self.conv2 = nn.Conv2d(32, 64, kernel_size=3, stride=1, padding=1)
        self.bn2 = nn.BatchNorm2d(64)
        self.pool2 = nn.MaxPool2d(2, 2)  # F: 32->16, T: 100->50

        # Calculated Feature Dimension for GRU
        # Frequency dimension becomes 16, Channels 64.
        # Flattened input to GRU: 16 * 64 = 1024
        self.gru_input_dim = 64 * (Config.N_MELS // 4)

        # ==========================================
        # 2. Temporal Context Modeling (BiGRU)
        # ==========================================
        self.gru = nn.GRU(
            input_size=self.gru_input_dim,
            hidden_size=Config.GRU_HIDDEN_DIM,
            num_layers=Config.GRU_NUM_LAYERS,
            batch_first=True,
            bidirectional=True,
            dropout=Config.DROPOUT if Config.GRU_NUM_LAYERS > 1 else 0,
        )

        # ==========================================
        # 3. Attention-Based Aggregation
        # ==========================================
        self.gru_out_dim = Config.GRU_HIDDEN_DIM * 2
        self.attention = AttentionPooling(self.gru_out_dim)

        # ==========================================
        # 4. Classifier
        # ==========================================
        self.classifier = nn.Sequential(
            nn.Dropout(Config.DROPOUT), nn.Linear(self.gru_out_dim, Config.NUM_CLASSES)
        )

    def forward(self, x):
        # Input x: (Batch, 1, F, T)

        # 1. CNN Frontend
        x = self.conv1(x)
        x = self.bn1(x)
        x = self.relu(x)
        x = self.pool1(x)

        x = self.conv2(x)
        x = self.bn2(x)
        x = self.relu(x)
        x = self.pool2(x)

        # 2. Prepare for RNN
        # Current shape: (Batch, 64, 16, 50)
        B, C, F, T = x.size()
        # Permute to (Batch, T, C, F)
        x = x.permute(0, 3, 1, 2).contiguous()
        # Flatten C and F -> (Batch, T, C*F)
        x = x.view(B, T, C * F)

        # 3. Temporal Modeling
        x, _ = self.gru(x)

        # 4. Attention Aggregation
        x = self.attention(x)

        # 5. Classification
        logits = self.classifier(x)

        return logits
